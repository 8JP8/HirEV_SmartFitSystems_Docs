19 public override Point3D Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options)
20 {
21 double x = 0, y = 0, z = 0;
22 if (reader.TokenType != JsonTokenType.StartObject) throw new JsonException();
23
24 while (reader.Read())
25 {
26 if (reader.TokenType == JsonTokenType.EndObject) break;
27 if (reader.TokenType == JsonTokenType.PropertyName)
28 {
29 string propertyName = reader.GetString()?.ToLowerInvariant() ?? "";
30 reader.Read();
31 switch (propertyName)
32 {
33 case "x": x = reader.GetDouble(); break;
34 case "y": y = reader.GetDouble(); break;
35 case "z": z = reader.GetDouble(); break;
36 }
37 }
38 }
39 return new Point3D(x, y, z);
40 }
