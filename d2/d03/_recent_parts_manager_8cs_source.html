36 private void LoadRecentParts()
37 {
38 try
39 {
40 if (File.Exists(_recentPartsFilePath))
41 {
42 string json = File.ReadAllText(_recentPartsFilePath);
43 var allParts = JsonSerializer.Deserialize<List<string>>(json) ?? new List<string>();
44
45 // Validate each file and keep only existing ones
46 _recentParts = allParts.Where(path => File.Exists(path)).ToList();
47
48 // If any files were removed, save the cleaned list
49 if (_recentParts.Count != allParts.Count)
50 {
51 SaveRecentParts();
52 Debug.WriteLine($"Cleaned recent parts list: removed {allParts.Count - _recentParts.Count} invalid entries");
53 }
54 }
55 }
56 catch (Exception ex)

59 _recentParts = new List<string>();

86 public void AddRecentFile(string filePath)
87 {
88 if (string.IsNullOrWhiteSpace(filePath) || !File.Exists(filePath))
89 return;
90
91 try
92 {
93 // Normalize the path
94 filePath = Path.GetFullPath(filePath);
95
96 // Remove if already in list
97 _recentParts.Remove(filePath);
98
99 // Add to the beginning
100 _recentParts.Insert(0, filePath);
101
102 // Trim to max size
103 if (_recentParts.Count > MAX_RECENT_FILES)
104 {
105 _recentParts = _recentParts.Take(MAX_RECENT_FILES).ToList();
106 }
107
108 SaveRecentParts();
109 Debug.WriteLine($"Added recent part: {Path.GetFileName(filePath)}");
110 }
111 catch (Exception ex)

113 Debug.WriteLine($"Error adding recent file: {ex.Message}");
