203 public void TestDetailedCollisionSampling(
204 Point3D start,
205 Point3D end,
206 List<Brep> obstacles,
207 Func<Point3D, List<Brep>, bool> isInsideSolidFunc,
208 double sampleInterval = 0.5)
209 {
210 _report.AppendLine("┌──────────────────────────────────────────────────────────────────────────────┐");
211 _report.AppendLine("│ DETAILED COLLISION SAMPLING TEST │");
212 _report.AppendLine("└──────────────────────────────────────────────────────────────────────────────┘");
213 _report.AppendLine();
214
215 Vector3D direction = new Vector3D(end.X - start.X, end.Y - start.Y, end.Z - start.Z);
216 double length = direction.Length;
217
218 if (length < 0.001)
219 {
220 _report.AppendLine(" ⚠️ WARNING: Segment too short for sampling test");
221 _testsWarnings++;
222 return;
223 }
224
225 direction.Normalize();
226 int samples = Math.Max(3, (int)(length / sampleInterval));
227
228 _report.AppendLine($" Segment Start: ({start.X:F3}, {start.Y:F3}, {start.Z:F3})");
229 _report.AppendLine($" Segment End: ({end.X:F3}, {end.Y:F3}, {end.Z:F3})");
230 _report.AppendLine($" Segment Length: {length:F3} mm");
231 _report.AppendLine($" Sample Interval: {sampleInterval} mm");
232 _report.AppendLine($" Total Samples: {samples}");
233 _report.AppendLine();
234
235 var collisionPoints = new List<(int index, Point3D point)>();
236 var brepObstacles = obstacles.Cast<Brep?>().ToList();
237
238 for (int i = 0; i <= samples; i++)
239 {
240 double t = i / (double)samples;
241 var samplePoint = new Point3D(
242 start.X + direction.X * length * t,
243 start.Y + direction.Y * length * t,
244 start.Z + direction.Z * length * t);
245
246 if (isInsideSolidFunc(samplePoint, obstacles))
247 {
248 collisionPoints.Add((i, samplePoint));
249 }
250 }
251
252 if (collisionPoints.Count > 0)
253 {
254 _report.AppendLine($" ❌ COLLISION POINTS DETECTED: {collisionPoints.Count}");
255 _report.AppendLine();
256
257 foreach (var (index, point) in collisionPoints.Take(20)) // Limit output
258 {
259 double t = index / (double)samples;
260 _report.AppendLine($" Sample {index} (t={t:F3}): ({point.X:F3}, {point.Y:F3}, {point.Z:F3})");
261 }
262
263 if (collisionPoints.Count > 20)
264 {
265 _report.AppendLine($" ... and {collisionPoints.Count - 20} more collision points");
266 }
267
268 _testsFailed++;
269 }
270 else
271 {
272 _report.AppendLine($" ✅ No collisions detected along segment");
273 _testsPassed++;
274 }
275 _report.AppendLine();
276 }

285 public void TestSurfaceMeasurementSequence(
286 List<Point3D> pathPoints,
287 List<Point3D> approachPoints,
288 List<Point3D> measurementPoints,
289 double tolerance = 1.0)
290 {
291 _report.AppendLine("┌──────────────────────────────────────────────────────────────────────────────┐");
292 _report.AppendLine("│ SURFACE MEASUREMENT SEQUENCE TEST │");
293 _report.AppendLine("│ Expected Pattern: Approach → Measure → Approach │");
294 _report.AppendLine("└──────────────────────────────────────────────────────────────────────────────┘");
295 _report.AppendLine();
296
297 if (pathPoints == null || approachPoints == null || measurementPoints == null)
298 {
299 _report.AppendLine(" ⚠️ WARNING: Missing data for sequence validation");
300 _testsWarnings++;
301 return;
302 }
303
304 _report.AppendLine($" Path Points: {pathPoints.Count}");
305 _report.AppendLine($" Approach Points: {approachPoints.Count}");
306 _report.AppendLine($" Measurement Points: {measurementPoints.Count}");
307 _report.AppendLine($" Tolerance: {tolerance} mm");
308 _report.AppendLine();
309
310 // Classify each path point
311 var classifications = new List<(int index, Point3D point, string type)>();
312
313 foreach (var (point, index) in pathPoints.Select((p, i) => (p, i)))
314 {
315 bool isApproach = approachPoints.Any(ap => ap.DistanceTo(point) < tolerance);
316 bool isMeasure = measurementPoints.Any(mp => mp.DistanceTo(point) < tolerance);
317
318 string type = "UNKNOWN";
319 if (isApproach && isMeasure)
320 type = "BOTH"; // Could be coincident approach/measure
321 else if (isApproach)
322 type = "APPROACH";
323 else if (isMeasure)
324 type = "MEASURE";
325 else
326 type = "TRANSIT";
327
328 classifications.Add((index, point, type));
329 }
330
331 // Validate sequence patterns
332 int validSequences = 0;
333 int invalidSequences = 0;
334 var sequenceViolations = new List<string>();
335
336 for (int i = 0; i < classifications.Count - 2; i++)
337 {
338 var p1 = classifications[i];
339 var p2 = classifications[i + 1];
340 var p3 = classifications[i + 2];
341
342 // Check for valid Approach-Measure-Approach pattern
343 if (p2.type == "MEASURE")
344 {
345 bool validPattern = (p1.type == "APPROACH" || p1.type == "BOTH") &&
346 (p3.type == "APPROACH" || p3.type == "BOTH");
347
348 var result = new SequenceTestResult
349 {
350 PointIndices = new[] { p1.index, p2.index, p3.index },
351 PointTypes = new[] { p1.type, p2.type, p3.type },
352 IsValid = validPattern
353 };
354 _sequenceResults.Add(result);
355
356 if (validPattern)
357 {
358 validSequences++;
359 }
360 else
361 {
362 invalidSequences++;
363 sequenceViolations.Add(
364 $"Index {p2.index}: MEASURE preceded by {p1.type} followed by {p3.type}");
365 }
366 }
367 }
368
369 _report.AppendLine($" ────────────────────────────────────────────────────────");
370 _report.AppendLine($" Sequence Validation Results:");
371 _report.AppendLine($" Valid A→M→A sequences: {validSequences}");
372 _report.AppendLine($" Invalid sequences: {invalidSequences}");
373 _report.AppendLine();
374
375 if (sequenceViolations.Count > 0)
376 {
377 _report.AppendLine($" ❌ SEQUENCE VIOLATIONS:");
378 foreach (var violation in sequenceViolations.Take(10))
379 {
380 _report.AppendLine($" • {violation}");
381 }
382 if (sequenceViolations.Count > 10)
383 {
384 _report.AppendLine($" ... and {sequenceViolations.Count - 10} more violations");
385 }
386 _testsFailed++;
387 }
388 else
389 {
390 _report.AppendLine($" ✅ TEST PASSED: All measurement sequences are valid");
391 _testsPassed++;
392 }
393 _report.AppendLine();
394 }

400 public void TestHoleMeasurementSequence(
401 List<Point3D> holePathPoints,
402 Point3D holeAxisTop,
403 Point3D holeAxisBottom,
404 List<Point3D> holeMeasurementPoints,
405 double tolerance = 2.0)
406 {
407 _report.AppendLine("┌──────────────────────────────────────────────────────────────────────────────┐");
408 _report.AppendLine("│ HOLE MEASUREMENT SEQUENCE TEST │");
409 _report.AppendLine("│ Expected: AxisTop → Descend → (Axis→Measure→Axis)* → Ascend │");
410 _report.AppendLine("└──────────────────────────────────────────────────────────────────────────────┘");
411 _report.AppendLine();
412
413 if (holePathPoints == null || holePathPoints.Count < 4)
414 {
415 _report.AppendLine(" ⚠️ WARNING: Insufficient hole path points for sequence validation");
416 _testsWarnings++;
417 return;
418 }
419
420 _report.AppendLine($" Hole Path Points: {holePathPoints.Count}");
421 _report.AppendLine($" Axis Top: ({holeAxisTop.X:F3}, {holeAxisTop.Y:F3}, {holeAxisTop.Z:F3})");
422 _report.AppendLine($" Axis Bottom: ({holeAxisBottom.X:F3}, {holeAxisBottom.Y:F3}, {holeAxisBottom.Z:F3})");
423 _report.AppendLine($" Measurement Points: {holeMeasurementPoints?.Count ?? 0}");
424 _report.AppendLine();
425
426 // Calculate hole axis direction
427 var axisDir = new Vector3D(
428 holeAxisTop.X - holeAxisBottom.X,
429 holeAxisTop.Y - holeAxisBottom.Y,
430 holeAxisTop.Z - holeAxisBottom.Z);
431 axisDir.Normalize();
432
433 // Check that path starts and ends near axis top
434 bool startsAtTop = holePathPoints.First().DistanceTo(holeAxisTop) < tolerance * 2;
435 bool endsAtTop = holePathPoints.Last().DistanceTo(holeAxisTop) < tolerance * 2;
436
437 _report.AppendLine($" Path Start near Axis Top: {(startsAtTop ? "✅ YES" : "❌ NO")}");
438 _report.AppendLine($" Path End near Axis Top: {(endsAtTop ? "✅ YES" : "❌ NO")}");
439 _report.AppendLine();
440
441 // Check radial pattern for measurement points
442 int radialPatternCount = 0;
443 int radialViolations = 0;
444
445 if (holeMeasurementPoints != null)
446 {
447 for (int i = 1; i < holePathPoints.Count - 1; i++)
448 {
449 var point = holePathPoints[i];
450 bool isMeasure = holeMeasurementPoints.Any(mp => mp.DistanceTo(point) < tolerance);
451
452 if (isMeasure)
453 {
454 // Check if points before and after are on the axis
455 var prev = holePathPoints[i - 1];
456 var next = i + 1 < holePathPoints.Count ? holePathPoints[i + 1] : holePathPoints[i - 1];
457
458 // Calculate distance from axis for prev and next
459 double prevAxisDist = PointToLineDistance(prev, holeAxisBottom, axisDir);
460 double nextAxisDist = PointToLineDistance(next, holeAxisBottom, axisDir);
461
462 if (prevAxisDist < tolerance && nextAxisDist < tolerance)
463 {
464 radialPatternCount++;
465 }
466 else
467 {
468 radialViolations++;
469 _report.AppendLine($" ⚠️ Measure at index {i}: prev axis dist={prevAxisDist:F2}, next axis dist={nextAxisDist:F2}");
470 }
471 }
472 }
473 }
474
475 _report.AppendLine();
476 _report.AppendLine($" Radial Pattern Validations: {radialPatternCount}");
477 _report.AppendLine($" Radial Pattern Violations: {radialViolations}");
478
479 bool testPassed = startsAtTop && endsAtTop && radialViolations == 0;
480
481 if (testPassed)
482 {
483 _report.AppendLine();
484 _report.AppendLine($" ✅ TEST PASSED: Hole measurement sequence is valid");
485 _testsPassed++;
486 }
487 else
488 {
489 _report.AppendLine();
490 _report.AppendLine($" ❌ TEST FAILED: Hole sequence violations detected");
491 _testsFailed++;
492 }
493 _report.AppendLine();
494 }

499 private double PointToLineDistance(Point3D point, Point3D linePoint, Vector3D lineDir)
500 {
501 var v = new Vector3D(point.X - linePoint.X, point.Y - linePoint.Y, point.Z - linePoint.Z);
502 var cross = Vector3D.Cross(v, lineDir);
503 return cross.Length / lineDir.Length;
504 }

513 public void TestFaceAdjacencyTransitions(
514 List<(Point3D point, int faceIndex)> pathWithFaces,
515 Dictionary<int, HashSet<int>> faceAdjacencyMap)
516 {
517 _report.AppendLine("┌──────────────────────────────────────────────────────────────────────────────┐");
518 _report.AppendLine("│ FACE ADJACENCY TRANSITION TEST │");
519 _report.AppendLine("└──────────────────────────────────────────────────────────────────────────────┘");
520 _report.AppendLine();
521
522 if (pathWithFaces == null || pathWithFaces.Count < 2)
523 {
524 _report.AppendLine(" ⚠️ WARNING: Insufficient path data for adjacency testing");
525 _testsWarnings++;
526 return;
527 }
528
529 _report.AppendLine($" Path Points with Face Info: {pathWithFaces.Count}");
530 _report.AppendLine($" Faces in Adjacency Map: {faceAdjacencyMap?.Count ?? 0}");
531 _report.AppendLine();
532
533 int validTransitions = 0;
534 int invalidTransitions = 0;
535 int sameFaceStays = 0;
536 var violations = new List<string>();
537
538 for (int i = 0; i < pathWithFaces.Count - 1; i++)
539 {
540 int currentFace = pathWithFaces[i].faceIndex;
541 int nextFace = pathWithFaces[i + 1].faceIndex;
542
543 if (currentFace == nextFace)
544 {
545 sameFaceStays++;
546 continue;
547 }
548
549 // Check if faces are adjacent
550 bool areAdjacent = false;
551 if (faceAdjacencyMap != null && faceAdjacencyMap.TryGetValue(currentFace, out var adjacentFaces))
552 {
553 areAdjacent = adjacentFaces.Contains(nextFace);
554 }
555
556 var result = new FaceAdjacencyTestResult
557 {
558 PathIndex = i,
559 FromFace = currentFace,
560 ToFace = nextFace,
561 AreAdjacent = areAdjacent
562 };
563 _adjacencyResults.Add(result);
564
565 if (areAdjacent)
566 {
567 validTransitions++;
568 }
569 else
570 {
571 invalidTransitions++;
572 violations.Add($"Index {i}: Face {currentFace} → Face {nextFace} (NOT ADJACENT)");
573 }
574 }
575
576 _report.AppendLine($" ────────────────────────────────────────────────────────");
577 _report.AppendLine($" Transition Analysis:");
578 _report.AppendLine($" Same-face movements: {sameFaceStays}");
579 _report.AppendLine($" Valid adjacent jumps: {validTransitions}");
580 _report.AppendLine($" Non-adjacent transitions: {invalidTransitions}");
581 _report.AppendLine();
582
583 if (violations.Count > 0)
584 {
585 _report.AppendLine($" ⚠️ NON-ADJACENT TRANSITIONS (may require intermediate waypoints):");
586 foreach (var v in violations.Take(10))
587 {
588 _report.AppendLine($" • {v}");
589 }
590 if (violations.Count > 10)
591 {
592 _report.AppendLine($" ... and {violations.Count - 10} more");
593 }
594
595 // Non-adjacent transitions are warnings, not failures (they may use intermediate waypoints)
596 _testsWarnings++;
597 _report.AppendLine();
598 _report.AppendLine($" ⚠️ WARNING: Non-adjacent transitions detected - verify intermediate waypoints");
599 }
600 else
601 {
602 _report.AppendLine($" ✅ TEST PASSED: All face transitions are valid");
603 _testsPassed++;
604 }
605 _report.AppendLine();
606 }

615 public void TestNormalParallelMovements(
616 List<(Point3D point, Vector3D normal)> pathWithNormals,
617 double maxDeviationDegrees = 30.0)
618 {
619 _report.AppendLine("┌──────────────────────────────────────────────────────────────────────────────┐");
620 _report.AppendLine("│ NORMAL-PARALLEL MOVEMENT TEST │");
621 _report.AppendLine("└──────────────────────────────────────────────────────────────────────────────┘");
622 _report.AppendLine();
623
624 if (pathWithNormals == null || pathWithNormals.Count < 2)
625 {
626 _report.AppendLine(" ⚠️ WARNING: Insufficient path data for normal testing");
627 _testsWarnings++;
628 return;
629 }
630
631 _report.AppendLine($" Path Points with Normals: {pathWithNormals.Count}");
632 _report.AppendLine($" Max Allowed Deviation: {maxDeviationDegrees}°");
633 _report.AppendLine();
634
635 int movementsChecked = 0;
636 int movementsAligned = 0;
637 int movementsMisaligned = 0;
638 var misalignments = new List<(int index, double deviation)>();
639
640 for (int i = 0; i < pathWithNormals.Count - 1; i++)
641 {
642 var current = pathWithNormals[i];
643 var next = pathWithNormals[i + 1];
644
645 // Calculate movement direction
646 var movement = new Vector3D(
647 next.point.X - current.point.X,
648 next.point.Y - current.point.Y,
649 next.point.Z - current.point.Z);
650
651 if (movement.Length < 0.001) continue; // Skip stationary points
652
653 movement.Normalize();
654 movementsChecked++;
655
656 // Calculate angle between movement and normal
657 double dotProduct = Math.Abs(Vector3D.Dot(movement, current.normal));
658 double angleRad = Math.Acos(Math.Min(1.0, dotProduct));
659 double angleDeg = angleRad * 180.0 / Math.PI;
660
661 // Movement should be roughly parallel to normal (either same direction or opposite)
662 double deviation = Math.Min(angleDeg, 180.0 - angleDeg);
663
664 if (deviation <= maxDeviationDegrees)
665 {
666 movementsAligned++;
667 }
668 else
669 {
670 movementsMisaligned++;
671 misalignments.Add((i, deviation));
672 }
673 }
674
675 _report.AppendLine($" Movements Checked: {movementsChecked}");
676 _report.AppendLine($" Movements Aligned: {movementsAligned}");
677 _report.AppendLine($" Movements Misaligned: {movementsMisaligned}");
678 _report.AppendLine();
679
680 if (misalignments.Count > 0)
681 {
682 _report.AppendLine($" ⚠️ MISALIGNED MOVEMENTS (deviation > {maxDeviationDegrees}°):");
683 foreach (var (index, deviation) in misalignments.Take(10))
684 {
685 _report.AppendLine($" • Index {index}: {deviation:F1}° deviation");
686 }
687 _testsWarnings++;
688 }
689 else
690 {
691 _report.AppendLine($" ✅ TEST PASSED: All approach movements align with surface normals");
692 _testsPassed++;
693 }
694 _report.AppendLine();
695 }
