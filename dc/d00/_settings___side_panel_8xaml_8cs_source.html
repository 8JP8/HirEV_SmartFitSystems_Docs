388 public void ChangeLanguage()
389 {
390 string languagecombo_selectedcode = ConfigurationManager.AppSettings["Language"] ?? "EN";
391 switch (currLangIndex)
392 {
393 case 0:
394 languagecombo_selectedcode = "PT";
395 break;
396 case 1:
397 languagecombo_selectedcode = "EN";
398 break;
399 }
400
401 if (languagecombo_selectedcode != LanguageCode)
402 {
403 if (languagecombo_selectedcode != config.AppSettings.Settings["Language"].Value)
404 {
405 config.AppSettings.Settings["Language"].Value = languagecombo_selectedcode;
406
407 config.Save(ConfigurationSaveMode.Modified);
408 ConfigurationManager.RefreshSection("appSettings");
409
410 PromptRestart();
411 }

443 public void Login()
444 {
445 if (mvm != null)
446 {
447 if (!IsAdmin)
448 {
449 PasswordInputDialog password_dialog = new PasswordInputDialog(mvm);
450 password_dialog.ShowDialog();
451
452 if (password_dialog.DialogResult == true)
453 {
454 IsAdmin = true;
455 LoginBT_Text = GetXMLElementValue(DocLanguage, "LoggedAs")?.Replace("\\n", "\n") ?? "Logged as";
456 LoginBT_Text2 = GetXMLElementValue(DocLanguage, "Admin")?.Replace("\\n", "\n") ?? "Admin";

458 }
459 else
460 {
461 MessageBoxResult result = MessageBox.Show(
462 GetXMLElementValue(DocLanguage, "MessageBoxQuestion_Logout"),
463 GetXMLElementValue(DocLanguage, "MessageBox_AuthPrompt"),
464 MessageBoxButton.YesNo,
465 MessageBoxImage.Question
466 );
467 if (result == MessageBoxResult.Yes)
468 {
469 IsAdmin = false;
470 LoginBT_Text = GetXMLElementValue(DocLanguage, "Login")?.Replace("\\n", "\n") ?? "Login";
471 LoginBT_Text2 = "";

474 mvm?.IsAdminStateUpdate(IsAdmin: IsAdmin, Settings: this, ConvertToSecureString(Key));
475 ToggleADMInterface();
476 }
477 else MessageBox.Show(GetXMLElementValue(DocLanguage, "MessageBox_InternalError"), $"Error: {GetXMLElementValue(DocLanguage, "ErrorMessageBox_MainViewModelNotFound")}", MessageBoxButton.OK, MessageBoxImage.Error);
