50 public List<ApproachMeasurementPair> GenerateApproachPoints(
51 List<(Point3D point, Vector3D normal, CADFace face)> measurementPoints,
52 HashSet<int> excludedFaceIndices,
53 HashSet<int> flippedFaceIndices,
54 bool enableProjection = false,
55 System.Threading.CancellationToken cancellationToken = default)
56 {
57 var pairs = new List<ApproachMeasurementPair>();
58 int skippedExcluded = 0;
59
60 foreach (var (point, normal, face) in measurementPoints)
61 {
62 cancellationToken.ThrowIfCancellationRequested();
63 // EXCLUSION CHECK
64 if (excludedFaceIndices.Contains(face.FaceGlobalIndex))
65 {
66 skippedExcluded++;
67 continue;
68 }
69
70 // INVERSION CHECK
71 bool isInverted = flippedFaceIndices.Contains(face.FaceGlobalIndex);
72 double inversionMultiplier = isInverted ? -1.0 : 1.0;
73
74 var usedNormal = normal;
75 if (usedNormal.Length < 0.0001) usedNormal = Vector3D.AxisZ;
76 else usedNormal.Normalize();
77
78 Vector3D adjustedNormal = new Vector3D(
79 usedNormal.X * inversionMultiplier,
80 usedNormal.Y * inversionMultiplier,
81 usedNormal.Z * inversionMultiplier);
82
83 // 1. Calculate Theoretical Approach Point
84 var theoreticalPoint = new Point3D(
85 point.X + adjustedNormal.X * _offsetDistance,
86 point.Y + adjustedNormal.Y * _offsetDistance,
87 point.Z + adjustedNormal.Z * _offsetDistance);
88
89 Point3D finalApproachPoint = theoreticalPoint;
90
91 // 2. Projection and Validation DISABLED per user request
92
93 bool isHoleFace = face.FaceType == FaceTypeEnum.Cylindrical || face.FaceType == FaceTypeEnum.Conical;
94 var pair = new ApproachMeasurementPair(point, finalApproachPoint, adjustedNormal, face, isHoleFace, isInverted);
95 pairs.Add(pair);
96 }
97
98 string format = GetXMLElement(DocLanguage, "PathManager_ApproachPointsGeneratedCount", "✅ Generated {0} approach points (skipped {1} excluded faces).");
99 Debug.WriteLine(string.Format(format, pairs.Count, skippedExcluded));
100 return pairs;
101 }
