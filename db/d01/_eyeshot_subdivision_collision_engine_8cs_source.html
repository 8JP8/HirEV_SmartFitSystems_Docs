26 public void Initialize(
27 IReadOnlyList<(Entity Entity, Mesh Mesh)> partMeshes,
28 IReadOnlyList<(Point3D Min, Point3D Max)> partBoxes)
29 {
30 _parts.Clear();
31 if (partMeshes == null || partMeshes.Count == 0)
32 return;
33
34 for (int i = 0; i < partMeshes.Count; i++)
35 {
36 var (ent, mesh) = partMeshes[i];
37 var (min, max) = (i < partBoxes.Count) ? partBoxes[i] : (mesh.BoxMin, mesh.BoxMax);
38 string name = ent.EntityData as string ?? ent.LayerName ?? $"Part_{i}";
39 _parts.Add((ent, mesh, min, max, name));
40 }
41 }

43 public ProbeCollisionEvaluationResult EvaluateSample(
44 PathManager.ProbeCollisionSample sample,
45 PathManager.ProbeToolGeometry toolGeom,
46 double tolerance,
47 CancellationToken token)
48 {
49 var result = new ProbeCollisionEvaluationResult();
50 if (_parts.Count == 0 || toolGeom == null)
51 return result;
52
53 token.ThrowIfCancellationRequested();
54
55 var transform = toolGeom.IsKinematicCmm
56 ? PathManager.GetKinematicProbeTransformation(sample.Point, sample.Normal, toolGeom.Reach, toolGeom.HeadPivotOffset)
57 : PathManager.GetProbeTransformation(sample.Point, sample.Normal, sample.Up);
58
59 Transformation? headTransform = (toolGeom.IsKinematicCmm && toolGeom.HeadMesh != null)
60 ? PathManager.GetKinematicHeadTransformation(sample.Point, sample.Normal, toolGeom.Reach, toolGeom.HeadPivotOffset)
61 : null;
62
63 double boxMargin = Math.Max(0.0, tolerance) + 0.5;
64
65 Point3D toolArmWorldMin = Point3D.Origin, toolArmWorldMax = Point3D.Origin;
66 bool haveArmBox = false;
67 if (toolGeom.LocalBoxMin != null && toolGeom.LocalBoxMax != null)
68 {
69 haveArmBox = PathManager.TryGetTransformedBox(transform, toolGeom.LocalBoxMin, toolGeom.LocalBoxMax, boxMargin, out toolArmWorldMin, out toolArmWorldMax);
70 }
71
72 Point3D headWorldMin = Point3D.Origin, headWorldMax = Point3D.Origin;
73 bool haveHeadBox = false;
74 if (headTransform != null && toolGeom.HeadBoxMin != null && toolGeom.HeadBoxMax != null)
75 {
76 haveHeadBox = PathManager.TryGetTransformedBox(headTransform, toolGeom.HeadBoxMin, toolGeom.HeadBoxMax, boxMargin, out headWorldMin, out headWorldMax);
77 }
78
79 // Find candidate parts overlapping arm or head
80 var candidateParts = new List<(Entity Entity, Mesh Mesh, Point3D Min, Point3D Max, string Name)>();
81 for (int i = 0; i < _parts.Count; i++)
82 {
83 var p = _parts[i];
84 bool hitsArm = haveArmBox && PathManager.BoxesIntersect(toolArmWorldMin, toolArmWorldMax, p.Min, p.Max);
85 bool hitsHead = haveHeadBox && PathManager.BoxesIntersect(headWorldMin, headWorldMax, p.Min, p.Max);
86 if (!haveArmBox && !haveHeadBox) hitsArm = true;
87
88 if (hitsArm || hitsHead)
89 {
90 candidateParts.Add(p);
91 }
92 }
93
94 if (candidateParts.Count == 0)
95 return result;
96
97 bool hasExactCollision = false;
98
99 // Prepare transformed tool mesh
100 Mesh? transformedTool = null;
101 if (toolGeom.BaseMesh != null)
102 {
103 transformedTool = (Mesh)toolGeom.BaseMesh.Clone();
104 transformedTool.TransformBy(transform);
105 }
106
107 Mesh? transformedHead = null;
108 if (toolGeom.HeadMesh != null && headTransform != null)
109 {
110 transformedHead = (Mesh)toolGeom.HeadMesh.Clone();
111 transformedHead.TransformBy(headTransform);
112 }
113
114 // Check each candidate part strictly pairwise to prevent part-part and arm-head checks
115 foreach (var part in candidateParts)
116 {
117 token.ThrowIfCancellationRequested();
118
119 // 1. Tool Arm vs Part
120 if (transformedTool != null && haveArmBox && PathManager.BoxesIntersect(toolArmWorldMin, toolArmWorldMax, part.Min, part.Max))
121 {
122 bool armHit = false;
123 lock (part.Mesh)
124 {
125 var cdArm = new devDept.Eyeshot.CollisionDetection(
126 new Entity[] { part.Mesh, transformedTool },
127 _emptyBlocks,
128 true,
129 collisionCheckType.SubdivisionTree,
130 nodeMaxItemCount: 8);
131 cdArm.DoWork();
132
133 if (cdArm.Result != null && cdArm.Result.Length > 0)
134 {
135 armHit = cdArm.Result.Any(r =>
136 (r.CollidedEntities.Item1.Entity == transformedTool && r.CollidedEntities.Item2.Entity == part.Mesh) ||
137 (r.CollidedEntities.Item2.Entity == transformedTool && r.CollidedEntities.Item1.Entity == part.Mesh));
138 }
139 }
140
141 if (armHit)
142 {
143 hasExactCollision = true;
144 AddCollidedPart(result.CollidedParts, part.Name);
145 break;
146 }
147 }
148
149 // 2. CMM Head vs Part (Separately!)
150 if (transformedHead != null && haveHeadBox && PathManager.BoxesIntersect(headWorldMin, headWorldMax, part.Min, part.Max))
151 {
152 bool headHit = false;
153 lock (part.Mesh)
154 {
155 var cdHead = new devDept.Eyeshot.CollisionDetection(
156 new Entity[] { part.Mesh, transformedHead },
157 _emptyBlocks,
158 true,
159 collisionCheckType.SubdivisionTree,
160 nodeMaxItemCount: 8);
161 cdHead.DoWork();
162
163 if (cdHead.Result != null && cdHead.Result.Length > 0)
164 {
165 headHit = cdHead.Result.Any(r =>
166 (r.CollidedEntities.Item1.Entity == transformedHead && r.CollidedEntities.Item2.Entity == part.Mesh) ||
167 (r.CollidedEntities.Item2.Entity == transformedHead && r.CollidedEntities.Item1.Entity == part.Mesh));
168 }
169 }
170
171 if (headHit)
172 {
173 hasExactCollision = true;
174 AddCollidedPart(result.CollidedParts, part.Name);
175 break;
176 }
177 }
178 }
179
180 result.HasExactCollision = hasExactCollision;
181
182 // Pass 2: Tolerance Offset Check
183 bool hasToleranceCollision = false;
184 if (hasExactCollision && tolerance > 0.0)
185 {
186 Vector3D moveVec = sample.Normal * tolerance;
187
188 Mesh? tolArm = null;
189 if (transformedTool != null)
190 {
191 tolArm = (Mesh)transformedTool.Clone();
192 tolArm.Translate(moveVec);
193 }
194
195 Mesh? tolHead = null;
196 if (transformedHead != null)
197 {
198 tolHead = (Mesh)transformedHead.Clone();
199 tolHead.Translate(moveVec);
200 }
201
202 Point3D armTolMin = new Point3D(toolArmWorldMin.X + moveVec.X, toolArmWorldMin.Y + moveVec.Y, toolArmWorldMin.Z + moveVec.Z);
203 Point3D armTolMax = new Point3D(toolArmWorldMax.X + moveVec.X, toolArmWorldMax.Y + moveVec.Y, toolArmWorldMax.Z + moveVec.Z);
204 Point3D headTolMin = new Point3D(headWorldMin.X + moveVec.X, headWorldMin.Y + moveVec.Y, headWorldMin.Z + moveVec.Z);
205 Point3D headTolMax = new Point3D(headWorldMax.X + moveVec.X, headWorldMax.Y + moveVec.Y, headWorldMax.Z + moveVec.Z);
206
207 foreach (var part in candidateParts)
208 {
209 if (tolArm != null && haveArmBox && PathManager.BoxesIntersect(armTolMin, armTolMax, part.Min, part.Max))
210 {
211 lock (part.Mesh)
212 {
213 var cdTol = new devDept.Eyeshot.CollisionDetection(
214 new Entity[] { part.Mesh, tolArm },
215 _emptyBlocks,
216 true,
217 collisionCheckType.SubdivisionTree,
218 nodeMaxItemCount: 8);
219 cdTol.DoWork();
220
221 if (cdTol.Result != null && cdTol.Result.Length > 0)
222 {
223 hasToleranceCollision = cdTol.Result.Any(r =>
224 (r.CollidedEntities.Item1.Entity == tolArm && r.CollidedEntities.Item2.Entity == part.Mesh) ||
225 (r.CollidedEntities.Item2.Entity == tolArm && r.CollidedEntities.Item1.Entity == part.Mesh));
226 }
227 }
228 }
229 if (hasToleranceCollision) break;
230
231 if (tolHead != null && haveHeadBox && PathManager.BoxesIntersect(headTolMin, headTolMax, part.Min, part.Max))
232 {
233 lock (part.Mesh)
234 {
235 var cdTolHead = new devDept.Eyeshot.CollisionDetection(
236 new Entity[] { part.Mesh, tolHead },
237 _emptyBlocks,
238 true,
239 collisionCheckType.SubdivisionTree,
240 nodeMaxItemCount: 8);
241 cdTolHead.DoWork();
242
243 if (cdTolHead.Result != null && cdTolHead.Result.Length > 0)
244 {
245 hasToleranceCollision = cdTolHead.Result.Any(r =>
246 (r.CollidedEntities.Item1.Entity == tolHead && r.CollidedEntities.Item2.Entity == part.Mesh) ||
247 (r.CollidedEntities.Item2.Entity == tolHead && r.CollidedEntities.Item1.Entity == part.Mesh));
248 }
249 }
250 }
251 if (hasToleranceCollision) break;
252 }
253 }
254
255 result.HasToleranceCollision = hasToleranceCollision;
256 return result;
257 }
