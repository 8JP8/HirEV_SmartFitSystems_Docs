73 private void ShowErrorWithLogOption(string source, Exception ex)
74 {
75 string exceptionName = ex.GetType().Name;
76 string exceptionMessage = ex.Message;
77 string? lang = System.Configuration.ConfigurationManager.AppSettings["Language"] ?? "EN";
78
79 string title;
80 string message;
81
82 if (lang == "PT")
83 {
84 title = source == "App" ? "Exceção Não Tratada" : (source == "Task" ? "Exceção de Tarefa" : "Exceção Fatal");
85 message = $"Erro na Aplicação ({exceptionName}):\n{exceptionMessage}\n\nConsulte crash_log.txt para mais detalhes. Deseja abrir o log agora?";
86 }
87 else
88 {
89 title = source == "App" ? "Unhandled Exception" : (source == "Task" ? "Task Exception" : "Fatal Exception");
90 message = $"Application Error ({exceptionName}):\n{exceptionMessage}\n\nSee crash_log.txt for details. Would you like to open the log file now?";
91 }
92
93 if (MessageBox.Show(message, title, MessageBoxButton.OKCancel, MessageBoxImage.Error) == MessageBoxResult.OK)
94 {
95 try
96 {
97 string logPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "crash_log.txt");
98 if (File.Exists(logPath))
99 {
100 Process.Start(new ProcessStartInfo(logPath) { UseShellExecute = true });
101 }
102 }
103 catch (Exception openEx)
104 {
105 Debug.WriteLine($"Failed to open log file: {openEx.Message}");
106 }
107 }
