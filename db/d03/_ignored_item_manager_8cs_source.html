282 public void ReindexParentEntities(IReadOnlyDictionary<int, int> indexMap)
283 {
284 if (indexMap == null || indexMap.Count == 0) return;
285
286 bool changed = false;
287
288 var maps = new[] { _ignoredFaceDataMap, _flippedFaceDataMap, _unmeasuredFaceDataMap };
289
290 foreach (var map in maps)
291 {
292 if (map.Count == 0) continue;
293
294 var rekeyed = new Dictionary<(int, int), IgnoredFaceData>(map.Count);
295 foreach (var kvp in map)
296 {
297 var faceData = kvp.Value;
298 if (!indexMap.TryGetValue(faceData.ParentEntityIndex, out int newParentIndex))
299 {
300 // Index wasn't touched by this shift; keep as-is.
301 rekeyed[kvp.Key] = faceData;
302 continue;
303 }
304
305 if (newParentIndex < 0)
306 {
307 // Parent entity was removed; drop its overlay data.
308 changed = true;
309 continue;
310 }
311
312 if (newParentIndex != faceData.ParentEntityIndex)
313 {
314 changed = true;
315 }
316 faceData.ParentEntityIndex = newParentIndex;
317 rekeyed[(newParentIndex, kvp.Key.Item2)] = faceData;
318 }
319
320 map.Clear();
321 foreach (var kvp in rekeyed)
322 map[kvp.Key] = kvp.Value;
323 }
324
325 if (changed)
326 {
327 MarkOverlayDataChanged();
328 RebuildIgnoredOverlayBlock();
329 }

382 public void AddFlippedFaceOverlays(List<int> flippedFaceIndices)
383 {
384 // Clear previous flipped overlays
385 bool dataChanged = _flippedFaceDataMap.Count > 0;
386 _flippedFaceDataMap.Clear();
387
388 if (flippedFaceIndices == null || flippedFaceIndices.Count == 0)
389 {
390 Debug.WriteLine("[IgnoredItemManager] AddFlippedFaceOverlays: No flipped faces provided, clearing");
391 if (dataChanged)
392 {
393 MarkOverlayDataChanged();
394 RebuildIgnoredOverlayBlock();
395 }
396 return;
397 }
398
399 Debug.WriteLine($"[IgnoredItemManager] AddFlippedFaceOverlays: Processing {flippedFaceIndices.Count} flipped faces");
400
401 // Find the first BREP in the design
402 Brep? brep = null;
403 int entityIndex = -1;
404 for (int i = 0; i < _designControl.Entities.Count; i++)
405 {
406 if (_designControl.Entities[i] is Brep b)
407 {
408 brep = b;
409 entityIndex = i;
410 Debug.WriteLine($"[IgnoredItemManager] Found BREP at index {entityIndex}");
411 break;
412 }
413 }
414
415 if (brep == null)
416 {
417 Debug.WriteLine("[IgnoredItemManager] AddFlippedFaceOverlays: No BREP found");
418 if (dataChanged)
419 {
420 MarkOverlayDataChanged();
421 RebuildIgnoredOverlayBlock();
422 }
423 return;
424 }
425
426 foreach (int faceIndex in flippedFaceIndices)
427 {
428 var key = (entityIndex, faceIndex);
429
430 // Skip if already ignored
431 if (_ignoredFaceDataMap.ContainsKey(key))
432 {
433 Debug.WriteLine($"[IgnoredItemManager] Face {faceIndex} already ignored, skipping flipped overlay");
434 continue;
435 }
436
437 // Skip if unmeasured (unmeasured RED takes priority over flipped YELLOW)
438 if (_unmeasuredFaceDataMap.ContainsKey(key))
439 {
440 Debug.WriteLine($"[IgnoredItemManager] Face {faceIndex} is unmeasured, keeping RED overlay instead of flipped YELLOW");
441 continue;
442 }
443
444 Debug.WriteLine($"[IgnoredItemManager] Generating YELLOW overlay for face {faceIndex}");
445
446 // Generate YELLOW overlay
447 List<Entity> overlayEntities = GenerateOverlayForFaceWithColor(brep, faceIndex, Color.Yellow);
448 if (overlayEntities != null && overlayEntities.Count > 0)
449 {
450 _flippedFaceDataMap.Add(key, new IgnoredFaceData(entityIndex, faceIndex, overlayEntities));
451 dataChanged = true;
452 Debug.WriteLine($"[IgnoredItemManager] Added YELLOW overlay for face {faceIndex}: {overlayEntities.Count} entities");
453 }
454 else
455 {
456 Debug.WriteLine($"[IgnoredItemManager] Failed to generate overlay for face {faceIndex}");
457 }
458 }
459
460 Debug.WriteLine($"[IgnoredItemManager] Total flipped overlays: {_flippedFaceDataMap.Count}");
461 if (dataChanged)
462 {
463 MarkOverlayDataChanged();
464 RebuildIgnoredOverlayBlock();
465 }

690 private List<Entity> GenerateOverlayForFace(Brep brep, int faceIndex)
691 {
692 if (brep == null || faceIndex < 0 || faceIndex >= brep.Faces.Length) return new List<Entity>();
693
694 switch (_currentStyle)
695 {
696 case IgnoredFaceStyle.Transparency:
697 var mesh = GenerateTransparencyMeshForFace(brep, faceIndex);
698 return mesh != null ? new List<Entity> { mesh } : new List<Entity>();
699
700 case IgnoredFaceStyle.StripeTexture:
701 case IgnoredFaceStyle.SquareTexture:
702 case IgnoredFaceStyle.GrayMesh:
703 var texMesh = GenerateTexturedMeshForFace(brep, faceIndex, GetMaterialNameForStyle());
704 return texMesh != null ? new List<Entity> { texMesh } : new List<Entity>();
705
706 case IgnoredFaceStyle.CylinderMesh:
707 return (GenerateDiagonalLinesForFace(brep, faceIndex) ?? new List<Mesh>())
708 .Where(c => c != null && c.Vertices != null && c.Vertices.Length > 0 && c.Triangles != null && c.Triangles.Length > 0)
709 .Cast<Entity>().ToList();
710
711 case IgnoredFaceStyle.CubeMesh:
712 return (GenerateCubesForFace(brep, faceIndex) ?? new List<Solid>())
713 .Where(c => c != null)
714 .Cast<Entity>().ToList();
715
716 default:
717 return new List<Entity>();
718 }
719 }

739 private List<Entity> GenerateOverlayForFaceWithColor(Brep brep, int faceIndex, Color color)
740 {
741 if (brep == null || faceIndex < 0 || faceIndex >= brep.Faces.Length) return new List<Entity>();
742
743 switch (_currentStyle)
744 {
745 case IgnoredFaceStyle.Transparency:
746 var mesh = GenerateTransparencyMeshForFace(brep, faceIndex);
747 if (mesh != null)
748 {
749 mesh.Color = color;
750 return new List<Entity> { mesh };
751 }
752 return new List<Entity>();
753
754 case IgnoredFaceStyle.StripeTexture:
755 case IgnoredFaceStyle.SquareTexture:
756 case IgnoredFaceStyle.GrayMesh:
757 var texMesh = GenerateTexturedMeshForFace(brep, faceIndex, GetMaterialNameForColor(color));
758 return texMesh != null ? new List<Entity> { texMesh } : new List<Entity>();
759
760 case IgnoredFaceStyle.CylinderMesh:
761 var cylinders = GenerateDiagonalLinesForFace(brep, faceIndex);
762 if (cylinders != null)
763 {
764 var validCyls = new List<Entity>();
765 foreach (var cyl in cylinders)
766 {
767 if (cyl != null && cyl.Vertices != null && cyl.Vertices.Length > 0 && cyl.Triangles != null && cyl.Triangles.Length > 0)
768 {
769 cyl.Color = color;
770 validCyls.Add(cyl);
771 }
772 }
773 return validCyls;
774 }
775 return new List<Entity>();
776
777 case IgnoredFaceStyle.CubeMesh:
778 var cubes = GenerateCubesForFace(brep, faceIndex);
779 if (cubes != null)
780 {
781 var validCubes = new List<Entity>();
782 foreach (var cube in cubes)
783 {
784 if (cube != null)
785 {
786 cube.Color = color;
787 validCubes.Add(cube);
788 }
789 }
790 return validCubes;
791 }
792 return new List<Entity>();
793
794 default:
795 return new List<Entity>();
796 }

1502 private void GenerateLineSegmentsOnSurface(Mesh faceMesh, Brep.Face face, Point3D lineStart, Point3D lineEnd,

1504 {

1506 Point3D lastValidPoint = null!;
1507 for (int i = 0; i <= samples; i++)
1508 {
1509 double t = (double)i / samples;
1510 Point3D samplePoint = lineStart + (lineEnd - lineStart) * t;
1511 var projectionRay = new Segment3D(samplePoint - projectionAxis * projectionLength * 2, samplePoint + projectionAxis * projectionLength * 2);
1512 bool intersectionFound = false;
1513 foreach (var triangle in faceMesh.Triangles)
1514 {
1515 Point3D v1 = faceMesh.Vertices[triangle.V1];
1516 Point3D v2 = faceMesh.Vertices[triangle.V2];
1517 Point3D v3 = faceMesh.Vertices[triangle.V3];
1518 if (projectionRay.IntersectWith(v1, v2, v3, true, out Point3D intersectionPoint, out _, out _))
1519 {
1520 Vector3D normal = face.Normal(intersectionPoint);
1521 Point3D currentValidPoint = intersectionPoint + normal * _lineOffset;
1522 if (lastValidPoint != null)
1523 {
1524 double segmentLength = new Vector3D(lastValidPoint, currentValidPoint).Length;
1525 if (segmentLength > 1e-6)
1526 {

1528 cylinder.LightWeight = true;
1529 Vector3D direction = new Vector3D(lastValidPoint, currentValidPoint);
1530 direction.Normalize();
1531 Vector3D zAxis = Vector3D.AxisZ;
1532 Vector3D rotAxis = Vector3D.Cross(zAxis, direction);
1533 if (rotAxis.Length > 1e-6)
1534 {
1535 double angle = Math.Acos(Math.Max(-1, Math.Min(1, Vector3D.Dot(zAxis, direction))));
1536 cylinder.Rotate(angle, rotAxis, Point3D.Origin);
1537 }
1538 else if (Vector3D.Dot(zAxis, direction) < 0)
1539 {
1540 cylinder.Rotate(Math.PI, Vector3D.AxisX, Point3D.Origin);
1541 }
1542 cylinder.Translate(lastValidPoint.X, lastValidPoint.Y, lastValidPoint.Z);
