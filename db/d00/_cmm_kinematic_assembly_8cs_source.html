77 public static CmmKinematicState ComputeKinematics(
78 Point3D targetTipCenter,
79 double angleADeg,
80 double angleBDeg,
81 CmmToolMontage montage)
82 {
83 double aRad = angleADeg * Math.PI / 180.0;
84 double bRad = angleBDeg * Math.PI / 180.0;
85
86 // Probe axis direction in world (from A-pivot toward tip):
87 // neutral (A=0, B=0): (0, 0, -1) probe straight down
88 // A tilt first (around a horizontal axis perpendicular to B direction):
89 // then B rotation around vertical Z
90 // Result: probeDir = (sin(A)·cos(B), sin(A)·sin(B), –cos(A))
91 double ux = Math.Sin(aRad) * Math.Cos(bRad);
92 double uy = Math.Sin(aRad) * Math.Sin(bRad);
93 double uz = -Math.Cos(aRad);
94 var probeVec = new Vector3D(ux, uy, uz);
95 if (probeVec.Length > 1e-9) probeVec.Normalize();
96
97 double reach = Math.Max(1.0, montage.TotalReachLength);
98 double headH = montage.Head.TotalHeadHeight;
99 double pivAOff = montage.Head.PivotOffsetA; // negative (below head face), e.g. –25 mm
100
101 // A-pivot position (tip – reach · probeDir)
102 var aPivotPos = new Point3D(
103 targetTipCenter.X - reach * probeVec.X,
104 targetTipCenter.Y - reach * probeVec.Y,
105 targetTipCenter.Z - reach * probeVec.Z);
106
107 // Quill face position: directly above A-pivot by |PivotAOffsetZ|
108 // (head is vertical so quill is directly above the B-pivot, which is above A-pivot)
109 double pivAZ = Math.Abs(pivAOff); // distance from head quill face to A-pivot
110 var quillPos = new Point3D(aPivotPos.X, aPivotPos.Y, aPivotPos.Z + pivAZ);
111
112 // Head base transform: translation only — head stays vertical
113 var headTrans = Transformation.CreateTranslation(quillPos.X, quillPos.Y, quillPos.Z);
114
115 // Probe transform: origin at quill face, rotated B then A around their pivots
116 // Step 1: translate to quill face
117 var transToQuill = Transformation.CreateTranslation(quillPos.X, quillPos.Y, quillPos.Z);
118 // Step 2: rotate B around Z through B-pivot (at quillZ + PivotBOffsetZ)
119 // Step 3: rotate A around the tilt axis through A-pivot
120 // For the entity positioning we use a combined world-space rotation around the A-pivot:
121 var rotB = Transformation.CreateRotation(bRad, Vector3D.AxisZ);
122 // A-tilt axis is perpendicular to B-direction in the XY plane
123 var tiltAxis = new Vector3D(-Math.Sin(bRad), Math.Cos(bRad), 0);
124 if (tiltAxis.Length < 1e-9) tiltAxis = new Vector3D(1, 0, 0);
125 tiltAxis.Normalize();
126 var rotA = Transformation.CreateRotation(aRad, tiltAxis);
127 var transToAPivot = Transformation.CreateTranslation(aPivotPos.X, aPivotPos.Y, aPivotPos.Z);
128 // Combined: first rotate (B then A) around the A-pivot, then translate to its world position
129 var probeTrans = transToAPivot * rotA * rotB;
130
131 return new CmmKinematicState
132 {
133 QuillPosition = quillPos,
134 KnucklePosition = aPivotPos,
135 TipCenterPosition = targetTipCenter,
136 ProbeDirectionVector = probeVec,
137 AngleA = angleADeg,
138 AngleB = angleBDeg,
139 HeadBaseTransform = headTrans,
140 ProbeTransform = probeTrans
141 };
142 }

159 public static Block BuildKinematicBlock(
160 string blockName,
161 CmmToolMontage montage,
162 CmmKinematicState? kinState = null,
163 List<Entity>? headEntities = null,
164 List<Entity>? tipEntities = null,
165 List<Entity>? stylusEntities = null)
166 {
167 var block = new Block(blockName);
168
169 double aRad = (kinState?.AngleA ?? 0.0) * Math.PI / 180.0;
170 double bRad = (kinState?.AngleB ?? 0.0) * Math.PI / 180.0;
171
172 double pivAOff = montage.Head.PivotOffsetA; // negative, e.g. –25 mm
173 double reach = Math.Max(1.0, montage.TotalReachLength);
174 double ballR = montage.Stylus.BallDiameter > 0.01 ? montage.Stylus.BallDiameter / 2.0 : 1.0;
175 double headH = montage.Head.TotalHeadHeight;
176
177 // ── 1. HEAD BODY — always vertical, rendered above quill face at Z = 0 ──
178 if (headEntities != null && headEntities.Count > 0)
179 {
180 foreach (var e in headEntities)
181 {
182 var clone = (Entity)e.Clone();
183 // Head already in correct orientation from loader (bottom face at Z=0, pointing up)
184 clone.LayerName = "Default";
185 block.Entities.Add(clone);
186 }
187 }
188 else
189 {
190 // Synthesised head if no CAD provided
191 var synthHead = SynthesisHeadMesh(montage);
192 block.Entities.Add(synthHead);
193 }
194
195 // ── 2. PROBE + STYLUS CHAIN — rotated by B then A about the A-pivot ──
196 // In local space, A-pivot is at (0, 0, pivAOff) from quill face
197 // Probe direction after A,B: (sin(A)cos(B), sin(A)sin(B), –cos(A))
198 double ux = Math.Sin(aRad) * Math.Cos(bRad);
199 double uy = Math.Sin(aRad) * Math.Sin(bRad);
200 double uz = -Math.Cos(aRad);
201
202 // Ruby ball centre in local block space (from quill face origin):
203 double tipLocalZ = pivAOff - reach * Math.Cos(aRad);
204 double tipLocalX = reach * ux;
205 double tipLocalY = reach * uy;
206
207 // Tilt axis for Axis-A: perpendicular to B direction in XY plane
208 var tiltAxis = new Vector3D(-Math.Sin(bRad), Math.Cos(bRad), 0.0);
209 if (tiltAxis.Length < 1e-9) tiltAxis = Vector3D.AxisX;
210 tiltAxis.Normalize();
211
212 if (tipEntities != null && tipEntities.Count > 0)
213 {
214 foreach (var e in tipEntities)
215 {
216 var clone = (Entity)e.Clone();
217 // Probe tip CAD: modelled with top at Z=0, tip pointing down –Z.
218 // Apply B-pan around Z, then A-tilt around the lateral tilt axis.
219 clone.Rotate(bRad, Vector3D.AxisZ);
220 clone.Rotate(aRad, tiltAxis);
221 clone.Translate(0, 0, pivAOff); // shift to A-pivot in local block frame
222 clone.LayerName = "Default";
223 block.Entities.Add(clone);
224 }
225 }
226 else
227 {
228 var synthProbe = SynthesisProbeMesh(montage, aRad, bRad, pivAOff);
229 block.Entities.Add(synthProbe);
230 }
231
232 if (stylusEntities != null && stylusEntities.Count > 0)
233 {
234 double probeBodyL = montage.Probe.BodyLength > 0.1 ? montage.Probe.BodyLength : 38.0;
235 // Bottom of probe body from A-pivot along probe direction
236 foreach (var e in stylusEntities)
237 {
238 var clone = (Entity)e.Clone();
239 // Stylus is in canonical –Z orientation. Apply same B then A rotation.
240 clone.Rotate(bRad, Vector3D.AxisZ);
241 clone.Rotate(aRad, tiltAxis);
242 // Translate to probe body bottom: A-pivot offset + probeBodyL along probe dir
243 clone.Translate(
244 ux * probeBodyL,
245 uy * probeBodyL,
246 pivAOff + uz * probeBodyL);
247 clone.LayerName = "Default";
248 block.Entities.Add(clone);
249 }
250 }
251
252 // ── 3. RUBY BALL — at exact tip contact point ──
253 var ruby = Mesh.CreateSphere(ballR, 24, 24);
254 ruby.Translate(tipLocalX, tipLocalY, tipLocalZ);
255 ruby.Color = Color.FromArgb(225, 20, 50); // Gemstone ruby red
256 ruby.ColorMethod = colorMethodType.byEntity;
257 ruby.LayerName = "Default";
258 block.Entities.Add(ruby);
259
260 return block;
261 }

269 public static Mesh SynthesisHeadMesh(CmmToolMontage montage)
270 {
271 var parts = new List<Mesh>();
272
273 double hw = montage.Head.BodyWidth > 0.1 ? montage.Head.BodyWidth : 62.0;
274 double hd = montage.Head.BodyDepth > 0.1 ? montage.Head.BodyDepth : 62.0;
275 double hh = montage.Head.BodyHeight > 0.1 ? montage.Head.BodyHeight : 60.0;
276 double kr = montage.Head.KnuckleRadius > 0.1 ? montage.Head.KnuckleRadius : 20.0;
277 double kh = montage.Head.KnuckleHeight > 0.1 ? montage.Head.KnuckleHeight : 42.0;
278
279 // Upper housing (box), bottom face at Z=0 pointing up
280 Mesh headBox = Mesh.CreateBox(hw, hd, hh);
281 headBox.Translate(-hw / 2.0, -hd / 2.0, 0.0);
282 headBox.Color = Color.FromArgb(42, 44, 48);
283 headBox.ColorMethod = colorMethodType.byEntity;
284 parts.Add(headBox);
285
286 // Knuckle cylinder hanging below head bottom face (at Z=0, going to Z=–kh)
287 Mesh knuckle = Mesh.CreateCylinder(kr, kh, 24);
288 knuckle.Rotate(Math.PI, Vector3D.AxisX); // flip to point down
289 knuckle.Color = Color.FromArgb(70, 72, 78);
290 knuckle.ColorMethod = colorMethodType.byEntity;
291 parts.Add(knuckle);
292
293 Mesh composite = parts[0];
294 for (int i = 1; i < parts.Count; i++)
295 composite.MergeWith(parts[i]);
296 return composite;
297 }

303 public static Mesh SynthesisProbeMesh(CmmToolMontage montage,
304 double aRad, double bRad, double pivAOff)
305 {
306 var parts = new List<Mesh>();
307
308 double prR = montage.Probe.Diameter > 0.1 ? montage.Probe.Diameter / 2.0 : 6.6;
309 double prL = montage.Probe.BodyLength > 0.1 ? montage.Probe.BodyLength : 38.0;
310 double stylusL = montage.Stylus.OverallLength > 0.1 ? montage.Stylus.OverallLength : 20.0;
311 double ballR = montage.Stylus.BallDiameter > 0.1 ? montage.Stylus.BallDiameter / 2.0 : 1.0;
312 double stemR = montage.Stylus.StemDiameter > 0.1 ? montage.Stylus.StemDiameter / 2.0 : 0.6;
313
314 // In canonical pose: probe top at Z=0, tip at Z=–prL
315 double currentZ = 0.0;
316
317 // M8 boss
318 Mesh m8 = Mesh.CreateCylinder(4.75, 4.0, 24);
319 m8.Rotate(Math.PI, Vector3D.AxisX);
320 m8.Translate(0, 0, currentZ);
321 m8.Color = Color.FromArgb(80, 82, 88);
322 m8.ColorMethod = colorMethodType.byEntity;
323 parts.Add(m8);
324
325 // Main body
326 Mesh body = Mesh.CreateCylinder(prR, prL - 12.0, 32);
327 body.Rotate(Math.PI, Vector3D.AxisX);
328 body.Translate(0, 0, currentZ);
329 body.Color = Color.FromArgb(48, 50, 56);
330 body.ColorMethod = colorMethodType.byEntity;
331 parts.Add(body);
332 currentZ -= (prL - 12.0);
333
334 // Lower cone taper
335 Mesh cone = Mesh.CreateCone(prR, 3.5, 9.0, 32);
336 cone.Rotate(Math.PI, Vector3D.AxisX);
337 cone.Translate(0, 0, currentZ);
338 cone.Color = Color.FromArgb(42, 44, 50);
339 cone.ColorMethod = colorMethodType.byEntity;
340 parts.Add(cone);
341 currentZ -= 9.0;
342
343 // Nose chamfer
344 Mesh nose = Mesh.CreateCone(3.5, 2.5, 3.0, 32);
345 nose.Rotate(Math.PI, Vector3D.AxisX);
346 nose.Translate(0, 0, currentZ);
347 nose.Color = Color.FromArgb(36, 38, 42);
348 nose.ColorMethod = colorMethodType.byEntity;
349 parts.Add(nose);
350 currentZ -= 3.0;
351
352 // Stylus hub
353 Mesh hub = Mesh.CreateCylinder(1.5, 3.0, 24);
354 hub.Rotate(Math.PI, Vector3D.AxisX);
355 hub.Translate(0, 0, currentZ);
356 hub.Color = Color.FromArgb(200, 202, 208);
357 hub.ColorMethod = colorMethodType.byEntity;
358 parts.Add(hub);
359 currentZ -= 3.0;
360
361 // Stem
362 double stemL = Math.Max(2.0, stylusL - 4.5 - ballR);
363 Mesh stem = Mesh.CreateCylinder(stemR, stemL, 20);
364 stem.Rotate(Math.PI, Vector3D.AxisX);
365 stem.Translate(0, 0, currentZ);
366 stem.Color = Color.FromArgb(215, 218, 222);
367 stem.ColorMethod = colorMethodType.byEntity;
368 parts.Add(stem);
369 currentZ -= stemL;
370
371 // Ruby ball
372 Mesh ruby = Mesh.CreateSphere(ballR, 24, 24);
373 ruby.Translate(0, 0, currentZ - ballR);
374 ruby.Color = Color.FromArgb(225, 20, 50);
375 ruby.ColorMethod = colorMethodType.byEntity;
376 parts.Add(ruby);
377
378 // Merge
379 Mesh composite = parts[0];
380 for (int i = 1; i < parts.Count; i++)
381 composite.MergeWith(parts[i]);
382
383 // Apply kinematic rotation around A-pivot using Eyeshot's Rotate API:
384 // B-pan first (around world Z), then A-tilt (around the lateral tilt axis in B-frame)
385 var tiltAxis = new Vector3D(-Math.Sin(bRad), Math.Cos(bRad), 0.0);
386 if (tiltAxis.Length < 1e-9) tiltAxis = Vector3D.AxisX;
387 tiltAxis.Normalize();
388
389 composite.Rotate(bRad, Vector3D.AxisZ);
390 composite.Rotate(aRad, tiltAxis);
391 composite.Translate(0, 0, pivAOff); // shift to A-pivot in local block frame
392
393 return composite;
394 }

402 var headMesh = SynthesisHeadMesh(montage);
