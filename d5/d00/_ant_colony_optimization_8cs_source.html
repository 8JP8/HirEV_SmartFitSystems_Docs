107 public List<int> OptimizePathACO(
108 List<Point3D> points,
109 int numAnts = 10,
110 int numIterations = 50,
111 double alpha = 1.0,
112 double beta = 2.0,
113 double rho = 0.1)
114 {
115 _numCities = points.Count;
116
117 // Validations
118 if (_numCities < 2)
119 {
120 Debug.WriteLine("⚠️ ACO: Less than 2 points, no optimization needed");
121 return Enumerable.Range(0, _numCities).ToList();
122 }
123
124 if (_numCities > 500)
125 {
126 Debug.WriteLine($"⚠️ ACO: {_numCities} points is large, may be slow. Consider reducing numIterations.");
127 }
128
129 // Configure parameters
130 _alpha = alpha;
131 _beta = beta;
132 _rho = rho;
133
134 // Adjust ant count if needed
135 int effectiveAnts = Math.Min(numAnts, Math.Max(5, _numCities / 5));
136
137 Debug.WriteLine($"\n🐜 ACO: Starting optimization");
138 Debug.WriteLine($" Points: {_numCities}, Ants: {effectiveAnts}, Iterations: {numIterations}");
139 Debug.WriteLine($" Parameters: α={_alpha}, β={_beta}, ρ={_rho}");
140
141 // ===== PHASE 1: Build distance matrix =====
142 BuildDistanceMatrix(points);
143
144 // ===== PHASE 2: Initialize pheromone matrix =====
145 InitializePheromoneMatrix();
146
147 // ===== PHASE 3: Main ACO loop =====
148 AntPath? globalBestPath = null;
149 double globalBestDistance = double.MaxValue;
150 int iterationsSinceImprovement = 0;
151
152 Stopwatch sw = Stopwatch.StartNew();
153
154 for (int iteration = 0; iteration < numIterations; iteration++)
155 {
156 List<AntPath> iterationPaths = new List<AntPath>();
157
158 // Each ant constructs a solution
159 for (int ant = 0; ant < effectiveAnts; ant++)
160 {
161 int startCity = ant % _numCities; // Distribute starting points
162 AntPath path = ConstructAntSolution(startCity);
163 iterationPaths.Add(path);
164
165 // Update global best
166 if (path.TotalDistance < globalBestDistance)
167 {
168 globalBestDistance = path.TotalDistance;
169 globalBestPath = path.Clone();
170 iterationsSinceImprovement = 0;
171
172 Debug.WriteLine($" 🏆 Iteration {iteration + 1}: New best = {globalBestDistance:F2}mm");
173 }
174 }
175
176 // ===== PHASE 4: Update pheromone =====
177 UpdatePheromone(iterationPaths, globalBestPath);
178
179 iterationsSinceImprovement++;
180
181 // Early stopping if no improvement for too long
182 if (iterationsSinceImprovement > numIterations / 3)
183 {
184 Debug.WriteLine($" ⏩ Early stopping at iteration {iteration + 1} (no improvement)");
185 break;
186 }
187
188 // Periodic logging
189 if ((iteration + 1) % 10 == 0)
190 {
191 double avgDist = iterationPaths.Average(p => p.TotalDistance);
192 Debug.WriteLine($" 📊 Iteration {iteration + 1}: Avg={avgDist:F2}mm, Best={globalBestDistance:F2}mm");
193 }
194 }
195
196 sw.Stop();
197
198 // Compute improvement vs original order
199 double originalDistance = CalculateSequentialDistance(points);
200 double improvement = ((originalDistance - globalBestDistance) / originalDistance) * 100.0;
201
202 Debug.WriteLine($"\n ✅ ACO Completed in {sw.ElapsedMilliseconds}ms");
203 Debug.WriteLine($" Original: {originalDistance:F2}mm → Optimized: {globalBestDistance:F2}mm");
204 Debug.WriteLine($" Improvement: {improvement:F1}% reduction\n");
205
206 return globalBestPath?.CityOrder ?? Enumerable.Range(0, _numCities).ToList();
207 }

212 public double CalculatePathDistance(List<int> cityOrder)
213 {
214 if (cityOrder.Count < 2 || _distanceMatrix == null)
215 return 0;
216
217 double totalDistance = 0;
218 for (int i = 0; i < cityOrder.Count - 1; i++)
219 {
220 int from = cityOrder[i];
221 int to = cityOrder[i + 1];
222 totalDistance += _distanceMatrix[from, to];
223 }
224
225 return totalDistance;
226 }

235 private void BuildDistanceMatrix(List<Point3D> points)
236 {
237 _distanceMatrix = new double[_numCities, _numCities];
238
239 for (int i = 0; i < _numCities; i++)
240 {
241 for (int j = 0; j < _numCities; j++)
242 {
243 if (i == j)
244 {
245 _distanceMatrix[i, j] = double.MaxValue; // Avoid self-loops
246 }
247 else
248 {
249 double dx = points[i].X - points[j].X;
250 double dy = points[i].Y - points[j].Y;
251 double dz = points[i].Z - points[j].Z;
252 _distanceMatrix[i, j] = Math.Sqrt(dx * dx + dy * dy + dz * dz);
253 }
254 }
255 }

277 private AntPath ConstructAntSolution(int startCity)
278 {
279 var distanceMatrix = _distanceMatrix ?? throw new InvalidOperationException("Distance matrix is not initialized.");
280
281 var path = new AntPath();
282 var unvisited = new HashSet<int>(Enumerable.Range(0, _numCities));
283
284 int currentCity = startCity;
285 path.CityOrder.Add(currentCity);
286 unvisited.Remove(currentCity);
287
288 // Build path by visiting all cities
289 while (unvisited.Count > 0)
290 {
291 int nextCity = SelectNextCity(currentCity, unvisited);
292 path.CityOrder.Add(nextCity);
293 path.TotalDistance += distanceMatrix[currentCity, nextCity];
294 unvisited.Remove(nextCity);
295 currentCity = nextCity;
296 }
297
298 return path;
299 }

306 private int SelectNextCity(int currentCity, HashSet<int> unvisited)
307 {
308 var pheromoneMatrix = _pheromoneMatrix ?? throw new InvalidOperationException("Pheromone matrix is not initialized.");
309 var distanceMatrix = _distanceMatrix ?? throw new InvalidOperationException("Distance matrix is not initialized.");
310
311 double[] probabilities = new double[_numCities];
312 double sumProbability = 0.0;
313
314 // Compute probabilities for all unvisited cities
315 foreach (int city in unvisited)
316 {
317 double tau = pheromoneMatrix[currentCity, city]; // τ[i,j]
318 double distance = distanceMatrix[currentCity, city];
319 double eta = 1.0 / (distance + 1e-10); // η[i,j] = 1/d[i,j]
320
321 double probability = Math.Pow(tau, _alpha) * Math.Pow(eta, _beta);
322 probabilities[city] = probability;
323 sumProbability += probability;
324 }
325
326 // Fallback: if zero, choose nearest
327 if (sumProbability < 1e-10)
328 {
329 return unvisited.OrderBy(c => distanceMatrix[currentCity, c]).First();
330 }
331
332 // Roulette wheel selection
333 double random = _random.NextDouble() * sumProbability;
334 double cumulative = 0.0;
335
336 foreach (int city in unvisited)
337 {
338 cumulative += probabilities[city];
339 if (random <= cumulative)
340 {
341 return city;
342 }
343 }
344
345 // Fallback: return last unvisited
346 return unvisited.Last();
347 }

354 private void UpdatePheromone(List<AntPath> iterationPaths, AntPath? globalBest)
355 {
356 var pheromoneMatrix = _pheromoneMatrix ?? throw new InvalidOperationException("Pheromone matrix is not initialized.");
357
358 // ===== Global evaporation =====
359 for (int i = 0; i < _numCities; i++)
360 {
361 for (int j = 0; j < _numCities; j++)
362 {
363 pheromoneMatrix[i, j] *= (1.0 - _rho);
364 }
365 }
366
367 // ===== Pheromone deposit - all ants in this iteration =====
368 foreach (var path in iterationPaths)
369 {
370 double deltaPhero = _q / path.TotalDistance;
371
372 for (int i = 0; i < path.CityOrder.Count - 1; i++)
373 {
374 int from = path.CityOrder[i];
375 int to = path.CityOrder[i + 1];
376
377 pheromoneMatrix[from, to] += deltaPhero;
378 pheromoneMatrix[to, from] += deltaPhero; // Symmetric
379 }
380 }
381
382 // ===== Elitism: extra reinforcement for global best =====
383 if (globalBest != null)
384 {
385 double eliteBonus = _q / globalBest.TotalDistance * 2.0; // 2× boost
386
387 for (int i = 0; i < globalBest.CityOrder.Count - 1; i++)
388 {
389 int from = globalBest.CityOrder[i];
390 int to = globalBest.CityOrder[i + 1];
391
392 pheromoneMatrix[from, to] += eliteBonus;
393 pheromoneMatrix[to, from] += eliteBonus;
394 }
395 }
396
397 // ===== Limit maximum pheromone (avoid stagnation) =====
398 double maxPheromone = 10.0;
399 for (int i = 0; i < _numCities; i++)
400 {
401 for (int j = 0; j < _numCities; j++)
402 {
403 pheromoneMatrix[i, j] = Math.Min(pheromoneMatrix[i, j], maxPheromone);
404 }
405 }
406 }
