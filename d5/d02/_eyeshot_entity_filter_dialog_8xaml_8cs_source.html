369 private void BackupAndHidePathEntities()
370 {
371 if (_pathEntitiesBackedUp) return;
372
373 var design = _parent.MainParent.designControl;
374
375 // Find all path entities and base-avoidance visuals.
376 _pathEntityBackup = design.Entities
377 .Where(e => (e.LayerName != null && e.LayerName.StartsWith("PATH_")) ||
378 MainViewModel.IsBaseAvoidanceVisualEntity(e))
379 .ToList();
380
381 // Remove from design temporarily
382 foreach (var entity in _pathEntityBackup)
383 {
384 design.Entities.Remove(entity);
385 }
386
387 if (_pathEntityBackup.Any())
388 {
389 design.Invalidate();
390 System.Diagnostics.Debug.WriteLine($"Backed up {_pathEntityBackup.Count} transient visual entities");
391 }
392
393 _pathEntitiesBackedUp = true;
394 }

399 private void RestorePathEntities()
400 {
401 if (!_pathEntitiesBackedUp) return;
402
403 var design = _parent.MainParent.designControl;
404
405 // Restore entities
406 foreach (var entity in _pathEntityBackup)
407 {
408 if (!design.Entities.Contains(entity))
409 {
410 design.Entities.Add(entity);
411 }
412 }
413
414 design.Entities.Regen();
415 design.Invalidate();
416
417 System.Diagnostics.Debug.WriteLine($"Restored {_pathEntityBackup.Count} transient visual entities");
418 _pathEntityBackup.Clear();
419 _pathEntitiesBackedUp = false;
420 }

427 private void UpdateTimer_Tick(object? sender, EventArgs e)
428 {
429 var design = _parent.MainParent.designControl;
430 int currentCount = design.Entities.Count;
431
432 // Check if entity count changed significantly (model added/removed)
433 // Ignore small changes from path entities
434 int delta = Math.Abs(currentCount - _lastEntityCount);
435
436 if (delta > 10) // Significant change indicates model modification
437 {
438 // Clear path backup if model changed - paths need to be regenerated
439 if (_pathEntitiesBackedUp)
440 {
441 _pathEntityBackup.Clear();
442 _pathEntitiesBackedUp = false;
443 System.Diagnostics.Debug.WriteLine("Model changed - cleared path backup (paths need regeneration)");
444 }
445
446 // Store initial state for new entities only
447 for (int i = 0; i < currentCount; i++)
448 {
449 var entity = design.Entities[i];
450 if (entity != null &&
451 (entity.EntityData as string) != "IgnoredOverlay_Ref" &&
452 !MainViewModel.IsBaseAvoidanceVisualEntity(entity) &&
453 !_initialVisibilityStates.ContainsKey(entity))
454 {
455 _initialVisibilityStates[entity] = entity.Visible;
456 }
457 }
458
459 RefreshEntitiesFromDesign();
460 _lastEntityCount = currentCount;
461 }
462 }

567 private void RefreshEntitiesFromDesign()
568 {
569 Application.Current.Dispatcher.Invoke(() =>
570 {
571 Entities.Clear();
572
573 var design = _parent.MainParent.designControl;
574
575 int entityCounter = 1;
576
577 // Start loop from 0
578 for (int i = 0; i < design.Entities.Count; i++)
579 {
580 var ent = design.Entities[i];
581
582 if (ent.EntityData as string == "IgnoredOverlay_Ref") continue;
583 if (MainViewModel.IsBaseAvoidanceVisualEntity(ent)) continue;
584
585 if (_parent.IsControlPointEntity(ent))
586 {
587 continue;
588 }
589
590 // Skip shell entities (debug/visualization only)
591 if (ent.LayerName?.StartsWith("SHELL_", StringComparison.OrdinalIgnoreCase) == true)
592 {
593 continue;
594 }
595
596 // Skip origin symbols
597 if (ent.EntityData as string == "PART_ORIGIN")
598 {
599 continue;
600 }
601
602 // Strict filter: only allow actual Brep or Mesh CAD objects
603 // This explicitly ignores dynamically added points, path lines, and tool block references.
604 if (!(ent is Brep) && !(ent is Mesh))
605 {
606 continue;
607 }
608
609 // Calculate metadata
610 double volume = 0;
611 int facesCount = 0;
612 int holesCount = 0;
613 string cadFile = string.Empty;
614
615 if (ent is Brep brep)
616 {
617 var mp = new VolumeProperties();
618 var convertedMesh = brep.ConvertToMesh();
619 if (convertedMesh != null)
620 {
621 mp.Add(convertedMesh);
622 volume = mp.Volume;
623 }
624 facesCount = brep.Faces.Length;
625
626 // Count holes associated with this entity from parent VM
627 var faceIndicesForEntity = _parent._allFacesByColor.Values.SelectMany(f => f)
628 .Where(f => f.EntityIndex == i)
629 .Select(f => f.FaceGlobalIndex)
630 .ToHashSet();
631
632 holesCount = _parent._holesByColor.Values.SelectMany(h => h)
633 .Count(h => h.FaceGlobalIndices.Any(idx => faceIndicesForEntity.Contains(idx)));
634
635 // Try to get CAD file from EntityData if it was set during import
636 cadFile = ent.EntityData as string ?? string.Empty;
637 }
638 else if (ent is Mesh mesh)
639 {
640 var mp = new VolumeProperties();
641 mp.Add(mesh);
642 volume = mp.Volume;
643 facesCount = 1; // Mesh is typically one entity
644 }
645
646 string displayName = $"{GetXMLElementValue(DocLanguage, "Entity") ?? "Entity"} {entityCounter++}";
647 string fileNameOnly = !string.IsNullOrEmpty(cadFile) ? System.IO.Path.GetFileName(cadFile) : string.Empty;
648
649 string formattedName = $"{displayName} -> Faces: {facesCount} | Volume: {volume:F2}mm³";
650 string toolTipText = $"Faces: {facesCount} | {GetXMLElementValue(DocLanguage, "Holes")} {holesCount} | Volume: {volume:F2}mm³";
651 if (!string.IsNullOrEmpty(fileNameOnly)) toolTipText += $" | CAD File: {fileNameOnly}";
652
653 Entities.Add(new EntityViewModel(_ignoredItemManager, _parent)
654 {
655 Name = formattedName,
656 ToolTip = toolTipText,
657 EntityRef = ent,
658 IsVisible = ent.Visible
659 });
660 }
661
662 OnPropertyChanged(nameof(Entities));
663 });
664 }

692 public static string? GetXMLElementValue(XmlDocument doc, string key)
693 {
694 // Check if the document is null
695 if (doc.DocumentElement == null)
696 {
697 return null; // or handle the error appropriately
698 }
699
700 // Use XPath to find the element with the specified key
701 XmlNode? node = doc.SelectSingleNode($"//{key}");
702 return node?.Attributes?["Value"]?.Value; // Return the value or an empty string if not found

773 private static T? FindVisualParent<T>(DependencyObject child) where T : DependencyObject
774 {
775 if (child == null || child is not Visual && child is not Visual3D) return null;
776
777 DependencyObject parentObject = VisualTreeHelper.GetParent(child);
778 if (parentObject == null) return null;
779
780 if (parentObject is T parent)
781 return parent;
782
783 return FindVisualParent<T>(parentObject);
