352 private void BackupAndHidePathEntities()
353 {
354 if (_pathEntitiesBackedUp) return;
355
356 var design = _parent.MainParent.designControl;
357
358 // Find all path entities and base-avoidance visuals.
359 _pathEntityBackup = design.Entities
360 .Where(e => (e.LayerName != null && e.LayerName.StartsWith("PATH_")) ||
361 MainViewModel.IsBaseAvoidanceVisualEntity(e))
362 .ToList();
363
364 // Remove from design temporarily
365 foreach (var entity in _pathEntityBackup)
366 {
367 design.Entities.Remove(entity);
368 }
369
370 if (_pathEntityBackup.Any())
371 {
372 design.Invalidate();
373 System.Diagnostics.Debug.WriteLine($"Backed up {_pathEntityBackup.Count} transient visual entities");
374 }
375
376 _pathEntitiesBackedUp = true;
377 }

382 private void RestorePathEntities()
383 {
384 if (!_pathEntitiesBackedUp) return;
385
386 var design = _parent.MainParent.designControl;
387
388 // Restore entities
389 foreach (var entity in _pathEntityBackup)
390 {
391 if (!design.Entities.Contains(entity))
392 {
393 design.Entities.Add(entity);
394 }
395 }
396
397 design.Entities.Regen();
398 design.Invalidate();
399
400 System.Diagnostics.Debug.WriteLine($"Restored {_pathEntityBackup.Count} transient visual entities");
401 _pathEntityBackup.Clear();
402 _pathEntitiesBackedUp = false;
403 }

410 private void UpdateTimer_Tick(object? sender, EventArgs e)
411 {
412 var design = _parent.MainParent.designControl;
413 int currentCount = design.Entities.Count;
414
415 // Check if entity count changed significantly (model added/removed)
416 // Ignore small changes from path entities
417 int delta = Math.Abs(currentCount - _lastEntityCount);
418
419 if (delta > 10) // Significant change indicates model modification
420 {
421 // Clear path backup if model changed - paths need to be regenerated
422 if (_pathEntitiesBackedUp)
423 {
424 _pathEntityBackup.Clear();
425 _pathEntitiesBackedUp = false;
426 System.Diagnostics.Debug.WriteLine("Model changed - cleared path backup (paths need regeneration)");
427 }
428
429 // Store initial state for new entities only
430 for (int i = 0; i < currentCount; i++)
431 {
432 var entity = design.Entities[i];
433 if (entity != null &&
434 (entity.EntityData as string) != "IgnoredOverlay_Ref" &&
435 !MainViewModel.IsBaseAvoidanceVisualEntity(entity) &&
436 !_initialVisibilityStates.ContainsKey(entity))
437 {
438 _initialVisibilityStates[entity] = entity.Visible;
439 }
440 }
441
442 RefreshEntitiesFromDesign();
443 _lastEntityCount = currentCount;
444 }
445 }

550 private void RefreshEntitiesFromDesign()
551 {
552 Application.Current.Dispatcher.Invoke(() =>
553 {
554 Entities.Clear();
555
556 var design = _parent.MainParent.designControl;
557
558 int entityCounter = 1;
559
560 // Start loop from 0
561 for (int i = 0; i < design.Entities.Count; i++)
562 {
563 var ent = design.Entities[i];
564
565 if (ent.EntityData as string == "IgnoredOverlay_Ref") continue;
566 if (MainViewModel.IsBaseAvoidanceVisualEntity(ent)) continue;
567
568 if (_parent.IsControlPointEntity(ent))
569 {
570 continue;
571 }
572
573 // Skip shell entities (debug/visualization only)
574 if (ent.LayerName?.StartsWith("SHELL_", StringComparison.OrdinalIgnoreCase) == true)
575 {
576 continue;
577 }
578
579 // Skip origin symbols
580 if (ent.EntityData as string == "PART_ORIGIN")
581 {
582 continue;
583 }
584
585 // Strict filter: only allow actual Brep or Mesh CAD objects
586 // This explicitly ignores dynamically added points, path lines, and tool block references.
587 if (!(ent is Brep) && !(ent is Mesh))
588 {
589 continue;
590 }
591
592 // Calculate metadata
593 double volume = 0;
594 int facesCount = 0;
595 int holesCount = 0;
596 string cadFile = string.Empty;
597
598 if (ent is Brep brep)
599 {
600 var mp = new VolumeProperties();
601 var convertedMesh = brep.ConvertToMesh();
602 if (convertedMesh != null)
603 {
604 mp.Add(convertedMesh);
605 volume = mp.Volume;
606 }
607 facesCount = brep.Faces.Length;
608
609 // Count holes associated with this entity from parent VM
610 var faceIndicesForEntity = _parent._allFacesByColor.Values.SelectMany(f => f)
611 .Where(f => f.EntityIndex == i)
612 .Select(f => f.FaceGlobalIndex)
613 .ToHashSet();
614
615 holesCount = _parent._holesByColor.Values.SelectMany(h => h)
616 .Count(h => h.FaceGlobalIndices.Any(idx => faceIndicesForEntity.Contains(idx)));
617
618 // Try to get CAD file from EntityData if it was set during import
619 cadFile = ent.EntityData as string ?? string.Empty;
620 }
621 else if (ent is Mesh mesh)
622 {
623 var mp = new VolumeProperties();
624 mp.Add(mesh);
625 volume = mp.Volume;
626 facesCount = 1; // Mesh is typically one entity
627 }
628
629 string displayName = $"{GetXMLElementValue(DocLanguage, "Entity") ?? "Entity"} {entityCounter++}";
630 string fileNameOnly = !string.IsNullOrEmpty(cadFile) ? System.IO.Path.GetFileName(cadFile) : string.Empty;
631
632 string formattedName = $"{displayName} -> Faces: {facesCount} | Volume: {volume:F2}mm³";
633 string toolTipText = $"Faces: {facesCount} | {GetXMLElementValue(DocLanguage, "Holes")} {holesCount} | Volume: {volume:F2}mm³";
634 if (!string.IsNullOrEmpty(fileNameOnly)) toolTipText += $" | CAD File: {fileNameOnly}";
635
636 Entities.Add(new EntityViewModel(_ignoredItemManager, _parent)
637 {
638 Name = formattedName,
639 ToolTip = toolTipText,
640 EntityRef = ent,
641 IsVisible = ent.Visible
642 });
643 }
644
645 OnPropertyChanged(nameof(Entities));
646 });
647 }

675 public static string? GetXMLElementValue(XmlDocument doc, string key)
676 {
677 // Check if the document is null
678 if (doc.DocumentElement == null)
679 {

681 }

683 // Use XPath to find the element with the specified key

756 private static T? FindVisualParent<T>(DependencyObject child) where T : DependencyObject
757 {
758 if (child == null || child is not Visual && child is not Visual3D) return null;
759
760 DependencyObject parentObject = VisualTreeHelper.GetParent(child);
761 if (parentObject == null) return null;
762
763 if (parentObject is T parent)
764 return parent;
765
766 return FindVisualParent<T>(parentObject);
