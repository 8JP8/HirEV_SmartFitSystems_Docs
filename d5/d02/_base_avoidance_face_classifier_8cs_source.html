73 public static bool HasIgnoredBottomFace(IEnumerable<CADFace>? faces)
74 {
75 if (faces == null)
76 return false;
77
78 var samples = faces
79 .Where(face => face != null)
80 .Select(face => BuildFaceSample(face, null))
81 .Where(sample => sample != null && sample.Points.Count > 0)
82 .Cast<FaceSample>()
83 .ToList();
84
85 foreach (var entityGroup in samples.GroupBy(sample => sample.Face.EntityIndex))
86 {
87 double baseZ = entityGroup.Min(sample => sample.MinZ);
88 double baseLimit = baseZ + BasePlaneTolerance + BaseAvoidanceClearance.ZTolerance;
89 if (entityGroup.Any(sample =>
90 sample.Face.IsIgnored &&
91 sample.Face.FaceType == FaceTypeEnum.Planar &&
92 sample.MinZ <= baseLimit &&
93 sample.MaxZ <= baseLimit))
94 {
95 return true;
96 }
97 }
98
99 return false;
100 }

110 private static IReadOnlyList<CADFace> FindFacesToIgnore(
111 IEnumerable<CADFace>? faces,
112 HashSet<int> holeFaceIndices,
113 Func<CADFace, Brep?>? brepResolver,
114 BaseAvoidanceConfig config)
115 {
116 var result = new List<CADFace>();
117 if (faces == null)
118 return result;
119
120 var faceList = faces
121 .Where(face => face != null)
122 .ToList();
123
124 config = BaseAvoidanceClearance.EnableForIgnoredBottomFace(
125 config,
126 HasIgnoredBottomFace(faceList));
127 if (!config.Enabled)
128 return result;
129
130 var samples = faceList
131 .Select(face => BuildFaceSample(face, brepResolver))
132 .Where(sample => sample != null && sample.Points.Count > 0)
133 .Cast<FaceSample>()
134 .ToList();
135
136 if (samples.Count == 0)
137 return result;
138
139 foreach (var entityGroup in samples.GroupBy(sample => sample.Face.EntityIndex))
140 {
141 var entitySamples = entityGroup.ToList();
142 double baseZ = entitySamples.Min(sample => sample.MinZ);
143 double seedMaxZ = baseZ + BasePlaneTolerance + BaseAvoidanceClearance.ZTolerance;
144 double seedMinZ = baseZ + BasePlaneTolerance + BaseAvoidanceClearance.ZTolerance;
145 var baseSeedSamples = entitySamples
146 .Where(sample =>
147 sample.Face.FaceType == FaceTypeEnum.Planar &&
148 sample.MinZ <= seedMinZ &&
149 sample.MaxZ <= seedMaxZ)
150 .ToList();
151
152 var baseSeedSet = new HashSet<FaceSample>(baseSeedSamples);
153 double containedMaxZ = baseZ + config.SafeHeight + BaseAvoidanceClearance.ZTolerance;
154
155 foreach (var sample in entitySamples)
156 {
157 var face = sample.Face;
158 if (face.IsIgnored ||
159 face.FaceGlobalIndex <= 0 ||
160 holeFaceIndices.Contains(face.FaceGlobalIndex))
161 {
162 continue;
163 }
164
165 bool isBaseSeedFace = baseSeedSet.Contains(sample);
166 bool isBaseTransitionFace = !isBaseSeedFace &&
167 IsBaseTransitionFaceCandidate(
168 sample,
169 entitySamples,
170 baseSeedSamples,
171 containedMaxZ);
172
173 if (!isBaseSeedFace && !isBaseTransitionFace)
174 continue;
175
176 result.Add(face);
177 Debug.WriteLine(
178 $" Base avoidance classifier selected face {face.FaceGlobalIndex} ({face.FaceType}) " +
179 $"entity={face.EntityIndex}, minZ={sample.MinZ:F3}, maxZ={sample.MaxZ:F3}, baseZ={baseZ:F3}, " +
180 $"reason={(isBaseSeedFace ? "bottom plane" : "base chamfer/fillet")}.");
181 }
182 }
183
184 return result;
185 }

195 private static bool IsBaseTransitionFaceCandidate(
196 FaceSample sample,
197 IReadOnlyList<FaceSample> entitySamples,
198 IReadOnlyList<FaceSample> baseSeedSamples,
199 double containedMaxZ)
200 {
201 if (sample.MaxZ > containedMaxZ)
202 return false;
203
204 switch (sample.Face.FaceType)
205 {
206 case FaceTypeEnum.Cylindrical:
207 {
208 var axis = sample.Face.AxisOrNormal ?? EyeshotGeometryUtils.GetFaceAxis(sample.Face);
209 if (axis.Length < 1e-9)
210 return false;
211
212 axis.Normalize();
213
214 if (Math.Abs(axis.Z) >= 0.65)
215 {
216 return IsVerticalCylindricalBaseTransition(
217 sample,
218 entitySamples,
219 baseSeedSamples);
220 }
221
222 return true;
223 }
224
225 case FaceTypeEnum.Conical:
226 return true;
227
228 case FaceTypeEnum.Planar:
229 {
230 var normal = sample.Face.AxisOrNormal;
231 if (normal == null && !TryParseFaceNormal(sample.Face, out normal))
232 return false;
233
234 double normalAbsZ = Math.Abs(normal.Z);
235 return normalAbsZ > 0.15 && normalAbsZ < 0.95;
236 }
237
238 case FaceTypeEnum.Toroidal:
239 case FaceTypeEnum.Spherical:
240 case FaceTypeEnum.BSpline:
241 case FaceTypeEnum.Revolved:
242 case FaceTypeEnum.Tabulated:
243 case FaceTypeEnum.Unknown:
244 return true;
245
246 default:
247 return false;
248 }
249 }

258 private static bool IsVerticalCylindricalBaseTransition(
259 FaceSample candidate,
260 IReadOnlyList<FaceSample> entitySamples,
261 IReadOnlyList<FaceSample> baseSeedSamples)
262 {
263 const double TouchTolerance = 0.35;
264
265 if (baseSeedSamples.Count == 0 ||
266 !SamplesTouch(candidate, baseSeedSamples, TouchTolerance))
267 {
268 return false;
269 }
270
271 double candidateRadius = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(candidate.Face);
272 if (candidateRadius <= 0.0)
273 return false;
274
275 return entitySamples.Any(other =>
276 {
277 if (ReferenceEquals(other, candidate) ||
278 other.Face.FaceType != FaceTypeEnum.Cylindrical)
279 {
280 return false;
281 }
282
283 var otherAxis = other.Face.AxisOrNormal ?? EyeshotGeometryUtils.GetFaceAxis(other.Face);
284 if (otherAxis.Length < 1e-9)
285 return false;
286
287 otherAxis.Normalize();
288 if (Math.Abs(otherAxis.Z) < 0.65)
289 return false;
290
291 double otherRadius = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(other.Face);
292 double radiusTolerance = Math.Max(0.05, candidateRadius * 0.01);
293 if (otherRadius <= 0.0 || Math.Abs(otherRadius - candidateRadius) <= radiusTolerance)
294 return false;
295
296 if (other.MaxZ <= candidate.MaxZ + TouchTolerance)
297 return false;
298
299 return SamplesTouchNearTop(candidate, other, TouchTolerance);
300 });
301 }

350 private static bool TryParseFaceNormal(CADFace face, out Vector3D normal)
351 {
352 normal = Vector3D.AxisZ;
353 if (string.IsNullOrWhiteSpace(face?.GeometricInfo))
354 return false;
355
356 var match = System.Text.RegularExpressions.Regex.Match(
357 face.GeometricInfo,
358 @"Normal:\s*\‍(([-0-9.+Ee]+),\s*([-0-9.+Ee]+),\s*([-0-9.+Ee]+)\‍)",
359 System.Text.RegularExpressions.RegexOptions.IgnoreCase);
360
361 if (!match.Success ||
362 !double.TryParse(match.Groups[1].Value, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double x) ||
363 !double.TryParse(match.Groups[2].Value, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double y) ||
364 !double.TryParse(match.Groups[3].Value, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double z))
365 {
366 return false;
367 }
368
369 normal = new Vector3D(x, y, z);
370 if (normal.Length < 1e-9)
371 return false;
372
373 normal.Normalize();
374 return true;
375 }

453 private static void AddBrepFaceSamplePoints(List<Point3D> points, CADFace face, Brep? brep)
454 {
455 if (brep == null ||
456 face.FaceIndex < 0 ||
457 face.FaceIndex >= brep.Faces.Length)
458 {
459 return;
460 }
461
462 try
463 {
464 var brepFace = brep.Faces[face.FaceIndex];
465 if (brepFace == null)
466 return;
467
468 Mesh? mesh = brepFace.Tessellation?.ConvertToMesh();
469 if (mesh == null)
470 {
471 try
472 {
473 mesh = brepFace.ConvertToMesh();
474 }
475 catch
476 {
477 mesh = null;
478 }
479 }
480
481 if (mesh?.Vertices != null)
482 AddPoint3DSamples(points, mesh.Vertices);
483
484 if (brepFace.Parametric == null)
485 return;
486
487 foreach (var surface in brepFace.Parametric)
488 {
489 if (surface == null)
490 continue;
491
492 try
493 {
494 if (surface.Vertices == null)
495 surface.Regen(0.01);
496
497 if (surface.Vertices != null)
498 AddPoint3DSamples(points, surface.Vertices);
499 }
500 catch
501 {
502 // Parametric samples are supplemental; tessellation/control/key points remain available.
503 }
504 }
505 }
506 catch (Exception ex)
507 {
508 Debug.WriteLine($"WARNING: Could not sample BREP face {face.FaceGlobalIndex} for base avoidance: {ex.Message}");
509 }
510 }

568 private static List<Point3D> Deduplicate(List<Point3D> points)
569 {
570 if (points.Count <= 1)
571 return points;
572
573 var result = new List<Point3D>(points.Count);
574 var seen = new HashSet<(long X, long Y, long Z)>();
575
576 foreach (var point in points)
577 {
578 var key = (
579 (long)Math.Round(point.X / SampleDedupTolerance),
580 (long)Math.Round(point.Y / SampleDedupTolerance),
581 (long)Math.Round(point.Z / SampleDedupTolerance));
582
583 if (seen.Add(key))
584 result.Add(point);
585 }
586
587 return result;
588 }
