152 public void SetObstacles(List<Brep?> obstacles)
153 {
154 _obstacles = obstacles ?? new List<Brep?>();
155
156 // Auto-calculate bounds from obstacles
157 if (_obstacles.Count > 0)
158 {
159 double minX = double.MaxValue, minY = double.MaxValue, minZ = double.MaxValue;
160 double maxX = double.MinValue, maxY = double.MinValue, maxZ = double.MinValue;
161
162 foreach (var brep in _obstacles)
163 {
164 if (brep?.BoxMin == null || brep?.BoxMax == null) continue;
165
166 minX = Math.Min(minX, brep.BoxMin.X);
167 minY = Math.Min(minY, brep.BoxMin.Y);
168 minZ = Math.Min(minZ, brep.BoxMin.Z);
169 maxX = Math.Max(maxX, brep.BoxMax.X);
170 maxY = Math.Max(maxY, brep.BoxMax.Y);
171 maxZ = Math.Max(maxZ, brep.BoxMax.Z);
172 }
173
174 if (minX != double.MaxValue)
175 {
176 // Add margin around obstacles for sampling space
177 double margin = 100.0;
178 _boundsMin = new Point3D(minX - margin, minY - margin, minZ - margin);
179 _boundsMax = new Point3D(maxX + margin, maxY + margin, maxZ + margin);
180 }

187 public PathResult FindPath(Point3D start, Point3D goal)
188 {
189 var result = new PathResult();
190 var sw = Stopwatch.StartNew();
191
192 Debug.WriteLine($"🌳 RRT*: Finding path from ({start.X:F1}, {start.Y:F1}, {start.Z:F1}) to ({goal.X:F1}, {goal.Y:F1}, {goal.Z:F1})");
193
194 // Check if direct path is clear
195 if (IsSegmentCollisionFree(start, goal))
196 {
197 result.Success = true;
198 result.Path = new List<Point3D> { start, goal };
199 result.TotalDistance = start.DistanceTo(goal);
200 result.Message = "Direct path is clear";
201 Debug.WriteLine($" ✅ Direct path clear: {result.TotalDistance:F1}mm");
202 return result;
203 }
204
205 // Validate start and goal are collision-free
206 if (IsPointInCollision(start))
207 {
208 result.Success = false;
209 result.Message = "Start point is in collision";
210 Debug.WriteLine($" ❌ Start point in collision");
211 return result;
212 }
213
214 if (IsPointInCollision(goal))
215 {
216 result.Success = false;
217 result.Message = "Goal point is in collision";
218 Debug.WriteLine($" ❌ Goal point in collision");
219 return result;
220 }
221
222 // Initialize tree with start node
223 var tree = new List<TreeNode>();
224 var root = new TreeNode(start);
225 tree.Add(root);
226
227 TreeNode? goalNode = null;
228 double bestGoalCost = double.MaxValue;
229
230 // Expand bounds to include start and goal
231 ExpandBoundsToInclude(start);
232 ExpandBoundsToInclude(goal);
233
234 // Main RRT* loop
235 for (int iteration = 0; iteration < _maxIterations; iteration++)
236 {
237 result.Iterations = iteration + 1;
238
239 // Sample random point (with goal bias)
240 Point3D randomPoint;
241 if (_random.NextDouble() < _goalBias)
242 {
243 randomPoint = goal;
244 }
245 else
246 {
247 randomPoint = SampleRandomPoint();
248 }
249
250 // Find nearest node in tree
251 var nearestNode = FindNearestNode(tree, randomPoint);
252 if (nearestNode == null) continue;
253
254 // Steer towards random point
255 Point3D newPoint = Steer(nearestNode.Position, randomPoint, _stepSize);
256
257 // Check if path to new point is collision-free
258 if (!IsSegmentCollisionFree(nearestNode.Position, newPoint))
259 continue;
260
261 // Find near nodes for potential rewiring
262 double radius = CalculateRewireRadius(tree.Count);
263 var nearNodes = FindNodesWithinRadius(tree, newPoint, radius);
264
265 // Find best parent among near nodes
266 TreeNode? bestParent = nearestNode;
267 double bestCost = nearestNode.CostFromStart + nearestNode.Position.DistanceTo(newPoint);
268
269 foreach (var nearNode in nearNodes)
270 {
271 double cost = nearNode.CostFromStart + nearNode.Position.DistanceTo(newPoint);
272 if (cost < bestCost && IsSegmentCollisionFree(nearNode.Position, newPoint))
273 {
274 bestParent = nearNode;
275 bestCost = cost;
276 }
277 }
278
279 // Create new node with best parent
280 var newNode = new TreeNode(newPoint, bestParent);
281 newNode.CostFromStart = bestCost;
282 bestParent.Children.Add(newNode);
283 tree.Add(newNode);
284 result.NodesCreated++;
285
286 // Rewire: check if new node provides better path to near nodes
287 foreach (var nearNode in nearNodes)
288 {
289 if (nearNode == bestParent) continue;
290
291 double costThroughNew = newNode.CostFromStart + newNode.Position.DistanceTo(nearNode.Position);
292 if (costThroughNew < nearNode.CostFromStart &&
293 IsSegmentCollisionFree(newNode.Position, nearNode.Position))
294 {
295 // Rewire: change parent of nearNode to newNode
296 if (nearNode.Parent != null)
297 {
298 nearNode.Parent.Children.Remove(nearNode);
299 }
300 nearNode.Parent = newNode;
301 nearNode.CostFromStart = costThroughNew;
302 newNode.Children.Add(nearNode);
303
304 // Propagate cost update to descendants
305 PropagateCoastUpdate(nearNode);
306 }
307 }
308
309 // Check if we can reach goal from new node
310 if (newPoint.DistanceTo(goal) < _goalTolerance)
311 {
312 if (newNode.CostFromStart < bestGoalCost)
313 {
314 goalNode = newNode;
315 bestGoalCost = newNode.CostFromStart;
316 }
317 }
318 else if (IsSegmentCollisionFree(newPoint, goal))
319 {
320 double totalCost = newNode.CostFromStart + newPoint.DistanceTo(goal);
321 if (totalCost < bestGoalCost)
322 {
323 // Create goal node
324 var gNode = new TreeNode(goal, newNode);
325 gNode.CostFromStart = totalCost;
326 newNode.Children.Add(gNode);
327 tree.Add(gNode);
328 goalNode = gNode;
329 bestGoalCost = totalCost;
330 }
331 }
332
333 // Early termination if we found a good path
334 if (goalNode != null && iteration > _maxIterations / 4)
335 {
336 if (iteration > _maxIterations / 2)
337 break;
338 }
339 }
340
341 sw.Stop();
342
343 if (goalNode != null)
344 {
345 result.Path = ReconstructPath(goalNode);
346 result.TotalDistance = goalNode.CostFromStart;
347 result.Success = true;
348 result.Message = $"Path found with {result.NodesCreated} nodes";
349
350 // Smooth path to remove unnecessary waypoints
351 result.Path = SmoothPath(result.Path);
352
353 Debug.WriteLine($" ✅ RRT* found path: {result.TotalDistance:F1}mm, {result.Path.Count} waypoints, {sw.ElapsedMilliseconds}ms");
354 }
355 else
356 {
357 result.Success = false;
358 result.Message = $"No path found after {result.Iterations} iterations";
359 Debug.WriteLine($" ❌ RRT* failed: {result.Message} ({sw.ElapsedMilliseconds}ms)");
360 }
361
362 return result;
363 }

428 private Point3D Steer(Point3D from, Point3D towards, double maxDistance)
429 {
430 Vector3D direction = new Vector3D(
431 towards.X - from.X,
432 towards.Y - from.Y,
433 towards.Z - from.Z);
434
435 double distance = direction.Length;
436 if (distance <= maxDistance)
437 return towards;
438
439 direction.Normalize();
440 return new Point3D(
441 from.X + direction.X * maxDistance,
442 from.Y + direction.Y * maxDistance,
443 from.Z + direction.Z * maxDistance);
444 }

449 private double CalculateRewireRadius(int numNodes)
450 {
451 const double gamma = 100.0;
452 if (numNodes < 2) return _rewireRadius;
453
454 double radius = gamma * Math.Pow(Math.Log(numNodes) / numNodes, 1.0 / 3.0);
455 return Math.Max(radius, _stepSize * 1.5);
456 }

461 private void PropagateCoastUpdate(TreeNode node)
462 {
463 var stack = new Stack<TreeNode>();
464 foreach (var child in node.Children)
465 stack.Push(child);
466
467 while (stack.Count > 0)
468 {
469 var current = stack.Pop();
470 if (current.Parent != null)
471 {
472 current.CostFromStart = current.Parent.CostFromStart +
473 current.Position.DistanceTo(current.Parent.Position);
474 }
475 foreach (var child in current.Children)
476 stack.Push(child);
477 }
478 }

501 private List<Point3D> SmoothPath(List<Point3D> path)
502 {
503 if (path.Count <= 2)
504 return path;
505
506 var smoothed = new List<Point3D> { path[0] };
507 int current = 0;
508
509 while (current < path.Count - 1)
510 {
511 int furthest = current + 1;
512
513 // Find furthest point we can reach directly
514 for (int i = path.Count - 1; i > current + 1; i--)
515 {
516 if (IsSegmentCollisionFree(path[current], path[i]))
517 {
518 furthest = i;
519 break;
520 }
521 }
522
523 smoothed.Add(path[furthest]);
524 current = furthest;
525 }
526
527 return smoothed;
528 }

533 private bool IsPointInCollision(Point3D point)
534 {
535 foreach (var brep in _obstacles)
536 {
537 if (brep == null) continue;
538
539 try
540 {
541 var min = brep.BoxMin;
542 var max = brep.BoxMax;
543 if (min == null || max == null) continue;
544
545 // Quick bounding box check with margin
546 if (point.X < min.X - _safetyMargin || point.X > max.X + _safetyMargin ||
547 point.Y < min.Y - _safetyMargin || point.Y > max.Y + _safetyMargin ||
548 point.Z < min.Z - _safetyMargin || point.Z > max.Z + _safetyMargin)
549 continue;
550
551 // Ray casting inside/outside test
552 int intersectionCount = 0;
553 foreach (var face in brep.Faces)
554 {
555 if (face == null) continue;
556 try
557 {
558 var mesh = face.ConvertToMesh();
559 if (mesh?.Triangles == null || mesh?.Vertices == null) continue;
560
561 foreach (var tri in mesh.Triangles)
562 {
563 if (tri.V1 >= mesh.Vertices.Length ||
564 tri.V2 >= mesh.Vertices.Length ||
565 tri.V3 >= mesh.Vertices.Length)
566 continue;
567
568 if (RayIntersectsTriangle(point, Vector3D.AxisZ,
569 mesh.Vertices[tri.V1],
570 mesh.Vertices[tri.V2],
571 mesh.Vertices[tri.V3]))
572 {
573 intersectionCount++;
574 }
575 }
576 }
577 catch { }
578 }
579
580 if (intersectionCount % 2 == 1)
581 return true;
582 }
583 catch { }
584 }
585
586 return false;
587 }

592 private bool IsSegmentCollisionFree(Point3D start, Point3D end)
593 {
594 if (_obstacles == null || _obstacles.Count == 0)
595 return true;
596
597 // Sample points along segment
598 for (int i = 0; i <= SEGMENT_SAMPLE_COUNT; i++)
599 {
600 double t = (double)i / SEGMENT_SAMPLE_COUNT;
601 Point3D sample = new Point3D(
602 start.X + (end.X - start.X) * t,
603 start.Y + (end.Y - start.Y) * t,
604 start.Z + (end.Z - start.Z) * t);
605
606 if (IsPointInCollision(sample))
607 return false;
608 }
609
610 // Check triangle intersections
611 foreach (var brep in _obstacles)
612 {
613 if (brep == null) continue;
614
615 try
616 {
617 // Quick bounding box check
618 var min = brep.BoxMin;
619 var max = brep.BoxMax;
620 if (min == null || max == null) continue;
621
622 double segMinX = Math.Min(start.X, end.X) - _safetyMargin;
623 double segMaxX = Math.Max(start.X, end.X) + _safetyMargin;
624 double segMinY = Math.Min(start.Y, end.Y) - _safetyMargin;
625 double segMaxY = Math.Max(start.Y, end.Y) + _safetyMargin;
626 double segMinZ = Math.Min(start.Z, end.Z) - _safetyMargin;
627 double segMaxZ = Math.Max(start.Z, end.Z) + _safetyMargin;
628
629 if (segMaxX < min.X || segMinX > max.X ||
630 segMaxY < min.Y || segMinY > max.Y ||
631 segMaxZ < min.Z || segMinZ > max.Z)
632 continue;
633
634 // Triangle intersection check
635 foreach (var face in brep.Faces)
636 {
637 if (face == null) continue;
638 try
639 {
640 var mesh = face.ConvertToMesh();
641 if (mesh?.Triangles == null || mesh?.Vertices == null) continue;
642
643 foreach (var tri in mesh.Triangles)
644 {
645 if (tri.V1 >= mesh.Vertices.Length ||
646 tri.V2 >= mesh.Vertices.Length ||
647 tri.V3 >= mesh.Vertices.Length)
648 continue;
649
650 if (SegmentIntersectsTriangle(start, end,
651 mesh.Vertices[tri.V1],
652 mesh.Vertices[tri.V2],
653 mesh.Vertices[tri.V3]))
654 {
655 return false;
656 }
657 }
658 }
659 catch { }
660 }
661 }
662 catch { }
663 }
664
665 return true;
666 }

671 private bool RayIntersectsTriangle(Point3D origin, Vector3D direction, Point3D v0, Point3D v1, Point3D v2)
672 {
673 const double EPSILON = 1e-7;
674
675 Vector3D edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
676 Vector3D edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
677
678 Vector3D h = Vector3D.Cross(direction, edge2);
679 double a = Vector3D.Dot(edge1, h);
680
681 if (a > -EPSILON && a < EPSILON)
682 return false;
683
684 double f = 1.0 / a;
685 Vector3D s = new Vector3D(origin.X - v0.X, origin.Y - v0.Y, origin.Z - v0.Z);
686 double u = f * Vector3D.Dot(s, h);
687
688 if (u < 0.0 || u > 1.0)
689 return false;
690
691 Vector3D q = Vector3D.Cross(s, edge1);
692 double v = f * Vector3D.Dot(direction, q);
693
694 if (v < 0.0 || u + v > 1.0)
695 return false;
696
697 double t = f * Vector3D.Dot(edge2, q);
698 return t > EPSILON;
699 }

704 private bool SegmentIntersectsTriangle(Point3D start, Point3D end, Point3D v0, Point3D v1, Point3D v2)
705 {
706 Vector3D direction = new Vector3D(
707 end.X - start.X,
708 end.Y - start.Y,
709 end.Z - start.Z);
710
711 const double EPSILON = 1e-7;
712
713 Vector3D edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
714 Vector3D edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
715
716 Vector3D h = Vector3D.Cross(direction, edge2);
717 double a = Vector3D.Dot(edge1, h);
718
719 if (a > -EPSILON && a < EPSILON)
720 return false;
721
722 double f = 1.0 / a;
723 Vector3D s = new Vector3D(start.X - v0.X, start.Y - v0.Y, start.Z - v0.Z);
724 double u = f * Vector3D.Dot(s, h);
725
726 if (u < 0.0 || u > 1.0)
727 return false;
728
729 Vector3D q = Vector3D.Cross(s, edge1);
730 double v = f * Vector3D.Dot(direction, q);
731
732 if (v < 0.0 || u + v > 1.0)
733 return false;
734
735 double t = f * Vector3D.Dot(edge2, q);
736 return t >= -EPSILON && t <= 1.0 + EPSILON;
737 }
