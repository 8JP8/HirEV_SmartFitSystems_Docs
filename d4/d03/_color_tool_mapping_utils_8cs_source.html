28 public static Color ParseRgbString(string rgbString)
29 {
30 try
31 {
32 // Remove "RGB(" prefix and ")" suffix, handle both formats
33 string values = rgbString.Replace("RGB(", "").Replace("(", "").Replace(")", "").Trim();
34 string[] components = values.Split(',');
35
36 if (components.Length == 3)
37 {
38 int r = int.Parse(components[0].Trim());
39 int g = int.Parse(components[1].Trim());
40 int b = int.Parse(components[2].Trim());
41 return Color.FromArgb(r, g, b);
42 }
43 }
44 catch { }
45
46 return Color.White; // Fallback
47 }

54 public static Color ParseHexString(string hexString)

65 int b = Convert.ToInt32(hexString.Substring(4, 2), 16);

132 public static void LoadToolColorMapping(string xmlFilePath)
133 {
134 _colorKeyToToolMap.Clear();
135
136 if (!File.Exists(xmlFilePath))
137 {
138 System.Diagnostics.Debug.WriteLine($"Warning: MeasuringToolByColor.xml not found at {xmlFilePath}");
139 return;
140 }
141
142 try
143 {
144 XmlDocument doc = new XmlDocument();
145 doc.Load(xmlFilePath);
146
147 XmlNodeList? nodes = doc.SelectNodes("//add");
148 if (nodes == null) return;
149
150 foreach (XmlNode node in nodes)
151 {
152 string? key = node.Attributes?["key"]?.Value;
153 string? tool = node.Attributes?["tool"]?.Value;
154
155 if (!string.IsNullOrEmpty(key) && !string.IsNullOrEmpty(tool))
156 {
157 // Support both RGB(r,g,b) and hex formats for backwards compatibility
158 string normalizedKey = key;
159 if (key.StartsWith("#") || (key.Length == 6 && !key.Contains("RGB")))
160 {
161 // Convert hex to RGB format
162 normalizedKey = HexToRgbString(key);
163 }
164
165 // Keep the tool string as is
166 _colorKeyToToolMap[normalizedKey] = tool;
167 }
168 }
169
170 System.Diagnostics.Debug.WriteLine($"Loaded {_colorKeyToToolMap.Count} tool-color mappings from configuration.");
171 _isInitialized = true;
172 }
173 catch (Exception ex)
174 {
175 System.Diagnostics.Debug.WriteLine($"Error loading MeasuringToolByColor.xml: {ex.Message}");
176 }
177 }
