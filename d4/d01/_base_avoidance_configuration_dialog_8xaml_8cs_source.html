38 private void Save_Click(object sender, RoutedEventArgs e)
39 {
40 if (!TryReadValues(out double height, out double angle))
41 return;
42
43 try
44 {
45 var config = ConfigurationManager.OpenExeConfiguration(ConfigurationUserLevel.None);
46 SetAppSetting(config, BaseAvoidanceClearance.SafeHeightSettingKey, height.ToString("0.###", CultureInfo.InvariantCulture));
47 SetAppSetting(config, BaseAvoidanceClearance.SafeAngleSettingKey, angle.ToString("0.###", CultureInfo.InvariantCulture));

49 ConfigurationManager.RefreshSection("appSettings");
50
51 DialogResult = true;
52 Close();
53 }
54 catch (ConfigurationErrorsException ex)
55 {
56 ShowValidationMessage(
57 string.Format(
58 GetText("BaseAvoidanceConfig_SaveFailed", "Failed to save base avoidance configuration: {0}"),
59 ex.Message));
60 }
61 }

63 private bool TryReadValues(out double height, out double angle)
64 {
65 height = 0;
66 angle = 0;
67
68 if (!double.TryParse(HeightTextBox.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out height) &&
69 !double.TryParse(HeightTextBox.Text, NumberStyles.Float, CultureInfo.CurrentCulture, out height))
70 {
71 ShowValidationMessage(GetText("BaseAvoidanceConfig_InvalidHeight", "Enter a valid avoidance height."));
72 return false;
73 }
74
75 if (height < 0)
76 {
77 ShowValidationMessage(GetText("BaseAvoidanceConfig_InvalidHeightRange", "Avoidance height must be greater than or equal to 0."));
78 return false;
79 }
80
81 if (!double.TryParse(AngleTextBox.Text, NumberStyles.Float, CultureInfo.InvariantCulture, out angle) &&
82 !double.TryParse(AngleTextBox.Text, NumberStyles.Float, CultureInfo.CurrentCulture, out angle))
83 {
84 ShowValidationMessage(GetText("BaseAvoidanceConfig_InvalidAngle", "Enter a valid avoidance angle."));
85 return false;
86 }
87
88 if (angle < 0 || angle > 90)
89 {
90 ShowValidationMessage(GetText("BaseAvoidanceConfig_InvalidAngleRange", "Avoidance angle must be between 0 and 90 degrees."));
91 return false;
92 }
93
94 return true;
95 }
