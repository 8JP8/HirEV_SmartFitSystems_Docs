255 public static (double A, double B) VectorToHeadAngles(Vector3D normal, ProbeHeadKinematicsConfig config)
256 {
257 if (normal == null || normal.Length < NumericalTolerance)
258 return (0.0, 0.0);
259
260 var norm = new Vector3D(normal.X, normal.Y, normal.Z);
261 norm.Normalize();
262
263 if (config.HeadType == ProbeHeadType.Indexable && config.StepADeg > 0.05 && config.StepBDeg > 0.05)
264 {
265 return ProbeAngleStandardizer.VectorToHeadAngles(norm, config.StepADeg, config.StepBDeg, config.MaxADeg);
266 }
267
268 // Continuous angles
269 double zClamped = Math.Clamp(norm.Z, -1.0, 1.0);
270 double a = Math.Acos(zClamped) * 180.0 / Math.PI;
271 double b = Math.Atan2(norm.Y, norm.X) * 180.0 / Math.PI;
272
273 a = Math.Clamp(a, 0.0, config.MaxADeg);
274 if (a < 0.1) b = 0.0;
275
276 return (a, b);
277 }

282 public static double AngleBetweenDeg(Vector3D v1, Vector3D v2)
283 {
284 if (v1 == null || v2 == null || v1.Length < NumericalTolerance || v2.Length < NumericalTolerance)
285 return 0.0;
286
287 double dot = (v1.X * v2.X + v1.Y * v2.Y + v1.Z * v2.Z) / (v1.Length * v2.Length);
288 dot = Math.Clamp(dot, -1.0, 1.0);
289 return Math.Acos(dot) * 180.0 / Math.PI;
290 }

295 public static double CalculateRotationTime(Vector3D n1, Vector3D n2, ProbeHeadKinematicsConfig config)
296 {
297 if (n1 == null || n2 == null)
298 return 0.0;
299
300 double devDeg = AngleBetweenDeg(n1, n2);
301 if (devDeg <= config.AngleReuseThresholdDeg + NumericalTolerance)
302 return 0.0;
303
304 var (a1, b1) = VectorToHeadAngles(n1, config);
305 var (a2, b2) = VectorToHeadAngles(n2, config);
306
307 double deltaA = Math.Abs(a1 - a2);
308 double deltaB = Math.Abs(b1 - b2);
309 if (deltaB > 180.0)
310 deltaB = 360.0 - deltaB;
311
312 if (deltaA < 0.05 && deltaB < 0.05)
313 return 0.0;
314
315 if (config.HeadType == ProbeHeadType.Indexable)
316 {
317 double stepA = Math.Max(0.1, config.StepADeg);
318 double stepB = Math.Max(0.1, config.StepBDeg);
319
320 int stepsA = (int)Math.Ceiling(deltaA / stepA);
321 int stepsB = (int)Math.Ceiling(deltaB / stepB);
322 int maxSteps = Math.Max(stepsA, stepsB);
323
324 return config.RotationTimeBaseSec + (maxSteps * config.TimePerStepSec);
325 }
326 else // Continuous (e.g. PH20)
327 {
328 double maxDelta = Math.Max(deltaA, deltaB);
329 double omega = Math.Max(1.0, config.MaxAngularVelocityDegPerSec);
330 double alpha = Math.Max(1.0, config.AngularAccelerationDegPerSec2);
331
332 double criticalAngle = (omega * omega) / alpha;
333 if (maxDelta <= criticalAngle)
334 {
335 // Triangular velocity profile
336 return 2.0 * Math.Sqrt(maxDelta / alpha);
337 }
338 else
339 {
340 // Trapezoidal velocity profile
341 return (omega / alpha) + ((maxDelta - criticalAngle) / omega);
342 }
343 }

349 public static double CalculateTranslationTime(Point3D p1, Point3D p2, ProbeHeadKinematicsConfig config)
350 {
351 if (p1 == null || p2 == null)
352 return 0.0;
353
354 double dist = p1.DistanceTo(p2);
355 double v = Math.Max(1.0, config.LinearVelocityMmPerSec);
356 const double accelTransitOverheadSec = 0.05;
357
358 return (dist / v) + accelTransitOverheadSec;

365 public static double CalculateTransitionCost(
366 Point3D p1, Vector3D n1,
367 Point3D p2, Vector3D n2,

399 private static List<ApproachMeasurementPair> OrderPairsByIsoOrientation(
400 List<ApproachMeasurementPair> pairs,
401 ProbeHeadKinematicsConfig config)
402 {
403 // 1. Group pairs into orientation bins
404 var clusters = new List<(long AngleKey, double A, double B, Vector3D Normal, List<ApproachMeasurementPair> Items)>();
405
406 foreach (var pair in pairs)
407 {
408 var (a, b) = VectorToHeadAngles(pair.Normal, config);
409 long key = ((long)Math.Round(a * 10.0) * 100000L) + (long)Math.Round((b + 360.0) * 10.0);
410
411 var existingCluster = clusters.FirstOrDefault(c =>
412 {
413 if (c.AngleKey == key) return true;
414 // Check if within reuse tolerance
415 double dev = AngleBetweenDeg(pair.Normal, c.Normal);
416 return dev <= config.AngleReuseThresholdDeg + NumericalTolerance;
417 });
418
419 if (existingCluster.Items != null)
420 {
421 existingCluster.Items.Add(pair);
422 }
423 else
424 {
425 clusters.Add((key, a, b, pair.Normal, new List<ApproachMeasurementPair> { pair }));
426 }
427 }
428
429 // 2. Sort internal points of each cluster by spatial proximity (T_rot = 0)
430 for (int i = 0; i < clusters.Count; i++)
431 {
432 var clusterItems = clusters[i].Items;
433 if (clusterItems.Count > 1)
434 {
435 clusters[i] = (clusters[i].AngleKey, clusters[i].A, clusters[i].B, clusters[i].Normal, OrderPairsBySpatialProximity(clusterItems));
436 }
437 }
438
439 // 3. Sequence clusters to minimize inter-cluster transition cost (angular delta + distance)
440 var orderedClusters = new List<List<ApproachMeasurementPair>>();
441 var remainingClusters = new List<(long AngleKey, double A, double B, Vector3D Normal, List<ApproachMeasurementPair> Items)>(clusters);
442
443 // Start with cluster closest to vertical (+Z, A=0)
444 var currentCluster = remainingClusters
445 .OrderBy(c => c.A)
446 .ThenBy(c => Math.Abs(c.B))
447 .First();
448
449 orderedClusters.Add(currentCluster.Items);
450 remainingClusters.Remove(currentCluster);
451
452 while (remainingClusters.Count > 0)
453 {
454 var lastPair = currentCluster.Items.Last();
455 var nextCluster = remainingClusters
456 .OrderBy(c =>
457 {
458 var firstPair = c.Items.First();
459 return CalculateTransitionCost(
460 lastPair.ApproachPoint, lastPair.Normal,
461 firstPair.ApproachPoint, firstPair.Normal,
462 config);
463 })
464 .First();
465
466 orderedClusters.Add(nextCluster.Items);
467 remainingClusters.Remove(nextCluster);
468 currentCluster = nextCluster;
469 }
470
471 // Flatten
472 var result = new List<ApproachMeasurementPair>(pairs.Count);
473 foreach (var cluster in orderedClusters)
474 {
475 result.AddRange(cluster);
476 }
477
478 return result;
479 }

485 private static List<ApproachMeasurementPair> OrderPairsByCombinedCost(
486 List<ApproachMeasurementPair> pairs,
487 ProbeHeadKinematicsConfig config)
488 {
489 var ordered = new List<ApproachMeasurementPair>(pairs.Count);
490 var remaining = new List<ApproachMeasurementPair>(pairs);
491
492 // Start with deterministic pair (min X/Y/Z)
493 var current = remaining
494 .OrderBy(p => p.ApproachPoint.X)
495 .ThenBy(p => p.ApproachPoint.Y)
496 .ThenBy(p => p.ApproachPoint.Z)
497 .First();
498
499 ordered.Add(current);
500 remaining.Remove(current);
501
502 while (remaining.Count > 0)
503 {
504 var next = remaining
505 .OrderBy(r => CalculateTransitionCost(
506 current.ApproachPoint, current.Normal,
507 r.ApproachPoint, r.Normal,
508 config))
509 .First();
510
511 ordered.Add(next);
512 remaining.Remove(next);
513 current = next;
514 }
515
516 // 2-Opt local refinement (cap passes to 3 for performance)
517 Apply2OptRefinement(ordered, config, maxPasses: 3);
518
519 return ordered;
520 }

525 private static List<ApproachMeasurementPair> OrderPairsBySpatialProximity(List<ApproachMeasurementPair> pairs)
526 {
527 if (pairs.Count <= 1) return pairs;
528 var ordered = new List<ApproachMeasurementPair>(pairs.Count);
529 var remaining = new List<ApproachMeasurementPair>(pairs);
530 var current = remaining[0];
531 ordered.Add(current);
532 remaining.RemoveAt(0);
533
534 while (remaining.Count > 0)
535 {
536 var next = remaining.OrderBy(r => current.ApproachPoint.DistanceTo(r.ApproachPoint)).First();
537 ordered.Add(next);
538 remaining.Remove(next);
539 current = next;
540 }
541 return ordered;

547 private static void Apply2OptRefinement(
548 List<ApproachMeasurementPair> route,
549 ProbeHeadKinematicsConfig config,
550 int maxPasses = 3)
551 {
552 if (route == null || route.Count < 4) return;
553 int n = route.Count;
554 bool improved = true;
555 int passes = 0;
556
557 while (improved && passes < maxPasses)
558 {
559 improved = false;
560 passes++;
561
562 for (int i = 0; i < n - 2; i++)
563 {
564 for (int j = i + 2; j < n; j++)
565 {
566 var p1 = route[i];
567 var p2 = route[i + 1];
568 var p3 = route[j];
569 var p4 = (j + 1 < n) ? route[j + 1] : null;
570
571 double currentCost = CalculateTransitionCost(p1.ApproachPoint, p1.Normal, p2.ApproachPoint, p2.Normal, config);
572 if (p4 != null)
573 currentCost += CalculateTransitionCost(p3.ApproachPoint, p3.Normal, p4.ApproachPoint, p4.Normal, config);
574
575 double newCost = CalculateTransitionCost(p1.ApproachPoint, p1.Normal, p3.ApproachPoint, p3.Normal, config);
576 if (p4 != null)
577 newCost += CalculateTransitionCost(p2.ApproachPoint, p2.Normal, p4.ApproachPoint, p4.Normal, config);
578
579 if (newCost < currentCost - 1e-4)
580 {
581 // Reverse subsegment [i+1 .. j]
582 route.Reverse(i + 1, j - i);
583 improved = true;
584 }
585 }
586 }
587 }

593 public static List<T> OrderPointsWithPAO<T>(
594 List<T> items,
595 Func<T, Point3D> positionProvider,
596 Func<T, Vector3D> normalProvider,
597 ProbeHeadKinematicsConfig config)
598 {
599 if (items == null || items.Count <= 1 || !config.EnablePAOOptimization)
600 return items ?? new List<T>();
601
602 if (config.EnableIsoOrientationClustering)
603 {
604 // Group by discrete angles
605 var clusters = new List<(long AngleKey, Vector3D Normal, List<T> Items)>();
606
607 foreach (var item in items)
608 {
609 var norm = normalProvider(item);
610 var (a, b) = VectorToHeadAngles(norm, config);
611 long key = ((long)Math.Round(a * 10.0) * 100000L) + (long)Math.Round((b + 360.0) * 10.0);
612
613 var existing = clusters.FirstOrDefault(c =>
614 {
615 if (c.AngleKey == key) return true;
616 return AngleBetweenDeg(norm, c.Normal) <= config.AngleReuseThresholdDeg + NumericalTolerance;
617 });
618
619 if (existing.Items != null)
620 {
621 existing.Items.Add(item);
622 }
623 else
624 {
625 clusters.Add((key, norm, new List<T> { item }));
626 }
627 }
628
629 // Sort intra-cluster by spatial distance
630 for (int i = 0; i < clusters.Count; i++)
631 {
632 var list = clusters[i].Items;
633 if (list.Count > 1)
634 {
635 var orderedList = new List<T>(list.Count);
636 var rem = new List<T>(list);
637 var cur = rem[0];
638 orderedList.Add(cur);
639 rem.RemoveAt(0);
640
641 while (rem.Count > 0)
642 {
643 var pt = positionProvider(cur);
644 var next = rem.OrderBy(r => pt.DistanceTo(positionProvider(r))).First();
645 orderedList.Add(next);
646 rem.Remove(next);
647 cur = next;
648 }
649 clusters[i] = (clusters[i].AngleKey, clusters[i].Normal, orderedList);
650 }
651 }
652
653 // Sequence clusters
654 var orderedClusters = new List<List<T>>();
655 var remainingClusters = new List<(long AngleKey, Vector3D Normal, List<T> Items)>(clusters);
656
657 var curCluster = remainingClusters[0];
658 orderedClusters.Add(curCluster.Items);
659 remainingClusters.RemoveAt(0);
660
661 while (remainingClusters.Count > 0)
662 {
663 var lastItem = curCluster.Items.Last();
664 var lastPos = positionProvider(lastItem);
665 var lastNorm = normalProvider(lastItem);
666
667 var nextCluster = remainingClusters
668 .OrderBy(c =>
669 {
670 var firstItem = c.Items.First();
671 return CalculateTransitionCost(
672 lastPos, lastNorm,
673 positionProvider(firstItem), normalProvider(firstItem),
674 config);
675 })
676 .First();
677
678 orderedClusters.Add(nextCluster.Items);
679 remainingClusters.Remove(nextCluster);
680 curCluster = nextCluster;
681 }
682
683 var res = new List<T>(items.Count);
684 foreach (var c in orderedClusters)
685 res.AddRange(c);
686
687 return res;
688 }
689
690 // Fallback greedy combined cost
691 var ordered = new List<T>(items.Count);
692 var remaining = new List<T>(items);
693 var current = remaining[0];
694 ordered.Add(current);
695 remaining.RemoveAt(0);
696
697 while (remaining.Count > 0)
698 {
699 var curPos = positionProvider(current);
700 var curNorm = normalProvider(current);
701
702 var next = remaining
703 .OrderBy(r => CalculateTransitionCost(
704 curPos, curNorm,
705 positionProvider(r), normalProvider(r),
706 config))
707 .First();
708
709 ordered.Add(next);
710 remaining.Remove(next);
711 current = next;
712 }
713
714 return ordered;
715 }
