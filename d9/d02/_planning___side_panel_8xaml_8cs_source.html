129 public List<string> GetMainTreeViewItems(ObservableCollection<TreeViewItem> treeViewItems)
130 {
131 var mainItems = new List<string>();
132 if (treeViewItems == null) return mainItems;
133
134 foreach (var item in treeViewItems)
135 {
136 // Check if the header is a StackPanel (new structure with color square)
137 if (item.Header is StackPanel headerPanel)
138 {
139 // Find the TextBlock in the StackPanel (second child after Border)
140 var textBlock = headerPanel.Children.OfType<TextBlock>().FirstOrDefault();
141 if (textBlock != null)
142 {
143 // Extract the full color text including RGB (e.g., "Red (R255_G0_B0) (5 faces, 2 holes)")
144 string fullText = textBlock.Text;
145 if (!string.IsNullOrEmpty(fullText))
146 {
147 // Extract just the "ColorName (R#_G#_B#)" part (before the face/hole count)
148 // Pattern: "Red (R255_G0_B0) (5 faces, 2 holes)"
149 // ^^^^^^^^^^^^^^ we want this part
150 var match = System.Text.RegularExpressions.Regex.Match(fullText, @"^(.+?\‍(R\d+_G\d+_B\d+\‍))");
151 if (match.Success)
152 {
153 string colorWithRGB = match.Groups[1].Value.Trim();
154 mainItems.Add(colorWithRGB);
155 }
156 else
157 {
158 // Fallback: just get the color name before first parenthesis
159 int firstParenIndex = fullText.IndexOf('(');
160 string colorName = firstParenIndex > 0
161 ? fullText.Substring(0, firstParenIndex).Trim()
162 : fullText.Trim();
163
164 if (!string.IsNullOrEmpty(colorName))
165 {
166 mainItems.Add(colorName);
167 }
168 }
169 }
170 }
171 }
172 // Check if the header is a TextBlock (old structure - for backwards compatibility)
173 else if (item.Header is TextBlock headerTextBlock)
174 {
175 if (headerTextBlock.Inlines.FirstOrDefault() is Run colorRun)
176 {
177 // Get the text from the Run and trim the trailing space we added
178 string faceColorName = colorRun.Text.Trim();
179
180 if (!string.IsNullOrEmpty(faceColorName))
181 {
182 mainItems.Add(faceColorName);
183 }
184 }
185 }
186 // Fallback just in case some headers are still simple strings
187 else if (item.Header is string stringHeader)
188 {
189 if (!string.IsNullOrEmpty(stringHeader))
190 {
191 mainItems.Add(stringHeader);
192 }
193 }
194 }
195
196 return mainItems;
197 }

257 public void GenerateToleranceSliders(List<string> treeItemNames)
258 {
259 ToleranceSliders.Clear();
260
261 foreach (string itemName in treeItemNames)
262 {
263 // Remove spaces from the name for config key
264 string configKey1 = itemName.Replace(" ", "") + "_NegativeTolerance";
265 string configKey2 = itemName.Replace(" ", "") + "_PositiveTolerance";
266
267 // Get default value from ToleranceConfigManager
268 (double, double) defaultTolerance = GetTolerancesFromConfig(configKey1, configKey2);
269
270 // Get min/max Tolerance Values from the ProtectedConfig
271 double negativeToleranceThreshold = Convert.ToDouble(ProtectedConfigManager.ReadConfig("NegativeOperatorToleranceThreshold"));
272 double positiveToleranceThreshold = Convert.ToDouble(ProtectedConfigManager.ReadConfig("PositiveOperatorToleranceThreshold"));
273
274 // Create slider model
275 var sliderModel = new ToleranceSlidersViewModel
276 {
277 FaceColor = itemName,
278 SliderValue1 = defaultTolerance.Item1,
279 SliderValue2 = defaultTolerance.Item2,
280 ConfigKey1 = configKey1,
281 ConfigKey2 = configKey2,
282 DefaultValue1 = defaultTolerance.Item1,
283 DefaultValue2 = defaultTolerance.Item2,
284 MinValue = negativeToleranceThreshold,
285 MaxValue = positiveToleranceThreshold,
286 SlidersToolTip = string.Format(GetXMLElementValue(DocLanguage, "Tooltip_ToleranceSliders") ?? "Set the negative and positive tolerance values from ({0} to {1})mm for the {2} colored faces.", negativeToleranceThreshold, positiveToleranceThreshold, ConvertColorToRGBFormat(itemName)),
287 };
288
289 ToleranceSliders.Add(sliderModel);
290 }

314 private (double, double) GetTolerancesFromConfig(string configKey1, string configKey2)
315 {
316 double tolerance1 = 0;
317 double tolerance2 = 0;
318
319 try
320 {
321 // --- Process the first key ---
322 string? value1 = ToleranceConfigManager.GetValue(configKey1);
323 if (!string.IsNullOrEmpty(value1) && double.TryParse(value1, out double parsedValue1))
324 {
325 // Ensure the value is within the allowed range [0, 5]
326 tolerance1 = Math.Max(0, Math.Min(5, parsedValue1));
327 }
328
329 // --- Process the second key ---
330 string? value2 = ToleranceConfigManager.GetValue(configKey2);
331 if (!string.IsNullOrEmpty(value2) && double.TryParse(value2, out double parsedValue2))
332 {
333 // Ensure the value is within the allowed range [0, 5]
334 tolerance2 = Math.Max(0, Math.Min(5, parsedValue2));
335 }
336 }
337 catch (Exception ex)
338 {
339 // Use a more general error message since the error could be from either key
340 MessageBox.Show($"Error reading tolerance configuration: {ex.Message}",
341 GetXMLElementValue(DocLanguage, "MessageBox_Error") ?? "Error",
342 MessageBoxButton.OK, MessageBoxImage.Warning);
343
344 // Return default values in case of any exception
345 return (0, 0);
346 }
347
348 // Return the retrieved (or default) tolerance values
349 return (tolerance1, tolerance2);
350 }

579 private static T? FindVisualChild<T>(DependencyObject obj) where T : DependencyObject
580 {
581 if (obj == null) return null;
582 for (int i = 0; i < System.Windows.Media.VisualTreeHelper.GetChildrenCount(obj); i++)
583 {
584 var child = System.Windows.Media.VisualTreeHelper.GetChild(obj, i);
585 if (child != null && child is T t)
586 return t;
587 else
588 {
589 var childOfChild = FindVisualChild<T>(child!);
590 if (childOfChild != null)
591 return childOfChild;
592 }
593 }
594 return null;
