338 private void LoadScannerProperties(string scannerType)
339 {
340 string scannerKey = scannerType;
341
342 if (_scannerConfigurations.TryGetValue(scannerKey, out var props))
343 {
344 _originalProperties = props;
345 FovX = props.FovX;
346 FovY = props.FovY;
347 OptimalStandoff = props.OptimalStandoff;
348 MinStandoff = props.MinStandoff;
349 MaxStandoff = props.MaxStandoff;
350 RaycastingTolerance = props.RaycastingTolerance;
351
352 // Load sensor transformation properties
353 SensorOffsetX = props.SensorOffsetX;
354 SensorOffsetY = props.SensorOffsetY;
355 SensorOffsetZ = props.SensorOffsetZ;
356 SensorRotationX = props.SensorRotationX;
357 SensorRotationY = props.SensorRotationY;
358 SensorRotationZ = props.SensorRotationZ;
359 SensorWidth = props.SensorWidth;
360 SensorHeight = props.SensorHeight;
361 FovAngleHorizontal = props.FovAngleHorizontal;
362 FovAngleVertical = props.FovAngleVertical;
363 }
364 else
365 {
366 // Use default values
367 _originalProperties = null;
368 FovX = 100.0;
369 FovY = 80.0;
370 OptimalStandoff = 150.0;
371 MinStandoff = 100.0;
372 MaxStandoff = 200.0;
373 RaycastingTolerance = 0.5;
374 SensorOffsetX = 0.0;
375 SensorOffsetY = 0.0;
376 SensorOffsetZ = 0.0;
377 SensorRotationX = 0.0;
378 SensorRotationY = 0.0;
379 SensorRotationZ = 0.0;
380 SensorWidth = 20.0;
381 SensorHeight = 15.0;
382
383 // Calculate FOV angles
384 FovAngleHorizontal = 2.0 * Math.Atan(FovX / (2.0 * OptimalStandoff)) * (180.0 / Math.PI);
385 FovAngleVertical = 2.0 * Math.Atan(FovY / (2.0 * OptimalStandoff)) * (180.0 / Math.PI);
386 }

404 private string GetLanguageValue(string key, string fallback = "")
405 {
406 try
407 {
408 if (DocLanguage == null) return fallback ?? string.Empty;
409 var nodes = DocLanguage.GetElementsByTagName(key);
410 if (nodes != null && nodes.Count > 0)
411 {
412 if (nodes[0] is XmlNode node)
413 {
414 var attr = node.Attributes?["Value"];
415 if (attr != null) return attr.Value ?? fallback ?? string.Empty;
416 }
417 }
418 }
419 catch
420 {
421 // ignore and return fallback
422 }
423 return fallback ?? string.Empty;
424 }

429 private void SaveConfiguration()
430 {
431 try
432 {
433 // Validate inputs
434 if (FovX <= 0 || FovY <= 0)
435 {
436 string errorTitle = GetLanguageValue("MessageBox_DefaultTitle", "Error");
437 string errorMessage = GetLanguageValue("ScannerConfig_Error_FovZero", "FOV dimensions must be greater than zero.");
438 MessageBox.Show(errorMessage, errorTitle, MessageBoxButton.OK, MessageBoxImage.Error);
439 return;
440 }
441
442 if (MinStandoff > OptimalStandoff || OptimalStandoff > MaxStandoff)
443 {
444 string errorTitle = GetLanguageValue("MessageBox_DefaultTitle", "Error");
445 string errorMessage = GetLanguageValue("ScannerConfig_Error_StandoffRange", "Standoff distances must be: Minimum ≤ Optimal ≤ Maximum");
446 MessageBox.Show(errorMessage, errorTitle, MessageBoxButton.OK, MessageBoxImage.Error);
447 return;
448 }
449
450 if (RaycastingTolerance < 0)
451 {
452 string errorTitle = GetLanguageValue("MessageBox_DefaultTitle", "Error");
453 string errorMessage = GetLanguageValue("ScannerConfig_Error_RaycastingNegative", "Raycasting tolerance cannot be negative.");
454 MessageBox.Show(errorMessage, errorTitle, MessageBoxButton.OK, MessageBoxImage.Error);
455 return;
456 }
457
458 // Create updated properties
459 var updatedProperties = new ScannerProperties
460 {
461 Name = _originalProperties?.Name ?? SelectedScannerType.ToString(),
462 FovX = FovX,
463 FovY = FovY,
464 OptimalStandoff = OptimalStandoff,
465 MinStandoff = MinStandoff,
466 MaxStandoff = MaxStandoff,
467 AspectRatio = AspectRatio,
468 RaycastingTolerance = RaycastingTolerance,
469 ModelFile = _originalProperties?.ModelFile ?? "",
470
471 // Sensor transformation settings
472 SensorOffsetX = SensorOffsetX,
473 SensorOffsetY = SensorOffsetY,
474 SensorOffsetZ = SensorOffsetZ,
475 SensorRotationX = SensorRotationX,
476 SensorRotationY = SensorRotationY,
477 SensorRotationZ = SensorRotationZ,
478 SensorWidth = SensorWidth,
479 SensorHeight = SensorHeight,
480 FovAngleHorizontal = HorizontalAngle,
481 FovAngleVertical = VerticalAngle
482 };
483
484 // Save to file
485 ScannerConfigurationSaver.SaveScannerConfiguration(_configFilePath, SelectedScannerType.ToString(), updatedProperties);
486 SaveSelectedScannerType(SelectedScannerType.ToString());
487
488 // Update in-memory configuration
489 _scannerConfigurations[SelectedScannerType.ToString()] = updatedProperties;
490
491 // Show success message
492 string successTitle = GetLanguageValue("MessageBox_DefaultTitle", "Success");
493 string template = GetLanguageValue("ScannerConfig_Success_ConfigSaved", "Scanner configuration for '{0}' saved successfully.");
494 string successMessage = string.Format(template, SelectedScannerType);
495 MessageBox.Show(successMessage, successTitle, MessageBoxButton.OK, MessageBoxImage.Information);
496
497 DialogResult = true;
498 Close();
499 }
500 catch (Exception ex)
501 {
502 string errorTitle = GetLanguageValue("MessageBox_DefaultTitle", "Error");
503 string template = GetLanguageValue("ScannerConfig_Error_SaveFailed", "Failed to save configuration: {0}");
504 string errorMessage = string.Format(template, ex.Message);
505 MessageBox.Show(errorMessage, errorTitle, MessageBoxButton.OK, MessageBoxImage.Error);
506 }
507 }

513 private static void SaveSelectedScannerType(string scannerType)
514 {
515 if (string.IsNullOrWhiteSpace(scannerType))
516 return;
517
518 try
519 {
520 var config = ConfigurationManager.OpenExeConfiguration(ConfigurationUserLevel.None);
521 if (config.AppSettings.Settings["Scanner_Type"] == null)
522 config.AppSettings.Settings.Add("Scanner_Type", scannerType);
523 else
524 config.AppSettings.Settings["Scanner_Type"].Value = scannerType;
525
526 config.Save(ConfigurationSaveMode.Modified);
527 ConfigurationManager.RefreshSection("appSettings");
528 }
529 catch (Exception ex)
530 {
531 System.Diagnostics.Debug.WriteLine($"⚠️ Could not persist selected scanner type: {ex.Message}");
532 }
533 }
