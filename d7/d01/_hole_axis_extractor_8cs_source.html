106 public static HoleAxisInfo? DetectHoleAxis(CADHole hole)
107 {
108 if (hole == null)
109 return null;
110
111 // Use pre-computed hole properties if available
112 if (hole.TopCenterPoint != null && hole.BottomCenterPoint != null && hole.AxisDirection != null)
113 {
114 Debug.WriteLine($" ✅ Using pre-computed hole axis for Hole {hole.HoleId}");
115 return new HoleAxisInfo(
116 hole.TopCenterPoint,
117 hole.BottomCenterPoint,
118 hole.AxisDirection,
119 hole.TopDiameter,
120 hole.BottomDiameter);
121 }
122
123 Debug.WriteLine($" ⚠️ Hole {hole.HoleId} missing axis data, attempting reconstruction...");
124
125 // Fallback: try to reconstruct from geometric info
126 // This is a simplified reconstruction - in production, would need more robust logic
127 if (!string.IsNullOrEmpty(hole.GeometricInfo))
128 {
129 // Try to parse axis from GeometricInfo
130 var axis = TryParseAxisFromGeometricInfo(hole.GeometricInfo);
131 if (axis != null)
132 {
133 // Estimate top/bottom from depth
134 Point3D estimatedTop = new Point3D(0, 0, 0);
135 Point3D estimatedBottom = new Point3D(
136 estimatedTop.X + axis.X * hole.Depth,
137 estimatedTop.Y + axis.Y * hole.Depth,
138 estimatedTop.Z + axis.Z * hole.Depth);
139
140 return new HoleAxisInfo(estimatedTop, estimatedBottom, axis, hole.TopDiameter, hole.BottomDiameter);
141 }
142 }
143
144 return null;
145 }

150 private static Vector3D? TryParseAxisFromGeometricInfo(string geometricInfo)
151 {
152 if (string.IsNullOrEmpty(geometricInfo))
153 return null;
154
155 try
156 {
157 if (geometricInfo.Contains("Axis:"))
158 {
159 int startIdx = geometricInfo.IndexOf("Axis:");
160 startIdx = geometricInfo.IndexOf('(', startIdx);
161 int endIdx = geometricInfo.IndexOf(')', startIdx);
162 if (startIdx > 0 && endIdx > startIdx)
163 {
164 string coords = geometricInfo.Substring(startIdx + 1, endIdx - startIdx - 1);
165 string[] parts = coords.Split(',');
166 if (parts.Length >= 3)
167 {
168 double x = double.Parse(parts[0].Trim(), System.Globalization.CultureInfo.InvariantCulture);
169 double y = double.Parse(parts[1].Trim(), System.Globalization.CultureInfo.InvariantCulture);
170 double z = double.Parse(parts[2].Trim(), System.Globalization.CultureInfo.InvariantCulture);
171 var axis = new Vector3D(x, y, z);
172 if (axis.Length > 0.0001)
173 {
174 axis.Normalize();
175 return axis;
176 }
177 }
178 }
179 }
180 }
181 catch
182 {
183 // Parsing failed
184 }
185
186 return null;
187 }

196 public static Point3D ComputeAxisEntranceFromShell(HoleAxisInfo holeAxis, double shellSafeDistance = 5.0)
197 {
198 // The entrance point is along the axis, above the hole top by the safe distance
199 return new Point3D(
200 holeAxis.AxisTop.X - holeAxis.AxisDirection.X * shellSafeDistance,
201 holeAxis.AxisTop.Y - holeAxis.AxisDirection.Y * shellSafeDistance,
202 holeAxis.AxisTop.Z - holeAxis.AxisDirection.Z * shellSafeDistance);
203 }

218 public static List<HoleProbeStep> GenerateHoleProbeSequence(
219 CADHole hole,
220 List<Point3D> measurementPoints,
221 HoleAxisInfo? axisInfo = null,
222 double shellSafeDistance = 5.0)
223 {
224 var steps = new List<HoleProbeStep>();
225
226 if (measurementPoints == null || measurementPoints.Count == 0)
227 return steps;
228
229 // Get or compute axis info
230 axisInfo ??= DetectHoleAxis(hole);
231 if (axisInfo == null)
232 {
233 Debug.WriteLine($" ❌ Cannot generate hole sequence: axis detection failed for Hole {hole?.HoleId}");
234 return steps;
235 }
236
237 Debug.WriteLine($"\n 🔧 Generating hole probe sequence for Hole {hole?.HoleId}: {measurementPoints.Count} points");
238
239 // Step 1: Move to axis entrance (on shell, above hole)
240 Point3D axisEntrance = ComputeAxisEntranceFromShell(axisInfo, shellSafeDistance);
241 steps.Add(new HoleProbeStep(HoleProbeStepType.MoveToAxisEntrance, axisEntrance, Vector3D.AxisZ));
242
243 // Sort measurement points by depth along axis (process from top to bottom)
244 var sortedPoints = measurementPoints
245 .OrderBy(p => GetDepthAlongAxis(p, axisInfo))
246 .ToList();
247
248 Point3D lastAxisPosition = axisEntrance;
249
250 foreach (var measurePt in sortedPoints)
251 {
252 // Calculate axis position at the Z level of this measurement
253 Point3D axisAtMeasureZ = GetAxisPointAtZ(axisInfo, measurePt.Z);
254
255 // Step 2a: Move along axis to measurement Z level (if different from last position)
256 if (lastAxisPosition.DistanceTo(axisAtMeasureZ) > 0.1)
257 {
258 steps.Add(new HoleProbeStep(HoleProbeStepType.MoveAlongAxis, axisAtMeasureZ, axisInfo.AxisDirection));
259 }
260
261 // Step 2b: Move radially from axis to measurement point
262 Vector3D radialDir = ComputeRadialDirection(axisAtMeasureZ, measurePt);
263 steps.Add(new HoleProbeStep(HoleProbeStepType.MoveRadialToMeasurement, measurePt, radialDir));
264
265 // Step 2c: Move radially back to axis (after measurement)
266 steps.Add(new HoleProbeStep(HoleProbeStepType.MoveRadialToAxis, axisAtMeasureZ, radialDir));
267
268 lastAxisPosition = axisAtMeasureZ;
269 }
270
271 // Step 3: Exit hole via axis
272 steps.Add(new HoleProbeStep(HoleProbeStepType.ExitToShell, axisEntrance, Vector3D.AxisZ));
273
274 Debug.WriteLine($" ✅ Generated {steps.Count} probe steps for hole");
275 return steps;
276 }

281 private static double GetDepthAlongAxis(Point3D point, HoleAxisInfo axis)
282 {
283 Vector3D fromTop = new Vector3D(
284 point.X - axis.AxisTop.X,
285 point.Y - axis.AxisTop.Y,
286 point.Z - axis.AxisTop.Z);
287
288 return Vector3D.Dot(fromTop, axis.AxisDirection);
289 }
290
295 private static Point3D GetAxisPointAtZ(HoleAxisInfo axis, double z)
296 {
297 // Simple case: vertical axis (common)
298 if (Math.Abs(axis.AxisDirection.Z) > 0.99)
299 {
300 return new Point3D(axis.AxisTop.X, axis.AxisTop.Y, z);
301 }
302
303 // General case: project Z onto axis
304 double t = (z - axis.AxisTop.Z) / axis.AxisDirection.Z;
305 return new Point3D(
306 axis.AxisTop.X + axis.AxisDirection.X * t,
307 axis.AxisTop.Y + axis.AxisDirection.Y * t,
308 z);
309 }

314 private static Vector3D ComputeRadialDirection(Point3D axisPoint, Point3D measurePoint)
315 {
316 var radial = new Vector3D(
317 measurePoint.X - axisPoint.X,
318 measurePoint.Y - axisPoint.Y,
319 measurePoint.Z - axisPoint.Z);
320
321 if (radial.Length > 0.0001)
322 radial.Normalize();
323 else
324 radial = Vector3D.AxisX;
325
326 return radial;
327 }
