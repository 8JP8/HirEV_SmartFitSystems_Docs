173 private void ValidateGraphEdgesForSafety(System.Threading.CancellationToken cancellationToken = default)
174 {
175 if (_navigationGraph == null || _partBrep == null) return;

203 private bool IsSegmentSafe(Point3D p1, Point3D p2, double minClearance)
204 {
205 if (_partBrep == null) return true;
206
207 Point3D mid = new Point3D((p1.X + p2.X) * 0.5, (p1.Y + p2.Y) * 0.5, (p1.Z + p2.Z) * 0.5);
208 if (GetDistanceToPart(mid) < minClearance) return false;
209
210 Point3D q1 = new Point3D((p1.X + mid.X) * 0.5, (p1.Y + mid.Y) * 0.5, (p1.Z + mid.Z) * 0.5);
211 if (GetDistanceToPart(q1) < minClearance) return false;
212
213 Point3D q3 = new Point3D((mid.X + p2.X) * 0.5, (mid.Y + p2.Y) * 0.5, (mid.Z + p2.Z) * 0.5);
214 if (GetDistanceToPart(q3) < minClearance) return false;
215
216 return true;
217 }

252 private void BuildNavigationGraph()
253 {
254 _navigationGraph = new Dictionary<int, List<(int, double)>>();
255 if (_approachPoints == null || _approachPoints.Count < 2) return;
256
257 int n = _approachPoints.Count;
258 int k = Math.Min(Math.Max(K_NEIGHBORS, n / 10), n - 1);
259
260 for (int i = 0; i < n; i++)
261 {
262 _navigationGraph[i] = new List<(int, double)>();
263 var distances = new List<(int idx, double dist)>();
264 for (int j = 0; j < n; j++)
265 if (i != j) distances.Add((j, _approachPoints[i].DistanceTo(_approachPoints[j])));
266
267 foreach (var neighbor in distances.OrderBy(d => d.dist).Take(k))
268 _navigationGraph[i].Add((neighbor.idx, neighbor.dist));
269 }
270
271 for (int i = 0; i < n; i++)
272 foreach (var (neighbor, dist) in _navigationGraph[i].ToList())
273 if (!_navigationGraph[neighbor].Any(x => x.Item1 == i)) _navigationGraph[neighbor].Add((i, dist));

325 {
326 if (_faceAdjacencyMap == null || _triangleToFaceMap == null || _shellMesh == null) return;
327 var faceToVertices = new Dictionary<int, HashSet<int>>();
328 for (int triIdx = 0; triIdx < _shellMesh.Triangles.Length; triIdx++)
329 {
330 if (_triangleToFaceMap.TryGetValue(triIdx, out int faceIdx))
331 {
332 if (!faceToVertices.ContainsKey(faceIdx)) faceToVertices[faceIdx] = new HashSet<int>();

334 faceToVertices[faceIdx].Add(tri.V1); faceToVertices[faceIdx].Add(tri.V2); faceToVertices[faceIdx].Add(tri.V3);

365 private void EnsureGraphConnectivity(System.Threading.CancellationToken cancellationToken = default)
366 {
367 if (_navigationGraph == null || _navigationGraph.Count == 0) return;
368 int n = _approachPoints.Count;
369 int[] parent = new int[n];
370 for (int i = 0; i < n; i++) parent[i] = i;
371
372 int Find(int x) => parent[x] == x ? x : (parent[x] = Find(parent[x]));
373 void Union(int x, int y) { int px = Find(x), py = Find(y); if (px != py) parent[px] = py; }
374
375 foreach (var kvp in _navigationGraph)
376 {
377 cancellationToken.ThrowIfCancellationRequested();
378 foreach (var (neighbor, _) in kvp.Value) Union(kvp.Key, neighbor);
379 }
380
381 var components = new Dictionary<int, List<int>>();
382 for (int i = 0; i < n; i++)
383 {
384 int root = Find(i);
385 if (!components.ContainsKey(root)) components[root] = new List<int>();
386 components[root].Add(i);
387 }
388
389 if (components.Count <= 1) return;
390 var componentList = components.Values.ToList();
391 for (int c1 = 0; c1 < componentList.Count - 1; c1++)
392 {
393 cancellationToken.ThrowIfCancellationRequested();
394 int c2 = c1 + 1;
395 double minDist = double.MaxValue;
396 int bestI = -1, bestJ = -1;
397 foreach (int i in componentList[c1])
398 foreach (int j in componentList[c2])
399 {
400 double dist = _approachPoints[i].DistanceTo(_approachPoints[j]);
401 if (dist < minDist) { minDist = dist; bestI = i; bestJ = j; }
402 }
403 if (bestI >= 0) { _navigationGraph[bestI].Add((bestJ, minDist)); _navigationGraph[bestJ].Add((bestI, minDist)); Union(bestI, bestJ); componentList[c2].AddRange(componentList[c1]); }
404 }

415 public List<Point3D> FindShortestPath(Point3D startPoint, Point3D endPoint, int numIntermediatePoints = 5, System.Threading.CancellationToken cancellationToken = default)
416 {
417 if (_navigationGraph == null || _navigationGraph.Count == 0 || _approachPoints.Count < 2) return new List<Point3D> { startPoint, endPoint };
418 int startIdx = FindNearestApproachPoint(startPoint);
419 int endIdx = FindNearestApproachPoint(endPoint);
420 if (startIdx == -1 || endIdx == -1) return new List<Point3D> { startPoint, endPoint };
421 if (startPoint.DistanceTo(endPoint) < _directTravelThreshold) return new List<Point3D> { startPoint, endPoint };
422 if (startIdx == endIdx) return new List<Point3D> { startPoint, _approachPoints[startIdx], endPoint };
423
424 var path = DijkstraShortestPath(startIdx, endIdx, cancellationToken);
425 if (path == null) return new List<Point3D> { startPoint, endPoint };
426
427 var finalPath = new List<Point3D> { startPoint };
428 foreach (var idx in path)
429 if (finalPath.Last().DistanceTo(_approachPoints[idx]) > 1.0) finalPath.Add(_approachPoints[idx]);
430 if (finalPath.Last().DistanceTo(endPoint) > 1.0) finalPath.Add(endPoint);
431
432 return InterpolateWithMeshPoints(finalPath, cancellationToken: cancellationToken);
433 }

442 private List<Point3D> InterpolateWithMeshPoints(List<Point3D> rawPath, double maxCorridor = 10.0, System.Threading.CancellationToken cancellationToken = default)
443 {
444 var result = new List<Point3D> { rawPath[0] };
445 for (int i = 0; i < rawPath.Count - 1; i++)
446 {
447 cancellationToken.ThrowIfCancellationRequested();
448 var p1 = rawPath[i]; var p2 = rawPath[i + 1];
449 var dir = new Vector3D(p1, p2); double len = dir.Length;
450 if (len > 2.0 && _approachPoints != null)
451 {
452 dir.Normalize();
453 var candidates = new List<(double d, Point3D p)>();
454 foreach (var ap in _approachPoints)
455 {
456 Vector3D v = new Vector3D(p1, ap); double d = Vector3D.Dot(v, dir);
457 if (d > 1.0 && d < len - 1.0 && ap.DistanceTo(p1 + dir * d) <= maxCorridor) candidates.Add((d, ap));
458 }
459 foreach (var c in candidates.OrderBy(x => x.d))
460 if (c.p.DistanceTo(result.Last()) > 2.0) result.Add(c.p);
461 }
462 result.Add(p2);
463 }
464 return result;
465 }

491 private List<int>? DijkstraShortestPath(int startIdx, int endIdx, System.Threading.CancellationToken cancellationToken = default)
492 {
493 if (_navigationGraph == null || !_navigationGraph.ContainsKey(startIdx)) return null;
494 var dists = new Dictionary<int, double>();
495 var prevs = new Dictionary<int, int>();
496 var pq = new SortedSet<(double d, int n)>(Comparer<(double d, int n)>.Create((a, b) => a.d != b.d ? a.d.CompareTo(b.d) : a.n.CompareTo(b.n)));
497
498 foreach (var n in _navigationGraph.Keys) dists[n] = double.MaxValue;
499 dists[startIdx] = 0; pq.Add((0, startIdx));
500
501 while (pq.Count > 0)
502 {
503 cancellationToken.ThrowIfCancellationRequested();
504 var curr = pq.Min; pq.Remove(pq.Min);
505 if (curr.d > dists[curr.n]) continue;
506 if (curr.n == endIdx)
507 {
508 var path = new List<int>(); int c = endIdx;
509 while (prevs.TryGetValue(c, out int p)) { path.Insert(0, c); c = p; }
510 path.Insert(0, startIdx); return path;
511 }
512 if (_navigationGraph.TryGetValue(curr.n, out var neighbors))
513 foreach (var (nb, ed) in neighbors)
514 {
515 double nd = dists[curr.n] + ed;
516 if (nd < dists[nb]) { pq.Remove((dists[nb], nb)); dists[nb] = nd; prevs[nb] = curr.n; pq.Add((nd, nb)); }
517 }
518 }
519 return null;
520 }
