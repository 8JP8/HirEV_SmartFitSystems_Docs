342 private bool IsSegmentSafe(Point3D p1, Point3D p2, double minClearance)
343 {
344 if (_shellMesh == null) return true;
345
346 Point3D mid = new Point3D((p1.X + p2.X) * 0.5, (p1.Y + p2.Y) * 0.5, (p1.Z + p2.Z) * 0.5);
347 if (!IsPointSafe(mid, minClearance)) return false;
348
349 Point3D q1 = new Point3D((p1.X + mid.X) * 0.5, (p1.Y + mid.Y) * 0.5, (p1.Z + mid.Z) * 0.5);
350 if (!IsPointSafe(q1, minClearance)) return false;
351
352 Point3D q3 = new Point3D((mid.X + p2.X) * 0.5, (mid.Y + p2.Y) * 0.5, (mid.Z + p2.Z) * 0.5);
353 if (!IsPointSafe(q3, minClearance)) return false;
354
355 return true;
356 }

361 private bool IsPointSafe(Point3D pt, double minClearance)
362 {
363 if (_shellMesh == null) return true;
364
365 double maxDist = _safetyOffset > 0 ? Math.Max(1.0, _safetyOffset - minClearance) : 1.5;
366 double maxDistSq = maxDist * maxDist;
367
368 if (_shellMeshVertexGrid != null)
369 {
370 int cx = (int)Math.Floor(pt.X / SHELL_MESH_GRID_CELL_SIZE);
371 int cy = (int)Math.Floor(pt.Y / SHELL_MESH_GRID_CELL_SIZE);
372 int cz = (int)Math.Floor(pt.Z / SHELL_MESH_GRID_CELL_SIZE);
373
374 int radius = Math.Max(1, (int)Math.Ceiling(maxDist / SHELL_MESH_GRID_CELL_SIZE));
375
376 for (int dx = -radius; dx <= radius; dx++)
377 {
378 for (int dy = -radius; dy <= radius; dy++)
379 {
380 for (int dz = -radius; dz <= radius; dz++)
381 {
382 if (_shellMeshVertexGrid.TryGetValue((cx + dx, cy + dy, cz + dz), out var list))
383 {
384 foreach (var v in list)
385 {
386 double dSq = (pt.X - v.X) * (pt.X - v.X) +
387 (pt.Y - v.Y) * (pt.Y - v.Y) +
388 (pt.Z - v.Z) * (pt.Z - v.Z);
389 if (dSq <= maxDistSq)
390 {
391 return true; // Found a shell vertex within clearance limit - safe!
392 }
393 }
394 }
395 }
396 }
397 }
398
399 return false;
400 }
401
402 var nearestOnShell = ShellProjectionUtils.FindNearestShellPoint(pt, _shellMesh);
403 double distToShell = pt.DistanceTo(nearestOnShell);
404 return distToShell <= maxDist;
405 }

427 private void BuildNavigationGraph()
428 {
429 _navigationGraph = new Dictionary<int, List<(int, double)>>();
430 if (_approachPoints == null || _approachPoints.Count < 2) return;
431
432 int n = _approachPoints.Count;
433 int k = Math.Min(Math.Max(K_NEIGHBORS, n / 10), n - 1);
434
435 for (int i = 0; i < n; i++)
436 {
437 _navigationGraph[i] = new List<(int, double)>();
438 var distances = new List<(int idx, double dist)>();
439 for (int j = 0; j < n; j++)
440 if (i != j) distances.Add((j, _approachPoints[i].DistanceTo(_approachPoints[j])));
441
442 foreach (var neighbor in distances.OrderBy(d => d.dist).Take(k))
443 _navigationGraph[i].Add((neighbor.idx, neighbor.dist));
444 }
445
446 for (int i = 0; i < n; i++)
447 foreach (var (neighbor, dist) in _navigationGraph[i].ToList())
448 if (!_navigationGraph[neighbor].Any(x => x.Item1 == i)) _navigationGraph[neighbor].Add((i, dist));
449
450 EnsureGraphConnectivity();
451 }

499 private void AddCrossFaceConnections(HashSet<(int, int)> addedEdges)
500 {
501 if (_faceAdjacencyMap == null || _triangleToFaceMap == null || _shellMesh == null) return;
502 var faceToVertices = new Dictionary<int, HashSet<int>>();
503 for (int triIdx = 0; triIdx < _shellMesh.Triangles.Length; triIdx++)
504 {
505 if (_triangleToFaceMap.TryGetValue(triIdx, out int faceIdx))
506 {
507 if (!faceToVertices.ContainsKey(faceIdx)) faceToVertices[faceIdx] = new HashSet<int>();
508 var tri = _shellMesh.Triangles[triIdx];
509 faceToVertices[faceIdx].Add(tri.V1); faceToVertices[faceIdx].Add(tri.V2); faceToVertices[faceIdx].Add(tri.V3);
510 }
511 }
512
513 foreach (var kvp in _faceAdjacencyMap)
514 {
515 if (!faceToVertices.TryGetValue(kvp.Key, out var verts1)) continue;
516 foreach (int face2 in kvp.Value)
517 {
518 if (!faceToVertices.TryGetValue(face2, out var verts2)) continue;
519 foreach (int v1 in verts1)
520 {
521 var pt1 = _shellMesh.Vertices[v1];
522 foreach (int v2 in verts2)
523 {
524 double dist = pt1.DistanceTo(_shellMesh.Vertices[v2]);
525 if (dist < 15.0)
526 {
527 if (IsSegmentSafe(pt1, _shellMesh.Vertices[v2], _safetyOffset > 0 ? _safetyOffset * 0.45 : 1.0))
528 {
529 var edge = v1 < v2 ? (v1, v2) : (v2, v1);
530 if (!addedEdges.Contains(edge)) AddMeshEdge(v1, v2, addedEdges);
531 }
532 }
533 }
534 }
535 }

544 private void EnsureGraphConnectivity(System.Threading.CancellationToken cancellationToken = default)
545 {
546 if (_navigationGraph == null || _navigationGraph.Count == 0) return;
547 int n = _approachPoints.Count;
548 int[] parent = new int[n];
549 for (int i = 0; i < n; i++) parent[i] = i;
550
551 int Find(int x) => parent[x] == x ? x : (parent[x] = Find(parent[x]));
552 void Union(int x, int y) { int px = Find(x), py = Find(y); if (px != py) parent[px] = py; }
553
554 foreach (var kvp in _navigationGraph)
555 {
556 cancellationToken.ThrowIfCancellationRequested();
557 foreach (var (neighbor, _) in kvp.Value) Union(kvp.Key, neighbor);
558 }
559
560 var components = new Dictionary<int, List<int>>();
561 for (int i = 0; i < n; i++)
562 {
563 int root = Find(i);
564 if (!components.ContainsKey(root)) components[root] = new List<int>();
565 components[root].Add(i);
566 }
567
568 if (components.Count <= 1) return;
569 var componentList = components.Values.ToList();
570 double minClearance = _safetyOffset > 0 ? _safetyOffset * 0.45 : 1.0;
571
572 // Maximum distance across which shell graph components can be bridged.
573 // Components further apart than this (e.g. separate parts in a multipart setup)
574 // must NOT be bridged on the shell graph, as part transitions are handled via AirTop bridge waypoints.
575 double maxBridgeDistance = Math.Max(30.0, _safetyOffset * 2.5);
576 double maxBridgeDistSq = maxBridgeDistance * maxBridgeDistance;
577
578 // Precompute bounding boxes for each component for O(1) early-rejection
579 var compBoxes = new (Point3D Min, Point3D Max)[componentList.Count];
580 for (int c = 0; c < componentList.Count; c++)
581 {
582 double minX = double.MaxValue, minY = double.MaxValue, minZ = double.MaxValue;
583 double maxX = double.MinValue, maxY = double.MinValue, maxZ = double.MinValue;
584 foreach (int idx in componentList[c])
585 {
586 var pt = _approachPoints[idx];
587 if (pt.X < minX) minX = pt.X;
588 if (pt.Y < minY) minY = pt.Y;
589 if (pt.Z < minZ) minZ = pt.Z;
590 if (pt.X > maxX) maxX = pt.X;
591 if (pt.Y > maxY) maxY = pt.Y;
592 if (pt.Z > maxZ) maxZ = pt.Z;
593 }
594 compBoxes[c] = (new Point3D(minX, minY, minZ), new Point3D(maxX, maxY, maxZ));
595 }
596
597 for (int c1 = 0; c1 < componentList.Count - 1; c1++)
598 {
599 cancellationToken.ThrowIfCancellationRequested();
600 for (int c2 = c1 + 1; c2 < componentList.Count; c2++)
601 {
602 if (Find(componentList[c1][0]) == Find(componentList[c2][0]))
603 continue;
604
605 // Check distance between component bounding boxes
606 double dx = Math.Max(0.0, Math.Max(compBoxes[c1].Min.X - compBoxes[c2].Max.X, compBoxes[c2].Min.X - compBoxes[c1].Min.X));
607 double dy = Math.Max(0.0, Math.Max(compBoxes[c1].Min.Y - compBoxes[c2].Max.Y, compBoxes[c2].Min.Y - compBoxes[c1].Min.Y));
608 double dz = Math.Max(0.0, Math.Max(compBoxes[c1].Min.Z - compBoxes[c2].Max.Z, compBoxes[c2].Min.Z - compBoxes[c1].Min.Z));
609 double boxDistSq = dx * dx + dy * dy + dz * dz;
610
611 // If bounding boxes are separated by more than maxBridgeDistance, they cannot be bridged
612 if (boxDistSq > maxBridgeDistSq)
613 {
614 continue;
615 }
616
617 // Only gather candidate pairs that are within maxBridgeDistance
618 var candidates = new List<(int i, int j, double dist)>();
619 foreach (int i in componentList[c1])
620 {
621 var p1 = _approachPoints[i];
622 foreach (int j in componentList[c2])
623 {
624 var p2 = _approachPoints[j];
625 double dSq = (p1.X - p2.X) * (p1.X - p2.X) +
626 (p1.Y - p2.Y) * (p1.Y - p2.Y) +
627 (p1.Z - p2.Z) * (p1.Z - p2.Z);
628 if (dSq <= maxBridgeDistSq)
629 {
630 candidates.Add((i, j, Math.Sqrt(dSq)));
631 }
632 }
633 }
634
635 if (candidates.Count == 0) continue;
636
637 candidates.Sort((a, b) => a.dist.CompareTo(b.dist));
638
639 int bestI = -1, bestJ = -1;
640 double minDist = double.MaxValue;
641 int testCount = Math.Min(10, candidates.Count);
642
643 for (int k = 0; k < testCount; k++)
644 {
645 var candidate = candidates[k];
646 if (IsSegmentSafe(_approachPoints[candidate.i], _approachPoints[candidate.j], minClearance))
647 {
648 bestI = candidate.i;
649 bestJ = candidate.j;
650 minDist = candidate.dist;
651 break;
652 }
653 }
654
655 if (bestI >= 0)
656 {
657 _navigationGraph[bestI].Add((bestJ, minDist));
658 _navigationGraph[bestJ].Add((bestI, minDist));
659 Union(bestI, bestJ);
660 componentList[c2].AddRange(componentList[c1]);
661 compBoxes[c2] = (
662 new Point3D(Math.Min(compBoxes[c1].Min.X, compBoxes[c2].Min.X), Math.Min(compBoxes[c1].Min.Y, compBoxes[c2].Min.Y), Math.Min(compBoxes[c1].Min.Z, compBoxes[c2].Min.Z)),
663 new Point3D(Math.Max(compBoxes[c1].Max.X, compBoxes[c2].Max.X), Math.Max(compBoxes[c1].Max.Y, compBoxes[c2].Max.Y), Math.Max(compBoxes[c1].Max.Z, compBoxes[c2].Max.Z))
664 );
665 }
666 }
667 }

678 public List<Point3D>? FindShortestPath(Point3D startPoint, Point3D endPoint, int numIntermediatePoints = 5, System.Threading.CancellationToken cancellationToken = default)
679 {
680 if (_navigationGraph == null || _navigationGraph.Count == 0 || _approachPoints.Count < 2) return null;
681 int startIdx = FindNearestApproachPoint(startPoint);
682 int endIdx = FindNearestApproachPoint(endPoint);
683 if (startIdx == -1 || endIdx == -1) return null;
684
685 double minClearance = _safetyOffset > 0 ? _safetyOffset * 0.45 : 1.0;
686 if (startPoint.DistanceTo(endPoint) < _directTravelThreshold)
687 {
688 if (IsSegmentSafe(startPoint, endPoint, minClearance))
689 {
690 return new List<Point3D> { startPoint, endPoint };
691 }
692 }
693
694 if (startIdx == endIdx) return new List<Point3D> { startPoint, _approachPoints[startIdx], endPoint };
695
696 var pathIndices = FindShortestPathInGraph(startIdx, endIdx, cancellationToken);
697 if (pathIndices == null) return null;
698
699 var finalPath = new List<Point3D> { startPoint };
700 foreach (var idx in pathIndices)
701 if (finalPath.Last().DistanceTo(_approachPoints[idx]) > 1.0) finalPath.Add(_approachPoints[idx]);
702 if (finalPath.Last().DistanceTo(endPoint) > 1.0) finalPath.Add(endPoint);
703
704 return finalPath;
705 }

714 private List<Point3D> InterpolateWithMeshPoints(List<Point3D> rawPath, double maxCorridor = 10.0, System.Threading.CancellationToken cancellationToken = default)
715 {
716 var result = new List<Point3D> { rawPath[0] };
717 for (int i = 0; i < rawPath.Count - 1; i++)
718 {
719 cancellationToken.ThrowIfCancellationRequested();
720 var p1 = rawPath[i]; var p2 = rawPath[i + 1];
721 var dir = new Vector3D(p1, p2); double len = dir.Length;
722
723 if (len > 5.0 && _shellMesh != null)
724 {
725 int steps = (int)Math.Ceiling(len / 5.0);
726 for (int s = 1; s < steps; s++)
727 {
728 double t = (double)s / steps;
729 Point3D interp = new Point3D(p1.X + (p2.X - p1.X) * t, p1.Y + (p2.Y - p1.Y) * t, p1.Z + (p2.Z - p1.Z) * t);
730 var projected = ShellProjectionUtils.FindNearestShellPoint(interp, _shellMesh);
731 if (projected.DistanceTo(result.Last()) > 2.0)
732 result.Add(projected);
733 }
734 }
735 else if (len > 2.0 && _approachPoints != null)
736 {
737 dir.Normalize();
738 var candidates = new List<(double d, Point3D p)>();
739 foreach (var ap in _approachPoints)
740 {
741 Vector3D v = new Vector3D(p1, ap); double d = Vector3D.Dot(v, dir);
742 if (d > 1.0 && d < len - 1.0 && ap.DistanceTo(p1 + dir * d) <= maxCorridor) candidates.Add((d, ap));
743 }
744 var ordered = candidates.OrderBy(x => x.d).ToList();
745 if (ordered.Count > 0)
746 {
747 foreach (var c in ordered)
748 if (c.p.DistanceTo(result.Last()) > 2.0) result.Add(c.p);
749 }
750 }
751
752 result.Add(p2);
753 }
754 return result;

762 private int FindNearestApproachPoint(Point3D point)
763 {
764 if (_approachPoints == null || _approachPoints.Count == 0) return -1;
765
766 if (_approachGrid != null)
767 {
768 int cx = (int)Math.Floor(point.X / APPROACH_GRID_CELL_SIZE);
769 int cy = (int)Math.Floor(point.Y / APPROACH_GRID_CELL_SIZE);
770 int cz = (int)Math.Floor(point.Z / APPROACH_GRID_CELL_SIZE);
771
772 double bestDistSq = double.MaxValue;
773 int bestIdx = -1;
774
775 // Search in 1-ring (3x3x3 = 27 cells)
776 for (int dx = -1; dx <= 1; dx++)
777 {
778 for (int dy = -1; dy <= 1; dy++)
779 {
780 for (int dz = -1; dz <= 1; dz++)
781 {
782 if (_approachGrid.TryGetValue((cx + dx, cy + dy, cz + dz), out var list))
783 {
784 foreach (int idx in list)
785 {
786 var ap = _approachPoints[idx];
787 double dSq = (point.X - ap.X) * (point.X - ap.X) +
788 (point.Y - ap.Y) * (point.Y - ap.Y) +
789 (point.Z - ap.Z) * (point.Z - ap.Z);
790 if (dSq < bestDistSq)
791 {
792 bestDistSq = dSq;
793 bestIdx = idx;
794 }
795 }
796 }
797 }
798 }
799 }
800
801 if (bestIdx >= 0) return bestIdx;
802 }
803
804 int fallbackIdx = -1;
805 double fallbackDistSq = double.MaxValue;
806 for (int i = 0; i < _approachPoints.Count; i++)
807 {
808 var ap = _approachPoints[i];
809 double dSq = (point.X - ap.X) * (point.X - ap.X) +
810 (point.Y - ap.Y) * (point.Y - ap.Y) +
811 (point.Z - ap.Z) * (point.Z - ap.Z);
812 if (dSq < fallbackDistSq)
813 {
814 fallbackDistSq = dSq;
815 fallbackIdx = i;
816 }
817 }
818 return fallbackIdx;
819 }

824 private static double DistancePointToLine(Point3D p, Point3D lineOrigin, Vector3D lineDir)
825 {
826 Vector3D v = new Vector3D(lineOrigin, p);
827 Vector3D cross = Vector3D.Cross(v, lineDir);
828 double dirLen = lineDir.Length;
829 if (dirLen < 1e-9) return p.DistanceTo(lineOrigin);
830 return cross.Length / dirLen;
831 }

841 private List<int>? FindShortestPathInGraph(int startIdx, int endIdx, System.Threading.CancellationToken cancellationToken = default)
842 {
843 if (_navigationGraph == null || !_navigationGraph.ContainsKey(startIdx)) return null;
844
845 string algorithm = System.Configuration.ConfigurationManager.AppSettings["PathPlanning_ShortestPathAlgorithm"] ?? "AStar";
846
847 bool isDijkstra = string.Equals(algorithm, "Dijkstra", StringComparison.OrdinalIgnoreCase);
848 bool isGreedy = string.Equals(algorithm, "Greedy", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "GreedyBFS", StringComparison.OrdinalIgnoreCase);
849 bool isBfs = string.Equals(algorithm, "BFS", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "BreadthFirst", StringComparison.OrdinalIgnoreCase);
850 bool isAStarLine = string.Equals(algorithm, "AStarLine", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "AStarPenalty", StringComparison.OrdinalIgnoreCase);
851 // Default is standard AStar ("AStar", "A*", or unrecognized setting)
852
853 var startPt = _approachPoints[startIdx];
854 var endPt = _approachPoints[endIdx];
855 var lineDir = new Vector3D(startPt, endPt);
856
857 var gScore = new Dictionary<int, double>();
858 var fScore = new Dictionary<int, double>();
859 var prevs = new Dictionary<int, int>();
860 var pq = new SortedSet<(double f, int n)>(Comparer<(double f, int n)>.Create((a, b) => a.f != b.f ? a.f.CompareTo(b.f) : a.n.CompareTo(b.n)));
861
862 foreach (var n in _navigationGraph.Keys)
863 {
864 gScore[n] = double.MaxValue;
865 fScore[n] = double.MaxValue;
866 }
867 double startH = (!isDijkstra && !isBfs) ? startPt.DistanceTo(endPt) : 0.0;
868 gScore[startIdx] = 0;
869 fScore[startIdx] = startH;
870 pq.Add((startH, startIdx));
871
872 while (pq.Count > 0)
873 {
874 cancellationToken.ThrowIfCancellationRequested();
875 var curr = pq.Min;
876 pq.Remove(curr);
877
878 if (curr.n == endIdx)
879 {
880 var path = new List<int>();
881 int c = endIdx;
882 while (prevs.TryGetValue(c, out int p)) { path.Insert(0, c); c = p; }
883 path.Insert(0, startIdx);
884 return path;
885 }
886
887 if (curr.f > fScore[curr.n]) continue;
888
889 if (_navigationGraph.TryGetValue(curr.n, out var neighbors))
890 {
891 foreach (var (nb, ed) in neighbors)
892 {
893 double stepCost = isBfs ? 1.0 : ed;
894 double tentative_g = gScore[curr.n] + stepCost;
895
896 if (tentative_g < gScore[nb])
897 {
898 pq.Remove((fScore[nb], nb));
899 prevs[nb] = curr.n;
900 gScore[nb] = tentative_g;
901
902 double fVal;
903 if (isDijkstra || isBfs)
904 {
905 fVal = tentative_g;
906 }
907 else if (isGreedy)
908 {
909 var nbPt = _approachPoints[nb];
910 fVal = nbPt.DistanceTo(endPt);
911 }
912 else if (isAStarLine)
913 {
914 var nbPt = _approachPoints[nb];
915 double h = nbPt.DistanceTo(endPt);
916 double linePenalty = DistancePointToLine(nbPt, startPt, lineDir) * 2.0;
917 fVal = tentative_g + h + linePenalty;
918 }
919 else
920 {
921 // Standard AStar / A*
922 var nbPt = _approachPoints[nb];
923 double h = nbPt.DistanceTo(endPt);
924 fVal = tentative_g + h;
925 }
926
927 fScore[nb] = fVal;
928 pq.Add((fVal, nb));
929 }
930 }
931 }
932 }
933 return null;
934 }
