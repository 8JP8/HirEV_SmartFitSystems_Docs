216 {

221 for (int i = 0; i < _shellMesh.Vertices.Length; i++)

284 private bool IsSegmentSafe(Point3D p1, Point3D p2, double minClearance)
285 {
286 if (_shellMesh == null) return true;
287
288 Point3D mid = new Point3D((p1.X + p2.X) * 0.5, (p1.Y + p2.Y) * 0.5, (p1.Z + p2.Z) * 0.5);
289 if (!IsPointSafe(mid, minClearance)) return false;
290
291 Point3D q1 = new Point3D((p1.X + mid.X) * 0.5, (p1.Y + mid.Y) * 0.5, (p1.Z + mid.Z) * 0.5);
292 if (!IsPointSafe(q1, minClearance)) return false;
293
294 Point3D q3 = new Point3D((mid.X + p2.X) * 0.5, (mid.Y + p2.Y) * 0.5, (mid.Z + p2.Z) * 0.5);
295 if (!IsPointSafe(q3, minClearance)) return false;
296
297 return true;
298 }

342 {
343 _navigationGraph = new Dictionary<int, List<(int, double)>>();
344 if (_approachPoints == null || _approachPoints.Count < 2) return;
345
346 int n = _approachPoints.Count;
347 int k = Math.Min(Math.Max(K_NEIGHBORS, n / 10), n - 1);
348
349 for (int i = 0; i < n; i++)
350 {
351 _navigationGraph[i] = new List<(int, double)>();
352 var distances = new List<(int idx, double dist)>();
353 for (int j = 0; j < n; j++)
354 if (i != j) distances.Add((j, _approachPoints[i].DistanceTo(_approachPoints[j])));
355
356 foreach (var neighbor in distances.OrderBy(d => d.dist).Take(k))
357 _navigationGraph[i].Add((neighbor.idx, neighbor.dist));
358 }
359
360 for (int i = 0; i < n; i++)
361 foreach (var (neighbor, dist) in _navigationGraph[i].ToList())

457 private void EnsureGraphConnectivity(System.Threading.CancellationToken cancellationToken = default)
458 {
459 if (_navigationGraph == null || _navigationGraph.Count == 0) return;
460 int n = _approachPoints.Count;
461 int[] parent = new int[n];
462 for (int i = 0; i < n; i++) parent[i] = i;
463
464 int Find(int x) => parent[x] == x ? x : (parent[x] = Find(parent[x]));
465 void Union(int x, int y) { int px = Find(x), py = Find(y); if (px != py) parent[px] = py; }
466
467 foreach (var kvp in _navigationGraph)
468 {
469 cancellationToken.ThrowIfCancellationRequested();
470 foreach (var (neighbor, _) in kvp.Value) Union(kvp.Key, neighbor);
471 }
472
473 var components = new Dictionary<int, List<int>>();
474 for (int i = 0; i < n; i++)
475 {
476 int root = Find(i);
477 if (!components.ContainsKey(root)) components[root] = new List<int>();
478 components[root].Add(i);
479 }
480
481 if (components.Count <= 1) return;
482 var componentList = components.Values.ToList();
483 double minClearance = _safetyOffset > 0 ? _safetyOffset * 0.45 : 1.0;
484
485 for (int c1 = 0; c1 < componentList.Count - 1; c1++)
486 {
487 cancellationToken.ThrowIfCancellationRequested();
488 int c2 = c1 + 1;
489
490 var candidates = new List<(int i, int j, double dist)>();
491 foreach (int i in componentList[c1])
492 {
493 foreach (int j in componentList[c2])
494 {
495 double dist = _approachPoints[i].DistanceTo(_approachPoints[j]);
496 candidates.Add((i, j, dist));
497 }
498 }
499
500 candidates.Sort((a, b) => a.dist.CompareTo(b.dist));
501
502 int bestI = -1, bestJ = -1;
503 double minDist = double.MaxValue;
504
505 foreach (var candidate in candidates)
506 {
507 if (IsSegmentSafe(_approachPoints[candidate.i], _approachPoints[candidate.j], minClearance))
508 {
509 bestI = candidate.i;
510 bestJ = candidate.j;
511 minDist = candidate.dist;
512 break;
513 }
514 }
515
516 if (bestI >= 0)
517 {
518 _navigationGraph[bestI].Add((bestJ, minDist));
519 _navigationGraph[bestJ].Add((bestI, minDist));
520 Union(bestI, bestJ);
521 componentList[c2].AddRange(componentList[c1]);
522 }
523 }

534 public List<Point3D>? FindShortestPath(Point3D startPoint, Point3D endPoint, int numIntermediatePoints = 5, System.Threading.CancellationToken cancellationToken = default)
535 {
536 if (_navigationGraph == null || _navigationGraph.Count == 0 || _approachPoints.Count < 2) return null;
537 int startIdx = FindNearestApproachPoint(startPoint);
538 int endIdx = FindNearestApproachPoint(endPoint);
539 if (startIdx == -1 || endIdx == -1) return null;
540
541 double minClearance = _safetyOffset > 0 ? _safetyOffset * 0.45 : 1.0;
542 if (startPoint.DistanceTo(endPoint) < _directTravelThreshold)
543 {
544 if (IsSegmentSafe(startPoint, endPoint, minClearance))
545 {
546 return new List<Point3D> { startPoint, endPoint };
547 }
548 }
549
550 if (startIdx == endIdx) return new List<Point3D> { startPoint, _approachPoints[startIdx], endPoint };
551
552 var pathIndices = FindShortestPathInGraph(startIdx, endIdx, cancellationToken);
553 if (pathIndices == null) return null;
554
555 var finalPath = new List<Point3D> { startPoint };
556 foreach (var idx in pathIndices)
557 if (finalPath.Last().DistanceTo(_approachPoints[idx]) > 1.0) finalPath.Add(_approachPoints[idx]);
558 if (finalPath.Last().DistanceTo(endPoint) > 1.0) finalPath.Add(endPoint);
559
560 return finalPath;
561 }

570 private List<Point3D> InterpolateWithMeshPoints(List<Point3D> rawPath, double maxCorridor = 10.0, System.Threading.CancellationToken cancellationToken = default)
571 {
572 var result = new List<Point3D> { rawPath[0] };
573 for (int i = 0; i < rawPath.Count - 1; i++)
574 {
575 cancellationToken.ThrowIfCancellationRequested();
576 var p1 = rawPath[i]; var p2 = rawPath[i + 1];
577 var dir = new Vector3D(p1, p2); double len = dir.Length;
578
579 if (len > 5.0 && _shellMesh != null)
580 {
581 int steps = (int)Math.Ceiling(len / 5.0);
582 for (int s = 1; s < steps; s++)
583 {
584 double t = (double)s / steps;
585 Point3D interp = new Point3D(p1.X + (p2.X - p1.X) * t, p1.Y + (p2.Y - p1.Y) * t, p1.Z + (p2.Z - p1.Z) * t);
586 var projected = ShellProjectionUtils.FindNearestShellPoint(interp, _shellMesh);
587 if (projected.DistanceTo(result.Last()) > 2.0)
588 result.Add(projected);
589 }
590 }
591 else if (len > 2.0 && _approachPoints != null)
592 {
593 dir.Normalize();
594 var candidates = new List<(double d, Point3D p)>();
595 foreach (var ap in _approachPoints)
596 {
597 Vector3D v = new Vector3D(p1, ap); double d = Vector3D.Dot(v, dir);
598 if (d > 1.0 && d < len - 1.0 && ap.DistanceTo(p1 + dir * d) <= maxCorridor) candidates.Add((d, ap));
599 }
600 var ordered = candidates.OrderBy(x => x.d).ToList();
601 if (ordered.Count > 0)
602 {
603 foreach (var c in ordered)
604 if (c.p.DistanceTo(result.Last()) > 2.0) result.Add(c.p);
605 }
606 }
607
608 result.Add(p2);
609 }
610 return result;
611 }

650 private List<int>? FindShortestPathInGraph(int startIdx, int endIdx, System.Threading.CancellationToken cancellationToken = default)
651 {
652 if (_navigationGraph == null || !_navigationGraph.ContainsKey(startIdx)) return null;
653
654 string algorithm = System.Configuration.ConfigurationManager.AppSettings["PathPlanning_ShortestPathAlgorithm"] ?? "AStar";
655
656 bool isDijkstra = string.Equals(algorithm, "Dijkstra", StringComparison.OrdinalIgnoreCase);
657 bool isGreedy = string.Equals(algorithm, "Greedy", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "GreedyBFS", StringComparison.OrdinalIgnoreCase);
658 bool isBfs = string.Equals(algorithm, "BFS", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "BreadthFirst", StringComparison.OrdinalIgnoreCase);
659 bool isAStarLine = string.Equals(algorithm, "AStarLine", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "AStarPenalty", StringComparison.OrdinalIgnoreCase);
660 // Default is standard AStar ("AStar", "A*", or unrecognized setting)
661
662 var startPt = _approachPoints[startIdx];
663 var endPt = _approachPoints[endIdx];
664 var lineDir = new Vector3D(startPt, endPt);
665
666 var gScore = new Dictionary<int, double>();
667 var fScore = new Dictionary<int, double>();
668 var prevs = new Dictionary<int, int>();
669 var pq = new SortedSet<(double f, int n)>(Comparer<(double f, int n)>.Create((a, b) => a.f != b.f ? a.f.CompareTo(b.f) : a.n.CompareTo(b.n)));
670
671 foreach (var n in _navigationGraph.Keys)
672 {
673 gScore[n] = double.MaxValue;
674 fScore[n] = double.MaxValue;
675 }
676 double startH = (!isDijkstra && !isBfs) ? startPt.DistanceTo(endPt) : 0.0;
677 gScore[startIdx] = 0;
678 fScore[startIdx] = startH;
679 pq.Add((startH, startIdx));
680
681 while (pq.Count > 0)
682 {
683 cancellationToken.ThrowIfCancellationRequested();
684 var curr = pq.Min;
685 pq.Remove(curr);
686
687 if (curr.n == endIdx)
688 {
689 var path = new List<int>();
690 int c = endIdx;
691 while (prevs.TryGetValue(c, out int p)) { path.Insert(0, c); c = p; }
692 path.Insert(0, startIdx);
693 return path;
694 }
695
696 if (curr.f > fScore[curr.n]) continue;
697
698 if (_navigationGraph.TryGetValue(curr.n, out var neighbors))
699 {
700 foreach (var (nb, ed) in neighbors)
701 {
702 double stepCost = isBfs ? 1.0 : ed;
703 double tentative_g = gScore[curr.n] + stepCost;
704
705 if (tentative_g < gScore[nb])
706 {
707 pq.Remove((fScore[nb], nb));
708 prevs[nb] = curr.n;
709 gScore[nb] = tentative_g;
710
711 double fVal;
712 if (isDijkstra || isBfs)
713 {
714 fVal = tentative_g;
715 }
716 else if (isGreedy)
717 {
718 var nbPt = _approachPoints[nb];
719 fVal = nbPt.DistanceTo(endPt);
720 }
721 else if (isAStarLine)
722 {
723 var nbPt = _approachPoints[nb];
724 double h = nbPt.DistanceTo(endPt);
725 double linePenalty = DistancePointToLine(nbPt, startPt, lineDir) * 2.0;
726 fVal = tentative_g + h + linePenalty;
727 }
728 else
729 {
730 // Standard AStar / A*
731 var nbPt = _approachPoints[nb];
732 double h = nbPt.DistanceTo(endPt);
733 fVal = tentative_g + h;
734 }
735
736 fScore[nb] = fVal;
737 pq.Add((fVal, nb));
738 }
739 }
740 }
741 }
742 return null;
743 }
