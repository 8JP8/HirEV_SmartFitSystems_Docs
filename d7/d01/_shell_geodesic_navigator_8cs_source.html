349 private bool IsSegmentSafe(Point3D p1, Point3D p2, double minClearance)
350 {
351 double dist = p1.DistanceTo(p2);
352 if (dist < 0.001) return true;
353
354 int steps = Math.Max(3, (int)Math.Ceiling(dist / 2.0));
355 for (int i = 1; i < steps; i++)
356 {
357 double t = (double)i / steps;
358 Point3D sample = new Point3D(
359 p1.X + (p2.X - p1.X) * t,
360 p1.Y + (p2.Y - p1.Y) * t,
361 p1.Z + (p2.Z - p1.Z) * t);
362
363 if (!IsPointSafe(sample, minClearance))
364 return false;
365 }
366
367 return true;
368 }

373 private bool IsPointSafe(Point3D pt, double minClearance)
374 {
375 // 1. Exact signed-distance check: point must not be inside part (d < 0) or violate minimum clearance
376 if (_shellGenerator != null)
377 {
378 double d = _shellGenerator.GetDistance(pt);
379 if (d != double.MaxValue && d < minClearance)
380 return false;
381 }
382
383 // 2. BREP solid containment check: point must not be inside part solid
384 if (_partBrep != null)
385 {
386 try
387 {
388 if (_partBrep.IsPointInside(pt) != pointStatusType.Outside)
389 return false;
390 }
391 catch
392 {
393 }
394 }
395
396 if (_shellMesh == null) return true;
397
398 double maxDist = Math.Max(30.0, _safetyOffset * 4.0);
399 double maxDistSq = maxDist * maxDist;
400
401 if (_shellMeshVertexGrid != null)
402 {
403 int cx = (int)Math.Floor(pt.X / SHELL_MESH_GRID_CELL_SIZE);
404 int cy = (int)Math.Floor(pt.Y / SHELL_MESH_GRID_CELL_SIZE);
405 int cz = (int)Math.Floor(pt.Z / SHELL_MESH_GRID_CELL_SIZE);
406
407 int radius = Math.Max(1, (int)Math.Ceiling(maxDist / SHELL_MESH_GRID_CELL_SIZE));
408
409 for (int dx = -radius; dx <= radius; dx++)
410 {
411 for (int dy = -radius; dy <= radius; dy++)
412 {
413 for (int dz = -radius; dz <= radius; dz++)
414 {
415 if (_shellMeshVertexGrid.TryGetValue((cx + dx, cy + dy, cz + dz), out var list))
416 {
417 foreach (var v in list)
418 {
419 double dSq = (pt.X - v.X) * (pt.X - v.X) +
420 (pt.Y - v.Y) * (pt.Y - v.Y) +
421 (pt.Z - v.Z) * (pt.Z - v.Z);
422 if (dSq <= maxDistSq)
423 {
424 return true; // Found a shell vertex within clearance limit - safe!
425 }
426 }
427 }
428 }
429 }
430 }
431
432 return false;
433 }
434
435 var nearestOnShell = ShellProjectionUtils.FindNearestShellPoint(pt, _shellMesh);
436 double distToShell = pt.DistanceTo(nearestOnShell);
437 return distToShell <= maxDist;
438 }

460 private void BuildNavigationGraph()
461 {
462 _navigationGraph = new Dictionary<int, List<(int, double)>>();
463 if (_approachPoints == null || _approachPoints.Count < 2) return;
464
465 int n = _approachPoints.Count;
466 int k = Math.Min(Math.Max(K_NEIGHBORS, n / 10), n - 1);
467
468 for (int i = 0; i < n; i++)
469 {
470 _navigationGraph[i] = new List<(int, double)>();
471 var distances = new List<(int idx, double dist)>();
472 for (int j = 0; j < n; j++)
473 if (i != j) distances.Add((j, _approachPoints[i].DistanceTo(_approachPoints[j])));
474
475 foreach (var neighbor in distances.OrderBy(d => d.dist).Take(k))
476 _navigationGraph[i].Add((neighbor.idx, neighbor.dist));
477 }
478
479 for (int i = 0; i < n; i++)
480 foreach (var (neighbor, dist) in _navigationGraph[i].ToList())
481 if (!_navigationGraph[neighbor].Any(x => x.Item1 == i)) _navigationGraph[neighbor].Add((i, dist));

577 private void EnsureGraphConnectivity(System.Threading.CancellationToken cancellationToken = default)
578 {
579 if (_navigationGraph == null || _navigationGraph.Count == 0) return;
580 int n = _approachPoints.Count;
581 int[] parent = new int[n];
582 for (int i = 0; i < n; i++) parent[i] = i;
583
584 int Find(int x) => parent[x] == x ? x : (parent[x] = Find(parent[x]));
585 void Union(int x, int y) { int px = Find(x), py = Find(y); if (px != py) parent[px] = py; }
586
587 foreach (var kvp in _navigationGraph)
588 {
589 cancellationToken.ThrowIfCancellationRequested();
590 foreach (var (neighbor, _) in kvp.Value) Union(kvp.Key, neighbor);
591 }
592
593 var components = new Dictionary<int, List<int>>();
594 for (int i = 0; i < n; i++)
595 {
596 int root = Find(i);
597 if (!components.ContainsKey(root)) components[root] = new List<int>();
598 components[root].Add(i);
599 }
600
601 if (components.Count <= 1) return;
602 var componentList = components.Values.ToList();
603 double minClearance = _safetyOffset > 0 ? _safetyOffset * 0.45 : 1.0;
604
605 // Maximum distance across which shell graph components can be bridged.
606 // Bridge candidate segments are individually tested by IsSegmentSafe to ensure no collision occurs.
607 double maxBridgeDistance = Math.Max(150.0, _safetyOffset * 10.0);
608 double maxBridgeDistSq = maxBridgeDistance * maxBridgeDistance;
609
610 // Precompute bounding boxes for each component for O(1) early-rejection
611 var compBoxes = new (Point3D Min, Point3D Max)[componentList.Count];
612 for (int c = 0; c < componentList.Count; c++)
613 {
614 double minX = double.MaxValue, minY = double.MaxValue, minZ = double.MaxValue;
615 double maxX = double.MinValue, maxY = double.MinValue, maxZ = double.MinValue;
616 foreach (int idx in componentList[c])
617 {
618 var pt = _approachPoints[idx];
619 if (pt.X < minX) minX = pt.X;
620 if (pt.Y < minY) minY = pt.Y;
621 if (pt.Z < minZ) minZ = pt.Z;
622 if (pt.X > maxX) maxX = pt.X;
623 if (pt.Y > maxY) maxY = pt.Y;
624 if (pt.Z > maxZ) maxZ = pt.Z;
625 }
626 compBoxes[c] = (new Point3D(minX, minY, minZ), new Point3D(maxX, maxY, maxZ));
627 }
628
629 for (int c1 = 0; c1 < componentList.Count - 1; c1++)
630 {
631 cancellationToken.ThrowIfCancellationRequested();
632 for (int c2 = c1 + 1; c2 < componentList.Count; c2++)
633 {
634 if (Find(componentList[c1][0]) == Find(componentList[c2][0]))
635 continue;
636
637 // Check distance between component bounding boxes
638 double dx = Math.Max(0.0, Math.Max(compBoxes[c1].Min.X - compBoxes[c2].Max.X, compBoxes[c2].Min.X - compBoxes[c1].Min.X));
639 double dy = Math.Max(0.0, Math.Max(compBoxes[c1].Min.Y - compBoxes[c2].Max.Y, compBoxes[c2].Min.Y - compBoxes[c1].Min.Y));
640 double dz = Math.Max(0.0, Math.Max(compBoxes[c1].Min.Z - compBoxes[c2].Max.Z, compBoxes[c2].Min.Z - compBoxes[c1].Min.Z));
641 double boxDistSq = dx * dx + dy * dy + dz * dz;
642
643 // If bounding boxes are separated by more than maxBridgeDistance, they cannot be bridged
644 if (boxDistSq > maxBridgeDistSq)
645 {
646 continue;
647 }
648
649 // Only gather candidate pairs that are within maxBridgeDistance
650 var candidates = new List<(int i, int j, double dist)>();
651 foreach (int i in componentList[c1])
652 {
653 var p1 = _approachPoints[i];
654 foreach (int j in componentList[c2])
655 {
656 var p2 = _approachPoints[j];
657 double dSq = (p1.X - p2.X) * (p1.X - p2.X) +
658 (p1.Y - p2.Y) * (p1.Y - p2.Y) +
659 (p1.Z - p2.Z) * (p1.Z - p2.Z);
660 if (dSq <= maxBridgeDistSq)
661 {
662 candidates.Add((i, j, Math.Sqrt(dSq)));
663 }
664 }
665 }
666
667 if (candidates.Count == 0) continue;
668
669 candidates.Sort((a, b) => a.dist.CompareTo(b.dist));
670
671 int bestI = -1, bestJ = -1;
672 double minDist = double.MaxValue;
673 int testCount = Math.Min(10, candidates.Count);
674
675 for (int k = 0; k < testCount; k++)
676 {
677 var candidate = candidates[k];
678 if (IsSegmentSafe(_approachPoints[candidate.i], _approachPoints[candidate.j], minClearance))
679 {
680 bestI = candidate.i;
681 bestJ = candidate.j;
682 minDist = candidate.dist;
683 break;
684 }
685 }
686
687 if (bestI >= 0)
688 {
689 _navigationGraph[bestI].Add((bestJ, minDist));
690 _navigationGraph[bestJ].Add((bestI, minDist));
691 Union(bestI, bestJ);
692 componentList[c2].AddRange(componentList[c1]);
693 compBoxes[c2] = (
694 new Point3D(Math.Min(compBoxes[c1].Min.X, compBoxes[c2].Min.X), Math.Min(compBoxes[c1].Min.Y, compBoxes[c2].Min.Y), Math.Min(compBoxes[c1].Min.Z, compBoxes[c2].Min.Z)),
695 new Point3D(Math.Max(compBoxes[c1].Max.X, compBoxes[c2].Max.X), Math.Max(compBoxes[c1].Max.Y, compBoxes[c2].Max.Y), Math.Max(compBoxes[c1].Max.Z, compBoxes[c2].Max.Z))
696 );
697 }
698 }
699 }

710 public List<Point3D>? FindShortestPath(Point3D startPoint, Point3D endPoint, int numIntermediatePoints = 5, System.Threading.CancellationToken cancellationToken = default)
711 {
712 if (_navigationGraph == null || _navigationGraph.Count == 0 || _approachPoints.Count < 2) return null;
713 int startIdx = FindNearestApproachPoint(startPoint);
714 int endIdx = FindNearestApproachPoint(endPoint);
715 if (startIdx == -1 || endIdx == -1) return null;
716
717 double minClearance = _safetyOffset > 0 ? _safetyOffset * 0.45 : 1.0;
718 if (startPoint.DistanceTo(endPoint) < _directTravelThreshold)
719 {
720 if (IsSegmentSafe(startPoint, endPoint, minClearance))
721 {
722 return new List<Point3D> { startPoint, endPoint };
723 }
724 }
725
726 if (startIdx == endIdx)
727 {
728 var sameNodeRoute = new List<Point3D> { startPoint, _approachPoints[startIdx], endPoint };
729 return InterpolateWithMeshPoints(sameNodeRoute, cancellationToken: cancellationToken);
730 }
731
732 var pathIndices = FindShortestPathInGraph(startIdx, endIdx, cancellationToken);
733 if (pathIndices == null) return null;
734
735 var finalPath = new List<Point3D> { startPoint };
736 foreach (var idx in pathIndices)
737 if (finalPath.Last().DistanceTo(_approachPoints[idx]) > 1.0) finalPath.Add(_approachPoints[idx]);
738 if (finalPath.Last().DistanceTo(endPoint) > 1.0) finalPath.Add(endPoint);
739
740 finalPath = InterpolateWithMeshPoints(finalPath, cancellationToken: cancellationToken);
741
742 return finalPath;
743 }

752 private List<Point3D> InterpolateWithMeshPoints(List<Point3D> rawPath, double maxCorridor = 10.0, System.Threading.CancellationToken cancellationToken = default)
753 {
754 if (rawPath == null || rawPath.Count < 2) return rawPath ?? new List<Point3D>();
755
756 var result = new List<Point3D> { rawPath[0] };
757 for (int i = 0; i < rawPath.Count - 1; i++)
758 {
759 cancellationToken.ThrowIfCancellationRequested();
760 var p1 = rawPath[i]; var p2 = rawPath[i + 1];
761 var dir = new Vector3D(p1, p2); double len = dir.Length;
762
763 if (len > 3.0 && (_shellGenerator != null || _shellMesh != null))
764 {
765 int steps = Math.Max(2, (int)Math.Ceiling(len / 2.5));
766 for (int s = 1; s < steps; s++)
767 {
768 double t = (double)s / steps;
769 Point3D interp = new Point3D(p1.X + (p2.X - p1.X) * t, p1.Y + (p2.Y - p1.Y) * t, p1.Z + (p2.Z - p1.Z) * t);
770 Point3D? projected = null;
771 if (_shellGenerator != null)
772 projected = _shellGenerator.FindNearestShellPoint(interp);
773 else if (_shellMesh != null)
774 projected = ShellProjectionUtils.FindNearestShellPoint(interp, _shellMesh);
775
776 if (projected != null && projected.DistanceTo(result.Last()) > 1.0)
777 result.Add(projected);
778 }
779 }
780 else if (len > 2.0 && _approachPoints != null)
781 {
782 dir.Normalize();
783 var candidates = new List<(double d, Point3D p)>();
784 foreach (var ap in _approachPoints)
785 {
786 Vector3D v = new Vector3D(p1, ap); double d = Vector3D.Dot(v, dir);
787 if (d > 1.0 && d < len - 1.0 && ap.DistanceTo(p1 + dir * d) <= maxCorridor) candidates.Add((d, ap));
788 }
789 var ordered = candidates.OrderBy(x => x.d).ToList();
790 if (ordered.Count > 0)
791 {
792 foreach (var c in ordered)
793 if (c.p.DistanceTo(result.Last()) > 2.0) result.Add(c.p);
794 }
795 }
796
797 if (result.Last().DistanceTo(p2) > 0.001)
798 result.Add(p2);
799 }
800 return result;
801 }

808 public int FindNearestApproachPoint(Point3D point)
809 {
810 if (_approachPoints == null || _approachPoints.Count == 0) return -1;
811
812 if (_approachGrid != null)
813 {
814 int cx = (int)Math.Floor(point.X / APPROACH_GRID_CELL_SIZE);
815 int cy = (int)Math.Floor(point.Y / APPROACH_GRID_CELL_SIZE);
816 int cz = (int)Math.Floor(point.Z / APPROACH_GRID_CELL_SIZE);
817
818 double bestDistSq = double.MaxValue;
819 int bestIdx = -1;
820
821 // Search in 1-ring (3x3x3 = 27 cells)
822 for (int dx = -1; dx <= 1; dx++)
823 {
824 for (int dy = -1; dy <= 1; dy++)
825 {
826 for (int dz = -1; dz <= 1; dz++)
827 {
828 if (_approachGrid.TryGetValue((cx + dx, cy + dy, cz + dz), out var list))
829 {
830 foreach (int idx in list)
831 {
832 var ap = _approachPoints[idx];
833 double dSq = (point.X - ap.X) * (point.X - ap.X) +
834 (point.Y - ap.Y) * (point.Y - ap.Y) +
835 (point.Z - ap.Z) * (point.Z - ap.Z);
836 if (dSq < bestDistSq)
837 {
838 bestDistSq = dSq;
839 bestIdx = idx;
840 }
841 }
842 }
843 }
844 }
845 }
846
847 if (bestIdx >= 0) return bestIdx;
848 }
849
850 int fallbackIdx = -1;
851 double fallbackDistSq = double.MaxValue;
852 for (int i = 0; i < _approachPoints.Count; i++)
853 {
854 var ap = _approachPoints[i];
855 double dSq = (point.X - ap.X) * (point.X - ap.X) +
856 (point.Y - ap.Y) * (point.Y - ap.Y) +
857 (point.Z - ap.Z) * (point.Z - ap.Z);
858 if (dSq < fallbackDistSq)
859 {
860 fallbackDistSq = dSq;
861 fallbackIdx = i;
862 }
863 }
864 return fallbackIdx;
865 }

870 private static double DistancePointToLine(Point3D p, Point3D lineOrigin, Vector3D lineDir)
871 {
872 Vector3D v = new Vector3D(lineOrigin, p);
873 Vector3D cross = Vector3D.Cross(v, lineDir);
874 double dirLen = lineDir.Length;
875 if (dirLen < 1e-9) return p.DistanceTo(lineOrigin);
876 return cross.Length / dirLen;
877 }

886 public List<int>? FindShortestPathInGraph(int startIdx, int endIdx, System.Threading.CancellationToken cancellationToken = default)
887 {
888 if (_navigationGraph == null || !_navigationGraph.ContainsKey(startIdx)) return null;
889
890 string algorithm = System.Configuration.ConfigurationManager.AppSettings["PathPlanning_ShortestPathAlgorithm"] ?? "AStar";
891
892 bool isDijkstra = string.Equals(algorithm, "Dijkstra", StringComparison.OrdinalIgnoreCase);
893 bool isGreedy = string.Equals(algorithm, "Greedy", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "GreedyBFS", StringComparison.OrdinalIgnoreCase);
894 bool isBfs = string.Equals(algorithm, "BFS", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "BreadthFirst", StringComparison.OrdinalIgnoreCase);
895 bool isAStarLine = string.Equals(algorithm, "AStarLine", StringComparison.OrdinalIgnoreCase) || string.Equals(algorithm, "AStarPenalty", StringComparison.OrdinalIgnoreCase);
896 // Default is standard AStar ("AStar", "A*", or unrecognized setting)
897
898 var startPt = _approachPoints[startIdx];
899 var endPt = _approachPoints[endIdx];
900 var lineDir = new Vector3D(startPt, endPt);
901
902 var gScore = new Dictionary<int, double>();
903 var fScore = new Dictionary<int, double>();
904 var prevs = new Dictionary<int, int>();
905 var pq = new SortedSet<(double f, int n)>(Comparer<(double f, int n)>.Create((a, b) => a.f != b.f ? a.f.CompareTo(b.f) : a.n.CompareTo(b.n)));
906
907 foreach (var n in _navigationGraph.Keys)
908 {
909 gScore[n] = double.MaxValue;
910 fScore[n] = double.MaxValue;
911 }
912 double startH = (!isDijkstra && !isBfs) ? startPt.DistanceTo(endPt) : 0.0;
913 gScore[startIdx] = 0;
914 fScore[startIdx] = startH;
915 pq.Add((startH, startIdx));
916
917 while (pq.Count > 0)
918 {
919 cancellationToken.ThrowIfCancellationRequested();
920 var curr = pq.Min;
921 pq.Remove(curr);
922
923 if (curr.n == endIdx)
924 {
925 var path = new List<int>();
926 int c = endIdx;
927 while (prevs.TryGetValue(c, out int p)) { path.Insert(0, c); c = p; }
928 path.Insert(0, startIdx);
929 return path;
930 }
931
932 if (curr.f > fScore[curr.n]) continue;
933
934 if (_navigationGraph.TryGetValue(curr.n, out var neighbors))
935 {
936 foreach (var (nb, ed) in neighbors)
937 {
938 double stepCost = isBfs ? 1.0 : ed;
939 double tentative_g = gScore[curr.n] + stepCost;
940
941 if (tentative_g < gScore[nb])
942 {
943 pq.Remove((fScore[nb], nb));
944 prevs[nb] = curr.n;
945 gScore[nb] = tentative_g;
946
947 double fVal;
948 if (isDijkstra || isBfs)
949 {
950 fVal = tentative_g;
951 }
952 else if (isGreedy)
953 {
954 var nbPt = _approachPoints[nb];
955 fVal = nbPt.DistanceTo(endPt);
956 }
957 else if (isAStarLine)
958 {
959 var nbPt = _approachPoints[nb];
960 double h = nbPt.DistanceTo(endPt);
961 double linePenalty = DistancePointToLine(nbPt, startPt, lineDir) * 2.0;
962 fVal = tentative_g + h + linePenalty;
963 }
964 else
965 {
966 // Standard AStar / A*
967 var nbPt = _approachPoints[nb];
968 double h = nbPt.DistanceTo(endPt);
969 fVal = tentative_g + h;
970 }
971
972 fScore[nb] = fVal;
973 pq.Add((fVal, nb));
974 }
975 }
976 }
977 }
978 return null;
979 }
