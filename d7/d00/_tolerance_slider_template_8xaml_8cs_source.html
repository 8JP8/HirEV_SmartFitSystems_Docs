116 private static T? FindVisualParent<T>(DependencyObject child) where T : DependencyObject
117 {
118 if (child == null || child is not Visual && child is not Visual3D) return null;
119
120 DependencyObject parentObject = VisualTreeHelper.GetParent(child);
121 if (parentObject == null) return null;
122
123 if (parentObject is T parent)
124 return parent;
125
126 return FindVisualParent<T>(parentObject);

312 private void UpdateDerivedProperties(string? fullName)
313 {
314 if (string.IsNullOrEmpty(fullName))
315 {
316 FaceColorShort = null;
317 FaceColorRGBCode = null;
318 FaceColorRGB = null;
319 FaceColorForeground = Brushes.Black;
320 return;
321 }
322
323 // --- 1. Parse for the Short Name ---
324 var nameMatch = Regex.Match(fullName, @"^(.*?)\s*\‍(");
325 if (nameMatch.Success && nameMatch.Groups.Count > 1)
326 {
327 FaceColorShort = nameMatch.Groups[1].Value.Trim();
328
329 // --- Extract RGB code properly using regex ---
330 var rgbCodeMatch = Regex.Match(fullName, @"\‍((.*?)\‍)");
331 if (rgbCodeMatch.Success)
332 {
333 FaceColorRGBCode = rgbCodeMatch.Groups[1].Value.Trim();
334 }
335 else
336 {
337 FaceColorRGBCode = null;
338 }
339 }
340 else
341 {
342 FaceColorShort = fullName;
343 FaceColorRGBCode = null;
344 }
345
346 // --- 2. Parse for the Color from R#_G#_B# format ---
347 try
348 {
349 var colorMatch = Regex.Match(fullName ?? "", @"R(\d{1,3})_G(\d{1,3})_B(\d{1,3})");
350 if (colorMatch.Success)
351 {
352 byte r = byte.Parse(colorMatch.Groups[1].Value);
353 byte g = byte.Parse(colorMatch.Groups[2].Value);
354 byte b = byte.Parse(colorMatch.Groups[3].Value);
355 FaceColorForeground = new SolidColorBrush(Color.FromRgb(r, g, b));
356 FaceColorRGB = $"RGB({r},{g},{b})";
357 }
358 else
359 {
360 FaceColorForeground = Brushes.Black;
361 FaceColorRGB = null;
362 }
363 }
364 catch (Exception)
365 {
366 FaceColorForeground = Brushes.Black;
367 FaceColorRGB = null;
368 }
369 }
370 #endregion
