161 private static T? FindVisualParent<T>(DependencyObject child) where T : DependencyObject
162 {
163 if (child == null || child is not System.Windows.Media.Visual && child is not Visual3D) return null;
164
165 DependencyObject parentObject = System.Windows.Media.VisualTreeHelper.GetParent(child);
166 if (parentObject == null) return null;
167
168 if (parentObject is T parent)
169 return parent;
170
171 return FindVisualParent<T>(parentObject);
172 }

364 #region Path Loading
368 private void LoadPaths()
369 {
370 Paths.Clear();
371 _initialVisibilityStates.Clear();
372 _initialExecutionOrder.Clear();
373 _initialExecutionOrder.AddRange(_pathManager.ExecutionOrder);
374
375 // Populate list in current execution order
376 foreach (var color in _pathManager.ExecutionOrder)
377 {
378 if (_pathManager.ColorPaths.TryGetValue(color, out var colorPath))
379 {
380 // Format color name consistently with TreeView: "ColorName (R#_G#_B#)"
381 string colorName = ColorNamer.GetColorName(color);
382 string displayName = $"{colorName} (R{color.R}_G{color.G}_B{color.B})";
383
384 var pathItem = new PathItemViewModel(displayName, color, colorPath.IsVisible);
385
386 // Subscribe to visibility changes to update PathManager and refresh visuals in real-time
387 pathItem.VisibilityChanged += (s, e) =>
388 {
389 if (s is PathItemViewModel item)
390 {
391 _pathManager.SetPathVisibility(item.OriginalColor, item.IsVisible);
392
393 // Regenerate path visuals to show/hide immediately
394 _pathManager.RegeneratePathVisuals();
395 }
396 };
397
398 Paths.Add(pathItem);
399 _initialVisibilityStates[color] = colorPath.IsVisible;
400 }
401 }
402 }

531 private void DiscardAllPaths()
532 {
533 string message = GetXMLElement("PathEditDialog_ConfirmDiscardAll", "Are you sure you want to discard all paths? This action cannot be undone.");
534 string title = GetXMLElement("PathEditDialog_DiscardAllTitle", "Confirm Discard All Paths");

559 private void RemovePath()
560 {
561 if (SelectedPath == null) return;
562
563 // Confirm removal
564 string message = string.Format(
565 GetXMLElement("PathEditDialog_ConfirmRemove", "Are you sure you want to remove the path for color {0}? This action cannot be undone."),
566 SelectedPath.Name);
567
568 string title = GetXMLElement("PathEditDialog_RemovePathTitle", "Confirm Path Removal");
569
570 var result = MessageBox.Show(message, title, MessageBoxButton.YesNo, MessageBoxImage.Warning);
571
572 if (result == MessageBoxResult.Yes)
573 {
574 var colorToRemove = SelectedPath.OriginalColor;
575 int indexToRemove = Paths.IndexOf(SelectedPath);
576
577 // Remove from PathManager (this removes visuals, path data, and execution order entry)
578 bool removed = _pathManager.RemovePath(colorToRemove);
579
580 if (removed)
581 {
582 // Remove from local collection
583 Paths.RemoveAt(indexToRemove);
584
585 // Update selection
586 if (Paths.Count > 0)
587 {
588 int newIndex = Math.Min(indexToRemove, Paths.Count - 1);
589 SelectedPath = Paths[newIndex];
590 }
591 else
592 {
593 SelectedPath = null;
594 }
595
596 // Refresh design to show changes
597 _design.Invalidate();
598
599 System.Diagnostics.Debug.WriteLine($"✅ Path removed from dialog for color: {colorToRemove.Name}");
600 }
601 else
602 {
603 System.Diagnostics.Debug.WriteLine($"❌ Failed to remove path for color: {colorToRemove.Name}");
604 MessageBox.Show(
605 GetXMLElement("PathEditDialog_RemoveError", "Failed to remove path. Please try again."),
606 GetXMLElement("Error", "Error"),
607 MessageBoxButton.OK,
608 MessageBoxImage.Error);
609 }
610 }

644 private void OnCancel()
645 {
646 try
647 {
648 // Revert any visibility changes
649 foreach (var state in _initialVisibilityStates)
650 {
651 _pathManager.SetPathVisibility(state.Key, state.Value);
652 }
653
654 // Revert global toggles
655 _pathManager.SetAllPathLinesVisibility(_initialShowPathLines);
656 _pathManager.SetAllApproachPointsVisibility(_initialShowApproachPoints);
657 _pathManager.SetAllMeasurePointsVisibility(_initialShowMeasurePoints);
658
659 // Revert execution order
660 if (_initialExecutionOrder.Count > 0 && _initialExecutionOrder.All(c => _pathManager.ColorPaths.ContainsKey(c)))
661 {
662 _pathManager.SetExecutionOrder(_initialExecutionOrder);
663 }
664
665 // Regenerate to reflect reverted state
666 _pathManager.RegeneratePathVisuals();
667
668 // Close dialog with cancel
669 DialogResult = false;
670 }
671 catch (Exception ex)
672 {
673 System.Diagnostics.Debug.WriteLine($"Error reverting visibility changes: {ex.Message}");
674 DialogResult = false;
675 }

686 private string GetXMLElement(string key, string fallback)
687 {

690 try
691 {
692 var nodes = _docLanguage.GetElementsByTagName(key);
693 if (nodes.Count > 0 && nodes[0]!.Attributes!["Value"] != null)

696 }
697 }
698 catch { }
699
700 return fallback;
