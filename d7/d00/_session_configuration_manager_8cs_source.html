100 public static void CreateNewSessionFile()
101 {
102 try
103 {
104 string dir = GetSaveDirectory();
105
106 // Timestamp-based name: Session_20260520_200613.json
107 string timestamp = DateTime.Now.ToString("yyyyMMdd_HHmmss");
108 string candidatePath = Path.Combine(dir, $"Session_{timestamp}.json");
109
110 // Guard against two sessions starting within the exact same second
111 int counter = 1;
112 while (File.Exists(candidatePath))
113 {
114 candidatePath = Path.Combine(dir, $"Session_{timestamp}_{counter}.json");
115 counter++;
116 }
117
118 CurrentSessionFilePath = candidatePath;
119
120 // Prune oldest files to keep within limit
121 PruneOldSessionFiles(dir);
122
123 System.Diagnostics.Debug.WriteLine($"📂 SessionConfigurationManager: New session file → {CurrentSessionFilePath}");
124 }
125 catch (Exception ex)
126 {
127 System.Diagnostics.Debug.WriteLine($"Failed to create new session file: {ex.Message}");
128 }
129 }

135 public static void SaveSession(SessionConfig config)
136 {
137 string? tempFile = null;
138 try
139 {
140 if (string.IsNullOrEmpty(CurrentSessionFilePath))
141 CreateNewSessionFile();
142
143 System.Diagnostics.Debug.WriteLine($"💾 SessionConfigurationManager: Saving session to {CurrentSessionFilePath}");
144
145 string dir = Path.GetDirectoryName(CurrentSessionFilePath)!;
146 tempFile = Path.Combine(dir, Guid.NewGuid().ToString() + ".tmp");
147
148 string json = JsonSerializer.Serialize(config, GetOptions());
149
150 // Write to a temp file first, then atomically replace the target to prevent corruption
151 File.WriteAllText(tempFile, json);
152 File.Move(tempFile, CurrentSessionFilePath, overwrite: true);
153
154 System.Diagnostics.Debug.WriteLine(
155 $"✅ SessionConfigurationManager: Save successful. Size: {new FileInfo(CurrentSessionFilePath).Length} bytes");
156 }
157 catch (Exception ex)
158 {
159 System.Diagnostics.Debug.WriteLine($"❌ SessionConfigurationManager: Failed to save session config: {ex.Message}");
160 if (tempFile != null && File.Exists(tempFile))
161 try { File.Delete(tempFile); } catch { }
162 }
163 }

249 public static string CalculateFileHash(string filePath)
250 {
251 if (!File.Exists(filePath)) return string.Empty;
252 using var sha256 = SHA256.Create();
253 using var stream = File.OpenRead(filePath);
254 byte[] hashBytes = sha256.ComputeHash(stream);
255 return BitConverter.ToString(hashBytes).Replace("-", "").ToLowerInvariant();
256 }
