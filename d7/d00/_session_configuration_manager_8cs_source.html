115 private static void NormalizePathExport(PathExportDTO? pathExport)
116 {
117 if (pathExport == null)
118 return;
119
120 pathExport.ColorPaths ??= new List<ColorPathExportDTO>();
121 pathExport.ExecutionOrder ??= new List<string>();
122 pathExport.EditHistory ??= new List<PointEditEventDTO>();
123
124 foreach (var colorPath in pathExport.ColorPaths)
125 {
126 colorPath.PathPoints ??= new List<PathPointExportDTO>();
127 colorPath.ColorName = NormalizeSerializedColorName(colorPath);
128 colorPath.ToolName ??= string.Empty;
129 colorPath.CmmType ??= string.Empty;
130
131 for (int i = 0; i < colorPath.PathPoints.Count; i++)
132 {
133 var point = colorPath.PathPoints[i];
134 if (string.IsNullOrWhiteSpace(point.PointType))
135 {
136 point.PointType = InferSerializedPointType(i, colorPath.PathPoints.Count);
137 }
138 else
139 {
140 point.PointType = point.PointType.Trim();
141 }
142 }
143 }
144
145 var colorNames = pathExport.ColorPaths
146 .Select(NormalizeSerializedColorName)
147 .Where(name => !string.IsNullOrWhiteSpace(name))
148 .Distinct(StringComparer.OrdinalIgnoreCase)
149 .ToList();
150
151 if (pathExport.ExecutionOrder.Count == 0)
152 {
153 pathExport.ExecutionOrder = colorNames;
154 }
155 else
156 {
157 var existingOrder = new HashSet<string>(pathExport.ExecutionOrder, StringComparer.OrdinalIgnoreCase);
158 foreach (var colorName in colorNames)
159 {
160 if (existingOrder.Add(colorName))
161 {
162 pathExport.ExecutionOrder.Add(colorName);
163 }
164 }
165 }
166 }

187 private static bool IsScannerPathExport(PathExportDTO? pathExport)
188 {
189 if (pathExport == null)
190 return false;
191
192 if (string.Equals(pathExport.CmmType, "Scanner", StringComparison.OrdinalIgnoreCase))
193 return true;
194
195 return pathExport.ColorPaths?.Any(cp =>
196 string.Equals(cp.CmmType, "Scanner", StringComparison.OrdinalIgnoreCase) ||
197 cp.PathPoints?.Any(p => string.Equals(p.PointType, "scanpoints", StringComparison.OrdinalIgnoreCase)) == true) == true;
198 }

241 public static void CreateNewSessionFile()
242 {
243 try
244 {
245 string dir = GetSaveDirectory();
246
247 // Timestamp-based name: Session_20260520_200613.json
248 string timestamp = DateTime.Now.ToString("yyyyMMdd_HHmmss");
249 string candidatePath = Path.Combine(dir, $"Session_{timestamp}.json");
250
251 // Guard against two sessions starting within the exact same second
252 int counter = 1;
253 while (File.Exists(candidatePath))
254 {
255 candidatePath = Path.Combine(dir, $"Session_{timestamp}_{counter}.json");
256 counter++;
257 }
258
259 CurrentSessionFilePath = candidatePath;
260
261 // Prune oldest files to keep within limit
262 PruneOldSessionFiles(dir);
263
264 System.Diagnostics.Debug.WriteLine($"📂 SessionConfigurationManager: New session file → {CurrentSessionFilePath}");
265 }
266 catch (Exception ex)
267 {
268 System.Diagnostics.Debug.WriteLine($"Failed to create new session file: {ex.Message}");
269 }
270 }

276 public static void SaveSession(SessionConfig config)
277 {
278 string? tempFile = null;
279 try
280 {
281 if (string.IsNullOrEmpty(CurrentSessionFilePath))
282 CreateNewSessionFile();
283
284 System.Diagnostics.Debug.WriteLine(string.Format(GetLoc("Session_SavingSession", "SessionConfigurationManager: Saving session to {0}"), CurrentSessionFilePath));
285
286 string dir = Path.GetDirectoryName(CurrentSessionFilePath)!;
287 tempFile = Path.Combine(dir, Guid.NewGuid().ToString() + ".tmp");
288
289 NormalizeSessionConfig(config);
290 string json = JsonSerializer.Serialize(config, GetOptions());
291
292 // Write to a temp file first, then atomically replace the target to prevent corruption
293 File.WriteAllText(tempFile, json);
294 File.Move(tempFile, CurrentSessionFilePath, overwrite: true);
295
296 System.Diagnostics.Debug.WriteLine(
297 string.Format(GetLoc("Session_SaveSuccessful", "SessionConfigurationManager: Save successful. Size: {0} bytes"), new FileInfo(CurrentSessionFilePath).Length));
298 }
299 catch (Exception ex)
300 {
301 System.Diagnostics.Debug.WriteLine(string.Format(GetLoc("Session_SaveFailed", "SessionConfigurationManager: Failed to save session config: {0}"), ex.Message));
302 if (tempFile != null && File.Exists(tempFile))
303 try { File.Delete(tempFile); } catch { }
304 }
305 }

311 public static SessionConfig? LoadSpecificConfiguration(string filePath)
312 {
313 try
314 {
315 if (File.Exists(filePath))
316 {
317 string json = File.ReadAllText(filePath);
318 var config = JsonSerializer.Deserialize<SessionConfig>(json, GetOptions());
319 if (config != null)
320 {
321 NormalizeSessionConfig(config);
322 // Update current session timestamp so it reflects the moment it was copied
323 config.Timestamp = DateTime.Now;
324
325 if (string.IsNullOrEmpty(CurrentSessionFilePath))
326 {
327 CreateNewSessionFile();
328 }
329
330 SaveSession(config);
331
332 System.Diagnostics.Debug.WriteLine(
333 $"📂 SessionConfigurationManager: Loaded save → {Path.GetFileName(filePath)} (copied to active session {Path.GetFileName(CurrentSessionFilePath)})");
334 }
335 return config;
336 }
337 }
338 catch (Exception ex)

340 System.Diagnostics.Debug.WriteLine($"Failed to load session config: {ex.Message}");

392 public static string CalculateFileHash(string filePath)
393 {
394 if (!File.Exists(filePath)) return string.Empty;
395 using var sha256 = SHA256.Create();
396 using var stream = File.OpenRead(filePath);
397 byte[] hashBytes = sha256.ComputeHash(stream);
398 return BitConverter.ToString(hashBytes).Replace("-", "").ToLowerInvariant();
399 }
