71 public static Vector3D ComputeOutwardNormal(CADFace face, Point3D partCentroid, Point3D? faceCentroid = null)
72 {
73 if (face == null)
74 return Vector3D.AxisZ;
75
76 // PHASE 2 COMPLIANCE: Extract normal ONLY from face geometry (GeometricInfo)
77 // Do NOT use centroid-based direction validation
78 Vector3D faceNormal = ExtractNormalFromGeometricInfo(face);
79
80 // Ensure normalized
81 if (faceNormal.Length > 0.0001)
82 {
83 faceNormal.Normalize();
84 }
85 else
86 {
87 // Fallback: use Z-axis (no centroid dependency)
88 faceNormal = Vector3D.AxisZ;
89 }
90
91 // PHASE 2 COMPLIANCE: Return normal as-is from geometry.
92 // Normal flipping (if needed) is handled by:
93 // 1. Manual flip via user control (FlippedFaceNormals)
94 // 2. Shell mesh tessellation winding order
95 // Do NOT auto-flip based on centroid heuristics.
96
97 return faceNormal;
98 }

103 private static Vector3D ExtractNormalFromGeometricInfo(CADFace face)
104 {
105 if (string.IsNullOrEmpty(face.GeometricInfo))
106 return Vector3D.AxisZ;
107
108 try
109 {
110 // GeometricInfo format for planar: "Normal: (X, Y, Z)"
111 if (face.GeometricInfo.Contains("Normal:"))
112 {
113 int startIdx = face.GeometricInfo.IndexOf('(');
114 int endIdx = face.GeometricInfo.IndexOf(')');
115 if (startIdx > 0 && endIdx > startIdx)
116 {
117 string coords = face.GeometricInfo.Substring(startIdx + 1, endIdx - startIdx - 1);
118 string[] parts = coords.Split(',');
119 if (parts.Length >= 3)
120 {
121 double x = double.Parse(parts[0].Trim(), System.Globalization.CultureInfo.InvariantCulture);
122 double y = double.Parse(parts[1].Trim(), System.Globalization.CultureInfo.InvariantCulture);
123 double z = double.Parse(parts[2].Trim(), System.Globalization.CultureInfo.InvariantCulture);
124 var normal = new Vector3D(x, y, z);
125 if (normal.Length > 0.0001)
126 {
127 normal.Normalize();
128 return normal;
129 }
130 }
131 }
132 }
133
134 // For cylindrical/conical faces with Axis info
135 if (face.GeometricInfo.Contains("Axis:"))
136 {
137 int startIdx = face.GeometricInfo.IndexOf("Axis:");
138 startIdx = face.GeometricInfo.IndexOf('(', startIdx);
139 int endIdx = face.GeometricInfo.IndexOf(')', startIdx);
140 if (startIdx > 0 && endIdx > startIdx)
141 {
142 string coords = face.GeometricInfo.Substring(startIdx + 1, endIdx - startIdx - 1);
143 string[] parts = coords.Split(',');
144 if (parts.Length >= 3)
145 {
146 double x = double.Parse(parts[0].Trim(), System.Globalization.CultureInfo.InvariantCulture);
147 double y = double.Parse(parts[1].Trim(), System.Globalization.CultureInfo.InvariantCulture);
148 double z = double.Parse(parts[2].Trim(), System.Globalization.CultureInfo.InvariantCulture);
149 var axis = new Vector3D(x, y, z);
150 if (axis.Length > 0.0001)
151 {
152 axis.Normalize();
153 return axis;
154 }
155 }
156 }
157 }
158 }
159 catch
160 {
161 // Parsing failed, use fallback
162 }
163
164 return Vector3D.AxisZ;
165 }

170 private static Point3D ComputeFaceCentroid(CADFace face)
171 {
172 var points = new List<Point3D>();
173
174 // Use KeyPoints if available
175 if (face.KeyPoints != null && face.KeyPoints.Count > 0)
176 {
177 foreach (var kp in face.KeyPoints)
178 {
179 points.Add(new Point3D(kp.X, kp.Y, kp.Z));
180 }
181 }
182 // Fallback to ControlPoints
183 else if (face.ControlPoints != null && face.ControlPoints.Count > 0)
184 {
185 foreach (var cp in face.ControlPoints)
186 {
187 points.Add(new Point3D(cp.X, cp.Y, cp.Z));
188 }
189 }
190 // Fallback to MeasurementPoints
191 else if (face.MeasurementPoints != null && face.MeasurementPoints.Count > 0)
192 {
193 foreach (var mp in face.MeasurementPoints)
194 {
195 points.Add(new Point3D(mp.X, mp.Y, mp.Z));
196 }
197 }
198
199 if (points.Count == 0)
200 return new Point3D(0, 0, 0);
201
202 double sumX = 0, sumY = 0, sumZ = 0;
203 foreach (var pt in points)
204 {
205 sumX += pt.X;
206 sumY += pt.Y;
207 sumZ += pt.Z;
208 }
209
210 return new Point3D(sumX / points.Count, sumY / points.Count, sumZ / points.Count);
211 }

241 public static List<ApproachMeasurementPair> GenerateApproachPointsForFace(
242 CADFace face,
243 Point3D partCentroid,
244 double offsetDistance)
245 {
246 var pairs = new List<ApproachMeasurementPair>();
247
248 if (face == null || face.MeasurementPoints == null || face.MeasurementPoints.Count == 0)
249 return pairs;
250
251 // Compute face normal once for all points on this face
252 Vector3D faceNormal = ComputeOutwardNormal(face, partCentroid);
253
254 // Determine if this is a hole face (cylindrical/conical)
255 bool isHoleFace = face.FaceType == FaceTypeEnum.Cylindrical ||
256 face.FaceType == FaceTypeEnum.Conical;
257
258 int debugCount = 0;
259 foreach (var mp in face.MeasurementPoints)
260 {
261 Point3D measurePt = new Point3D(mp.X, mp.Y, mp.Z);
262
263 // For hole faces, we might need to compute per-point normals (radial from axis)
264 // For planar faces, all points use the same face normal
265 Vector3D pointNormal = faceNormal;
266
267 if (isHoleFace)
268 {
269 // For cylindrical/conical faces, compute radial normal from hole axis to point
270 pointNormal = ComputeRadialNormalForHolePoint(measurePt, face, partCentroid);
271 }
272
273 Point3D approachPt = ProjectMeasurementPointToApproachPoint(measurePt, pointNormal, offsetDistance);
274 pairs.Add(new ApproachMeasurementPair(measurePt, approachPt, pointNormal, face, isHoleFace));
275
276 // Debug first point
277 if (debugCount == 0)
278 {
279 double distance = measurePt.DistanceTo(approachPt);
280 Debug.WriteLine($" DEBUG: Measure=({measurePt.X:F2},{measurePt.Y:F2},{measurePt.Z:F2}) Normal=({pointNormal.X:F3},{pointNormal.Y:F3},{pointNormal.Z:F3}) Offset={offsetDistance:F2}mm Distance={distance:F2}mm");
281 }
282 debugCount++;
283 }
284
285 Debug.WriteLine($" ✅ Generated {pairs.Count} approach points for face {face.FaceGlobalIndex} ({face.FaceType})");
286 return pairs;
287 }

292 private static Vector3D ComputeRadialNormalForHolePoint(Point3D point, CADFace face, Point3D partCentroid)
293 {
294 // For cylindrical holes, we need the radial normal from the HOLE AXIS, not part centroid
295 // Try to extract hole center from KeyPoints or ControlPoints
296
297 Point3D holeCenter = partCentroid; // Fallback
298
299 if (face.KeyPoints != null && face.KeyPoints.Count >= 2)
300 {
301 // Estimate hole center as average of KeyPoints XY (holes are usually vertical)
302 double sumX = 0, sumY = 0;
303 foreach (var kp in face.KeyPoints)
304 {
305 sumX += kp.X;
306 sumY += kp.Y;
307 }
308 holeCenter = new Point3D(sumX / face.KeyPoints.Count, sumY / face.KeyPoints.Count, point.Z);
309 }
310 else if (face.ControlPoints != null && face.ControlPoints.Count >= 2)
311 {
312 // Estimate from control points
313 double sumX = 0, sumY = 0;
314 foreach (var cp in face.ControlPoints)
315 {
316 sumX += cp.X;
317 sumY += cp.Y;
318 }
319 holeCenter = new Point3D(sumX / face.ControlPoints.Count, sumY / face.ControlPoints.Count, point.Z);
320 }
321
322 // Radial normal from hole center to point (in XY plane)
323 Vector3D radial = new Vector3D(
324 point.X - holeCenter.X,
325 point.Y - holeCenter.Y,
326 0); // Cylindrical holes: radial in XY plane only
327
328 if (radial.Length > 0.0001)
329 {
330 radial.Normalize();
331 }
332 else
333 {
334 radial = Vector3D.AxisX; // Fallback
335 }
336
337 return radial;
338 }

385 public static Point3D? RaycastToShellSurface(Point3D origin, Vector3D direction, Mesh shellMesh)
386 {
387 if (shellMesh?.Vertices == null || shellMesh?.Triangles == null)
388 return null;
389
390 Point3D? closestHit = null;
391 double closestDist = double.MaxValue;
392
393 foreach (var triangle in shellMesh.Triangles)
394 {
395 if (triangle.V1 >= shellMesh.Vertices.Length ||
396 triangle.V2 >= shellMesh.Vertices.Length ||
397 triangle.V3 >= shellMesh.Vertices.Length)
398 continue;
399
400 var v0 = shellMesh.Vertices[triangle.V1];
401 var v1 = shellMesh.Vertices[triangle.V2];
402 var v2 = shellMesh.Vertices[triangle.V3];
403
404 var hit = RayTriangleIntersection(origin, direction, v0, v1, v2);
405 if (hit != null)
406 {
407 double dist = origin.DistanceTo(hit);
408 if (dist < closestDist && dist > 0.001) // Avoid self-intersection
409 {
410 closestDist = dist;
411 closestHit = hit;
412 }
413 }
414 }
415
416 return closestHit;
417 }

422 private static Point3D? RayTriangleIntersection(Point3D rayOrigin, Vector3D rayDir, Point3D v0, Point3D v1, Point3D v2)
423 {
424 const double EPSILON = 0.0000001;
425
426 var edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
427 var edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
428 var h = Vector3D.Cross(rayDir, edge2);
429 var a = Vector3D.Dot(edge1, h);
430
431 if (a > -EPSILON && a < EPSILON)
432 return null; // Ray parallel to triangle
433
434 double f = 1.0 / a;
435 var s = new Vector3D(rayOrigin.X - v0.X, rayOrigin.Y - v0.Y, rayOrigin.Z - v0.Z);
436 double u = f * Vector3D.Dot(s, h);
437
438 if (u < 0.0 || u > 1.0)
439 return null;
440
441 var q = Vector3D.Cross(s, edge1);
442 double v = f * Vector3D.Dot(rayDir, q);
443
444 if (v < 0.0 || u + v > 1.0)
445 return null;
446
447 double t = f * Vector3D.Dot(edge2, q);
448
449 if (t > EPSILON) // Ray intersection
450 {
451 return new Point3D(
452 rayOrigin.X + rayDir.X * t,
453 rayOrigin.Y + rayDir.Y * t,
454 rayOrigin.Z + rayDir.Z * t);
455 }
456
457 return null;
458 }

466 public static Point3D FindNearestShellPoint(Point3D point, Mesh shellMesh)
467 {
468 if (shellMesh?.Vertices == null || shellMesh.Vertices.Length == 0)
469 return point;
470
471 double minDist = double.MaxValue;
472 Point3D nearest = point;
473
474 foreach (var vertex in shellMesh.Vertices)
475 {
476 double dist = point.DistanceTo(vertex);
477 if (dist < minDist)
478 {
479 minDist = dist;
480 nearest = vertex;
481 }
482 }
483
484 return nearest;
485 }
