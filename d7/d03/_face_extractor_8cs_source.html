68 public List<CADFace> ExtractFaces(Action<int, int>? progressCallback = null, System.Threading.CancellationToken cancellationToken = default)
69 {
70 var results = new List<CADFace>();
71 int globalFaceIndex = 0;
72 int totalEntities = _entities.Count;
73 int lastReportedPercentage = -1;
74 int lastReportedFaceCount = 0;
75 long lastUpdateTime = 0;
76 const int UPDATE_INTERVAL_MS = 50; // Update UI at most every 50ms
77
78 for (int i = 0; i < _entities.Count; i++)
79 {
80 cancellationToken.ThrowIfCancellationRequested();
81 var ent = _entities[i];
82 if (ent == null) continue;
83 Brep? brep = ent as Brep ?? TryConvertToBrep(ent);
84 if (brep == null) continue;
85
86 // Try to ensure tessellation / parametric info available
87 try { brep.Rebuild(); } catch { /* ignore if not available */ }
88
89 for (int j = 0; j < brep.Faces.Length; j++)
90 {
91 cancellationToken.ThrowIfCancellationRequested();
92 var face = brep.Faces[j];
93 var cadFace = ExtractFaceData(brep, face, ++globalFaceIndex, ent.Color);
94 cadFace.EntityIndex = i;
95 cadFace.FaceIndex = j;
96 cadFace.ModelFilePath = ent.EntityData as string;
97 results.Add(cadFace);
98 }
99
100 // Report progress after processing each entity with throttling
101 if (progressCallback != null && totalEntities > 0)
102 {
103 int percentage = (int)((i + 1) * 100.0 / totalEntities);
104 long currentTime = DateTime.Now.Ticks / TimeSpan.TicksPerMillisecond;
105 bool shouldUpdate = (percentage != lastReportedPercentage) ||
106 (currentTime - lastUpdateTime >= UPDATE_INTERVAL_MS) ||
107 (i == _entities.Count - 1); // Always update on last entity
108
109 if (shouldUpdate && results.Count != lastReportedFaceCount)
110 {
111 lastReportedPercentage = percentage;
112 lastReportedFaceCount = results.Count;
113 lastUpdateTime = currentTime;
114 progressCallback(percentage, results.Count);
115 }
116 }
117 }
118 return results;
119 }

147 private CADFace ExtractFaceData(Brep brep, Brep.Face face, int index, Color entityColor)
148 {
149 var cadFace = new CADFace
150 {
151 FaceGlobalIndex = index,
152 FaceColor = (face?.Color ?? entityColor), // prefer face color if present
153 KeyPoints = new List<FacePoint>(),
154 ControlPoints = new List<Point4D>()
155 };
156
157 // FIXED CONTROL POINT EXTRACTION
158 try
159 {
160 var surfaces = face?.ConvertToSurface();
161 if (surfaces != null && surfaces.Length > 0)
162 {
163 var surface = surfaces[0];
164 if (surface.ControlPoints != null)
165 {
166 var allPts = surface.ControlPoints.Cast<Point4D>().ToList();
167
168 // Deduplicate within a tolerance
169 var uniquePts = allPts
170 .GroupBy(p => (Math.Round(p.X, 3), Math.Round(p.Y, 3), Math.Round(p.Z, 3)))
171 .Select(g => g.First())
172 .ToList();
173
174 // Filter out any points far outside the brep bounding box
175 var boxMin = brep.BoxMin;
176 var boxMax = brep.BoxMax;
177 uniquePts = uniquePts
178 .Where(p =>
179 p.X >= boxMin.X - 1 && p.X <= boxMax.X + 1 &&
180 p.Y >= boxMin.Y - 1 && p.Y <= boxMax.Y + 1 &&
181 p.Z >= boxMin.Z - 1 && p.Z <= boxMax.Z + 1)
182 .ToList();
183
184 // Optional: ensure consistent ordering
185 uniquePts = uniquePts
186 .OrderBy(p => p.Z)
187 .ThenBy(p => p.X)
188 .ThenBy(p => p.Y)
189 .ToList();
190
191 cadFace.ControlPoints = uniquePts;
192 }
193 }
194 }
195 catch
196 {
197 cadFace.ControlPoints = new List<Point4D>();
198 }
199
200 // Extract corner points with vertex indices
201 if (face != null)
202 {
203 var corners = ExtractCornerPointsWithIndices(brep, face);
204 foreach (var (pt, vidx) in corners)
205 {
206 cadFace.KeyPoints.Add(new FacePoint
207 {
208 X = pt.X,
209 Y = pt.Y,
210 Z = pt.Z,
211 PointType = PointTypeEnum.Corner,
212 VertexIndex = vidx
213 });
214 }
215
216 cadFace.FaceArea = GetFaceArea(face);
217 cadFace.FaceType = InferFaceType(face.Surface, out string geoInfo, out devDept.Geometry.Vector3D? axisOrNormal);
218 cadFace.GeometricInfo = geoInfo;
219 cadFace.AxisOrNormal = axisOrNormal;
220
221 // Store only the scalar required by HoleDetector. Positive means that the
222 // oriented cylindrical face normal points towards the cylinder middle axis;
223 // negative means it points outwards and the face is convex.
224 if (cadFace.FaceType == FaceTypeEnum.Cylindrical &&
225 face.Surface is CylindricalSurf cylinder &&
226 TryGetCylinderInwardNormalDot(face, cylinder, out double inwardNormalDot))
227 {
228 cadFace.GeometricInfo += string.Format(
229 CultureInfo.InvariantCulture,
230 ", CylinderInwardNormalDot: {0:R}",
231 inwardNormalDot);
232 }
233
234 var center = GetFaceCentroid(face);
235 if (center != null)
236 {
237 cadFace.KeyPoints.Add(new FacePoint
238 {
239 X = center.X,
240 Y = center.Y,
241 Z = center.Z,
242 PointType = PointTypeEnum.Center,
243 VertexIndex = -1
244 });
245 }
246
247 // Note: Measurement points are generated later by PathManager.PopulateMeasurementPointsForFaces()
248 // This ensures consistency with path generation and allows PathManager configuration to be used
249 }
250
251 return cadFace;
252 }

275 private List<(Point3D pt, int vertexIndex)> ExtractCornerPointsWithIndices(Brep brep, Brep.Face face)
276 {
277 var set = new HashSet<int>();
278 foreach (var loop in face.Loops)
279 {
280 foreach (var seg in loop.Segments)
281 {
282 var edge = brep.Edges[seg.CurveIndex];
283 if (edge.StartPointIndex >= 0) set.Add(edge.StartPointIndex);
284 if (edge.EndPointIndex >= 0) set.Add(edge.EndPointIndex);
285 }
286 }
287
288 var res = new List<(Point3D, int)>();
289 foreach (var vidx in set)
290 {
291 var pt = brep.Vertices[vidx];
292 res.Add((pt, vidx));
293 }
294 return res;
295 }

355 private Point3D? GetFaceCentroid(Brep.Face face)
356 {
357 if (face.Parametric == null) return null;
358 var points = new List<Point3D>();
359 foreach (var surface in face.Parametric)
360 {
361 if (surface.Vertices == null)
362 surface.Regen(0.01);
363 if (surface.Vertices != null)
364 points.AddRange(surface.Vertices);
365 }
366 if (points.Count == 0) return null;
367 return new Point3D(
368 points.Average(p => p.X),
369 points.Average(p => p.Y),
370 points.Average(p => p.Z)
371 );

405 private FaceTypeEnum InferFaceType(object surface, out string geometricInfo, out devDept.Geometry.Vector3D? axisOrNormal)
406 {
407 geometricInfo = "N/A";
408 axisOrNormal = null;
409 if (surface == null) return FaceTypeEnum.Unknown;
410 try
411 {
412 if (surface is SphericalSurf sph)
413 {
414 geometricInfo = $"Radius: {sph.Radius:F2}";
415 return FaceTypeEnum.Spherical;
416 }
417 else if (surface is CylindricalSurf cyl)
418 {
419 geometricInfo = $"Radius: {cyl.Radius:F2}, Origin: ({cyl.Plane.Origin.X:F2}, {cyl.Plane.Origin.Y:F2}, {cyl.Plane.Origin.Z:F2}), Axis: ({cyl.Plane.AxisZ.X:F2}, {cyl.Plane.AxisZ.Y:F2}, {cyl.Plane.AxisZ.Z:F2})";
420 axisOrNormal = cyl.Plane.AxisZ;
421 return FaceTypeEnum.Cylindrical;
422 }
423 else if (surface is ConicalSurf cone)
424 {
425 geometricInfo = $"Radius: {cone.Radius:F2}, Angle: {Utility.RadToDeg(cone.HalfAngle):F2}";
426 axisOrNormal = cone.Plane.AxisZ;
427 return FaceTypeEnum.Conical;
428 }
429 else if (surface is ToroidalSurf tor)
430 {
431 geometricInfo = $"Major R: {tor.MajorRadius:F2}, Minor R: {tor.MinorRadius:F2}";
432 return FaceTypeEnum.Toroidal;
433 }
434 else if (surface is NurbsSurf)
435 {
436 geometricInfo = "NURBS Surface";
437 return FaceTypeEnum.BSpline;
438 }
439 else if (surface is RevolvedSurf rev)
440 {
441 geometricInfo = $"Axis: ({rev.Plane.AxisZ.X:F2}, {rev.Plane.AxisZ.Y:F2}, {rev.Plane.AxisZ.Z:F2})";
442 axisOrNormal = rev.Plane.AxisZ;
443 return FaceTypeEnum.Revolved;
444 }
445 else if (surface is TabulatedSurf tab)
446 {
447 geometricInfo = $"Direction: ({tab.Generatrix.X:F2}, {tab.Generatrix.Y:F2}, {tab.Generatrix.Z:F2})";
448 axisOrNormal = tab.Generatrix;
449 return FaceTypeEnum.Tabulated;
450 }
451 else if (surface is PlanarSurf pl)
452 {
453 geometricInfo = $"Normal: ({pl.Plane.AxisZ.X:F2}, {pl.Plane.AxisZ.Y:F2}, {pl.Plane.AxisZ.Z:F2})";
454 axisOrNormal = pl.Plane.AxisZ;
455 return FaceTypeEnum.Planar;
456 }
457 }
458 catch (Exception ex)
459 {
460 System.Diagnostics.Debug.WriteLine($"Error inferring face type: {ex.Message}");
461 }
462 return FaceTypeEnum.Unknown;
463 }

471 private bool TryGetCylinderInwardNormalDot(
472 Brep.Face face,
473 CylindricalSurf cylinder,
474 out double inwardNormalDot)
475 {
476 inwardNormalDot = 0;
477
478 // Sense=true uses the cylinder's natural outward orientation; Sense=false
479 // reverses it and therefore represents an inward-facing cylindrical wall.
480 if (TryGetFaceSense(face, out bool sense))
481 {
482 inwardNormalDot = sense ? -1.0 : 1.0;
483 return true;
484 }
485
486 // Small compatibility fallback when Sense is not exposed by the Eyeshot build.
487 // This reuses the extractor's existing approximate tessellation normal.
488 Point3D? samplePoint = GetFaceCentroid(face);
489 Vector3D normal = GetFaceNormalApprox(face);
490 Vector3D axis = cylinder.Plane.AxisZ;
491 if (samplePoint == null || normal.Length < 1e-9 || axis.Length < 1e-9)
492 return false;
493
494 normal.Normalize();
495 axis.Normalize();
496
497 var originToPoint = new Vector3D(cylinder.Plane.Origin, samplePoint);
498 double axialDistance = Vector3D.Dot(originToPoint, axis);
499 var pointOnAxis = new Point3D(
500 cylinder.Plane.Origin.X + axis.X * axialDistance,
501 cylinder.Plane.Origin.Y + axis.Y * axialDistance,
502 cylinder.Plane.Origin.Z + axis.Z * axialDistance);
503 var directionToAxis = new Vector3D(samplePoint, pointOnAxis);
504
505 if (directionToAxis.Length < 1e-9)
506 return false;
507
508 directionToAxis.Normalize();
509 inwardNormalDot = Vector3D.Dot(normal, directionToAxis);
510 return !double.IsNaN(inwardNormalDot) && !double.IsInfinity(inwardNormalDot);
511 }

517 private static bool TryGetFaceSense(Brep.Face face, out bool sense)
518 {
519 sense = true;
520 if (face == null)
521 return false;
522
523 const BindingFlags flags = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;
524 Type type = face.GetType();
525
526 bool TryRead(string name, out bool value)
527 {
528 value = false;
529 try
530 {
531 if (type.GetProperty(name, flags)?.GetValue(face) is bool propertyValue)
532 {
533 value = propertyValue;
534 return true;
535 }
536
537 if (type.GetField(name, flags)?.GetValue(face) is bool fieldValue)
538 {
539 value = fieldValue;
540 return true;
541 }
542 }
543 catch
544 {
545 // Try the next compatible member name.
546 }
547
548 return false;
549 }
550
551 foreach (string name in new[] { "Sense", "sense", "_sense", "m_sense" })
552 {
553 if (TryRead(name, out sense))
554 return true;
555 }
556
557 foreach (string name in new[] { "OrientationIsReversed", "IsOrientationReversed", "IsReversed", "Reversed" })
558 {
559 if (TryRead(name, out bool reversed))
560 {
561 sense = !reversed;
562 return true;
563 }
564 }
565
566 return false;
567 }

594 private Vector3D GetFaceNormalApprox(Brep.Face face)
595 {
596 try
597 {
598 var list = new List<Point3D>();
599 if (face.Parametric != null)
600 {
601 foreach (var surf in face.Parametric)
602 {
603 if (surf.Vertices == null) surf.Regen(0.01);
604 if (surf.Vertices != null) list.AddRange(surf.Vertices);
605 }
606 }
607
608 if (list.Count < 3)
609 return new Vector3D(0, 0, 0);
610
611 Point3D a = list[0];
612 int idxB = 1;
613 for (int i = 1; i < list.Count; i++)
614 {
615 if (!PointsAreEqual(a, list[i])) { idxB = i; break; }
616 }
617 Point3D b = list[idxB];
618
619 for (int i = idxB + 1; i < list.Count; i++)
620 {
621 Point3D c = list[i];
622 var vx1 = b.X - a.X; var vy1 = b.Y - a.Y; var vz1 = b.Z - a.Z;
623 var vx2 = c.X - a.X; var vy2 = c.Y - a.Y; var vz2 = c.Z - a.Z;
624 var cx = vy1 * vz2 - vz1 * vy2;
625 var cy = vz1 * vx2 - vx1 * vz2;
626 var cz = vx1 * vy2 - vy1 * vx2;
627 double clen = Math.Sqrt(cx * cx + cy * cy + cz * cz);
628 if (clen > 1e-9)
629 return new Vector3D(cx / clen, cy / clen, cz / clen);
630 }
631 return new Vector3D(0, 0, 0);
632 }
633 catch
634 {
635 return new Vector3D(0, 0, 0);
636 }
637 }

721 public class Point3DComparer : IEqualityComparer<Point3D>
722 {
727 private const double Tolerance = 1e-6;
728
744 public bool Equals(Point3D? p1, Point3D? p2)
745 {
746 if (ReferenceEquals(p1, p2)) return true;
747 if (p1 is null || p2 is null) return false;
748
749 return Math.Abs(p1.X - p2.X) < Tolerance &&
750 Math.Abs(p1.Y - p2.Y) < Tolerance &&
751 Math.Abs(p1.Z - p2.Z) < Tolerance;
752 }
753
775 public int GetHashCode(Point3D? p)
776 {
777 if (p is null) return 0;
778 int hashX = Math.Round(p.X / Tolerance).GetHashCode();
779 int hashY = Math.Round(p.Y / Tolerance).GetHashCode();
780 int hashZ = Math.Round(p.Z / Tolerance).GetHashCode();
781 return hashX ^ (hashY << 1) ^ (hashZ << 2);
782 }
783 }
