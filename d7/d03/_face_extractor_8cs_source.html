67 public List<CADFace> ExtractFaces(Action<int, int>? progressCallback = null, System.Threading.CancellationToken cancellationToken = default)
68 {
69 var results = new List<CADFace>();
70 int globalFaceIndex = 0;
71 int totalEntities = _entities.Count;
72 int lastReportedPercentage = -1;
73 int lastReportedFaceCount = 0;
74 long lastUpdateTime = 0;
75 const int UPDATE_INTERVAL_MS = 50; // Update UI at most every 50ms
76
77 for (int i = 0; i < _entities.Count; i++)
78 {
79 cancellationToken.ThrowIfCancellationRequested();
80 var ent = _entities[i];
81 if (ent == null) continue;
82 Brep? brep = ent as Brep ?? TryConvertToBrep(ent);
83 if (brep == null) continue;
84
85 // Try to ensure tessellation / parametric info available
86 try { brep.Rebuild(); } catch { /* ignore if not available */ }
87
88 for (int j = 0; j < brep.Faces.Length; j++)
89 {
90 cancellationToken.ThrowIfCancellationRequested();
91 var face = brep.Faces[j];
92 var cadFace = ExtractFaceData(brep, face, ++globalFaceIndex, ent.Color);
93 cadFace.EntityIndex = i;
94 cadFace.FaceIndex = j;
95 cadFace.ModelFilePath = ent.EntityData as string;
96 results.Add(cadFace);
97 }
98
99 // Report progress after processing each entity with throttling
100 if (progressCallback != null && totalEntities > 0)
101 {
102 int percentage = (int)((i + 1) * 100.0 / totalEntities);
103 long currentTime = DateTime.Now.Ticks / TimeSpan.TicksPerMillisecond;
104 bool shouldUpdate = (percentage != lastReportedPercentage) ||
105 (currentTime - lastUpdateTime >= UPDATE_INTERVAL_MS) ||
106 (i == _entities.Count - 1); // Always update on last entity
107
108 if (shouldUpdate && results.Count != lastReportedFaceCount)
109 {
110 lastReportedPercentage = percentage;
111 lastReportedFaceCount = results.Count;
112 lastUpdateTime = currentTime;
113 progressCallback(percentage, results.Count);
114 }
115 }
116 }
117 return results;
118 }

146 private CADFace ExtractFaceData(Brep brep, Brep.Face face, int index, Color entityColor)
147 {
148 var cadFace = new CADFace
149 {
150 FaceGlobalIndex = index,
151 FaceColor = (face?.Color ?? entityColor), // prefer face color if present
152 KeyPoints = new List<FacePoint>(),
153 ControlPoints = new List<Point4D>()
154 };
155
156 // FIXED CONTROL POINT EXTRACTION
157 try
158 {
159 var surfaces = face?.ConvertToSurface();
160 if (surfaces != null && surfaces.Length > 0)
161 {
162 var surface = surfaces[0];
163 if (surface.ControlPoints != null)
164 {
165 var allPts = surface.ControlPoints.Cast<Point4D>().ToList();
166
167 // Deduplicate within a tolerance
168 var uniquePts = allPts
169 .GroupBy(p => (Math.Round(p.X, 3), Math.Round(p.Y, 3), Math.Round(p.Z, 3)))
170 .Select(g => g.First())
171 .ToList();
172
173 // Filter out any points far outside the brep bounding box
174 var boxMin = brep.BoxMin;
175 var boxMax = brep.BoxMax;
176 uniquePts = uniquePts
177 .Where(p =>
178 p.X >= boxMin.X - 1 && p.X <= boxMax.X + 1 &&
179 p.Y >= boxMin.Y - 1 && p.Y <= boxMax.Y + 1 &&
180 p.Z >= boxMin.Z - 1 && p.Z <= boxMax.Z + 1)
181 .ToList();
182
183 // Optional: ensure consistent ordering
184 uniquePts = uniquePts
185 .OrderBy(p => p.Z)
186 .ThenBy(p => p.X)
187 .ThenBy(p => p.Y)
188 .ToList();
189
190 cadFace.ControlPoints = uniquePts;
191 }
192 }
193 }
194 catch
195 {
196 cadFace.ControlPoints = new List<Point4D>();
197 }
198
199 // Extract corner points with vertex indices
200 if (face != null)
201 {
202 var corners = ExtractCornerPointsWithIndices(brep, face);
203 foreach (var (pt, vidx) in corners)
204 {
205 cadFace.KeyPoints.Add(new FacePoint
206 {
207 X = pt.X,
208 Y = pt.Y,
209 Z = pt.Z,
210 PointType = PointTypeEnum.Corner,
211 VertexIndex = vidx
212 });
213 }
214
215 cadFace.FaceArea = GetFaceArea(face);
216 cadFace.FaceType = InferFaceType(face.Surface, out string geoInfo);
217 cadFace.GeometricInfo = geoInfo;
218
219 var center = GetFaceCentroid(face);
220 if (center != null)
221 {
222 cadFace.KeyPoints.Add(new FacePoint
223 {
224 X = center.X,
225 Y = center.Y,
226 Z = center.Z,
227 PointType = PointTypeEnum.Center,
228 VertexIndex = -1
229 });
230 }
231
232 // Note: Measurement points are generated later by PathManager.PopulateMeasurementPointsForFaces()
233 // This ensures consistency with path generation and allows PathManager configuration to be used
234 }
235
236 return cadFace;
237 }

260 private List<(Point3D pt, int vertexIndex)> ExtractCornerPointsWithIndices(Brep brep, Brep.Face face)
261 {
262 var set = new HashSet<int>();
263 foreach (var loop in face.Loops)
264 {
265 foreach (var seg in loop.Segments)
266 {
267 var edge = brep.Edges[seg.CurveIndex];
268 if (edge.StartPointIndex >= 0) set.Add(edge.StartPointIndex);
269 if (edge.EndPointIndex >= 0) set.Add(edge.EndPointIndex);
270 }
271 }
272
273 var res = new List<(Point3D, int)>();
274 foreach (var vidx in set)
275 {
276 var pt = brep.Vertices[vidx];
277 res.Add((pt, vidx));
278 }
279 return res;
280 }

301 private double GetFaceArea(Brep.Face face)
302 {
303 double area = 0;
304 if (face.Parametric == null) return double.NaN;
305 foreach (var surface in face.Parametric)
306 {
307 if (surface.Vertices == null)
308 surface.Regen(0.01);

311 area += surface.GetArea(out _);

340 private Point3D? GetFaceCentroid(Brep.Face face)
341 {
342 if (face.Parametric == null) return null;
343 var points = new List<Point3D>();
344 foreach (var surface in face.Parametric)
345 {
346 if (surface.Vertices == null)
347 surface.Regen(0.01);
348 if (surface.Vertices != null)
349 points.AddRange(surface.Vertices);
350 }
351 if (points.Count == 0) return null;
352 return new Point3D(
353 points.Average(p => p.X),
354 points.Average(p => p.Y),
355 points.Average(p => p.Z)
356 );
357 }

390 private FaceTypeEnum InferFaceType(object surface, out string geometricInfo)
391 {
392 geometricInfo = "N/A";
393 if (surface == null) return FaceTypeEnum.Unknown;
394 try
395 {
396 if (surface is SphericalSurf sph)
397 {
398 geometricInfo = $"Radius: {sph.Radius:F2}";
399 return FaceTypeEnum.Spherical;
400 }
401 else if (surface is CylindricalSurf cyl)
402 {
403 geometricInfo = $"Radius: {cyl.Radius:F2}, Origin: ({cyl.Plane.Origin.X:F2}, {cyl.Plane.Origin.Y:F2}, {cyl.Plane.Origin.Z:F2}), Axis: ({cyl.Plane.AxisZ.X:F2}, {cyl.Plane.AxisZ.Y:F2}, {cyl.Plane.AxisZ.Z:F2})";
404 return FaceTypeEnum.Cylindrical;
405 }
406 else if (surface is ConicalSurf cone)
407 {
408 geometricInfo = $"Radius: {cone.Radius:F2}, Angle: {Utility.RadToDeg(cone.HalfAngle):F2}�";
409 return FaceTypeEnum.Conical;
410 }
411 else if (surface is ToroidalSurf tor)
412 {
413 geometricInfo = $"Major R: {tor.MajorRadius:F2}, Minor R: {tor.MinorRadius:F2}";
414 return FaceTypeEnum.Toroidal;
415 }
416 else if (surface is NurbsSurf)
417 {
418 geometricInfo = "NURBS Surface";
419 return FaceTypeEnum.BSpline;
420 }
421 else if (surface is RevolvedSurf rev)
422 {
423 geometricInfo = $"Axis: ({rev.Plane.AxisZ.X:F2}, {rev.Plane.AxisZ.Y:F2}, {rev.Plane.AxisZ.Z:F2})";
424 return FaceTypeEnum.Revolved;
425 }
426 else if (surface is TabulatedSurf tab)
427 {
428 geometricInfo = $"Direction: ({tab.Generatrix.X:F2}, {tab.Generatrix.Y:F2}, {tab.Generatrix.Z:F2})";
429 return FaceTypeEnum.Tabulated;
430 }
431 else if (surface is PlanarSurf pl)
432 {
433 geometricInfo = $"Normal: ({pl.Plane.AxisZ.X:F2}, {pl.Plane.AxisZ.Y:F2}, {pl.Plane.AxisZ.Z:F2})";
434 return FaceTypeEnum.Planar;
435 }
436 }
437 catch (Exception ex)
438 {
439 System.Diagnostics.Debug.WriteLine($"Error inferring face type: {ex.Message}");
440 }
441 return FaceTypeEnum.Unknown;
442 }

469 private Vector3D GetFaceNormalApprox(Brep.Face face)
470 {
471 try
472 {
473 var list = new List<Point3D>();
474 if (face.Parametric != null)
475 {
476 foreach (var surf in face.Parametric)
477 {
478 if (surf.Vertices == null) surf.Regen(0.01);
479 if (surf.Vertices != null) list.AddRange(surf.Vertices);
480 }
481 }
482
483 if (list.Count < 3)
484 return new Vector3D(0, 0, 0);
485
486 Point3D a = list[0];
487 int idxB = 1;
488 for (int i = 1; i < list.Count; i++)
489 {
490 if (!PointsAreEqual(a, list[i])) { idxB = i; break; }
491 }
492 Point3D b = list[idxB];
493
494 for (int i = idxB + 1; i < list.Count; i++)
495 {
496 Point3D c = list[i];
497 var vx1 = b.X - a.X; var vy1 = b.Y - a.Y; var vz1 = b.Z - a.Z;
498 var vx2 = c.X - a.X; var vy2 = c.Y - a.Y; var vz2 = c.Z - a.Z;
499 var cx = vy1 * vz2 - vz1 * vy2;
500 var cy = vz1 * vx2 - vx1 * vz2;
501 var cz = vx1 * vy2 - vy1 * vx2;
502 double clen = Math.Sqrt(cx * cx + cy * cy + cz * cz);
503 if (clen > 1e-9)
504 return new Vector3D(cx / clen, cy / clen, cz / clen);
505 }
506 return new Vector3D(0, 0, 0);
507 }
508 catch
509 {
510 return new Vector3D(0, 0, 0);
511 }
512 }

596 public class Point3DComparer : IEqualityComparer<Point3D>
597 {
602 private const double Tolerance = 1e-6;
603
619 public bool Equals(Point3D? p1, Point3D? p2)
620 {
621 if (ReferenceEquals(p1, p2)) return true;
622 if (p1 is null || p2 is null) return false;
623
624 return Math.Abs(p1.X - p2.X) < Tolerance &&
625 Math.Abs(p1.Y - p2.Y) < Tolerance &&
626 Math.Abs(p1.Z - p2.Z) < Tolerance;
627 }
628
650 public int GetHashCode(Point3D? p)
651 {
652 if (p is null) return 0;
653 int hashX = Math.Round(p.X / Tolerance).GetHashCode();
654 int hashY = Math.Round(p.Y / Tolerance).GetHashCode();
655 int hashZ = Math.Round(p.Z / Tolerance).GetHashCode();
656 return hashX ^ (hashY << 1) ^ (hashZ << 2);
657 }
