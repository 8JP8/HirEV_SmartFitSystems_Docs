125 private static devDept.Geometry.Transformation GetPlaneToXYTransform(devDept.Geometry.Plane sectionPlane)
126 {
127 // Build a frame from the section plane axes
128 var origin = sectionPlane.Origin;
129 var axisX = sectionPlane.AxisX;
130 var axisY = sectionPlane.AxisY;
131 var axisZ = sectionPlane.AxisZ;
132
133 // Matrix that takes section-plane coords → world coords (columns = plane axes)
134 double[,] m = {
135 { axisX.X, axisY.X, axisZ.X, origin.X },
136 { axisX.Y, axisY.Y, axisZ.Y, origin.Y },
137 { axisX.Z, axisY.Z, axisZ.Z, origin.Z },
138 { 0, 0, 0, 1 }
139 };
140
141 // We want the INVERSE: world → XY-aligned
142 var t = new devDept.Geometry.Transformation(m);
143 t.Invert();
144 return t;
145 }

577 private void CreateDimensionAndShow(Point3D ptA, Point3D ptB)
578 {
579 // Project the two world points onto the section drawing plane (returns Point2D)
580 Point2D p1_2d = sectionDesign.DrawingPlane.Project(ptA);
581 Point2D p2_2d = sectionDesign.DrawingPlane.Project(ptB);
582
583 // Lift to Point3D so we can do distance and vector arithmetic
584 var p1 = new Point3D(p1_2d.X, p1_2d.Y, 0);
585 var p2 = new Point3D(p2_2d.X, p2_2d.Y, 0);
586
587 // Build a custom Plane whose X-axis is the segment direction.
588 // LinearDim measures along the plane's X axis, so aligning X with the segment
589 // makes the dimension parallel to the segment instead of being locked to global X.
590 double segLen = p1.DistanceTo(p2);
591 if (segLen < 1e-6) return; // degenerate — nothing to dimension
592
593 var segDir = new Vector3D((p2.X - p1.X) / segLen, (p2.Y - p1.Y) / segLen, 0);
594 var perpDir = new Vector3D(-segDir.Y, segDir.X, 0); // 90° CCW in XY plane
595
596 // Dimension plane: origin at p1, X along segment, Y perpendicular, Z = screen-out
597 var dimPlane = new Plane(p1, segDir, perpDir);
598
599 // In dimPlane's 2D coordinate system:
600 // p1 → (0, 0) (plane origin)
601 // p2 → (segLen, 0) (along X = segment direction)
602 var lp1 = new Point2D(0, 0);
603 var lp2 = new Point2D(segLen, 0);
604
605 // Dim-line position: offset the midpoint 20 mm in the perpendicular (Y) direction
606 var dimLinePos = new Point2D(segLen / 2.0, 20.0);
607
608 var ldim = new LinearDim(dimPlane, lp1, lp2, dimLinePos, sectionDesign.DimTextHeight);
609
610 ldim.Color = System.Drawing.Color.Black;
611 ldim.ColorMethod = colorMethodType.byEntity;
612
613 // Store the original measured points as anchor data so AlignDimensionWithAxis
614 // can always recover the exact positions without reverse-engineering from the plane.
615 ldim.EntityData = new DimensionAnchor { P1 = p1, P2 = p2 };
616
617 // Always Regen before adding to scene to initialize internal graphics data (prevents crashes in v2025)
618 ldim.Regen(new devDept.Eyeshot.RegenParams(0.001, sectionDesign));
619 ldim.UpdateBoundingBox(new devDept.Eyeshot.TraversalParams());
620 sectionDesign.Entities.Add(ldim);
621
622 ShowMeasureDetails(ptA, ptB);
623 }

1171 private void AlignDimensionWithAxis(LinearDim ldim, string axis)
1172 {
1173 if (_xyToWorld == null) return;
1174
1175 // 1. Read the anchor points stored on the dimension (exact original measured positions)
1176 var anchor = ldim.EntityData as DimensionAnchor;
1177 if (anchor == null) return;
1178 var p1 = anchor.P1;
1179 var p2 = anchor.P2;
1180
1181 // 2. Determine the alignment direction in viewport space and the label prefix
1182 Vector3D alignmentDir;
1183 string prefix;
1184 if (_showCADCoordinates)
1185 {
1186 Vector3D cadAxis = axis switch { "X" => Vector3D.AxisX, "Y" => Vector3D.AxisY, _ => Vector3D.AxisZ };
1187 prefix = axis switch { "X" => "dX", "Y" => "dY", _ => "dZ" };
1188 var worldToLocal = (devDept.Geometry.Transformation)_xyToWorld.Clone();
1189 worldToLocal.Invert();
1190 alignmentDir = worldToLocal * cadAxis;
1191 }
1192 else
1193 {
1194 alignmentDir = axis switch { "X" => Vector3D.AxisX, "Y" => Vector3D.AxisY, _ => Vector3D.AxisZ };
1195 prefix = axis switch { "X" => "dX", "Y" => "dY", _ => "dZ" };
1196 }
1197
1198 // Project alignment direction onto XY and normalize
1199 alignmentDir.Z = 0;
1200 if (alignmentDir.Length < 1e-6) return;
1201 alignmentDir.Normalize();
1202
1203 // 3. Calculate the component distance
1204 var cp1 = _xyToWorld * p1;
1205 var cp2 = _xyToWorld * p2;
1206
1207 double dist;
1208 if (_showCADCoordinates)
1209 {
1210 dist = axis switch
1211 {
1212 "X" => Math.Abs(cp1.X - cp2.X),
1213 "Y" => Math.Abs(cp1.Y - cp2.Y),
1214 _ => Math.Abs(cp1.Z - cp2.Z)
1215 };
1216 }
1217 else
1218 {
1219 dist = axis switch
1220 {
1221 "X" => Math.Abs(p1.X - p2.X),
1222 "Y" => Math.Abs(p1.Y - p2.Y),
1223 _ => 0
1224 };
1225 }
1226
1227 // If the component distance is zero, show a topmost message and abort
1228 if (dist < 1e-4)
1229 {
1230 MessageBox.Show(this, $"The dimension returned 0 for the {axis} component.", "Invalid Dimension", MessageBoxButton.OK, MessageBoxImage.Information);
1231 return;
1232 }
1233
1234 // 4. Build the dimension anchored to the original measured points.
1235 // Plane origin at p1, AxisX = alignment direction, AxisY = perp in XY.
1236 var perpDir = new Vector3D(-alignmentDir.Y, alignmentDir.X, 0);
1237 var dimPlane = new Plane(p1, alignmentDir, perpDir);
1238
1239 // Project p1 and p2 into the dim plane's local 2D space
1240 var delta = new Vector3D(p2.X - p1.X, p2.Y - p1.Y, 0);
1241 double projX = Vector3D.Dot(delta, alignmentDir); // component along axis = dimension width
1242 double projY = Vector3D.Dot(delta, perpDir); // perpendicular offset
1243
1244 // Extension lines start exactly at the measured points
1245 var lp1 = new Point2D(0, 0);
1246 var lp2 = new Point2D(projX, projY);
1247
1248 // Dim-line centred on X span, offset 20mm perpendicular from the midpoint of the legs
1249 var dimLinePos = new Point2D(projX / 2.0, (projY / 2.0) + 20.0);
1250
1251 var newDim = new LinearDim(dimPlane, lp1, lp2, dimLinePos, sectionDesign.DimTextHeight);
1252 newDim.Color = ldim.Color;
1253 newDim.ColorMethod = ldim.ColorMethod;
1254 newDim.TextOverride = $"{prefix}: {dist:F3} mm";
1255 anchor.ActiveAxis = axis; // track which axis is active for context menu filtering
1256 newDim.EntityData = anchor; // preserve the anchor for future re-alignment
1257
1258 newDim.Regen(new devDept.Eyeshot.RegenParams(0.001, sectionDesign));
1259 newDim.UpdateBoundingBox(new devDept.Eyeshot.TraversalParams());
1260
1261 // Replace in scene
1262 sectionDesign.Entities.Remove(ldim);
1263 sectionDesign.Entities.Add(newDim);
1264
1265 // Update info panel if this dimension was selected
1266 if (ldim.Selected)
1267 {
1268 newDim.Selected = true;
1269 ShowDimensionDetails(newDim);
1270 }
1271 }
