128 private static devDept.Geometry.Transformation GetPlaneToXYTransform(devDept.Geometry.Plane sectionPlane)
129 {
130 // Build a frame from the section plane axes
131 var origin = sectionPlane.Origin;
132 var axisX = sectionPlane.AxisX;
133 var axisY = sectionPlane.AxisY;
134 var axisZ = sectionPlane.AxisZ;
135
136 // Matrix that takes section-plane coords → world coords (columns = plane axes)
137 double[,] m = {
138 { axisX.X, axisY.X, axisZ.X, origin.X },
139 { axisX.Y, axisY.Y, axisZ.Y, origin.Y },
140 { axisX.Z, axisY.Z, axisZ.Z, origin.Z },
141 { 0, 0, 0, 1 }
142 };
143
144 // We want the INVERSE: world → XY-aligned
145 var t = new devDept.Geometry.Transformation(m);
146 t.Invert();
147 return t;
148 }

815 private void CreateDimensionAndShow(Point3D ptA, Point3D ptB, bool isRadius = false, double radius = 0)
816 {
817 // Project the two world points onto the section drawing plane (returns Point2D)
818 Point2D p1_2d = sectionDesign.DrawingPlane.Project(ptA);
819 Point2D p2_2d = sectionDesign.DrawingPlane.Project(ptB);
820
821 // Lift to Point3D so we can do distance and vector arithmetic
822 var p1 = new Point3D(p1_2d.X, p1_2d.Y, 0);
823 var p2 = new Point3D(p2_2d.X, p2_2d.Y, 0);
824
825 // Build a custom Plane whose X-axis is the segment direction.
826 // LinearDim measures along the plane's X axis, so aligning X with the segment
827 // makes the dimension parallel to the segment instead of being locked to global X.
828 double segLen = p1.DistanceTo(p2);
829 if (segLen < 1e-6) return; // degenerate — nothing to dimension
830
831 var segDir = new Vector3D((p2.X - p1.X) / segLen, (p2.Y - p1.Y) / segLen, 0);
832 var perpDir = new Vector3D(-segDir.Y, segDir.X, 0); // 90° CCW in XY plane
833
834 // Dimension plane: origin at p1, X along segment, Y perpendicular, Z = screen-out
835 var dimPlane = new Plane(p1, segDir, perpDir);
836
837 // In dimPlane's 2D coordinate system:
838 // p1 → (0, 0) (plane origin)
839 // p2 → (segLen, 0) (along X = segment direction)
840 var lp1 = new Point2D(0, 0);
841 var lp2 = new Point2D(segLen, 0);
842
843 // Dim-line position: offset the midpoint 20 mm in the perpendicular (Y) direction
844 var dimLinePos = new Point2D(segLen / 2.0, 20.0);
845
846 var ldim = new LinearDim(dimPlane, lp1, lp2, dimLinePos, sectionDesign.DimTextHeight);
847
848 ldim.Color = System.Drawing.Color.Black;
849 ldim.ColorMethod = colorMethodType.byEntity;
850
851 double displayRad = radius > 0 ? radius : segLen;
852 if (isRadius)
853 {
854 ldim.TextOverride = $"⌀ {displayRad:0.###} mm";
855 }
856
857 // Store the original measured points as anchor data so AlignDimensionWithAxis
858 // can always recover the exact positions without reverse-engineering from the plane.
859 ldim.EntityData = new DimensionAnchor
860 {
861 P1 = p1,
862 P2 = p2,
863 IsRadiusMeasurement = isRadius,
864 Radius = displayRad
865 };
866
867 // Always Regen before adding to scene to initialize internal graphics data (prevents crashes in v2025)
868 ldim.Regen(new devDept.Eyeshot.RegenParams(0.001, sectionDesign));
869 ldim.UpdateBoundingBox(new devDept.Eyeshot.TraversalParams());
870 sectionDesign.Entities.Add(ldim);
871
872 ShowMeasureDetails(ptA, ptB, isRadius, displayRad);
873 }

1138 private bool TryGetCylindricalInfo(Entity ent, out Point3D centerPt, out double radius)
1139 {
1140 centerPt = Point3D.Origin;
1141 radius = 0.0;
1142
1143 if (ent == null) return false;
1144
1145 // Direct circle
1146 if (ent is Circle circle)
1147 {
1148 centerPt = circle.Center;
1149 radius = circle.Radius;
1150 return true;
1151 }
1152
1153 // Direct arc
1154 if (ent is devDept.Eyeshot.Entities.Arc arc)
1155 {
1156 centerPt = arc.Center;
1157 radius = arc.Radius;
1158 return true;
1159 }
1160
1161 // Extract expected CAD radius if available
1162 double expectedCadRadius = 0.0;
1163 if (ent.EntityData is CADFace cf && cf.FaceType == FaceTypeEnum.Cylindrical && cf.GeometricInfo != null)
1164 {
1165 var match = System.Text.RegularExpressions.Regex.Match(cf.GeometricInfo, @"Radius:\s*([\d.]+)");
1166 if (match.Success && double.TryParse(match.Groups[1].Value, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double r))
1167 {
1168 expectedCadRadius = r;
1169 }
1170 }
1171 else if (ent.EntityData is CADHole hole)
1172 {
1173 if (!double.IsNaN(hole.TopDiameter) && hole.TopDiameter > 0)
1174 {
1175 expectedCadRadius = hole.TopDiameter / 2.0;
1176 }
1177 else if (!double.IsNaN(hole.BottomDiameter) && hole.BottomDiameter > 0)
1178 {
1179 expectedCadRadius = hole.BottomDiameter / 2.0;
1180 }
1181 else if (!string.IsNullOrEmpty(hole.GeometricInfo))
1182 {
1183 var match = System.Text.RegularExpressions.Regex.Match(hole.GeometricInfo, @"(?:Radius|Raio):\s*([\d.]+)");
1184 if (!match.Success) match = System.Text.RegularExpressions.Regex.Match(hole.GeometricInfo, @"(?:Diameter|Di[âa]metro):\s*([\d.]+)");
1185 if (match.Success && double.TryParse(match.Groups[1].Value, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double val))
1186 {
1187 expectedCadRadius = hole.GeometricInfo.Contains("Diameter", StringComparison.OrdinalIgnoreCase) || hole.GeometricInfo.Contains("Diâmetro", StringComparison.OrdinalIgnoreCase) ? val / 2.0 : val;
1188 }
1189 }
1190 }
1191
1192 // Try circle fitting from curve geometry in section XY plane
1193 Point3D? p0 = null, p1 = null, p2 = null;
1194
1195 if (ent is ICurve ic)
1196 {
1197 bool isGeomClosed = ic.IsClosed || (ic.StartPoint != null && ic.EndPoint != null && ic.StartPoint.DistanceTo(ic.EndPoint) < 1e-4);
1198 if (isGeomClosed)
1199 {
1200 double len = ic.Domain.Max - ic.Domain.Min;
1201 p0 = ic.PointAt(ic.Domain.Min);
1202 p1 = ic.PointAt(ic.Domain.Min + len / 3.0);
1203 p2 = ic.PointAt(ic.Domain.Min + 2.0 * len / 3.0);
1204 }
1205 else
1206 {
1207 p0 = ic.PointAt(ic.Domain.Min);
1208 p1 = ic.PointAt((ic.Domain.Min + ic.Domain.Max) / 2.0);
1209 p2 = ic.PointAt(ic.Domain.Max);
1210 }
1211 }
1212 else if (ent.Vertices != null && ent.Vertices.Length >= 3)
1213 {
1214 bool isGeomClosed = ent.Vertices[0].DistanceTo(ent.Vertices[ent.Vertices.Length - 1]) < 1e-4;
1215 if (isGeomClosed)
1216 {
1217 p0 = ent.Vertices[0];
1218 p1 = ent.Vertices[ent.Vertices.Length / 3];
1219 p2 = ent.Vertices[2 * ent.Vertices.Length / 3];
1220 }
1221 else
1222 {
1223 p0 = ent.Vertices[0];
1224 p1 = ent.Vertices[ent.Vertices.Length / 2];
1225 p2 = ent.Vertices[ent.Vertices.Length - 1];
1226 }
1227 }
1228
1229 if (p0 != null && p1 != null && p2 != null)
1230 {
1231 double d = 2.0 * (p0.X * (p1.Y - p2.Y) + p1.X * (p2.Y - p0.Y) + p2.X * (p0.Y - p1.Y));
1232 if (Math.Abs(d) > 1e-6)
1233 {
1234 double ux = ((p0.X * p0.X + p0.Y * p0.Y) * (p1.Y - p2.Y) + (p1.X * p1.X + p1.Y * p1.Y) * (p2.Y - p0.Y) + (p2.X * p2.X + p2.Y * p2.Y) * (p0.Y - p1.Y)) / d;
1235 double uy = ((p0.X * p0.X + p0.Y * p0.Y) * (p2.X - p1.X) + (p1.X * p1.X + p1.Y * p1.Y) * (p0.X - p2.X) + (p2.X * p2.X + p2.Y * p2.Y) * (p1.X - p0.X)) / d;
1236 var candCenter = new Point3D(ux, uy, 0);
1237 double candRadius = candCenter.DistanceTo(p0);
1238
1239 // Validate reasonable radius
1240 if (candRadius > 0.05 && candRadius < 5000)
1241 {
1242 bool isValid = true;
1243 if (ent is ICurve icCheck && !icCheck.IsClosed)
1244 {
1245 Point3D pTest = icCheck.PointAt(icCheck.Domain.Min + (icCheck.Domain.Max - icCheck.Domain.Min) * 0.25);
1246 if (Math.Abs(candCenter.DistanceTo(pTest) - candRadius) > Math.Max(0.2, candRadius * 0.08))
1247 {
1248 isValid = false;
1249 }
1250 }
1251
1252 if (isValid || expectedCadRadius > 0)
1253 {
1254 centerPt = candCenter;
1255 radius = expectedCadRadius > 0 ? expectedCadRadius : candRadius;
1256 return true;
1257 }
1258 }
1259 }
1260 }
1261
1262 // Fallback for holes: intersect hole axis with section plane
1263 if (ent.EntityData is CADHole h && expectedCadRadius > 0 && h.TopCenterPoint != null)
1264 {
1265 Vector3D? axis = h.AxisDirection;
1266 if (axis == null && h.BottomCenterPoint != null)
1267 {
1268 axis = new Vector3D(h.TopCenterPoint, h.BottomCenterPoint);
1269 axis.Normalize();
1270 }
1271
1272 if (axis != null)
1273 {
1274 double denom = Vector3D.Dot(axis, _sectionPlane.AxisZ);
1275 if (Math.Abs(denom) > 1e-5)
1276 {
1277 double dist = Vector3D.Dot(_sectionPlane.Origin - h.TopCenterPoint, _sectionPlane.AxisZ);
1278 double t = dist / denom;
1279 Point3D intersectWorld = h.TopCenterPoint + t * axis;
1280
1281 var transform = GetPlaneToXYTransform(_sectionPlane);
1282 Point3D intersectXY = transform * intersectWorld;
1283 centerPt = new Point3D(intersectXY.X, intersectXY.Y, 0);
1284 radius = expectedCadRadius;
1285 return true;
1286 }
1287 }
1288 }
1289
1290 // Fallback if CAD face or hole has known radius but fitting was not possible
1291 if (expectedCadRadius > 0)
1292 {
1293 radius = expectedCadRadius;
1294 return true;
1295 }
1296
1297 return false;
1298 }

1692 private void AlignDimensionWithAxis(LinearDim ldim, string axis)
1693 {
1694 if (_xyToWorld == null) return;
1695
1696 // 1. Read the anchor points stored on the dimension (exact original measured positions)
1697 var anchor = ldim.EntityData as DimensionAnchor;
1698 if (anchor == null) return;
1699 var p1 = anchor.P1;
1700 var p2 = anchor.P2;
1701
1702 // 2. Determine the alignment direction in viewport space and the label prefix
1703 Vector3D alignmentDir;
1704 string prefix;
1705 if (_showCADCoordinates)
1706 {
1707 Vector3D cadAxis = axis switch { "X" => Vector3D.AxisX, "Y" => Vector3D.AxisY, _ => Vector3D.AxisZ };
1708 prefix = axis switch { "X" => "dX", "Y" => "dY", _ => "dZ" };
1709 var worldToLocal = (devDept.Geometry.Transformation)_xyToWorld.Clone();
1710 worldToLocal.Invert();
1711 alignmentDir = worldToLocal * cadAxis;
1712 }
1713 else
1714 {
1715 alignmentDir = axis switch { "X" => Vector3D.AxisX, "Y" => Vector3D.AxisY, _ => Vector3D.AxisZ };
1716 prefix = axis switch { "X" => "dX", "Y" => "dY", _ => "dZ" };
1717 }
1718
1719 // Project alignment direction onto XY and normalize
1720 alignmentDir.Z = 0;
1721 if (alignmentDir.Length < 1e-6) return;
1722 alignmentDir.Normalize();
1723
1724 // 3. Calculate the component distance
1725 var cp1 = _xyToWorld * p1;
1726 var cp2 = _xyToWorld * p2;
1727
1728 double dist;
1729 if (_showCADCoordinates)
1730 {
1731 dist = axis switch
1732 {
1733 "X" => Math.Abs(cp1.X - cp2.X),
1734 "Y" => Math.Abs(cp1.Y - cp2.Y),
1735 _ => Math.Abs(cp1.Z - cp2.Z)
1736 };
1737 }
1738 else
1739 {
1740 dist = axis switch
1741 {
1742 "X" => Math.Abs(p1.X - p2.X),
1743 "Y" => Math.Abs(p1.Y - p2.Y),
1744 _ => 0
1745 };
1746 }
1747
1748 // If the component distance is zero, show a topmost message and abort
1749 if (dist < 1e-4)
1750 {
1751 MessageBox.Show(this, $"The dimension returned 0 for the {axis} component.", "Invalid Dimension", MessageBoxButton.OK, MessageBoxImage.Information);
1752 return;
1753 }
1754
1755 // 4. Build the dimension anchored to the original measured points.
1756 // Plane origin at p1, AxisX = alignment direction, AxisY = perp in XY.
1757 var perpDir = new Vector3D(-alignmentDir.Y, alignmentDir.X, 0);
1758 var dimPlane = new Plane(p1, alignmentDir, perpDir);
1759
1760 // Project p1 and p2 into the dim plane's local 2D space
1761 var delta = new Vector3D(p2.X - p1.X, p2.Y - p1.Y, 0);
1762 double projX = Vector3D.Dot(delta, alignmentDir); // component along axis = dimension width
1763 double projY = Vector3D.Dot(delta, perpDir); // perpendicular offset
1764
1765 // Extension lines start exactly at the measured points
1766 var lp1 = new Point2D(0, 0);
1767 var lp2 = new Point2D(projX, projY);
1768
1769 // Dim-line centred on X span, offset 20mm perpendicular from the midpoint of the legs
1770 var dimLinePos = new Point2D(projX / 2.0, (projY / 2.0) + 20.0);
1771
1772 var newDim = new LinearDim(dimPlane, lp1, lp2, dimLinePos, sectionDesign.DimTextHeight);
1773 newDim.Color = ldim.Color;
1774 newDim.ColorMethod = ldim.ColorMethod;
1775 newDim.TextOverride = $"{prefix}: {dist:F3} mm";
1776 anchor.ActiveAxis = axis; // track which axis is active for context menu filtering
1777 newDim.EntityData = anchor; // preserve the anchor for future re-alignment
1778
1779 newDim.Regen(new devDept.Eyeshot.RegenParams(0.001, sectionDesign));
1780 newDim.UpdateBoundingBox(new devDept.Eyeshot.TraversalParams());
1781
1782 // Replace in scene
1783 sectionDesign.Entities.Remove(ldim);
1784 sectionDesign.Entities.Add(newDim);
1785
1786 // Update info panel if this dimension was selected
1787 if (ldim.Selected)
1788 {
1789 newDim.Selected = true;
1790 ShowDimensionDetails(newDim);
1791 }
