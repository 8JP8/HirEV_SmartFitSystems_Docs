51 public static List<string> GetAvailableToolNames(bool includeDefaultTool = true)
52 {
53 var modelNames = new SortedSet<string>(StringComparer.OrdinalIgnoreCase);
54
55 try
56 {
57 string toolsPath = ResolveModelsFolder(TOOLS_FOLDER);
58 if (Directory.Exists(toolsPath))
59 {
60 foreach (var modelFile in Directory.EnumerateFiles(toolsPath, "*.*", SearchOption.AllDirectories))
61 {
62 string extension = Path.GetExtension(modelFile);
63 if (!SupportedModelExtensions.Contains(extension, StringComparer.OrdinalIgnoreCase))
64 continue;
65
66 string modelName = Path.GetFileNameWithoutExtension(modelFile);
67 if (!string.IsNullOrWhiteSpace(modelName))
68 modelNames.Add(modelName.Trim());
69 }
70 }
71 }
72 catch (Exception ex)
73 {
74 Debug.WriteLine($"⚠️ Error scanning available tool models: {ex.Message}");
75 }
76
77 var tools = new List<string>();
78 if (includeDefaultTool)
79 tools.Add(MeasuringToolTypes.DefaultTool);
80
81 tools.AddRange(modelNames.Where(name =>
82 !string.Equals(name, MeasuringToolTypes.DefaultTool, StringComparison.OrdinalIgnoreCase)));
83
84 return tools;
85 }

170 private static void ValidateCacheFreshness(string modelPathOrName, bool isScanner)
171 {
172 if (string.IsNullOrWhiteSpace(modelPathOrName)) return;
173
174 string modelName = Path.GetFileNameWithoutExtension(modelPathOrName);
175 string? modelFile = ResolveModelFile(modelPathOrName, isScanner ? SCANNERS_FOLDER : TOOLS_FOLDER);
176 if (modelFile == null && isScanner)
177 modelFile = ResolveModelFile(modelPathOrName, TOOLS_FOLDER);
178 string lookupPath = modelFile ?? modelPathOrName;
179
180 string xmlPath = GetTransformationPath(lookupPath, ".xml", isScanner);
181 string jsonPath = GetTransformationPath(lookupPath, ".json", isScanner);
182
183 DateTime latestTime = DateTime.MinValue;
184 if (File.Exists(xmlPath))
185 {
186 DateTime t = File.GetLastWriteTimeUtc(xmlPath);
187 if (t > latestTime) latestTime = t;
188 }
189 if (File.Exists(jsonPath))
190 {
191 DateTime t = File.GetLastWriteTimeUtc(jsonPath);
192 if (t > latestTime) latestTime = t;
193 }
194
195 if (latestTime == DateTime.MinValue)
196 {
197 return;
198 }
199
200 if (_cachedModelTransformationTimestamps.TryGetValue(modelName, out DateTime cachedTime))
201 {
202 if (latestTime > cachedTime)
203 {
204 Debug.WriteLine($"🔄 Cache stale for model '{modelName}' (disk: {latestTime}, cached: {cachedTime}). Invalidating cache.");
205 _cachedToolModelsData.Remove(modelName);
206 _cachedToolModels.Remove(modelName);
207 _cachedTransformations.Remove(GetTransformationCacheKey(modelName, isScanner));
208 _cachedModelTransformationTimestamps[modelName] = latestTime;
209 }
210 }
211 else
212 {
213 // First time seeing this model's transformation timestamps. Just
214 // record the current disk timestamp so future calls can detect changes.
215 // Do NOT evict the cache here — the model was likely preloaded by
216 // LoadModelIntoCacheAsync which doesn't store transformation timestamps.
217 // Evicting here would nuke the preloaded data and force a full, expensive
218 // reload from disk on the UI thread.
219 _cachedModelTransformationTimestamps[modelName] = latestTime;
220 }

228 if (string.IsNullOrWhiteSpace(modelName))

231 if (modelName.EndsWith(".stp", StringComparison.OrdinalIgnoreCase) ||
232 modelName.EndsWith(".step", StringComparison.OrdinalIgnoreCase) ||
233 modelName.EndsWith(".stl", StringComparison.OrdinalIgnoreCase))
234 {
235 modelName = Path.GetFileNameWithoutExtension(modelName);
236 }
237
238 bool isScanner = false;
239 string? modelFile = ResolveModelFile(modelName, TOOLS_FOLDER);
240 if (modelFile == null)
241 {
242 modelFile = ResolveModelFile(modelName, SCANNERS_FOLDER);
243 isScanner = (modelFile != null);
244 }
245 string lookupPath = modelFile ?? modelName;
246
247 string xmlPath = GetTransformationPath(lookupPath, ".xml", isScanner: false);
248 string jsonPath = GetTransformationPath(lookupPath, ".json", isScanner: false);
249 if (!File.Exists(xmlPath) && !File.Exists(jsonPath))
250 {
251 xmlPath = GetTransformationPath(lookupPath, ".xml", isScanner: true);
252 jsonPath = GetTransformationPath(lookupPath, ".json", isScanner: true);
253 isScanner = true;
254 }
255
256 ValidateCacheFreshness(lookupPath, isScanner);
257
258 return _cachedToolModels.ContainsKey(modelName);

264 public static async Task<bool> LoadModelIntoCacheAsync(string modelPathOrName, bool isScanner, Action<double>? progressCallback = null)
265 {
266 if (string.IsNullOrWhiteSpace(modelPathOrName))
267 return false;
268
269 string modelName = Path.GetFileNameWithoutExtension(modelPathOrName);
270 if (string.IsNullOrWhiteSpace(modelName))
271 return false;
272
273 ValidateCacheFreshness(modelPathOrName, isScanner);
274
275 if (_cachedToolModelsData.ContainsKey(modelName))
276 return true;
277
278
279 try
280 {
281 string? resolvedFile = null;
282 if (isScanner)
283 {
284 resolvedFile = ResolveModelFile(modelPathOrName, SCANNERS_FOLDER);
285 if (resolvedFile == null)
286 resolvedFile = ResolveModelFile(modelPathOrName, TOOLS_FOLDER);
287 }
288 else
289 {
290 resolvedFile = ResolveModelFile(modelPathOrName, TOOLS_FOLDER);
291 }
292
293 if (resolvedFile == null)
294 {
295 Debug.WriteLine($"❌ No model file found for: {modelPathOrName}");
296 return false;
297 }
298
299 Debug.WriteLine($"📥 Background loading model from: {Path.GetFileName(resolvedFile)}");
300 CachedModelData? cachedData = null;
301
302 await Task.Run(() =>
303 {
304 cachedData = LoadCachedModelDataFromFile(resolvedFile, isScanner, progressCallback);
305 }).ConfigureAwait(false);
306
307 if (cachedData != null)
308 {
309 _cachedToolModelsData[modelName] = cachedData;
310 _cachedToolModels[modelName] = new Mesh(); // keep existing ContainsKey checks working
311 Debug.WriteLine($"✅ Background loaded and cached model data for: {modelName}");
312 return true;
313 }
314 }
315 catch (Exception ex)
316 {
317 Debug.WriteLine($"❌ Error in LoadModelIntoCacheAsync: {ex.Message}");
318 }
319
320 return false;
321 }

330 private static CachedModelData? LoadCachedModelDataFromFile(string filePath, bool isScanner, Action<double>? progressCallback = null)
331 {
332 try
333 {
334 Debug.WriteLine($"📖 Reading file: {Path.GetFileName(filePath)}");
335
336 string extension = Path.GetExtension(filePath).ToLowerInvariant();
337 string modelName = Path.GetFileNameWithoutExtension(filePath);
338
339 ReadSTL? stlReader = null;
340 ReadSTEP? stepReader = null;
341
342 if (extension == ".stl")
343 {
344 Debug.WriteLine($" Using STL reader...");
345 stlReader = new ReadSTL(filePath);
346 if (progressCallback != null)
347 stlReader.ProgressChanged += (s, e) => progressCallback(e.Progress);
348 }
349 else if (extension == ".stp" || extension == ".step")
350 {
351 Debug.WriteLine($" Using STEP reader...");
352 stepReader = new ReadSTEP(filePath);
353 if (progressCallback != null)
354 stepReader.ProgressChanged += (s, e) => progressCallback(e.Progress);
355 }
356 else
357 {
358 Debug.WriteLine($" ❌ Unsupported extension: {extension}");
359 return null;
360 }
361
362 if (stlReader != null)
363 stlReader.DoWork();
364 else if (stepReader != null)
365 stepReader.DoWork();
366
367 CachedModelData? cachedData = null;
368
369 System.Windows.Application.Current.Dispatcher.Invoke(() =>
370 {
371 try
372 {
373 Design tempDesign = new Design();
374 tempDesign.BeginInit();
375 tempDesign.EndInit();
376
377 if (stlReader != null)
378 stlReader.AddTo(tempDesign);
379 else if (stepReader != null)
380 stepReader.AddTo(tempDesign);
381
382 cachedData = new CachedModelData();
383 cachedData.Entities = tempDesign.Entities.Select(e => (Entity)e.Clone()).ToList();
384 cachedData.Blocks = tempDesign.Blocks.Select(b => (Block)b.Clone()).ToList();
385
386 CenterAndTransformCachedEntities(cachedData, filePath, isScanner);
387 }
388 catch (Exception ex)
389 {
390 Debug.WriteLine($"❌ Error in tempDesign processing on UI thread: {ex.Message}");
391 }
392 });
393
394 return cachedData;
395 }
396 catch (Exception ex)
397 {
398 Debug.WriteLine($"❌ Error loading cached model data from file: {ex.Message}");
399 return null;
400 }

409 private static void CenterAndTransformCachedEntities(CachedModelData cachedData, string modelFilePath, bool isScanner)
410 {
411 if (cachedData.Entities.Count == 0)
412 return;
413
414 try
415 {
416 string toolName = Path.GetFileNameWithoutExtension(modelFilePath);
417 // Calculate the overall bounding box of the entities
418 Point3D boxMin = new Point3D(double.MaxValue, double.MaxValue, double.MaxValue);
419 Point3D boxMax = new Point3D(double.MinValue, double.MinValue, double.MinValue);
420 bool hasBox = false;
421
422 foreach (var entity in cachedData.Entities)
423 {
424 try
425 {
426 Design tempDesign = new Design();
427 tempDesign.BeginInit();
428 tempDesign.EndInit();
429
430 if (!string.IsNullOrEmpty(entity.LayerName) && !tempDesign.Layers.Contains(entity.LayerName))
431 {
432 tempDesign.Layers.Add(new devDept.Eyeshot.Layer(entity.LayerName));
433 }
434
435 foreach (var b in cachedData.Blocks)
436 {
437 if (!tempDesign.Blocks.Contains(b.Name))
438 {
439 var clonedBlock = (Block)b.Clone();
440 foreach (var ent in clonedBlock.Entities)
441 {
442 if (!string.IsNullOrEmpty(ent.LayerName) && !tempDesign.Layers.Contains(ent.LayerName))
443 {
444 tempDesign.Layers.Add(new devDept.Eyeshot.Layer(ent.LayerName));
445 }
446 }
447 tempDesign.Blocks.Add(clonedBlock);
448 }
449 }
450
451 // Add entity to tempDesign so it can resolve block references correctly
452 tempDesign.Entities.Add(entity);
453 entity.UpdateBoundingBox(new devDept.Eyeshot.TraversalParams());
454
455 if (entity.BoxMin != null && entity.BoxMax != null)
456 {
457 boxMin.X = Math.Min(boxMin.X, entity.BoxMin.X);
458 boxMin.Y = Math.Min(boxMin.Y, entity.BoxMin.Y);
459 boxMin.Z = Math.Min(boxMin.Z, entity.BoxMin.Z);
460 boxMax.X = Math.Max(boxMax.X, entity.BoxMax.X);
461 boxMax.Y = Math.Max(boxMax.Y, entity.BoxMax.Y);
462 boxMax.Z = Math.Max(boxMax.Z, entity.BoxMax.Z);
463 hasBox = true;
464 }
465 }
466 catch (Exception entityEx)
467 {
468 Debug.WriteLine($"⚠️ Warning in CenterAndTransformCachedEntities for entity: {entityEx.Message}");
469 try
470 {
471 entity.UpdateBoundingBox(new devDept.Eyeshot.TraversalParams());
472 if (entity.BoxMin != null && entity.BoxMax != null)
473 {
474 boxMin.X = Math.Min(boxMin.X, entity.BoxMin.X);
475 boxMin.Y = Math.Min(boxMin.Y, entity.BoxMin.Y);
476 boxMin.Z = Math.Min(boxMin.Z, entity.BoxMin.Z);
477 boxMax.X = Math.Max(boxMax.X, entity.BoxMax.X);
478 boxMax.Y = Math.Max(boxMax.Y, entity.BoxMax.Y);
479 boxMax.Z = Math.Max(boxMax.Z, entity.BoxMax.Z);
480 hasBox = true;
481 }
482 }
483 catch
484 {
485 // Ignore if completely unable to compute bounding box
486 }
487 }
488 }
489
490 if (!hasBox)
491 {
492 Debug.WriteLine(" ⚠️ Bounding box calculation returned no valid box");
493 return;
494 }
495
496 double centerX = (boxMin.X + boxMax.X) / 2.0;
497 double centerY = (boxMin.Y + boxMax.Y) / 2.0;
498 double bottomZ = boxMin.Z;
499
500 Debug.WriteLine($" Original bounds: min({boxMin.X:F2},{boxMin.Y:F2},{boxMin.Z:F2}) max({boxMax.X:F2},{boxMax.Y:F2},{boxMax.Z:F2})");
501 Debug.WriteLine($" Translating to center: ({-centerX:F2}, {-centerY:F2}, {-bottomZ:F2})");
502
503 // Translate all entities
504 foreach (var entity in cachedData.Entities)
505 {
506 entity.Translate(-centerX, -centerY, -bottomZ);
507 }
508
509 // If it is a tool (not scanner), flip it top to bottom by default!
510 if (!isScanner)
511 {
512 Debug.WriteLine(" Flipping tool top-to-bottom by default (rotating 180 degrees around X axis)");
513 foreach (var entity in cachedData.Entities)
514 {
515 entity.Rotate(Math.PI, Vector3D.AxisX);
516 }
517 }
518
519 // Apply custom transformations
520 var transformation = LoadTransformation(modelFilePath, isScanner);
521 if (transformation != null)
522 {
523 Debug.WriteLine($" Applying custom transformation: scale={transformation.ScaleFactor}, rot=({transformation.RotateX},{transformation.RotateY},{transformation.RotateZ}), trans=({transformation.TranslateX},{transformation.TranslateY},{transformation.TranslateZ})");
524 foreach (var entity in cachedData.Entities)
525 {
526 if (Math.Abs(transformation.ScaleFactor - 1.0) > 0.001)
527 entity.Scale(transformation.ScaleFactor, transformation.ScaleFactor, transformation.ScaleFactor);
528
529 if (Math.Abs(transformation.RotateX) > 0.001)
530 entity.Rotate(transformation.RotateX * Math.PI / 180.0, Vector3D.AxisX);
531
532 if (Math.Abs(transformation.RotateY) > 0.001)
533 entity.Rotate(transformation.RotateY * Math.PI / 180.0, Vector3D.AxisY);
534
535 if (Math.Abs(transformation.RotateZ) > 0.001)
536 entity.Rotate(transformation.RotateZ * Math.PI / 180.0, Vector3D.AxisZ);
537
538 entity.Translate(transformation.TranslateX, transformation.TranslateY, transformation.TranslateZ);
539 }
540 }
541
542 // Strip layers to prevent 'Invalid Layer with name 1' exceptions when adding to active Design
543 foreach (var entity in cachedData.Entities)
544 {
545 entity.LayerName = "Default";
546 }
547 foreach (var block in cachedData.Blocks)
548 {
549 foreach (var entity in block.Entities)
550 {
551 entity.LayerName = "Default";
552 }
553 }
554 }
555 catch (Exception ex)
556 {
557 Debug.WriteLine($"❌ Error in CenterAndTransformCachedEntities: {ex.Message}");
558 }
559 }

626 }

628 Debug.WriteLine($"✅ Registered block {modelBlockName} in active design blocks");
629 }
630
631 // Return a BlockReference pointing to the block
632 BlockReference br = new BlockReference(modelBlockName);
633 return br;
634 }
635 catch (Exception ex)
636 {
637 Debug.WriteLine($"❌ Error loading machine model {modelPathOrName}: {ex.Message}");
638 return null;

753 private static ToolTransformation LoadTransformation(string modelFilePath, bool isScanner = false)
754 {
755 string toolName = Path.GetFileNameWithoutExtension(modelFilePath);
756 string cacheKey = GetTransformationCacheKey(toolName, isScanner);
757 if (_cachedTransformations.ContainsKey(cacheKey))
758 return _cachedTransformations[cacheKey];
759
760 try
761 {
762 EnsureTransformationFileExists(modelFilePath, isScanner);
763
764 string xmlPath = GetTransformationPath(modelFilePath, ".xml", isScanner);
765 string jsonPath = GetTransformationPath(modelFilePath, ".json", isScanner);
766
767 ToolTransformation? transformation = null;
768 if (File.Exists(xmlPath))
769 transformation = ReadXmlTransformation(xmlPath);
770 else if (File.Exists(jsonPath))
771 transformation = ReadJsonTransformation(jsonPath);
772
773 if (transformation != null)
774 {
775 _cachedTransformations[cacheKey] = transformation;
776 Debug.WriteLine($"✅ Loaded transformation for tool: {toolName}");
777 return transformation;
778 }
779 }
780 catch (Exception ex)
781 {
782 Debug.WriteLine($"⚠️ Error loading transformation for {toolName}: {ex.Message}");
783 }
784
785 // Return default transformation
786 var defaultTransform = new ToolTransformation { ToolName = toolName };
787 _cachedTransformations[cacheKey] = defaultTransform;
788 return defaultTransform;

830 private static void SynchronizeTransformationFiles(string modelFilePath, bool isScanner = false)
831 {
832 string toolName = Path.GetFileNameWithoutExtension(modelFilePath);
833 string xmlPath = GetTransformationPath(modelFilePath, ".xml", isScanner);
834 string jsonPath = GetTransformationPath(modelFilePath, ".json", isScanner);
835 bool hasXml = File.Exists(xmlPath);
836 bool hasJson = File.Exists(jsonPath);
837
838 if (!hasXml && hasJson)
839 {
840 var jsonTransform = ReadJsonTransformation(jsonPath) ?? CreateDefaultTransformation(toolName);
841 WriteXmlTransformation(xmlPath, jsonTransform);
842 _cachedTransformations.Remove(GetTransformationCacheKey(toolName, isScanner));
843 return;
844 }
845
846 if (!hasXml || !hasJson)
847 return;
848
849 DateTime xmlTime = File.GetLastWriteTimeUtc(xmlPath);
850 DateTime jsonTime = File.GetLastWriteTimeUtc(jsonPath);
851
852 if (Math.Abs((xmlTime - jsonTime).TotalSeconds) < 1.0)
853 return;
854
855 if (xmlTime >= jsonTime)
856 {
857 var xmlTransform = ReadXmlTransformation(xmlPath) ?? CreateDefaultTransformation(toolName);
858 WriteJsonTransformation(jsonPath, xmlTransform);
859 }
860 else
861 {
862 var jsonTransform = ReadJsonTransformation(jsonPath) ?? CreateDefaultTransformation(toolName);
863 WriteXmlTransformation(xmlPath, jsonTransform);
864 }
865
866 _cachedTransformations.Remove(GetTransformationCacheKey(toolName, isScanner));
867 }

1036 private static string? ResolveModelFile(string modelPathOrName, string preferredFolder)
1037 {
1038 if (string.IsNullOrWhiteSpace(modelPathOrName))
1039 return null;
1040
1041 if (Path.IsPathRooted(modelPathOrName) && File.Exists(modelPathOrName))
1042 return modelPathOrName;
1043
1044 string baseDir = AppDomain.CurrentDomain.BaseDirectory;
1045 string directRelative = Path.Combine(baseDir, modelPathOrName);
1046 if (File.Exists(directRelative))
1047 return directRelative;
1048
1049 string modelName = Path.GetFileNameWithoutExtension(modelPathOrName);
1050 string extension = Path.GetExtension(modelPathOrName);
1051 string folderPath = ResolveModelsFolder(preferredFolder);
1052 if (!Directory.Exists(folderPath))
1053 return null;

1055 if (!string.IsNullOrWhiteSpace(extension))

1057 string explicitPath = Path.Combine(folderPath, Path.GetFileName(modelPathOrName));
1058 if (File.Exists(explicitPath))
1059 return explicitPath;
1060 }
1061
1062 foreach (var ext in SupportedModelExtensions)
1063 {
1064 string candidate = Path.Combine(folderPath, modelName + ext);
1065 if (File.Exists(candidate))
1066 return candidate;
1067 }
1068
1069 return null;
