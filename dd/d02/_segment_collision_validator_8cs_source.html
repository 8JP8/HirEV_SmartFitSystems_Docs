394 public SegmentValidationResult ValidateSegment(Point3D startApproach, Point3D endApproach)
395 {
396 var result = new SegmentValidationResult
397 {
398 Start = startApproach,
399 End = endApproach,
400 IsValid = true
401 };
402
403 if (!_enabled)
404 {
405 result.SkipReason = "Validation disabled";
406 return result;
407 }
408
409 if (!IsMeshReady)
410 {
411 result.SkipReason = "Collision mesh not built";
412 return result;
413 }
414
415 // PHASE 1: Fast AABB pre-filter
416 if (!SegmentIntersectsBoundingBox(startApproach, endApproach))
417 {
418 result.SkipReason = "Outside bounding box";
419 return result;
420 }
421
422 // PHASE 2: Sample point collision checking
423 var collisionPoint = CheckSegmentCollision(startApproach, endApproach);
424 if (collisionPoint != null)
425 {
426 result.IsValid = false;
427 result.CollisionPoint = collisionPoint;
428 result.CollisionDistance = startApproach.DistanceTo(collisionPoint);
429 }
430
431 return result;
432 }

434 public List<SegmentValidationResult> ValidateSegments(
435 IReadOnlyList<SegmentValidationRequest> requests,
436 Action<int, int, int>? progressCallback = null,
437 System.Threading.CancellationToken cancellationToken = default)
438 {
439 var results = new SegmentValidationResult[requests?.Count ?? 0];
440 if (requests == null || requests.Count == 0)
441 return results.ToList();
442
443 int processed = 0;
444 int collisions = 0;
445
446 progressCallback?.Invoke(0, requests.Count, 0);
447
448 for (int i = 0; i < requests.Count; i++)
449 {
450 cancellationToken.ThrowIfCancellationRequested();
451 var request = requests[i];
452 var result = new SegmentValidationResult
453 {
454 SegmentIndex = request.SegmentIndex,
455 Start = request.Start,
456 End = request.End,
457 IsValid = true
458 };
459
460 if (!_enabled)
461 {
462 result.SkipReason = "Validation disabled";
463 results[i] = result;
464 processed++;
465 System.Windows.Application.Current?.Dispatcher.InvokeAsync(() => progressCallback?.Invoke(processed, requests.Count, collisions));
466 continue;
467 }
468
469 if (!IsMeshReady)
470 {
471 result.SkipReason = "Collision entities not built";
472 results[i] = result;
473 processed++;
474 System.Windows.Application.Current?.Dispatcher.InvokeAsync(() => progressCallback?.Invoke(processed, requests.Count, collisions));
475 continue;
476 }
477
478 // Use the TrimTolerance logic
479 Point3D trimmedStart = request.Start;
480 Point3D trimmedEnd = request.End;
481
482 if (TrimTolerance > 0.001)
483 {
484 Vector3D dir = new Vector3D(request.Start, request.End);
485 double len = dir.Length;
486 if (len <= TrimTolerance * 2.1)
487 {
488 result.SkipReason = "Segment too short for trim tolerance";
489 results[i] = result;
490 processed++;
491 System.Windows.Application.Current?.Dispatcher.InvokeAsync(() => progressCallback?.Invoke(processed, requests.Count, collisions));
492 continue;
493 }
494 dir.Normalize();
495 trimmedStart = request.Start + dir * TrimTolerance;
496 trimmedEnd = request.End - dir * TrimTolerance;
497 }
498
499 var internalResult = ValidateSegment(trimmedStart, trimmedEnd);
500 result.IsValid = internalResult.IsValid;
501 result.CollisionPoint = internalResult.CollisionPoint;
502 result.CollisionDistance = internalResult.CollisionDistance;
503 result.SkipReason = internalResult.SkipReason;
504
505 if (!result.IsValid)
506 collisions++;
507 else if (string.IsNullOrWhiteSpace(result.SkipReason))
508 result.SkipReason = "No collision";
509
510 results[i] = result;
511 processed++;
512 if (processed % 50 == 0 || processed == requests.Count)
513 {
514 System.Windows.Application.Current?.Dispatcher.InvokeAsync(() => progressCallback?.Invoke(processed, requests.Count, collisions));
515 }
516 }
517
518 return results.ToList();
519 }

548 private bool SegmentIntersectsBoundingBox(Point3D start, Point3D end)
549 {
550 if (_meshBBoxMin == null || _meshBBoxMax == null)
551 return true; // Assume intersection if no bbox
552
553 // Calculate segment AABB
554 double segMinX = Math.Min(start.X, end.X);
555 double segMaxX = Math.Max(start.X, end.X);
556 double segMinY = Math.Min(start.Y, end.Y);
557 double segMaxY = Math.Max(start.Y, end.Y);
558 double segMinZ = Math.Min(start.Z, end.Z);
559 double segMaxZ = Math.Max(start.Z, end.Z);
560
561 // AABB intersection test
562 return !(segMaxX < _meshBBoxMin.X || segMinX > _meshBBoxMax.X ||
563 segMaxY < _meshBBoxMin.Y || segMinY > _meshBBoxMax.Y ||
564 segMaxZ < _meshBBoxMin.Z || segMinZ > _meshBBoxMax.Z);
565 }

570 private Point3D? CheckSegmentCollision(Point3D start, Point3D end)
571 {
572 if (_vertices == null || _triangles == null)
573 return null;
574
575 // Calculate segment direction and length
576 double dx = end.X - start.X;
577 double dy = end.Y - start.Y;
578 double dz = end.Z - start.Z;
579 double segmentLength = Math.Sqrt(dx * dx + dy * dy + dz * dz);
580
581 if (segmentLength < 0.001)
582 return null;
583
584 // Normalize direction
585 double invLen = 1.0 / segmentLength;
586 dx *= invLen;
587 dy *= invLen;
588 dz *= invLen;
589
590 // Check triangle intersections along segment
591 double? closestT = null;
592 Point3D? closestCollision = null;
593
594 for (int i = 0; i < _triangles.Length; i++)
595 {
596 var tri = _triangles[i];
597 if (tri.V1 >= _vertices.Length || tri.V2 >= _vertices.Length || tri.V3 >= _vertices.Length)
598 continue;
599
600 var v0 = _vertices[tri.V1];
601 var v1 = _vertices[tri.V2];
602 var v2 = _vertices[tri.V3];
603
604 // Möller-Trumbore ray-triangle intersection
605 double? t = RayTriangleIntersection(start, dx, dy, dz, segmentLength, v0, v1, v2);
606
607 if (t.HasValue && t.Value >= 0 && t.Value <= segmentLength)
608 {
609 if (!closestT.HasValue || t.Value < closestT.Value)
610 {
611 closestT = t.Value;
612 closestCollision = new Point3D(
613 start.X + dx * t.Value,
614 start.Y + dy * t.Value,
615 start.Z + dz * t.Value);
616 }
617 }
618 }
619
620 return closestCollision;
621 }

627 private double? RayTriangleIntersection(
628 Point3D origin, double dirX, double dirY, double dirZ, double maxDist,
629 Point3D v0, Point3D v1, Point3D v2)
630 {
631 const double EPSILON = 0.0000001;
632
633 // Edge vectors
634 double edge1X = v1.X - v0.X;
635 double edge1Y = v1.Y - v0.Y;
636 double edge1Z = v1.Z - v0.Z;
637
638 double edge2X = v2.X - v0.X;
639 double edge2Y = v2.Y - v0.Y;
640 double edge2Z = v2.Z - v0.Z;
641
642 // Cross product: h = dir × edge2
643 double hX = dirY * edge2Z - dirZ * edge2Y;
644 double hY = dirZ * edge2X - dirX * edge2Z;
645 double hZ = dirX * edge2Y - dirY * edge2X;
646
647 // Dot product: a = edge1 · h
648 double a = edge1X * hX + edge1Y * hY + edge1Z * hZ;
649
650 if (a > -EPSILON && a < EPSILON)
651 return null; // Ray parallel to triangle
652
653 double f = 1.0 / a;
654
655 // Vector from v0 to ray origin
656 double sX = origin.X - v0.X;
657 double sY = origin.Y - v0.Y;
658 double sZ = origin.Z - v0.Z;
659
660 // Barycentric coordinate u
661 double u = f * (sX * hX + sY * hY + sZ * hZ);
662
663 if (u < 0.0 || u > 1.0)
664 return null;
665
666 // Cross product: q = s × edge1
667 double qX = sY * edge1Z - sZ * edge1Y;
668 double qY = sZ * edge1X - sX * edge1Z;
669 double qZ = sX * edge1Y - sY * edge1X;
670
671 // Barycentric coordinate v
672 double v = f * (dirX * qX + dirY * qY + dirZ * qZ);
673
674 if (v < 0.0 || u + v > 1.0)
675 return null;
676
677 // Distance t to intersection point
678 double t = f * (edge2X * qX + edge2Y * qY + edge2Z * qZ);
679
680 if (t > EPSILON && t <= maxDist)
681 return t;
682
683 return null;
684 }

699 public List<Point3D> FindAlternativeWaypoints(
700 Point3D start,
701 Point3D end,
702 Point3D collisionPoint,
703 List<Point3D> allApproachPoints)
704 {
705 if (allApproachPoints == null || allApproachPoints.Count == 0)
706 return new List<Point3D>();
707
708 // Calculate direction from collision point to "outside"
709 // We want waypoints that go around the collision
710 var midPoint = new Point3D(
711 (start.X + end.X) / 2,
712 (start.Y + end.Y) / 2,
713 (start.Z + end.Z) / 2);
714
715 // Score waypoints by:
716 // 1. Distance from collision point (prefer farther)
717 // 2. Angle deviation from direct path (prefer moderate deviations)
718 // 3. Total path length (prefer shorter total)
719
720 var directDist = start.DistanceTo(end);
721
722 var candidates = allApproachPoints
723 .Where(wp =>
724 wp.DistanceTo(start) > 1.0 && // Not too close to start
725 wp.DistanceTo(end) > 1.0 && // Not too close to end
726 wp.DistanceTo(collisionPoint) > 2.0) // Not too close to collision
727 .Select(wp => new
728 {
729 Point = wp,
730 DistFromCollision = wp.DistanceTo(collisionPoint),
731 TotalPathLength = start.DistanceTo(wp) + wp.DistanceTo(end),
732 LengthRatio = (start.DistanceTo(wp) + wp.DistanceTo(end)) / directDist
733 })
734 .Where(c => c.LengthRatio < 3.0) // Don't go more than 3x the direct distance
735 .OrderBy(c => c.LengthRatio) // Prefer shorter paths
736 .ThenByDescending(c => c.DistFromCollision) // Then prefer farther from collision
737 .Take(_maxAlternativeAttempts * 2)
738 .Select(c => c.Point)
739 .ToList();
740
741 return candidates;
742 }
