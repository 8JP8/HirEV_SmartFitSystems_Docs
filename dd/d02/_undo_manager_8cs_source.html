259 private string BuildDescription()
260 {
261 var movePartCommand = _commands.OfType<MovePartCommand>().FirstOrDefault();
262 int pathUpdateCount = _commands.OfType<UpdateColorPathCommand>().Count();
263
264 if (movePartCommand != null && pathUpdateCount > 0)
265 return $"{movePartCommand.Description} with related paths";

279 public void Undo()
280 {
281 Execute(isUndo: true);
282 }
283
287 public void Redo()
288 {
289 Execute(isUndo: false);

296 private void Execute(bool isUndo)
297 {
298 var commands = isUndo
299 ? _commands.AsEnumerable().Reverse()
300 : _commands;
301 var batchedPathRefreshes = new List<UpdateColorPathCommand>();
302
303 foreach (var command in commands)
304 {
305 if (command is UpdateColorPathCommand pathCommand && pathCommand.RegenerateAllPaths)
306 {
307 pathCommand.ApplyStateWithoutVisualRefresh(isUndo);
308 if (!batchedPathRefreshes.Any(existing => existing.UsesSameVisualContext(pathCommand)))
309 batchedPathRefreshes.Add(pathCommand);
310 continue;
311 }
312
313 if (isUndo)
314 command.Undo();
315 else
316 command.Redo();
317 }
318
319 foreach (var pathCommand in batchedPathRefreshes)
320 {
321 pathCommand.RefreshVisuals();
322 }
323 }

509 private void ApplyTranslation(Transformation transMatrix)
510 {
511 foreach (var cp in _translatedPaths)
512 {
513 for (int i = 0; i < cp.Points.Count; i++)
514 {
515 cp.Points[i].TransformBy(transMatrix);
516 }
517
518 if (cp.Normals != null)
519 {
520 for (int i = 0; i < cp.Normals.Count; i++)
521 {
522 cp.Normals[i].TransformBy(transMatrix);
523 cp.Normals[i].Normalize();
524 }
525 }
526
527 if (cp.PathEntity != null)
528 {
529 cp.PathEntity.TransformBy(transMatrix);
530 }
531 }
532
533 // We must also translate the visual Waypoint spheres
534 foreach (var ent in _designControl.Entities)
535 {
536 if (ent.EntityData is WaypointData wpData && _translatedPaths.Contains(wpData.Path))
537 {
538 ent.TransformBy(transMatrix);
539 wpData.OriginalPosition.TransformBy(transMatrix);
540 ent.Regen(0.1);
541 }
542 }
543
544 _designControl.Entities.Regen();
545 _designControl.Invalidate();
546 }
