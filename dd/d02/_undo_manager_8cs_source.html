373 {
374 var commands = isUndo

376 : _commands;
377 var batchedPathRefreshes = new List<UpdateColorPathCommand>();
378
379 foreach (var command in commands)
380 {

382 {
383 pathCommand.ApplyStateWithoutVisualRefresh(isUndo);
384 if (!batchedPathRefreshes.Any(existing => existing.UsesSameVisualContext(pathCommand)))
385 batchedPathRefreshes.Add(pathCommand);
386 continue;
387 }
388
389 if (isUndo)
390 command.Undo();
391 else
392 command.Redo();
393 }
394
395 foreach (var pathCommand in batchedPathRefreshes)
396 {
397 pathCommand.RefreshVisuals();
398 }
