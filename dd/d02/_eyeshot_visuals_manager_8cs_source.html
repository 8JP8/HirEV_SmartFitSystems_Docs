130 public void ShowCornerMarkersForFace(CADFace face, int highlightVertexIndex = -1)
131 {
132 ClearCornerMarkers();
133
134 if (face?.KeyPoints == null) return;
135
136 Task.Run(() =>
137 {
138 var markers = new List<Entity>();
139 foreach (var keyPoint in face.KeyPoints.Where(kp => kp.PointType == PointTypeEnum.Corner))
140 {
141 double s = _yellowCubeSize;
142 var cube = Mesh.CreateBox(s, s, s);
143 cube.Translate(keyPoint.X - s / 2.0, keyPoint.Y - s / 2.0, keyPoint.Z - s / 2.0);
144 cube.Color = System.Drawing.Color.Yellow;
145 cube.ColorMethod = colorMethodType.byEntity;
146 cube.LayerName = MARKERS_LAYER;
147 markers.Add(cube);
148 }
149
150 if (highlightVertexIndex >= 0)
151 {
152 var highlight = face.KeyPoints.FirstOrDefault(k => k.PointType == PointTypeEnum.Corner && k.VertexIndex == highlightVertexIndex);
153 if (highlight != null)
154 {
155 double s2 = _yellowCubeSize * 1.8;
156 var bigCube = Mesh.CreateBox(s2, s2, s2);
157 bigCube.Translate(highlight.X - s2 / 2.0, highlight.Y - s2 / 2.0, highlight.Z - s2 / 2.0);
158 bigCube.Color = System.Drawing.Color.FromArgb(255, 255, 200, 0);
159 bigCube.ColorMethod = colorMethodType.byEntity;
160 bigCube.LayerName = MARKERS_LAYER;
161 markers.Add(bigCube);
162 }
163 }
164
165 Application.Current.Dispatcher.Invoke(() =>
166 {
167 EnsureMarkersLayerExists();
168 foreach (var cube in markers)
169 {
170 CornerMarkers.Add(cube);
171 _designControl.Entities.Add(cube);
172 }
173 _designControl.Invalidate();
174 });
175 });
176 }

184 public void ShowHoleCenterAndDepth(CADHole hole)
185 {
186 ClearHoleVisuals();
187
188 if (hole == null || hole.TopCenterPoint == null || hole.BottomCenterPoint == null) return;
189
190 try
191 {
192 var top = hole.TopCenterPoint;
193 var bottom = hole.BottomCenterPoint;
194
195 // Center sphere at top (using configured values)
196 var topSphere = Mesh.CreateSphere(_holeSphereSize, 12, 12);
197 topSphere.Translate(top.X, top.Y, top.Z);
198 topSphere.Color = System.Drawing.Color.Cyan;
199 topSphere.ColorMethod = colorMethodType.byEntity;
200 topSphere.LayerName = MARKERS_LAYER;
201 HoleCenterMarker = topSphere;
202 _currentHoleVisuals.Add(topSphere);
203
204 // Bottom sphere (using configured values)
205 var bottomSphere = Mesh.CreateSphere(_holeSphereSize, 12, 12);
206 bottomSphere.Translate(bottom.X, bottom.Y, bottom.Z);
207 bottomSphere.Color = System.Drawing.Color.Blue;
208 bottomSphere.ColorMethod = colorMethodType.byEntity;
209 bottomSphere.LayerName = MARKERS_LAYER;
210 _currentHoleVisuals.Add(bottomSphere);
211
212 // Depth line (using configured values)
213 var depthLine = new devDept.Eyeshot.Entities.Line(top, bottom);
214 depthLine.Color = System.Drawing.Color.LightBlue;
215 depthLine.LineWeightMethod = colorMethodType.byEntity;
216 depthLine.LineWeight = (float)_holeDepthLineWidth;
217 depthLine.LayerName = MARKERS_LAYER;
218 HoleDepthLine = depthLine;
219 _currentHoleVisuals.Add(depthLine);
220
221 // Depth text at midpoint
222 var midpoint = new Point3D(
223 (top.X + bottom.X) / 2,
224 (top.Y + bottom.Y) / 2,
225 (top.Z + bottom.Z) / 2
226 );
227
228 string depthText = $"Depth: {hole.Depth:F2}mm";
229 var textEntity = new Text(
230 midpoint,
231 depthText,
232 _yellowCubeSize * 1.2
233 );
234 textEntity.Color = System.Drawing.Color.White;
235 textEntity.LayerName = MARKERS_LAYER;
236 HoleDepthText = textEntity;
237 _currentHoleVisuals.Add(textEntity);
238
239 // Add all entities to the design control for display
240 EnsureMarkersLayerExists();
241 _designControl.Entities.AddRange(_currentHoleVisuals);
242
243 _designControl.Invalidate();
244 }
245 catch (Exception ex)
246 {
247 System.Diagnostics.Debug.WriteLine($"ShowHoleCenterAndDepth error: {ex.Message}");
248 }

254 public void ShowControlPointsForFace(CADFace face)
255 {
256 ClearControlPointSpheres();
257
258 // If we're switching to a different face, disable control point selection
259 bool isSameFace = CurrentSelectedFace != null &&
260 CurrentSelectedFace.EntityIndex == face.EntityIndex &&
261 CurrentSelectedFace.FaceIndex == face.FaceIndex;
262
263 CurrentSelectedFace = face;
264 ControlPointSelectionEnabled = false; // Always reset when showing new spheres
265
266 // Use MeasurementPoints if available, otherwise fallback to ControlPoints
267 var pointsToShow = (face?.MeasurementPoints != null && face.MeasurementPoints.Any())
268 ? face.MeasurementPoints
269 : face?.ControlPoints;
270
271 if (pointsToShow == null || pointsToShow.Count == 0) return;
272
273 try
274 {
275 foreach (var pt in pointsToShow)
276 {
277 var sphere = Mesh.CreateSphere(_pinkSphereSize, 12, 12);
278 sphere.Translate(pt.X, pt.Y, pt.Z);
279 sphere.Color = System.Drawing.Color.Magenta;
280 sphere.ColorMethod = colorMethodType.byEntity;
281 sphere.LayerName = MARKERS_LAYER;
282 ControlPointSpheres.Add(sphere);
283 EnsureMarkersLayerExists();
284 _designControl.Entities.Add(sphere);
285 }
286 _designControl.Invalidate();
287
288 // Enable control point selection on the next event cycle
289 System.Windows.Application.Current.Dispatcher.BeginInvoke(
290 System.Windows.Threading.DispatcherPriority.Input,
291 new Action(() => ControlPointSelectionEnabled = true)
292 );
293 }
294 catch (Exception ex)
295 {
296 System.Diagnostics.Debug.WriteLine($"Error showing measurement points: {ex.Message}");
297 }

327 public void HighlightFaceVisualFallback(Brep? brep, int faceIndex)
328 {
329 RemoveHoverHighlight(); // Clear previous highlights
330
331 if (brep == null || faceIndex < 0 || faceIndex >= brep.Faces.Length) return;
332
333 try
334 {
335 var face = brep.Faces[faceIndex];
336 var pts = new List<Point3D>();
337 if (face.Parametric != null && face.Parametric.Length > 0)
338 {
339 foreach (var surf in face.Parametric)
340 {
341 if (surf == null) continue;
342 if (surf.Vertices == null) surf.Regen(0.01);
343 if (surf.Vertices != null) pts.AddRange(surf.Vertices);
344 }
345 }
346
347 if (pts.Count == 0) return;
348
349 int step = Math.Max(1, pts.Count / 50);
350 EnsureMarkersLayerExists();
351 for (int i = 0; i < pts.Count; i += step)
352 {
353 var p = pts[i];
354 var s = Mesh.CreateSphere(_yellowCubeSize * 0.75, 6, 6);
355 s.Translate(p.X, p.Y, p.Z);
356 s.Color = System.Drawing.Color.FromArgb(160, System.Drawing.Color.Yellow);
357 s.ColorMethod = colorMethodType.byEntity;
358 s.LayerName = MARKERS_LAYER;
359
360 _hoverHighlightEntities.Add(s); // Store all spheres
361 _designControl.Entities.Add(s);
362 }
363 _designControl.Invalidate();
364 }
365 catch (Exception ex)
366 {
367 System.Diagnostics.Debug.WriteLine($"Highlight fallback error: {ex.Message}");
368 }

536 {
537 // Use a simpler display mode. Shaded is much faster than Rendered.
538 design.ActiveViewport.DisplayMode = devDept.Eyeshot.displayType.Shaded;
539
540 // Setting a minimum framerate is the most effective way to keep the viewport responsive.
541 // Eyeshot will automatically simplify the scene (e.g., by hiding small objects)
542 // during camera movements to maintain this rate.
543 design.MinimumFramerate = 10;
544
545 // --- Settings for Shaded mode ---
546 // Disable shadows for Shaded mode.
547 design.Shaded.ShadowMode = shadowType.None;
548 // Hide edges for a slight performance gain.
549 design.Shaded.ShowEdges = false;
