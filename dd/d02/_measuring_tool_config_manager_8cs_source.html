50 {
51 string exeDirectory = AppDomain.CurrentDomain.BaseDirectory;
52 string exePath = Path.Combine(exeDirectory, CONFIG_FILE_NAME);
53
54 try
55 {
56 string? current = exeDirectory;
57 for (int i = 0; i < 8; i++)
58 {
59 if (current == null) break;
60 string candidate = Path.Combine(current, "HirEV_SmartFitSystemsApp", CONFIG_FILE_NAME);
61 if (File.Exists(candidate))
62 {
63 // If project source file is newer or different, copy to exePath
64 if (File.Exists(exePath))
65 {
66 if (File.GetLastWriteTimeUtc(candidate) > File.GetLastWriteTimeUtc(exePath))
67 {
68 File.Copy(candidate, exePath, overwrite: true);
69 }
70 }
71 else
72 {
73 File.Copy(candidate, exePath, overwrite: true);
74 }
75 return exePath;
76 }
77 current = Path.GetDirectoryName(current?.TrimEnd(Path.DirectorySeparatorChar));
78 }

152 private void LoadConfiguration()
153 {
154 lock (_lockObject)
155 {
156 try
157 {
158 _xmlDoc.Load(_configFilePath);
159 _colorToolMapping.Clear();
160
161 XmlNodeList? nodes = _xmlDoc.SelectNodes("//MeasuringToolByColorConfig/add");
162 if (nodes != null)
163 {
164 foreach (XmlNode node in nodes)
165 {
166 string? key = node.Attributes?["key"]?.Value;
167 string? tool = node.Attributes?["tool"]?.Value;
168
169 if (string.IsNullOrEmpty(key) || string.IsNullOrEmpty(tool))
170 continue;
171
172 if (key == "DEFAULT")
173 continue; // Skip the default entry
174
175 // Parse RGB(R,G,B) format
176 Color? color = ParseRGBKey(key);
177 if (color.HasValue && !_colorToolMapping.ContainsKey(color.Value))
178 {
179 _colorToolMapping[color.Value] = tool;
180 }
181 }
182 }
183 }
184 catch (Exception ex)
185 {
186 throw new InvalidOperationException($"Failed to load MeasuringToolByColor configuration: {ex.Message}", ex);
187 }
188 }

196 private Color? ParseRGBKey(string rgbKey)
197 {
198 try
199 {
200 // Expected format: RGB(R,G,B)
201 if (!rgbKey.StartsWith("RGB(") || !rgbKey.EndsWith(")"))
202 return null;
203
204 string rgbValues = rgbKey.Substring(4, rgbKey.Length - 5); // Extract "R,G,B"
205 string[] parts = rgbValues.Split(',');
206
207 if (parts.Length != 3)
208 return null;
209
210 int r = int.Parse(parts[0].Trim());
211 int g = int.Parse(parts[1].Trim());
212 int b = int.Parse(parts[2].Trim());
213
214 return Color.FromArgb(r, g, b);
215 }
216 catch
217 {
218 return null;
219 }
220 }

255 private void AddColorToConfiguration(Color color, string toolName)
256 {
257 lock (_lockObject)
258 {
259 try
260 {
261 // Add to in-memory dictionary
262 _colorToolMapping[color] = toolName;
263
264 // Add to XML document
265 XmlNode? root = _xmlDoc.SelectSingleNode("//MeasuringToolByColorConfig");
266 if (root == null)
267 return;
268
269 // Find the DEFAULT entry to insert before it
270 XmlNode? defaultNode = _xmlDoc.SelectSingleNode("//MeasuringToolByColorConfig/add[@key='DEFAULT']");
271
272 // Create new node
273 XmlElement newElement = _xmlDoc.CreateElement("add");
274 newElement.SetAttribute("key", FormatRGBKey(color));
275 newElement.SetAttribute("tool", toolName);
276 newElement.SetAttribute("colorName", GetColorName(color));
277 newElement.SetAttribute("description", $"Auto-generated entry for {GetColorName(color)} faces");
278
279 // Insert before DEFAULT or at the end
280 if (defaultNode != null)
281 {
282 root.InsertBefore(newElement, defaultNode);
283 }
284 else
285 {
286 root.AppendChild(newElement);
287 }
288
289 // Save the file
290 SaveConfigFile();
291 }
292 catch (Exception ex)
293 {
294 // Log error but don't crash the application

348 public bool UpdateToolForColor(Color color, string toolName)
349 {
350 lock (_lockObject)
351 {
352 try
353 {
354 // Update in-memory dictionary
355 _colorToolMapping[color] = toolName;
356
357 // Find or create the XML node for this color
358 string rgbKey = FormatRGBKey(color);
359 XmlNode? existingNode = _xmlDoc.SelectSingleNode($"//MeasuringToolByColorConfig/add[@key='{rgbKey}']");
360
361 if (existingNode != null)
362 {
363 // Update existing node
364 XmlAttribute? toolAttr = existingNode.Attributes?["tool"];
365 if (toolAttr != null)
366 {
367 toolAttr.Value = toolName;
368 }
369 else
370 {
371 // Add tool attribute if it doesn't exist
372 XmlAttribute newToolAttr = _xmlDoc.CreateAttribute("tool");
373 newToolAttr.Value = toolName;
374 existingNode.Attributes?.Append(newToolAttr);
375 }
376
377 // Update description
378 XmlAttribute? descAttr = existingNode.Attributes?["description"];
379 if (descAttr != null)
380 {
381 descAttr.Value = $"User-configured preset for {GetColorName(color)} faces";
382 }
383
384 // Update colorName to use friendly name from ColorNamer
385 XmlAttribute? colorNameAttr = existingNode.Attributes?["colorName"];
386 if (colorNameAttr != null)
387 {
388 colorNameAttr.Value = GetColorName(color);
389 }
390 }
391 else
392 {
393 // Add new node
394 XmlNode? root = _xmlDoc.SelectSingleNode("//MeasuringToolByColorConfig");
395 if (root == null)
396 return false;
397
398 // Find the DEFAULT entry to insert before it
399 XmlNode? defaultNode = _xmlDoc.SelectSingleNode("//MeasuringToolByColorConfig/add[@key='DEFAULT']");
400
401 // Create new node
402 XmlElement newElement = _xmlDoc.CreateElement("add");
403 newElement.SetAttribute("key", rgbKey);
404 newElement.SetAttribute("tool", toolName);
405 newElement.SetAttribute("colorName", GetColorName(color));
406 newElement.SetAttribute("description", $"User-configured preset for {GetColorName(color)} faces");
407
408 // Insert before DEFAULT or at the end
409 if (defaultNode != null)
410 {
411 root.InsertBefore(newElement, defaultNode);
412 }
413 else
414 {
415 root.AppendChild(newElement);
416 }
417 }
418
419 // Save the file
420 SaveConfigFile();
421 ConfigurationChanged?.Invoke();
422 return true;
423 }
424 catch (Exception ex)
