171 public static (double A, double B) VectorToHeadAngles(
172 Vector3D v,
173 double stepA = 7.5,
174 double stepB = 7.5,
175 double maxA = 105.0)
176 {
177 if (v == null || v.Length < NumericalTolerance)
178 return (0.0, 0.0);
179
180 var norm = new Vector3D(v.X, v.Y, v.Z);
181 norm.Normalize();
182
183 double zClamped = Math.Clamp(norm.Z, -1.0, 1.0);
184 double continuousA = Math.Acos(zClamped) * 180.0 / Math.PI;
185 double continuousB = Math.Atan2(norm.Y, norm.X) * 180.0 / Math.PI;
186
187 double safeStepA = Math.Max(0.1, stepA);
188 double safeStepB = Math.Max(0.1, stepB);
189
190 double discreteA = Math.Round(continuousA / safeStepA) * safeStepA;
191 discreteA = Math.Clamp(discreteA, 0.0, maxA);
192
193 // When A is exactly 0 (or 180), azimuth B is redundant; standardize to 0.
194 if (discreteA < safeStepA * 0.5 || discreteA > 180.0 - (safeStepA * 0.5))
195 return (discreteA < safeStepA * 0.5 ? 0.0 : 180.0, 0.0);
196
197 double discreteB = Math.Round(continuousB / safeStepB) * safeStepB;
198 if (discreteB > 180.0 - NumericalTolerance)
199 discreteB = 180.0;
200 else if (discreteB < -180.0 + NumericalTolerance)
201 discreteB = -180.0;
202
203 return (discreteA, discreteB);
204 }

213 public static Vector3D HeadAnglesToVector(double aDeg, double bDeg)
214 {
215 if (aDeg <= 1e-4)
216 return new Vector3D(0.0, 0.0, 1.0);
217
218 if (aDeg >= 180.0 - 1e-4)
219 return new Vector3D(0.0, 0.0, -1.0);
220
221 double aRad = aDeg * Math.PI / 180.0;
222 double bRad = bDeg * Math.PI / 180.0;
223
224 double sinA = Math.Sin(aRad);
225 double cosA = Math.Cos(aRad);
226 double cosB = Math.Cos(bRad);
227 double sinB = Math.Sin(bRad);
228
229 var vector = new Vector3D(sinA * cosB, sinA * sinB, cosA);
230 vector.Normalize();
231 return vector;
232 }

237 public static double AngleBetweenVectorsDeg(Vector3D v1, Vector3D v2)
238 {
239 if (v1 == null || v2 == null || v1.Length < NumericalTolerance || v2.Length < NumericalTolerance)
240 return 0.0;
241
242 double dot = (v1.X * v2.X + v1.Y * v2.Y + v1.Z * v2.Z) / (v1.Length * v2.Length);
243 dot = Math.Clamp(dot, -1.0, 1.0);
244 return Math.Acos(dot) * 180.0 / Math.PI;
245 }

286 public static (double A, double B, Vector3D Vector, bool WasReused, bool WasAdjustedForBase) SelectStandardProbeAngle(
287 Point3D point,
288 Vector3D currentNormal,
289 Vector3D surfaceNormal,
290 (double A, double B)? activePreviousAngle,
291 double baseZ,
292 ProbeAngleStandardizationConfig config)
293 {
294 double localHeight = point.Z - baseZ;
295
296 // 1. EVALUATE ANGULAR REUSE THRESHOLD: Check if previous calibrated head angle can be reused
297 if (activePreviousAngle.HasValue)
298 {
299 var (prevA, prevB) = activePreviousAngle.Value;
300 Vector3D prevVec = HeadAnglesToVector(prevA, prevB);
301 double devFromSurface = AngleBetweenVectorsDeg(prevVec, surfaceNormal);
302
303 if (devFromSurface <= config.AngleReuseThresholdDeg + NumericalTolerance &&
304 devFromSurface <= config.MeasuringAngleToleranceDeg + NumericalTolerance &&
305 IsHeadPoseSafeWithBaseAvoidance(prevA, prevB, localHeight, config.BaseAvoidance))
306 {
307 return (prevA, prevB, prevVec, WasReused: true, WasAdjustedForBase: false);
308 }
309 }
310
311 // 2. DIRECT SNAP TO NEAREST DISCRETE CMM HEAD POSE
312 var (directA, directB) = VectorToHeadAngles(
313 surfaceNormal,
314 config.StepADeg,
315 config.StepBDeg,
316 config.MaxADeg);
317
318 if (IsHeadPoseValidForSurface(directA, directB, surfaceNormal, localHeight, config))
319 {
320 Vector3D directVec = HeadAnglesToVector(directA, directB);
321 return (directA, directB, directVec, WasReused: false, WasAdjustedForBase: false);
322 }
323
324 // 3. BASE AVOIDANCE & TOLERANCE SEARCH: Search discrete (A_i, B_j) grid within measuring tolerance
325 // that satisfies base avoidance at localHeight.
326 var candidates = GenerateCandidateHeadAnglesWithinTolerance(
327 surfaceNormal,
328 config.MeasuringAngleToleranceDeg,
329 config.StepADeg,
330 config.StepBDeg,
331 config.MaxADeg);
332
333 double bestDeviation = double.MaxValue;
334 (double A, double B) bestCandidate = (directA, directB);
335 bool foundValidCandidate = false;
336
337 foreach (var (candA, candB) in candidates)
338 {
339 if (!IsHeadPoseSafeWithBaseAvoidance(candA, candB, localHeight, config.BaseAvoidance))
340 continue;
341
342 Vector3D candVec = HeadAnglesToVector(candA, candB);
343 double dev = AngleBetweenVectorsDeg(candVec, surfaceNormal);
344 if (dev <= config.MeasuringAngleToleranceDeg + NumericalTolerance && dev < bestDeviation)
345 {
346 bestDeviation = dev;
347 bestCandidate = (candA, candB);
348 foundValidCandidate = true;
349 }
350 }
351
352 if (foundValidCandidate)
353 {
354 Vector3D bestVec = HeadAnglesToVector(bestCandidate.A, bestCandidate.B);
355 return (bestCandidate.A, bestCandidate.B, bestVec, WasReused: false, WasAdjustedForBase: true);
356 }
357
358 // 4. FALLBACK: Return closest discrete direct snap
359 Vector3D fallbackVec = HeadAnglesToVector(directA, directB);
360 return (directA, directB, fallbackVec, WasReused: false, WasAdjustedForBase: false);
361 }

366 public static List<(double A, double B)> GenerateCandidateHeadAnglesWithinTolerance(
367 Vector3D targetNormal,
368 double toleranceDeg,
369 double stepA,
370 double stepB,
371 double maxA)
372 {
373 var result = new List<(double A, double B)>();
374 if (targetNormal == null || targetNormal.Length < NumericalTolerance)
375 {
376 result.Add((0.0, 0.0));
377 return result;
378 }
379
380 var (centerA, centerB) = VectorToHeadAngles(targetNormal, stepA, stepB, maxA);
381 double searchSpan = Math.Max(stepA, toleranceDeg);
382
383 int spanStepsA = (int)Math.Ceiling(searchSpan / Math.Max(0.1, stepA));
384 int spanStepsB = (int)Math.Ceiling(searchSpan / Math.Max(0.1, stepB));
385
386 var visited = new HashSet<(long, long)>();
387
388 for (int i = -spanStepsA; i <= spanStepsA; i++)
389 {
390 double a = Math.Clamp(centerA + i * stepA, 0.0, maxA);
391 a = Math.Round(a / stepA) * stepA;
392
393 for (int j = -spanStepsB; j <= spanStepsB; j++)
394 {
395 double b = centerB + j * stepB;
396 while (b > 180.0) b -= 360.0;
397 while (b < -180.0) b += 360.0;
398 b = Math.Round(b / stepB) * stepB;
399
400 if (a < stepA * 0.5)
401 b = 0.0;
402
403 long keyA = (long)Math.Round(a * 1000.0);
404 long keyB = (long)Math.Round(b * 1000.0);
405 if (!visited.Add((keyA, keyB)))
406 continue;
407
408 Vector3D candVec = HeadAnglesToVector(a, b);
409 if (AngleBetweenVectorsDeg(candVec, targetNormal) <= toleranceDeg + (stepA * 0.5))
410 {
411 result.Add((a, b));
412 }
413 }
414 }
415
416 if (result.Count == 0)
417 result.Add((centerA, centerB));
418
419 return result;
420 }

432 public static StandardizationPassSummary ApplyStandardization(
433 ColorPath path,
434 ProbeAngleStandardizationConfig config,
435 double defaultBaseZ = 0.0,
436 Func<int, bool>? isHolePointFunc = null,
437 Func<int, double>? getPointBaseZFunc = null)
438 {
439 var summary = new StandardizationPassSummary();
440 if (path == null || path.Points == null || path.Points.Count == 0 || path.PathCmmType != CMMType.Probe)
441 return summary;
442
443 summary.TotalPoints = path.Points.Count;
444 if (!config.Enabled)
445 return summary;
446
447 if (path.Normals == null || path.Normals.Count != path.Points.Count)
448 {
449 path.Normals = new List<Vector3D>(path.Points.Count);
450 for (int i = 0; i < path.Points.Count; i++)
451 path.Normals.Add(new Vector3D(0, 0, 1));
452 }
453
454 (double A, double B)? activeAngle = null;
455
456 for (int i = 0; i < path.Points.Count; i++)
457 {
458 Point3D pt = path.Points[i];
459 Vector3D currentNormal = path.Normals[i];
460 double baseZ = getPointBaseZFunc?.Invoke(i) ?? defaultBaseZ;
461
462 bool isHole = isHolePointFunc?.Invoke(i) == true ||
463 (path.PointTypes != null && i < path.PointTypes.Count &&
464 string.Equals(path.PointTypes[i], "Hole", StringComparison.OrdinalIgnoreCase));
465
466 // 1. HOLE VERTICAL PROBE RULE: Holes strictly remain fixed to vertical (+Z)
467 if (isHole)
468 {
469 path.Normals[i] = new Vector3D(0, 0, 1);
470 summary.HolePointsPreservedCount++;
471 activeAngle = (0.0, 0.0);
472 summary.UniqueAngles.Add((0.0, 0.0));
473 continue;
474 }
475
476 // 2. SELECT DISCRETE CALIBRATED CMM PROBE ANGLE (with reuse threshold & base avoidance)
477 var selection = SelectStandardProbeAngle(
478 pt,
479 currentNormal,
480 currentNormal,
481 activeAngle,
482 baseZ,
483 config);
484
485 path.Normals[i] = selection.Vector;
486 summary.StandardizedPoints++;
487
488 if (selection.WasReused)
489 {
490 summary.ReusedAnglesCount++;
491 }
492 else
493 {
494 if (activeAngle.HasValue &&
495 (Math.Abs(activeAngle.Value.A - selection.A) > 1e-4 ||
496 Math.Abs(activeAngle.Value.B - selection.B) > 1e-4))
497 {
498 summary.RotationsCount++;
499 }
500 }
501
502 if (selection.WasAdjustedForBase)
503 summary.BaseAvoidanceAdjustedCount++;
504
505 activeAngle = (selection.A, selection.B);
506 summary.UniqueAngles.Add((selection.A, selection.B));
507 }
508
509 summary.UniqueAnglesCount = summary.UniqueAngles.Count;
510 return summary;
511 }
