169 public CmmProgram ConvertDtoToIntermediate(PathExportDTO dto, DMISPostProcessorSettings? settings = null)
170 {
171 settings ??= new DMISPostProcessorSettings();
172
173 var program = new CmmProgram
174 {
175 ProgramName = string.IsNullOrWhiteSpace(settings.ProgramName) ? "INSPECTION_RUN" : settings.ProgramName,
176 ModelName = dto.ModelName ?? "UNKNOWN_MODEL",
177 CreatedAt = dto.ExportTimestamp != default ? dto.ExportTimestamp : DateTime.Now,
178 Units = settings.Units,
179 Decimals = settings.CoordinateDecimals,
180 WorkingPlane = settings.WorkingPlane
181 };
182
183 // Compute multipart and assembly bounding boxes from all inspection and travel points
184 double allMinX = double.MaxValue, allMaxX = double.MinValue;
185 double allMinY = double.MaxValue, allMaxY = double.MinValue;
186 double allMinZ = double.MaxValue, allMaxZ = double.MinValue;
187 bool hasPoints = false;
188
189 var partBoundsMap = new Dictionary<int, (double MinX, double MaxX, double MinY, double MaxY, double MinZ, double MaxZ)>();
190
191 foreach (var cp in dto.ColorPaths ?? Enumerable.Empty<ColorPathExportDTO>())
192 {
193 foreach (var pt in cp.PathPoints ?? Enumerable.Empty<PathPointExportDTO>())
194 {
195 hasPoints = true;
196 if (pt.X < allMinX) allMinX = pt.X; if (pt.X > allMaxX) allMaxX = pt.X;
197 if (pt.Y < allMinY) allMinY = pt.Y; if (pt.Y > allMaxY) allMaxY = pt.Y;
198 if (pt.Z < allMinZ) allMinZ = pt.Z; if (pt.Z > allMaxZ) allMaxZ = pt.Z;
199
200 int eIdx = pt.EntityIndex;
201 if (!partBoundsMap.TryGetValue(eIdx, out var b))
202 {
203 partBoundsMap[eIdx] = (pt.X, pt.X, pt.Y, pt.Y, pt.Z, pt.Z);
204 }
205 else
206 {
207 partBoundsMap[eIdx] = (
208 Math.Min(b.MinX, pt.X), Math.Max(b.MaxX, pt.X),
209 Math.Min(b.MinY, pt.Y), Math.Max(b.MaxY, pt.Y),
210 Math.Min(b.MinZ, pt.Z), Math.Max(b.MaxZ, pt.Z)
211 );
212 }
213 }
214 }
215
216 // Include explicit PartBounds from settings if provided
217 if (settings.HasPartBounds)
218 {
219 hasPoints = true;
220 allMinX = Math.Min(allMinX, settings.PartBoundsXMin!.Value);
221 allMaxX = Math.Max(allMaxX, settings.PartBoundsXMax!.Value);
222 allMinY = Math.Min(allMinY, settings.PartBoundsYMin!.Value);
223 allMaxY = Math.Max(allMaxY, settings.PartBoundsYMax!.Value);
224 allMinZ = Math.Min(allMinZ, settings.PartBoundsZMin!.Value);
225 allMaxZ = Math.Max(allMaxZ, settings.PartBoundsZMax!.Value);
226 }
227
228 // Clearance offset above top of part(s) matching transition bridge AirTop formula
229 double bridgeClearance = Math.Max(30.0, settings.CalibrationClearanceOffset > 0.1 ? settings.CalibrationClearanceOffset * 2.0 : 50.0);
230 double safeTopZ = hasPoints ? (allMaxZ + bridgeClearance) : 500.0;
231 double topCenterX = hasPoints ? (allMinX + allMaxX) * 0.5 : 0.0;
232 double topCenterY = hasPoints ? (allMinY + allMaxY) * 0.5 : 0.0;
233
234 program.SafeClearanceZ = safeTopZ;
235
236 foreach (var kvp in partBoundsMap)
237 {
238 program.MultipartBounds[kvp.Key] = new CmmPartBounds
239 {
240 EntityIndex = kvp.Key,
241 PartName = $"Part_{kvp.Key}",
242 MinX = kvp.Value.MinX,
243 MaxX = kvp.Value.MaxX,
244 MinY = kvp.Value.MinY,
245 MaxY = kvp.Value.MaxY,
246 MinZ = kvp.Value.MinZ,
247 MaxZ = kvp.Value.MaxZ
248 };
249 }
250
251 // Setup machine definition using Bounding Box Top Centerpoint as the Home coordinate
252 program.Machine = new CmmMachineDefinition
253 {
254 MachineName = "CMM_InspectionCenter",
255 Manufacturer = "AGT Metrology",
256 ControllerType = "DMIS 5.3 Compliant",
257 HomeX = topCenterX,
258 HomeY = topCenterY,
259 HomeZ = safeTopZ,
260 XMin = hasPoints ? allMinX - 200.0 : -1000.0,
261 XMax = hasPoints ? allMaxX + 200.0 : 2000.0,
262 YMin = hasPoints ? allMinY - 200.0 : -1000.0,
263 YMax = hasPoints ? allMaxY + 200.0 : 2000.0,
264 ZMin = hasPoints ? allMinZ - 100.0 : -800.0,
265 ZMax = hasPoints ? safeTopZ + 200.0 : 800.0,
266 Wrist = new CmmWristDefinition
267 {
268 WristName = "PH10",
269 Manufacturer = "Renishaw",
270 ModelName = settings.ProbeHeadModel,
271 AxisAMin = 0.0,
272 AxisAMax = 105.0,
273 AxisAStep = 7.5,
274 AxisBMin = -180.0,
275 AxisBMax = 180.0,
276 AxisBStep = 7.5
277 }
278 };
279
280 // Collect unique tools using standardized CMM catalog definitions
281 var toolMap = new Dictionary<string, CmmToolDefinition>(StringComparer.OrdinalIgnoreCase);
282 foreach (var cp in dto.ColorPaths ?? Enumerable.Empty<ColorPathExportDTO>())
283 {
284 string toolName = string.IsNullOrWhiteSpace(cp.ToolName) ? "DefaultTool" : cp.ToolName.Trim();
285 if (!toolMap.ContainsKey(toolName))
286 {
287 var montage = CmmStandardToolCatalog.ResolveToolMontage(toolName);
288 double tipRadius = montage.Stylus.BallDiameter > 0.05 ? montage.Stylus.BallDiameter / 2.0 : 1.0;
289 double stylusLength = montage.Stylus.OverallLength > 0.1 ? montage.Stylus.OverallLength : 20.0;
290 double extLength = montage.Extension?.Length ?? 0.0;
291
292 toolMap[toolName] = new CmmToolDefinition
293 {
294 ToolId = toolName,
295 Description = string.IsNullOrWhiteSpace(montage.Description) ? $"Measuring Tool {toolName}" : montage.Description,
296 TipRadius = tipRadius,
297 StylusLength = stylusLength,
298 ExtensionLength = extLength,
299 ToolHolderSlot = montage.DefaultRackSlot > 0 ? montage.DefaultRackSlot : (toolMap.Count + 1)
300 };
301 }
302 }
303
304 if (toolMap.Count == 0)
305 {
306 var defaultMontage = CmmStandardToolCatalog.ResolveToolMontage("DefaultTool");
307 toolMap["DefaultTool"] = new CmmToolDefinition
308 {
309 ToolId = "DefaultTool",
310 Description = defaultMontage.Description,
311 TipRadius = defaultMontage.Stylus.BallDiameter / 2.0,
312 StylusLength = defaultMontage.Stylus.OverallLength,
313 ExtensionLength = defaultMontage.Extension?.Length ?? 0.0,
314 ToolHolderSlot = 1
315 };
316 }
317
318 program.Tools = toolMap.Values.ToList();
319
320 // Order color paths based on execution order if present
321 var orderedPaths = new List<ColorPathExportDTO>();
322 var remainingPaths = (dto.ColorPaths ?? new List<ColorPathExportDTO>()).ToList();
323
324 if (dto.ExecutionOrder != null && dto.ExecutionOrder.Count > 0)
325 {
326 foreach (var orderName in dto.ExecutionOrder)
327 {
328 var match = remainingPaths.FirstOrDefault(p =>
329 string.Equals(p.ColorName, orderName, StringComparison.OrdinalIgnoreCase));
330 if (match != null)
331 {
332 orderedPaths.Add(match);
333 remainingPaths.Remove(match);
334 }
335 }
336 }
337 orderedPaths.AddRange(remainingPaths);
338
339 int commandCounter = 1;
340 int featureCounter = 1;
341
342 foreach (var colorPath in orderedPaths)
343 {
344 // If filtering to probe paths only, check CmmType
345 if (settings.ProbePathsOnly &&
346 !string.IsNullOrWhiteSpace(colorPath.CmmType) &&
347 string.Equals(colorPath.CmmType, "Scanner", StringComparison.OrdinalIgnoreCase))
348 {
349 continue;
350 }
351
352 string toolId = string.IsNullOrWhiteSpace(colorPath.ToolName) ? "DefaultTool" : colorPath.ToolName.Trim();
353 var sequence = new CmmSequence
354 {
355 SequenceName = $"Seq_{colorPath.ColorName}",
356 ColorName = colorPath.ColorName,
357 ColorR = colorPath.ColorR,
358 ColorG = colorPath.ColorG,
359 ColorB = colorPath.ColorB,
360 ToolId = toolId
361 };
362
363 sequence.Commands.Add(new CmmCommentCommand($"Color Path: {colorPath.ColorName} (RGB: {colorPath.ColorR},{colorPath.ColorG},{colorPath.ColorB}), Tool: {toolId}")
364 {
365 CommandId = commandCounter++
366 });
367
368 // Tool selection command
369 sequence.Commands.Add(new CmmToolChangeCommand(toolId)
370 {
371 CommandId = commandCounter++
372 });
373
374 double? activeA = null;
375 double? activeB = null;
376
377 var points = colorPath.PathPoints ?? new List<PathPointExportDTO>();
378 for (int i = 0; i < points.Count; i++)
379 {
380 var pt = points[i];
381 string ptType = pt.PointType?.Trim() ?? "Measurement";
382
383 // Check for head angle orientation change
384 if (pt.HeadAngleA.HasValue && pt.HeadAngleB.HasValue)
385 {
386 double targetA = pt.HeadAngleA.Value;
387 double targetB = pt.HeadAngleB.Value;
388
389 if (!activeA.HasValue || !activeB.HasValue ||
390 Math.Abs(activeA.Value - targetA) > 0.001 ||
391 Math.Abs(activeB.Value - targetB) > 0.001)
392 {
393 sequence.Commands.Add(new CmmSensorOrientationCommand
394 {
395 CommandId = commandCounter++,
396 ToolId = toolId,
397 AngleA = targetA,
398 AngleB = targetB,
399 NormalI = pt.OrientationX,
400 NormalJ = pt.OrientationY,
401 NormalK = pt.OrientationZ,
402 Comment = $"Orient wrist to A={targetA:F1} deg, B={targetB:F1} deg"
403 });
404
405 activeA = targetA;
406 activeB = targetB;
407 }
408 }
409
410 // Determine command type based on point role
411 if (string.Equals(ptType, "Measurement", StringComparison.OrdinalIgnoreCase))
412 {
413 string featId = $"{settings.FeaturePrefix}{featureCounter++:D4}";
414 sequence.Commands.Add(new CmmTouchPointCommand
415 {
416 CommandId = commandCounter++,
417 FeatureId = featId,
418 ContactX = pt.X,
419 ContactY = pt.Y,
420 ContactZ = pt.Z,
421 NormalI = pt.OrientationX,
422 NormalJ = pt.OrientationY,
423 NormalK = pt.OrientationZ,
424 HeadAngleA = pt.HeadAngleA,
425 HeadAngleB = pt.HeadAngleB,
426 SourceFaceGlobalIndex = pt.SourceFaceGlobalIndex,
427 EntityIndex = pt.EntityIndex,
428 Comment = $"Touch point on Face {pt.SourceFaceGlobalIndex}"
429 });
430 }
431 else
432 {
433 CmmMotionType motionType = CmmMotionType.Traverse;
434 if (string.Equals(ptType, "Approach", StringComparison.OrdinalIgnoreCase))
435 motionType = CmmMotionType.Approach;
436 else if (string.Equals(ptType, "Retract", StringComparison.OrdinalIgnoreCase))
437 motionType = CmmMotionType.Retract;
438 else if (string.Equals(ptType, "Safe", StringComparison.OrdinalIgnoreCase) ||
439 string.Equals(ptType, "Start", StringComparison.OrdinalIgnoreCase) ||
440 string.Equals(ptType, "End", StringComparison.OrdinalIgnoreCase))
441 motionType = CmmMotionType.Clearance;
442 else if (string.Equals(ptType, "Bridge", StringComparison.OrdinalIgnoreCase) ||
443 string.Equals(ptType, "ShellWaypoint", StringComparison.OrdinalIgnoreCase) ||
444 string.Equals(ptType, "BridgeIntermediary", StringComparison.OrdinalIgnoreCase))
445 motionType = CmmMotionType.Rapid;
446
447 sequence.Commands.Add(new CmmMotionCommand
448 {
449 CommandId = commandCounter++,
450 MotionType = motionType,
451 X = pt.X,
452 Y = pt.Y,
453 Z = pt.Z,
454 Comment = $"{ptType} move"
455 });
456 }
457 }
458
459 program.Sequences.Add(sequence);
460 }
461
462 return program;
463 }

468 public string GenerateDmisFromIntermediate(CmmProgram program, DMISPostProcessorSettings? settings = null)
469 {
470 settings ??= new DMISPostProcessorSettings();
471 var sb = new StringBuilder();
472 string fmt = "F" + settings.CoordinateDecimals;
473
474 // Format coordinates with culture invariance
475 string Fmt(double val) => val.ToString(fmt, Inv);
476 string FmtVec(double val) => val.ToString("F4", Inv);
477
478 // -----------------------------------------------------------------
479 // 1. Program Header (DMISMN, FILNAM, UNITS, MODE, PRCOMP)
480 // -----------------------------------------------------------------
481 if (settings.IncludeComments)
482 {
483 sb.AppendLine("$$ ====================================================================");
484 sb.AppendLine("$$ ANSI/DMIS 5.3 CMM Inspection Program");
485 sb.AppendLine($"$$ Part Model : {program.ModelName}");
486 sb.AppendLine($"$$ Generated : {program.CreatedAt:yyyy-MM-dd HH:mm:ss}");
487 sb.AppendLine("$$ Generator : Hi-rEV SmartFitSystems Metrology Post-Processor");
488 sb.AppendLine("$$ Standard : ANSI/DMIS 105.3-2016");
489 sb.AppendLine("$$ ====================================================================");
490 }
491
492 sb.AppendLine($"DMISMN/'{program.ProgramName}',05.3");
493 sb.AppendLine($"FILNAM/'{program.ModelName}',05.3");
494 sb.AppendLine("DVIFRM/05.3");
495 sb.AppendLine($"DECPL/ALL,{settings.CoordinateDecimals}");
496 sb.AppendLine($"UNITS/{settings.Units},ANGDEC");
497 sb.AppendLine($"WKPLAN/{settings.WorkingPlane}");
498 sb.AppendLine("PRCOMP/ON");
499 sb.AppendLine("MODE/AUTO,PROG,MAN");
500 sb.AppendLine();
501
502 // -----------------------------------------------------------------
503 // 2. Machine & Sensor Parameters (FEDRAT, SNSET)
504 // -----------------------------------------------------------------
505 if (settings.IncludeComments)
506 {
507 sb.AppendLine("$$ --------------------------------------------------------------------");
508 sb.AppendLine("$$ Motion Speeds and Probe Clearances");
509 sb.AppendLine("$$ --------------------------------------------------------------------");
510 }
511 sb.AppendLine($"FEDRAT/POSVEL,CART,{settings.TraverseFeedrate.ToString("F1", Inv)}");
512 sb.AppendLine($"FEDRAT/MESVEL,CART,{settings.MeasuringFeedrate.ToString("F1", Inv)}");
513 sb.AppendLine($"SNSET/APPRCH,{settings.ApproachDistance.ToString("F2", Inv)}");
514 sb.AppendLine($"SNSET/RETRCT,{settings.RetractDistance.ToString("F2", Inv)}");
515 sb.AppendLine($"SNSET/SEARCH,{settings.SearchDistance.ToString("F2", Inv)}");
516 sb.AppendLine($"SNSET/CLRSRF,{(settings.ApproachDistance * 2.0).ToString("F2", Inv)}");
517 sb.AppendLine();
518
519 // -----------------------------------------------------------------
520 // 3. Machine & Sensor Kinematics Definition (SNSMNT, WRIST, SENSOR, BUILD)
521 // -----------------------------------------------------------------
522 if (settings.IncludeMachineDefinitions)
523 {
524 if (settings.IncludeComments)
525 {
526 sb.AppendLine("$$ --------------------------------------------------------------------");
527 sb.AppendLine("$$ Sensor Reference Mount & Motorized Wrist Definition (Renishaw PH10)");
528 sb.AppendLine("$$ --------------------------------------------------------------------");
529 }
530 sb.AppendLine("SNSMNT/XVEC,1,0,0,ZVEC,0,0,1,MNTLEN,0,0,0");
531
532 var wrist = program.Machine.Wrist ?? new CmmWristDefinition();
533 sb.AppendLine($"SW({wrist.WristName})=WRIST/ROTCEN,0,0,-5,0,0,1,0,1,0,ANGLE,'RotAngle',{wrist.AxisBMin.ToString("F1", Inv)},{wrist.AxisBMax.ToString("F1", Inv)},{wrist.AxisBStep.ToString("F1", Inv)},$");
534 sb.AppendLine($"ROTCEN,0,0,-25,1,0,0,0,0,-1,ANGLE,'TiltAngle',{wrist.AxisAMin.ToString("F1", Inv)},{wrist.AxisAMax.ToString("F1", Inv)},{wrist.AxisAStep.ToString("F1", Inv)},$");
535 sb.AppendLine($"MNTLEN,{wrist.MountLengthX.ToString("F1", Inv)},{wrist.MountLengthY.ToString("F1", Inv)},{wrist.MountLengthZ.ToString("F1", Inv)},'{wrist.Manufacturer}','{wrist.ModelName}',1");
536 sb.AppendLine();
537
538 if (settings.IncludeSensorDefinitions)
539 {
540 if (settings.IncludeComments)
541 {
542 sb.AppendLine("$$ --------------------------------------------------------------------");
543 sb.AppendLine("$$ Measuring Tool & Stylus Assembly Definitions");
544 sb.AppendLine("$$ --------------------------------------------------------------------");
545 }
546 foreach (var tool in program.Tools)
547 {
548 string safeId = CleanLabel(tool.ToolId);
549 sb.AppendLine($"$$ Tool Assembly: {tool.ToolId}");
550 sb.AppendLine($"SX({safeId}_EXT)=EXTENS/0,0,-{tool.ExtensionLength.ToString("F1", Inv)}");
551 sb.AppendLine($"SS({safeId}_TIP)=SENSOR/PROBE,0,0,-{tool.StylusLength.ToString("F1", Inv)},0,0,-1,{(tool.TipRadius * 2.0).ToString("F1", Inv)},SPHERE");
552 sb.AppendLine($"SGS({safeId}_GRP)=SNSGRP/BUILD,SX({safeId}_EXT),SS({safeId}_TIP)");
553 sb.AppendLine($"THLDEF/SGS({safeId}_GRP),{tool.ToolHolderSlot}");
554 sb.AppendLine($"S({safeId})=SNSDEF/BUILD,SW({wrist.WristName}),SGS({safeId}_GRP)");
555 }
556 sb.AppendLine();
557 }
558 }
559
560 // -----------------------------------------------------------------
561 // 4. Initial Safe Positioning & Home (Bounding Box Top Centerpoint)
562 // -----------------------------------------------------------------
563 if (program.Tools.Count > 0)
564 {
565 string firstTool = ResolveToolLabel(program.Tools[0].ToolId, settings);
566 sb.AppendLine($"SNSLCT/S({firstTool})");
567 }
568 if (settings.IncludeComments)
569 {
570 sb.AppendLine("$$ ====================================================================");
571 sb.AppendLine(program.IsMultipart
572 ? $"$$ INITIAL HOME POSITION: Multipart Bounding Box Top Centerpoint ({program.MultipartBounds.Count} Parts)"
573 : "$$ INITIAL HOME POSITION: Part Bounding Box Top Centerpoint");
574 sb.AppendLine($"$$ Top Clearance: Z={Fmt(program.Machine.HomeZ)} mm, Center: X={Fmt(program.Machine.HomeX)}, Y={Fmt(program.Machine.HomeY)}");
575 sb.AppendLine("$$ ====================================================================");
576 }
577 sb.AppendLine($"GOTO/CART,{Fmt(program.Machine.HomeX)},{Fmt(program.Machine.HomeY)},{Fmt(program.Machine.HomeZ)}");
578 sb.AppendLine();
579
580 // -----------------------------------------------------------------
581 // 4b. Automated Part Position Calibration & Datum Verification Routine
582 // -----------------------------------------------------------------
583 if (settings.IncludePartCalibrationRoutine && settings.HasPartBounds)
584 {
585 double xMin = settings.PartBoundsXMin!.Value;
586 double xMax = settings.PartBoundsXMax!.Value;
587 double yMin = settings.PartBoundsYMin!.Value;
588 double yMax = settings.PartBoundsYMax!.Value;
589 double zMin = settings.PartBoundsZMin!.Value;
590 double zMax = settings.PartBoundsZMax!.Value;
591
592 double xMid = (xMin + xMax) / 2.0;
593 double yMid = (yMin + yMax) / 2.0;
594 double zTouch = zMin + (zMax - zMin) * Math.Clamp(settings.CalibrationTouchDepthRatio, 0.1, 0.9);
595 double clearance = Math.Max(5.0, settings.CalibrationClearanceOffset);
596 double safeZ = zMax + clearance;
597
598 if (settings.IncludeComments)
599 {
600 sb.AppendLine("$$ ====================================================================");
601 sb.AppendLine("$$ AUTOMATED PART POSITION CALIBRATION & DATUM VERIFICATION ROUTINE");
602 sb.AppendLine("$$ Measures Top limit (Z datum) and Lateral limits (+/-X, +/-Y) to verify");
603 sb.AppendLine("$$ part placement on the CMM table prior to executing inspection touches.");
604 sb.AppendLine("$$ ====================================================================");
605 }
606
607 // 1. Top Limit Verification Touch (Primary Z Datum)
608 if (settings.IncludeComments) sb.AppendLine("$$ 1. TOP SURFACE CALIBRATION (Z Limit / Height Verification)");
609 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMid)},{Fmt(safeZ)}");
610 sb.AppendLine($"F(CAL_TOP)=FEAT/POINT,CART,{Fmt(xMid)},{Fmt(yMid)},{Fmt(zMax)},0.0000,0.0000,1.0000");
611 sb.AppendLine("MEAS/POINT,F(CAL_TOP),1");
612 sb.AppendLine($"PTMEAS/CART,{Fmt(xMid)},{Fmt(yMid)},{Fmt(zMax)},0.0000,0.0000,1.0000");
613 sb.AppendLine("ENDMES");
614 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMid)},{Fmt(safeZ)}");
615
616 // 2. Lateral -X Limit Verification
617 if (settings.IncludeComments) sb.AppendLine("$$ 2. LATERAL -X LIMIT CALIBRATION (X- Bound Verification)");
618 sb.AppendLine($"GOTO/CART,{Fmt(xMin - clearance)},{Fmt(yMid)},{Fmt(safeZ)}");
619 sb.AppendLine($"GOTO/CART,{Fmt(xMin - clearance)},{Fmt(yMid)},{Fmt(zTouch)}");
620 sb.AppendLine($"F(CAL_XMIN)=FEAT/POINT,CART,{Fmt(xMin)},{Fmt(yMid)},{Fmt(zTouch)},-1.0000,0.0000,0.0000");
621 sb.AppendLine("MEAS/POINT,F(CAL_XMIN),1");
622 sb.AppendLine($"PTMEAS/CART,{Fmt(xMin)},{Fmt(yMid)},{Fmt(zTouch)},-1.0000,0.0000,0.0000");
623 sb.AppendLine("ENDMES");
624 sb.AppendLine($"GOTO/CART,{Fmt(xMin - clearance)},{Fmt(yMid)},{Fmt(zTouch)}");
625 sb.AppendLine($"GOTO/CART,{Fmt(xMin - clearance)},{Fmt(yMid)},{Fmt(safeZ)}");
626
627 // 3. Lateral +X Limit Verification
628 if (settings.IncludeComments) sb.AppendLine("$$ 3. LATERAL +X LIMIT CALIBRATION (X+ Bound Verification)");
629 sb.AppendLine($"GOTO/CART,{Fmt(xMax + clearance)},{Fmt(yMid)},{Fmt(safeZ)}");
630 sb.AppendLine($"GOTO/CART,{Fmt(xMax + clearance)},{Fmt(yMid)},{Fmt(zTouch)}");
631 sb.AppendLine($"F(CAL_XMAX)=FEAT/POINT,CART,{Fmt(xMax)},{Fmt(yMid)},{Fmt(zTouch)},1.0000,0.0000,0.0000");
632 sb.AppendLine("MEAS/POINT,F(CAL_XMAX),1");
633 sb.AppendLine($"PTMEAS/CART,{Fmt(xMax)},{Fmt(yMid)},{Fmt(zTouch)},1.0000,0.0000,0.0000");
634 sb.AppendLine("ENDMES");
635 sb.AppendLine($"GOTO/CART,{Fmt(xMax + clearance)},{Fmt(yMid)},{Fmt(zTouch)}");
636 sb.AppendLine($"GOTO/CART,{Fmt(xMax + clearance)},{Fmt(yMid)},{Fmt(safeZ)}");
637
638 // 4. Lateral -Y Limit Verification
639 if (settings.IncludeComments) sb.AppendLine("$$ 4. LATERAL -Y LIMIT CALIBRATION (Y- Bound Verification)");
640 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMin - clearance)},{Fmt(safeZ)}");
641 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMin - clearance)},{Fmt(zTouch)}");
642 sb.AppendLine($"F(CAL_YMIN)=FEAT/POINT,CART,{Fmt(xMid)},{Fmt(yMin)},{Fmt(zTouch)},0.0000,-1.0000,0.0000");
643 sb.AppendLine("MEAS/POINT,F(CAL_YMIN),1");
644 sb.AppendLine($"PTMEAS/CART,{Fmt(xMid)},{Fmt(yMin)},{Fmt(zTouch)},0.0000,-1.0000,0.0000");
645 sb.AppendLine("ENDMES");
646 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMin - clearance)},{Fmt(zTouch)}");
647 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMin - clearance)},{Fmt(safeZ)}");
648
649 // 5. Lateral +Y Limit Verification
650 if (settings.IncludeComments) sb.AppendLine("$$ 5. LATERAL +Y LIMIT CALIBRATION (Y+ Bound Verification)");
651 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMax + clearance)},{Fmt(safeZ)}");
652 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMax + clearance)},{Fmt(zTouch)}");
653 sb.AppendLine($"F(CAL_YMAX)=FEAT/POINT,CART,{Fmt(xMid)},{Fmt(yMax)},{Fmt(zTouch)},0.0000,1.0000,0.0000");
654 sb.AppendLine("MEAS/POINT,F(CAL_YMAX),1");
655 sb.AppendLine($"PTMEAS/CART,{Fmt(xMid)},{Fmt(yMax)},{Fmt(zTouch)},0.0000,1.0000,0.0000");
656 sb.AppendLine("ENDMES");
657 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMax + clearance)},{Fmt(zTouch)}");
658 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMax + clearance)},{Fmt(safeZ)}");
659
660 // Retract back to top safe clearance
661 sb.AppendLine($"GOTO/CART,{Fmt(xMid)},{Fmt(yMid)},{Fmt(safeZ)}");
662 sb.AppendLine();
663
664 // Establish Part Coordinate System (PCS) Alignment and Tolerance Verification
665 if (settings.IncludeComments)
666 {
667 sb.AppendLine("$$ --------------------------------------------------------------------");
668 sb.AppendLine("$$ Establish Part Coordinate System (PCS) via 3-2-1 Datum Alignment");
669 sb.AppendLine("$$ --------------------------------------------------------------------");
670 }
671 sb.AppendLine("DATDEF/F(CAL_TOP),DAT(A)");
672 sb.AppendLine("DATDEF/F(CAL_YMIN),DAT(B)");
673 sb.AppendLine("DATDEF/F(CAL_XMIN),DAT(C)");
674 sb.AppendLine("DATSET/DAT(A),ZDIR,ZORIG,DAT(B),YDIR,YORIG,DAT(C),XORIG");
675 sb.AppendLine();
676
677 if (settings.IncludeComments)
678 {
679 sb.AppendLine("$$ --------------------------------------------------------------------");
680 sb.AppendLine("$$ Workpiece Position Deviation Evaluation against Nominal Limits");
681 sb.AppendLine("$$ --------------------------------------------------------------------");
682 }
683 sb.AppendLine("TOL/POS,CART,-2.0000,2.0000");
684 sb.AppendLine("EVAL/F(CAL_TOP)");
685 sb.AppendLine("EVAL/F(CAL_XMIN)");
686 sb.AppendLine("EVAL/F(CAL_YMIN)");
687 sb.AppendLine();
688 }
689
690 // -----------------------------------------------------------------
691 // 5. Execution Sequences (Moves, Head Indexing, Contact Touch Measurements)
692 // -----------------------------------------------------------------
693 double? lastGotoX = null;
694 double? lastGotoY = null;
695 double? lastGotoZ = null;
696 string? activeToolId = null;
697 int? lastEntityIndex = null;
698
699 foreach (var seq in program.Sequences)
700 {
701 if (settings.IncludeComments)
702 {
703 sb.AppendLine("$$ --------------------------------------------------------------------");
704 sb.AppendLine($"$$ Sequence: {seq.SequenceName}");
705 sb.AppendLine($"$$ Assigned Tool: {seq.ToolId} | Face Color: {seq.ColorName}");
706 sb.AppendLine("$$ --------------------------------------------------------------------");
707 }
708
709 foreach (var cmd in seq.Commands)
710 {
711 switch (cmd)
712 {
713 case CmmCommentCommand commentCmd:
714 if (settings.IncludeComments && !string.IsNullOrWhiteSpace(commentCmd.Text))
715 {
716 sb.AppendLine($"$$ {commentCmd.Text}");
717 }
718 break;
719
720 case CmmToolChangeCommand toolChangeCmd:
721 string targetTool = ResolveToolLabel(toolChangeCmd.ToolId, settings);
722 if (!string.Equals(activeToolId, targetTool, StringComparison.OrdinalIgnoreCase))
723 {
724 // Retract to safe clearance height before tool change if currently lower
725 if (lastGotoZ.HasValue && lastGotoZ.Value < program.SafeClearanceZ - 1.0)
726 {
727 sb.AppendLine($"GOTO/CART,{Fmt(lastGotoX ?? program.Machine.HomeX)},{Fmt(lastGotoY ?? program.Machine.HomeY)},{Fmt(program.SafeClearanceZ)}");
728 lastGotoZ = program.SafeClearanceZ;
729 }
730
731 sb.AppendLine($"SNSLCT/S({targetTool})");
732 activeToolId = targetTool;
733 }
734 break;
735
736 case CmmSensorOrientationCommand orientCmd:
737 string orientTool = ResolveToolLabel(string.IsNullOrWhiteSpace(orientCmd.ToolId) ? (activeToolId ?? "DefaultTool") : orientCmd.ToolId, settings);
738 string wristLabel = program.Machine.Wrist?.WristName ?? "PH10";
739
740 // Retract to safe clearance height before wrist reorientation if currently lower
741 if (lastGotoZ.HasValue && lastGotoZ.Value < program.SafeClearanceZ - 1.0)
742 {
743 sb.AppendLine($"GOTO/CART,{Fmt(lastGotoX ?? program.Machine.HomeX)},{Fmt(lastGotoY ?? program.Machine.HomeY)},{Fmt(program.SafeClearanceZ)}");
744 lastGotoZ = program.SafeClearanceZ;
745 }
746
747 if (settings.WristAnglesMode == DmisWristAnglesMode.WristAngles ||
748 settings.WristAnglesMode == DmisWristAnglesMode.Both)
749 {
750 sb.AppendLine($"SNSLCT/S({orientTool}),SW({wristLabel}),'RotAngle',{orientCmd.AngleB.ToString("F1", Inv)},'TiltAngle',{orientCmd.AngleA.ToString("F1", Inv)}");
751 }
752 else
753 {
754 sb.AppendLine($"SNSLCT/S({orientTool}),HEADCS,{orientCmd.AngleA.ToString("F1", Inv)},{orientCmd.AngleB.ToString("F1", Inv)}");
755 }
756 break;
757
758 case CmmMotionCommand motionCmd:
759 // Check for redundant consecutive duplicate moves
760 if (settings.FilterDuplicateMoves &&
761 lastGotoX.HasValue && lastGotoY.HasValue && lastGotoZ.HasValue &&
762 Math.Abs(lastGotoX.Value - motionCmd.X) < settings.DuplicateTolerance &&
763 Math.Abs(lastGotoY.Value - motionCmd.Y) < settings.DuplicateTolerance &&
764 Math.Abs(lastGotoZ.Value - motionCmd.Z) < settings.DuplicateTolerance)
765 {
766 continue;
767 }
768
769 sb.AppendLine($"GOTO/CART,{Fmt(motionCmd.X)},{Fmt(motionCmd.Y)},{Fmt(motionCmd.Z)}");
770 lastGotoX = motionCmd.X;
771 lastGotoY = motionCmd.Y;
772 lastGotoZ = motionCmd.Z;
773 break;
774
775 case CmmTouchPointCommand touchCmd:
776 int curEntity = touchCmd.EntityIndex;
777 if (lastEntityIndex.HasValue && curEntity != lastEntityIndex.Value)
778 {
779 if (settings.IncludeComments)
780 {
781 sb.AppendLine("$$ --------------------------------------------------------------------");
782 sb.AppendLine($"$$ Multipart Bridge Transition: Moving from Part {lastEntityIndex.Value} to Part {curEntity}");
783 sb.AppendLine("$$ --------------------------------------------------------------------");
784 }
785
786 // 1. Ascend to safe clearance over previous part
787 if (lastGotoZ.HasValue && lastGotoZ.Value < program.SafeClearanceZ - 1.0)
788 {
789 sb.AppendLine($"GOTO/CART,{Fmt(lastGotoX ?? program.Machine.HomeX)},{Fmt(lastGotoY ?? program.Machine.HomeY)},{Fmt(program.SafeClearanceZ)}");
790 lastGotoZ = program.SafeClearanceZ;
791 }
792
793 // 2. Traverse at safe clearance to target part top centerpoint
794 if (program.MultipartBounds.TryGetValue(curEntity, out var targetPartBounds))
795 {
796 sb.AppendLine($"GOTO/CART,{Fmt(targetPartBounds.CenterX)},{Fmt(targetPartBounds.CenterY)},{Fmt(program.SafeClearanceZ)}");
797 lastGotoX = targetPartBounds.CenterX;
798 lastGotoY = targetPartBounds.CenterY;
799 }
800 }
801 lastEntityIndex = curEntity;
802
803 // Normalize normal vector if needed
804 double len = Math.Sqrt(touchCmd.NormalI * touchCmd.NormalI +
805 touchCmd.NormalJ * touchCmd.NormalJ +
806 touchCmd.NormalK * touchCmd.NormalK);
807 double ni = len > 1e-6 ? touchCmd.NormalI / len : 0.0;
808 double nj = len > 1e-6 ? touchCmd.NormalJ / len : 0.0;
809 double nk = len > 1e-6 ? touchCmd.NormalK / len : 1.0;
810
811 string featId = string.IsNullOrWhiteSpace(touchCmd.FeatureId)
812 ? $"{settings.FeaturePrefix}{touchCmd.CommandId:D4}"
813 : touchCmd.FeatureId;
814
815 if (settings.GenerateFeatureDefinitions)
816 {
817 sb.AppendLine($"F({featId})=FEAT/POINT,CART,{Fmt(touchCmd.ContactX)},{Fmt(touchCmd.ContactY)},{Fmt(touchCmd.ContactZ)},{FmtVec(ni)},{FmtVec(nj)},{FmtVec(nk)}");
818 sb.AppendLine($"MEAS/POINT,F({featId}),1");
819 }
820
821 if (settings.WristAnglesMode == DmisWristAnglesMode.Both &&
822 touchCmd.HeadAngleA.HasValue && touchCmd.HeadAngleB.HasValue)
823 {
824 sb.AppendLine($"PTMEAS/CART,{Fmt(touchCmd.ContactX)},{Fmt(touchCmd.ContactY)},{Fmt(touchCmd.ContactZ)},{FmtVec(ni)},{FmtVec(nj)},{FmtVec(nk)},HEADCS,{touchCmd.HeadAngleA.Value.ToString("F1", Inv)},{touchCmd.HeadAngleB.Value.ToString("F1", Inv)}");
825 }
826 else
827 {
828 sb.AppendLine($"PTMEAS/CART,{Fmt(touchCmd.ContactX)},{Fmt(touchCmd.ContactY)},{Fmt(touchCmd.ContactZ)},{FmtVec(ni)},{FmtVec(nj)},{FmtVec(nk)}");
829 }
830
831 if (settings.GenerateFeatureDefinitions)
832 {
833 sb.AppendLine("ENDMES");
834 }
835
836 // Touch point updates last machine position
837 lastGotoX = touchCmd.ContactX;
838 lastGotoY = touchCmd.ContactY;
839 lastGotoZ = touchCmd.ContactZ;
840 break;
841 }
842 }
843
844 sb.AppendLine();
845 }
846
847 // -----------------------------------------------------------------
848 // 6. Return Home & File Termination
849 // -----------------------------------------------------------------
850 if (settings.ReturnHomeAtEnd)
851 {
852 if (settings.IncludeComments)
853 {
854 sb.AppendLine("$$ --------------------------------------------------------------------");
855 sb.AppendLine(program.IsMultipart
856 ? $"$$ Return to Multipart Top Centerpoint Home Position ({program.MultipartBounds.Count} Parts)"
857 : "$$ Return to Machine Clearance / Home Position (Top Centerpoint)");
858 sb.AppendLine("$$ --------------------------------------------------------------------");
859 }
860 if (lastGotoZ.HasValue && lastGotoZ.Value < program.SafeClearanceZ - 1.0)
861 {
862 sb.AppendLine($"GOTO/CART,{Fmt(lastGotoX ?? program.Machine.HomeX)},{Fmt(lastGotoY ?? program.Machine.HomeY)},{Fmt(program.SafeClearanceZ)}");
863 }
864 sb.AppendLine($"GOTO/CART,{Fmt(program.Machine.HomeX)},{Fmt(program.Machine.HomeY)},{Fmt(program.Machine.HomeZ)}");
865 }
866
867 sb.AppendLine("ENDFIL");
868 return sb.ToString();
869 }

957 private static string ResolveToolLabel(string toolId, DMISPostProcessorSettings? settings)
958 {
959 if (settings?.ToolMappings != null &&
960 settings.ToolMappings.TryGetValue(toolId, out var mapping) &&
961 !string.IsNullOrWhiteSpace(mapping.TargetCmmSensorLabel))
962 {
963 return CleanLabel(mapping.TargetCmmSensorLabel);
964 }
965
966 if (CmmStandardToolCatalog.IsStandaloneStylusTool(toolId))
967 {
968 var montage = CmmStandardToolCatalog.GetBestMontageForStylus(toolId);
969 return CleanLabel(montage.MontageId);
970 }
971
972 var resolvedMontage = CmmStandardToolCatalog.StandardMontages.FirstOrDefault(m =>
973 string.Equals(m.MontageId, toolId, StringComparison.OrdinalIgnoreCase));
974 if (resolvedMontage != null)
975 {
976 return CleanLabel(resolvedMontage.MontageId);
977 }
978
979 return CleanLabel(toolId);
980 }
981
985 private static string CleanLabel(string label)
986 {
987 if (string.IsNullOrWhiteSpace(label)) return "TOOL1";
988 var sb = new StringBuilder();
989 foreach (char c in label)
990 {
991 if (char.IsLetterOrDigit(c) || c == '_')
992 sb.Append(c);
993 else
994 sb.Append('_');
995 }
996 string cleaned = sb.ToString();
997 if (char.IsDigit(cleaned[0]))
998 cleaned = "T_" + cleaned;
999 return cleaned;
1000 }
