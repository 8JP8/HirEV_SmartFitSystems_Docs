172 private static string ResolveFolder(string relativeFolder)
173 {
174 string baseDir = AppDomain.CurrentDomain.BaseDirectory;
175 string directPath = Path.Combine(baseDir, relativeFolder);
176
177 string? current = baseDir;
178 for (int i = 0; i < 8; i++)
179 {
180 current = Path.GetDirectoryName(current?.TrimEnd(Path.DirectorySeparatorChar));
181 if (current == null)
182 break;
183
184 string check = Path.Combine(current, relativeFolder);
185 if (Directory.Exists(check) && Directory.GetFiles(current, "*.csproj").Length > 0)
186 return check;
187 }
188
189 if (Directory.Exists(directPath))
190 return directPath;
191
192 current = baseDir;
193 for (int i = 0; i < 8; i++)
194 {
195 current = Path.GetDirectoryName(current?.TrimEnd(Path.DirectorySeparatorChar));
196 if (current == null)
197 break;
198
199 string check = Path.Combine(current, relativeFolder);
200 if (Directory.Exists(check))
201 return check;
202 }
203
204 return directPath;
205 }
