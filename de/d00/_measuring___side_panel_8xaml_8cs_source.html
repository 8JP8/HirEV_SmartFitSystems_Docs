109 public void AppendLog(string text)
110 {
111 Application.Current.Dispatcher.BeginInvoke(() =>
112 {
113 var lines = text.Split(new[] { "\r\n", "\r", "\n" }, StringSplitOptions.RemoveEmptyEntries);
114 foreach (var line in lines)
115 {
116 var trimmedLine = line.Trim();
117
118 // Completely ignore box drawing lines
119 if (trimmedLine.StartsWith("╔") || trimmedLine.StartsWith("║") || trimmedLine.StartsWith("╚") || trimmedLine.StartsWith("═") || trimmedLine.StartsWith("---"))
120 {
121 continue;
122 }
123
124 // Ignore intermediate "..." logs to keep the UI clean
125 if (trimmedLine.EndsWith("...") && !trimmedLine.StartsWith("STEP") && !trimmedLine.StartsWith("PASSO") && !trimmedLine.Contains("Configuração") && !trimmedLine.Contains("Configuration"))
126 {
127 continue;
128 }
129
130 string foreground = "Black";
131 FontWeight fontWeight = FontWeights.Normal;
132
133 if (trimmedLine.Contains("SUCCESS"))
134 {
135 foreground = "Green";
136 trimmedLine = trimmedLine.Replace("SUCCESS:", "").Replace("SUCCESS :", "").Trim();
137 }
138 else if (trimmedLine.Contains("WARNING"))
139 {
140 foreground = "Orange";
141 trimmedLine = trimmedLine.Replace("WARNING:", "").Replace("WARNING :", "").Trim();
142 }
143 else if (trimmedLine.Contains("ERROR"))
144 {
145 foreground = "Red";
146 trimmedLine = trimmedLine.Replace("ERROR:", "").Replace("ERROR :", "").Trim();
147 }
148 else if (trimmedLine.Contains("INFO"))
149 {
150 foreground = "Blue";
151 trimmedLine = trimmedLine.Replace("INFO:", "").Replace("INFO :", "").Trim();
152 }
153
154 // Remove emojis (surrogate pairs and common symbol blocks)
155 trimmedLine = System.Text.RegularExpressions.Regex.Replace(trimmedLine, @"[\uD800-\uDFFF\u2600-\u27BF]", "").Trim();
156 // Remove variation selectors that might be left behind
157 trimmedLine = trimmedLine.Replace("\uFE0F", "").Trim();
158
159 var newNode = new LogNode { Header = trimmedLine, Foreground = foreground, FontWeight = fontWeight };
160
161 if (trimmedLine.Contains("PATH PLANNING") || trimmedLine.Contains("PATH GENERATION") || trimmedLine.Contains("INÍCIO DO PLANEAMENTO") || trimmedLine.Contains("Log Start"))
162 {
163 LogNodes.Clear(); // Automatically clear previous run
164 newNode.Header = ViewModel.MainViewModel.GetXMLElementValue(DocLanguage, "PathManager_PathPlanningStarted") ?? "PATH PLANNING STARTED";
165 newNode.FontWeight = FontWeights.Bold;
166 _activeCategoryNode = null;
167 LogNodes.Add(newNode);
168 }
169 else if (trimmedLine.StartsWith("STEP") || trimmedLine.StartsWith("PASSO") || trimmedLine.StartsWith("COLOR") || trimmedLine.StartsWith("COR") || trimmedLine.Contains("Configuration") || trimmedLine.Contains("Configuração") || trimmedLine.StartsWith("RESUMO") || trimmedLine.StartsWith("SUMMARY"))
170 {
171 if (_activeCategoryNode != null)
172 {
173 _activeCategoryNode.Header = _activeCategoryNode.Header.ToString()!.Replace("🕒", "✔️").Replace("▶️", "✔️");
174 }
175
176 newNode.FontWeight = FontWeights.Normal; // Only title bold
177 newNode.Header = "▶️ " + newNode.Header;
178
179 LogNodes.Add(newNode);
180 _activeCategoryNode = newNode;
181 }
182 else
183 {
184 // Mark the final node as completed if the process finishes
185 if (trimmedLine.Contains("Save successful") || trimmedLine.Contains("Path exported to XML") || trimmedLine.Contains("cancelled by user") || trimmedLine.Contains("COMPLETED SUCCESSFULLY") || trimmedLine.Contains("CONCLUÍDO"))
186 {
187 if (_activeCategoryNode != null && _activeCategoryNode.Header.ToString()!.StartsWith("▶️"))
188 {
189 _activeCategoryNode.Header = _activeCategoryNode.Header.ToString()!.Replace("▶️", "✔️");
190 }
191 }

723 public void UpdateMeasurementCounts()
724 {
725 try
726 {
727 // Count non-ignored faces (faces not in ignored list and not part of ignored holes)
728 int totalNonIgnoredFaces = 0;
729 foreach (var colorGroup in mvm!._facesByColor.Values)
730 {
731 totalNonIgnoredFaces += colorGroup.Count(f => !mvm!.IgnoredFaces.Contains(f));
732 }
733
734 // Add hole faces that are not ignored
735 foreach (var holeList in mvm!._holesByColor.Values)
736 {
737 foreach (var hole in holeList.Where(h => !mvm!.IgnoredHoles.Contains(h)))
738 {
739 totalNonIgnoredFaces += hole.FaceGlobalIndices.Count;
740 }
741 }
742
743 SelectedFaces_Count = totalNonIgnoredFaces;
744
745 // Count total holes (non-ignored)
746 int totalNonIgnoredHoles = 0;
747 foreach (var holeList in mvm!._holesByColor.Values)
748 {
749 totalNonIgnoredHoles += holeList.Count(h => !mvm!.IgnoredHoles.Contains(h));
750 }
751 Holes_Count = totalNonIgnoredHoles;
752
753 // Count tool changes (number of unique colors with non-ignored items - 1)
754 var activeColors = new HashSet<string>();
755
756 // Add colors from non-ignored faces
757 foreach (var kvp in mvm!._facesByColor)
758 {
759 if (kvp.Value.Any(f => !mvm!.IgnoredFaces.Contains(f)))
760 {
761 activeColors.Add(kvp.Key);
762 }
763 }
764
765 // Add colors from non-ignored holes
766 foreach (var kvp in mvm!._holesByColor)
767 {
768 if (kvp.Value.Any(h => !mvm!.IgnoredHoles.Contains(h)))
769 {
770 activeColors.Add(kvp.Key);
771 }
772 }
773
774 ToolChanges_Count = (CMMType == "Probe") ? Math.Max(0, activeColors.Count - 1) : 0;
775
776 // Processes
777 //Processes_Count = 0;
778
779 Debug.WriteLine($"📊 Measurement counts updated:");
780 Debug.WriteLine($" Faces: {SelectedFaces_Count}");
781 Debug.WriteLine($" Holes: {Holes_Count}");
782 Debug.WriteLine($" Tool Changes: {ToolChanges_Count}");
783 Debug.WriteLine($" Processes: {Processes_Count}");
784 }
785 catch (Exception ex)
786 {
787 Debug.WriteLine($"❌ Error updating measurement counts: {ex.Message}");
788 }
789 }
