188 var lines = text.Split(new[] { "\r\n", "\r", "\n" }, StringSplitOptions.RemoveEmptyEntries);
189 foreach (var line in lines)
190 {
191 var trimmedLine = line.Trim();
192
193 if (IsRawScanPathDebugLine(trimmedLine))
194 {
195 continue;
196 }
197
198 // Completely ignore box drawing lines
199 if (trimmedLine.StartsWith("╔") || trimmedLine.StartsWith("║") || trimmedLine.StartsWith("╚") || trimmedLine.StartsWith("═") || trimmedLine.StartsWith("---"))
200 {
201 continue;
202 }
203
204 // Ignore intermediate "..." logs to keep the UI clean
205 if (trimmedLine.EndsWith("...") && !trimmedLine.StartsWith("STEP") && !trimmedLine.StartsWith("PASSO") && !trimmedLine.Contains("Configuração") && !trimmedLine.Contains("Configuration"))
206 {
207 continue;
208 }
209
210 // Ignore raw Visual Studio debug exceptions to organize the treeview better
211 if (trimmedLine.StartsWith("Exception thrown:") || (trimmedLine.StartsWith("The thread") && trimmedLine.Contains("has exited with code")))
212 {
213 continue;
214 }
215
216 string foreground = "Black";
217 FontWeight fontWeight = FontWeights.Normal;
218
219 if (trimmedLine.Contains("SUCCESS"))

221 foreground = "Green";
222 trimmedLine = trimmedLine.Replace("SUCCESS:", "").Replace("SUCCESS :", "").Trim();
223 }
224 else if (trimmedLine.Contains("WARNING"))
225 {
226 foreground = "Orange";
227 trimmedLine = trimmedLine.Replace("WARNING:", "").Replace("WARNING :", "").Trim();

229 else if (trimmedLine.Contains("ERROR"))

231 foreground = "Red";
232 trimmedLine = trimmedLine.Replace("ERROR:", "").Replace("ERROR :", "").Trim();
233 }
234 else if (trimmedLine.Contains("INFO"))

237 trimmedLine = trimmedLine.Replace("INFO:", "").Replace("INFO :", "").Trim();
238 }

240 // Remove emojis (surrogate pairs and common symbol blocks)
241 trimmedLine = System.Text.RegularExpressions.Regex.Replace(trimmedLine, @"[\uD800-\uDFFF\u2600-\u27BF]", "").Trim();
242 // Remove variation selectors that might be left behind
243 trimmedLine = trimmedLine.Replace("\uFE0F", "").Trim();

245 var newNode = new LogNode { Header = trimmedLine, Foreground = foreground, FontWeight = fontWeight };
246
247 bool isPathPlanningStart = trimmedLine.Contains("PATH PLANNING STARTED") || trimmedLine.Contains("PATH GENERATION") || trimmedLine.Contains("INÍCIO DO PLANEAMENTO") || trimmedLine.Contains("INICIO DO PLANEAMENTO") || trimmedLine.Contains("Log Start");
248 bool isSimulationStart = trimmedLine.Contains("STARTING PATH SIMULATION") || trimmedLine.Contains("INICIO DA SIMULA") || trimmedLine.Contains("INÍCIO DA SIMULA");
249 bool isCollisionCheckStart = trimmedLine.Contains("Running probe collision validation") || trimmedLine.Contains("executar valida") || trimmedLine.Contains("Running probe collision check") || trimmedLine.Contains("A executar verifica");
250 bool isScanVerificationStart = trimmedLine.Contains("Verification worker thread started") || trimmedLine.Contains("trabalhador de verifica");
251 bool isDiagnosticsStart = trimmedLine.Contains("RUNNING PATH PLANNING DIAGNOSTICS") || trimmedLine.Contains("A EXECUTAR DIAGNOSTICO");

253
254 if (isSimulationStart || isCollisionCheckStart || isScanVerificationStart || isDiagnosticsStart || isFaceExtractionStart)

256 _isCapturingPathGenerationLogs = false;

1792 {
1793 activeColors.Add(kvp.Key);

1810

1814 else

1818 foreach (var kvp in mvm!._facesByColor)
