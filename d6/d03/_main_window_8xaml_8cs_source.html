150 private void TreeScrollViewer_PreviewMouseLeftButtonDown(object sender, MouseButtonEventArgs e)
151 {
152 if (sender is TreeView tv)
153 {
154 // Get the ScrollViewer from the TreeView
155 if (treeScrollViewer == null)
156 {
157 treeScrollViewer = FindVisualChild<ScrollViewer>(tv);

160
161 // Check if user clicked on a scrollbar thumb
162 var depObj = e.OriginalSource as DependencyObject;
163 if (FindVisualParent<System.Windows.Controls.Primitives.Thumb>(depObj!) != null)
164 {
165 isDraggingScrollBar = true;
166 return;
167 }
168
169 // NEW: Check if clicked on a TreeViewItem - if so, don't interfere

171 if (treeViewItem != null)
172 {
173 treeViewItem.IsSelected = true;
174 if (DataContext is MainViewModel viewModel && treeViewItem.Tag != null)
175 viewModel.HighlightGeometryItem(treeViewItem.Tag);
176
177 // User clicked on an actual item, let TreeView handle expansion and focus.
178 return;
179 }
180
181 // User clicked on empty space or scrollviewer background
182 isDraggingScrollBar = false;
183 isScrollDragging = false; // Reset drag state
184 scrollMousePoint = e.GetPosition(tv);
185 verticalOffset = treeScrollViewer.VerticalOffset;

194 private void TreeScrollViewer_MouseMove(object sender, MouseEventArgs e)
195 {
196 if (isDraggingScrollBar) return;
197
198 if (sender is TreeView tv && e.LeftButton == MouseButtonState.Pressed)
199 {
200 if (treeScrollViewer == null) return;
201
202 var currentPosition = e.GetPosition(tv);
203 double distance = Math.Abs(currentPosition.Y - scrollMousePoint.Y);
204
205 // NEW: Only start dragging if moved beyond threshold
206 if (!isScrollDragging && distance > DragThreshold)
207 {
208 isScrollDragging = true;
209 tv.CaptureMouse();
210 }
211
212 // Only scroll if we're actually dragging
213 if (isScrollDragging)
214 {
215 double newVerticalOffset = verticalOffset + (scrollMousePoint.Y - currentPosition.Y);
216 treeScrollViewer.ScrollToVerticalOffset(newVerticalOffset);
217 }
218 }
219 }

264 private void grid_MouseMove(object sender, MouseEventArgs e)
265 {
266 // Check if the left mouse button is pressed
267 if (e.LeftButton == MouseButtonState.Pressed)
268 {
269 if (!_isDragging)
270 {
271 // Initialize dragging when first detected
272 _mousePosition = e.GetPosition(this); // Capture the initial mouse position
273 _isDragging = true;
274 }
275
276 // Calculate the difference from the original mouse position
277 var currentPosition = e.GetPosition(this);
278 var delta = currentPosition - _mousePosition;
279
280 // If the mouse has moved significantly, exit maximized state
281 if (Math.Abs(delta.X) > SystemParameters.MinimumHorizontalDragDistance ||
282 Math.Abs(delta.Y) > SystemParameters.MinimumVerticalDragDistance)
283 {
284 if (WindowState == WindowState.Maximized)
285 {
286 // Calculate the relative position to set the window's position accordingly
287 var mousePositionOnScreen = PointToScreen(_mousePosition);
288 double widthRatio = _mousePosition.X / ActualWidth;
289
290 // Restore the window state
291 WindowState = WindowState.Normal;
292
293 // Set the new window position to keep the cursor in the same position
294 Left = mousePositionOnScreen.X - (widthRatio * ActualWidth);
295 Top = mousePositionOnScreen.Y - _mousePosition.Y;
296 }
297
298 // Start dragging the window
299 DragMove();
300 _isDragging = false; // End dragging
301 }
302 }
303 else
304 {
305 // Reset dragging state if the left mouse button is not pressed
306 _isDragging = false;
307 }
308 }

319 private T FindVisualChild<T>(DependencyObject parent) where T : DependencyObject
320 {
321 if (parent == null) return null!;
322
323 for (int i = 0; i < VisualTreeHelper.GetChildrenCount(parent); i++)
324 {
325 var child = VisualTreeHelper.GetChild(parent, i);
326 if (child is T typedChild)
327 {
328 return typedChild;
329 }
330 var result = FindVisualChild<T>(child);
331 if (result != null)
332 {
333 return result;
334 }
335 }
336 return null!;
337 }

345 private T? FindVisualParent<T>(DependencyObject child) where T : DependencyObject
346 {
347 while (child != null)
348 {
349 try
350 {
351 child = VisualTreeHelper.GetParent(child);
352 if (child is T parent)
353 {
354 return parent;
355 }
356 }
357 catch (InvalidOperationException)
358 {
359 // Hit a non-visual element (like TextBlock content)
360 // This means we're inside a TreeViewItem, so return null to allow normal TreeView behavior
361 return null!;
362 }
363 }
364 return null!;
365 }
