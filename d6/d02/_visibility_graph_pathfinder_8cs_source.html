173 public void BuildGraph(List<Point3D> waypoints, List<Brep?> obstacles, double maxEdgeDistance = 300.0)
174 {
175 var sw = Stopwatch.StartNew();
176
177 _obstacles = obstacles ?? new List<Brep?>();
178 _nodes.Clear();
179 _nodeMap.Clear();
180
181 if (waypoints == null || waypoints.Count == 0)
182 {
183 Debug.WriteLine("⚠️ VisibilityGraph: No waypoints provided");
184 return;
185 }
186
187 // Remove duplicate waypoints (within 1mm tolerance)
188 var uniqueWaypoints = RemoveDuplicates(waypoints, 1.0);
189
190 Debug.WriteLine($"🔷 VisibilityGraph: Building graph with {uniqueWaypoints.Count} waypoints...");
191
192 // Create nodes
193 for (int i = 0; i < uniqueWaypoints.Count; i++)
194 {
195 var node = new GraphNode(uniqueWaypoints[i], i);
196 _nodes.Add(node);
197 _nodeMap[uniqueWaypoints[i]] = node;
198 }
199
200 // Build edges (visibility connections)
201 int totalEdges = 0;
202 int collisionChecks = 0;
203
204 for (int i = 0; i < _nodes.Count; i++)
205 {
206 var nodeA = _nodes[i];
207 var candidateEdges = new List<(GraphNode target, double dist)>();
208
209 for (int j = i + 1; j < _nodes.Count; j++)
210 {
211 var nodeB = _nodes[j];
212 double distance = nodeA.Position.DistanceTo(nodeB.Position);
213
214 // Skip if too far apart
215 if (distance > maxEdgeDistance)
216 continue;
217
218 candidateEdges.Add((nodeB, distance));
219 }
220
221 // Sort by distance and limit candidates for performance
222 candidateEdges = candidateEdges
223 .OrderBy(e => e.dist)
224 .Take(MAX_EDGES_PER_NODE * 2)
225 .ToList();
226
227 // Check visibility for candidates
228 int edgesAdded = 0;
229 foreach (var (target, dist) in candidateEdges)
230 {
231 if (edgesAdded >= MAX_EDGES_PER_NODE)
232 break;
233
234 collisionChecks++;
235 if (IsSegmentVisible(nodeA.Position, target.Position))
236 {
237 // Add bidirectional edge
238 nodeA.Edges.Add(new GraphEdge(target, dist));
239 target.Edges.Add(new GraphEdge(nodeA, dist));
240 totalEdges++;
241 edgesAdded++;
242 }
243 }
244 }
245
246 _graphBuilt = true;
247 sw.Stop();
248
249 Debug.WriteLine($" ✅ Graph built: {_nodes.Count} nodes, {totalEdges} edges");
250 Debug.WriteLine($" 📊 Collision checks: {collisionChecks}, Time: {sw.ElapsedMilliseconds}ms");
251 }

259 public PathResult FindPath(Point3D start, Point3D goal)
260 {
261 var result = new PathResult();
262 var sw = Stopwatch.StartNew();
263
264 // Check if direct path is clear
265 if (IsSegmentVisible(start, goal))
266 {
267 result.Success = true;
268 result.Path = new List<Point3D> { start, goal };
269 result.TotalDistance = start.DistanceTo(goal);
270 result.Message = "Direct path is clear";
271 return result;
272 }
273
274 if (!_graphBuilt || _nodes.Count == 0)
275 {
276 result.Success = false;
277 result.Message = "Graph not built or empty";
278 return result;
279 }
280
281 // Reset all nodes for new search
282 foreach (var node in _nodes)
283 {
284 node.Reset();
285 }
286
287 // Find or create start and goal nodes
288 var startNode = GetOrCreateTemporaryNode(start);
289 var goalNode = GetOrCreateTemporaryNode(goal);
290
291 if (startNode == null || goalNode == null)
292 {
293 result.Success = false;
294 result.Message = "Could not connect start or goal to graph";
295 return result;
296 }
297
298 // A* search
299 var openSet = new SortedSet<GraphNode>(Comparer<GraphNode>.Create((a, b) =>
300 {
301 int cmp = a.FScore.CompareTo(b.FScore);
302 return cmp != 0 ? cmp : a.Index.CompareTo(b.Index);
303 }));
304
305 startNode.GScore = 0;
306 startNode.FScore = Heuristic(startNode.Position, goal);
307 openSet.Add(startNode);
308
309 while (openSet.Count > 0)
310 {
311 var current = openSet.Min;
312 if (current == null) break;
313
314 openSet.Remove(current);
315 result.NodesExplored++;
316
317 if (current == goalNode || current.Position.DistanceTo(goal) < 0.1)
318 {
319 // Reconstruct path
320 result.Path = ReconstructPath(current, start, goal);
321 result.TotalDistance = current.GScore;
322 result.Success = true;
323 result.Message = $"Path found via {result.Path.Count - 2} intermediate waypoints";
324 sw.Stop();
325 Debug.WriteLine($" ✅ A* found path: {result.TotalDistance:F1}mm, {result.NodesExplored} nodes explored, {sw.ElapsedMilliseconds}ms");
326 return result;
327 }
328
329 current.IsVisited = true;
330
331 foreach (var edge in current.Edges)
332 {
333 if (edge.Target.IsVisited)
334 continue;
335
336 double tentativeGScore = current.GScore + edge.Distance;
337
338 if (tentativeGScore < edge.Target.GScore)
339 {
340 edge.Target.Parent = current;
341 edge.Target.GScore = tentativeGScore;
342 edge.Target.FScore = tentativeGScore + Heuristic(edge.Target.Position, goal);
343
344 if (!openSet.Contains(edge.Target))
345 {
346 openSet.Add(edge.Target);
347 }
348 }
349 }
350 }
351
352 sw.Stop();
353 result.Success = false;
354 result.Message = $"No path found after exploring {result.NodesExplored} nodes";
355 Debug.WriteLine($" ❌ A* failed: {result.Message} ({sw.ElapsedMilliseconds}ms)");
356 return result;
357 }

382 private bool IsSegmentVisible(Point3D start, Point3D end)
383 {
384 if (_obstacles == null || _obstacles.Count == 0)
385 return true;
386
387 // Sample points along segment
388 Vector3D direction = new Vector3D(
389 end.X - start.X,
390 end.Y - start.Y,
391 end.Z - start.Z);
392
393 double length = direction.Length;
394 if (length < 1e-6)
395 return true;
396
397 // Check sample points along segment
398 for (int i = 1; i < SEGMENT_SAMPLE_COUNT; i++)
399 {
400 double t = (double)i / SEGMENT_SAMPLE_COUNT;
401 Point3D samplePoint = new Point3D(
402 start.X + direction.X * t,
403 start.Y + direction.Y * t,
404 start.Z + direction.Z * t);
405
406 if (IsPointInsideObstacle(samplePoint))
407 return false;
408 }
409
410 // Check for triangle intersections
411 foreach (var brep in _obstacles)
412 {
413 if (brep == null) continue;
414
415 try
416 {
417 // Quick bounding box check
418 if (!SegmentIntersectsBoundingBox(start, end, brep.BoxMin, brep.BoxMax, _safetyMargin))
419 continue;
420
421 // Detailed triangle intersection check
422 foreach (var face in brep.Faces)
423 {
424 if (face == null) continue;
425
426 try
427 {
428 var mesh = face.ConvertToMesh();
429 if (mesh?.Triangles == null || mesh?.Vertices == null)
430 continue;
431
432 foreach (var tri in mesh.Triangles)
433 {
434 if (tri.V1 >= mesh.Vertices.Length ||
435 tri.V2 >= mesh.Vertices.Length ||
436 tri.V3 >= mesh.Vertices.Length)
437 continue;
438
439 if (SegmentIntersectsTriangle(start, end,
440 mesh.Vertices[tri.V1],
441 mesh.Vertices[tri.V2],
442 mesh.Vertices[tri.V3]))
443 {
444 return false;
445 }
446 }
447 }
448 catch { }
449 }
450 }
451 catch { }
452 }
453
454 return true;
455 }

460 private bool IsPointInsideObstacle(Point3D point)
461 {
462 foreach (var brep in _obstacles)
463 {
464 if (brep == null) continue;
465
466 try
467 {
468 // Quick bounding box check with margin
469 var min = brep.BoxMin;
470 var max = brep.BoxMax;
471
472 if (point.X < min.X - _safetyMargin || point.X > max.X + _safetyMargin ||
473 point.Y < min.Y - _safetyMargin || point.Y > max.Y + _safetyMargin ||
474 point.Z < min.Z - _safetyMargin || point.Z > max.Z + _safetyMargin)
475 continue;
476
477 // Ray casting for inside/outside test
478 int intersectionCount = 0;
479 Vector3D rayDir = Vector3D.AxisZ;
480
481 foreach (var face in brep.Faces)
482 {
483 if (face == null) continue;
484
485 try
486 {
487 var mesh = face.ConvertToMesh();
488 if (mesh?.Triangles == null || mesh?.Vertices == null)
489 continue;
490
491 foreach (var tri in mesh.Triangles)
492 {
493 if (tri.V1 >= mesh.Vertices.Length ||
494 tri.V2 >= mesh.Vertices.Length ||
495 tri.V3 >= mesh.Vertices.Length)
496 continue;
497
498 if (RayIntersectsTriangle(point, rayDir,
499 mesh.Vertices[tri.V1],
500 mesh.Vertices[tri.V2],
501 mesh.Vertices[tri.V3]))
502 {
503 intersectionCount++;
504 }
505 }
506 }
507 catch { }
508 }
509
510 // Odd count means inside
511 if (intersectionCount % 2 == 1)
512 return true;
513 }
514 catch { }
515 }
516
517 return false;
518 }

523 private GraphNode? GetOrCreateTemporaryNode(Point3D position)
524 {
525 // Check if position matches existing node
526 foreach (var node in _nodes)
527 {
528 if (node.Position.DistanceTo(position) < 1.0)
529 return node;
530 }
531
532 // Create temporary node and connect to visible nodes
533 var tempNode = new GraphNode(position, _nodes.Count);
534 var visibleNodes = new List<(GraphNode node, double dist)>();
535
536 foreach (var node in _nodes)
537 {
538 double dist = position.DistanceTo(node.Position);
539 if (dist < MAX_EDGE_DISTANCE && IsSegmentVisible(position, node.Position))
540 {
541 visibleNodes.Add((node, dist));
542 }
543 }
544
545 if (visibleNodes.Count == 0)
546 {
547 Debug.WriteLine($" ⚠️ Could not connect point ({position.X:F1}, {position.Y:F1}, {position.Z:F1}) to graph");
548 return null;
549 }
550
551 // Connect to nearest visible nodes
552 var nearest = visibleNodes.OrderBy(v => v.dist).Take(MAX_EDGES_PER_NODE).ToList();
553 foreach (var (node, dist) in nearest)
554 {
555 tempNode.Edges.Add(new GraphEdge(node, dist));
556 node.Edges.Add(new GraphEdge(tempNode, dist));
557 }
558
559 _nodes.Add(tempNode);
560 return tempNode;
561 }

566 private List<Point3D> ReconstructPath(GraphNode endNode, Point3D start, Point3D goal)
567 {
568 var path = new List<Point3D>();
569 var current = endNode;
570
571 while (current != null)
572 {
573 path.Add(current.Position);
574 current = current.Parent;
575 }
576
577 path.Reverse();
578
579 // Ensure start and goal are exact
580 if (path.Count > 0 && path[0].DistanceTo(start) > 0.1)
581 path.Insert(0, start);
582 if (path.Count > 0 && path[path.Count - 1].DistanceTo(goal) > 0.1)
583 path.Add(goal);
584
585 return path;
586 }

614 private bool SegmentIntersectsBoundingBox(Point3D start, Point3D end, Point3D boxMin, Point3D boxMax, double margin)
615 {
616 double segMinX = Math.Min(start.X, end.X) - margin;
617 double segMaxX = Math.Max(start.X, end.X) + margin;
618 double segMinY = Math.Min(start.Y, end.Y) - margin;
619 double segMaxY = Math.Max(start.Y, end.Y) + margin;
620 double segMinZ = Math.Min(start.Z, end.Z) - margin;
621 double segMaxZ = Math.Max(start.Z, end.Z) + margin;
622
623 return !(segMaxX < boxMin.X || segMinX > boxMax.X ||
624 segMaxY < boxMin.Y || segMinY > boxMax.Y ||
625 segMaxZ < boxMin.Z || segMinZ > boxMax.Z);
626 }

631 private bool RayIntersectsTriangle(Point3D origin, Vector3D direction, Point3D v0, Point3D v1, Point3D v2)
632 {
633 const double EPSILON = 1e-7;
634
635 Vector3D edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
636 Vector3D edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
637
638 Vector3D h = Vector3D.Cross(direction, edge2);
639 double a = Vector3D.Dot(edge1, h);
640
641 if (a > -EPSILON && a < EPSILON)
642 return false;
643
644 double f = 1.0 / a;
645 Vector3D s = new Vector3D(origin.X - v0.X, origin.Y - v0.Y, origin.Z - v0.Z);
646 double u = f * Vector3D.Dot(s, h);
647
648 if (u < 0.0 || u > 1.0)
649 return false;
650
651 Vector3D q = Vector3D.Cross(s, edge1);
652 double v = f * Vector3D.Dot(direction, q);
653
654 if (v < 0.0 || u + v > 1.0)
655 return false;
656
657 double t = f * Vector3D.Dot(edge2, q);
658 return t > EPSILON;
659 }

664 private bool SegmentIntersectsTriangle(Point3D start, Point3D end, Point3D v0, Point3D v1, Point3D v2)
665 {
666 Vector3D direction = new Vector3D(
667 end.X - start.X,
668 end.Y - start.Y,
669 end.Z - start.Z);
670
671 const double EPSILON = 1e-7;
672
673 Vector3D edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
674 Vector3D edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
675
676 Vector3D h = Vector3D.Cross(direction, edge2);
677 double a = Vector3D.Dot(edge1, h);
678
679 if (a > -EPSILON && a < EPSILON)
680 return false;
681
682 double f = 1.0 / a;
683 Vector3D s = new Vector3D(start.X - v0.X, start.Y - v0.Y, start.Z - v0.Z);
684 double u = f * Vector3D.Dot(s, h);
685
686 if (u < 0.0 || u > 1.0)
687 return false;
688
689 Vector3D q = Vector3D.Cross(s, edge1);
690 double v = f * Vector3D.Dot(direction, q);
691
692 if (v < 0.0 || u + v > 1.0)
693 return false;
694
695 double t = f * Vector3D.Dot(edge2, q);
696
697 // t must be in [0, 1] for segment intersection
698 return t >= -EPSILON && t <= 1.0 + EPSILON;
699 }
