344 public double GetDistance(double px, double py, double pz)
345 {
346 if (Sdf == null || GridOrigin == null) return double.MaxValue;
347
348 double gx = (px - GridOrigin.X) / VoxelSize - 0.5;
349 double gy = (py - GridOrigin.Y) / VoxelSize - 0.5;
350 double gz = (pz - GridOrigin.Z) / VoxelSize - 0.5;
351
352 int ix = (int)Math.Floor(gx);
353 int iy = (int)Math.Floor(gy);
354 int iz = (int)Math.Floor(gz);
355
356 if (ix < 0 || ix >= Nx - 1 || iy < 0 || iy >= Ny - 1 || iz < 0 || iz >= Nz - 1)
357 return double.MaxValue;
358
359 double tx = gx - ix;
360 double ty = gy - iy;
361 double tz = gz - iz;
362
363 int nyPitch = Nx;
364 int nzPitch = Nx * Ny;
365 int base00 = ix + nyPitch * iy + nzPitch * iz;
366 int base01 = base00 + nzPitch;
367
368 float c000 = Sdf[base00];
369 float c100 = Sdf[base00 + 1];
370 float c010 = Sdf[base00 + nyPitch];
371 float c110 = Sdf[base00 + 1 + nyPitch];
372 float c001 = Sdf[base01];
373 float c101 = Sdf[base01 + 1];
374 float c011 = Sdf[base01 + nyPitch];
375 float c111 = Sdf[base01 + 1 + nyPitch];
376
377 double c00 = c000 + (c100 - c000) * tx;
378 double c10 = c010 + (c110 - c010) * tx;
379 double c01 = c001 + (c101 - c001) * tx;
380 double c11 = c011 + (c111 - c011) * tx;
381
382 double c0 = c00 + (c10 - c00) * ty;
383 double c1 = c01 + (c11 - c01) * ty;
384
385 return c0 + (c1 - c0) * tz;
386 }

504 private static string CompressFloatArray(float[] values)
505 {
506 var bytes = new byte[values.Length * sizeof(float)];
507 Buffer.BlockCopy(values, 0, bytes, 0, bytes.Length);
508
509 using var output = new MemoryStream();
510 using (var gzip = new GZipStream(output, CompressionLevel.SmallestSize, leaveOpen: true))
511 {
512 gzip.Write(bytes, 0, bytes.Length);
513 }
514
515 return Convert.ToBase64String(output.ToArray());
516 }

526 private static bool TryDecompressFloatArray(string? base64, int expectedCount, out float[] values, out string reason)
527 {
528 values = Array.Empty<float>();
529 reason = string.Empty;
530
531 if (string.IsNullOrWhiteSpace(base64))
532 {
533 reason = "compressed SDF data is missing";
534 return false;
535 }
536
537 try
538 {
539 var compressed = Convert.FromBase64String(base64);
540 using var input = new MemoryStream(compressed);
541 using var gzip = new GZipStream(input, CompressionMode.Decompress);
542 using var output = new MemoryStream();
543 gzip.CopyTo(output);
544
545 var bytes = output.ToArray();
546 int expectedBytes = checked(expectedCount * sizeof(float));
547 if (bytes.Length != expectedBytes)
548 {
549 reason = "decompressed SDF byte count mismatch";
550 return false;
551 }
552
553 values = new float[expectedCount];
554 Buffer.BlockCopy(bytes, 0, values, 0, bytes.Length);
555
556 if (values.Any(float.IsNaN) || values.Any(float.IsInfinity))
557 {
558 reason = "SDF data contains invalid floating point values";
559 return false;
560 }
561
562 return true;
563 }
564 catch (Exception ex)
565 {
566 reason = $"compressed SDF data is corrupt: {ex.Message}";
567 values = Array.Empty<float>();
568 return false;
569 }

753 public async Task<Mesh?> BuildAsync(
754 List<Brep> partBreps,
755 HashSet<int>? holeFaceIndices = null,
756 HashSet<int>? flippedFaceNormals = null,
757 double voxelSize = 0.75,
758 Action<int, string>? progressCallback = null,
759 System.Threading.CancellationToken cancellationToken = default)
760 {
761 var dispatcher = System.Windows.Application.Current?.Dispatcher;
762 ResultMesh = null; _triangles.Clear();
763 if (partBreps == null || partBreps.Count == 0) return null;
764
765 holeFaceIndices ??= new HashSet<int>(); _breps = partBreps;
766 progressCallback?.Invoke(5, "InitializingFaceData");
767
768 int totalFaces = partBreps.Sum(b => b.Faces?.Length ?? 0);
769 _faceMeshCache = new Mesh?[totalFaces];
770
771 if (dispatcher == null || dispatcher.CheckAccess())
772 {
773 int cacheIdx = 0;
774 foreach (var brep in partBreps)
775 {
776 if (brep.Faces == null) continue;
777 for (int i = 0; i < brep.Faces.Length; i++)
778 {
779 cancellationToken.ThrowIfCancellationRequested();
780 try { _faceMeshCache[cacheIdx] = brep.Faces[i].Tessellation?.ConvertToMesh() ?? brep.Faces[i].ConvertToMesh(); } catch { }
781 cacheIdx++;
782 if (cacheIdx % 50 == 0) progressCallback?.Invoke(5 + (int)(5.0 * cacheIdx / Math.Max(1, totalFaces)), "InitializingFaceData");
783 }
784 }
785 }
786 else
787 {
788 int batchSize = 100;
789 int cacheIdx = 0;
790 for (int partIdx = 0; partIdx < partBreps.Count; partIdx++)
791 {
792 var brep = partBreps[partIdx];
793 if (brep.Faces == null) continue;
794 int faceCount = brep.Faces.Length;
795 for (int startIdx = 0; startIdx < faceCount; startIdx += batchSize)
796 {
797 cancellationToken.ThrowIfCancellationRequested();
798 int curStart = startIdx;
799 int curCount = Math.Min(batchSize, faceCount - curStart);
800 int curCacheStart = cacheIdx;
801 cacheIdx += curCount;
802
803 await dispatcher.InvokeAsync(() =>
804 {
805 for (int j = 0; j < curCount; j++)
806 {
807 try { _faceMeshCache[curCacheStart + j] = brep.Faces[curStart + j].Tessellation?.ConvertToMesh() ?? brep.Faces[curStart + j].ConvertToMesh(); } catch { }
808 }
809 }, System.Windows.Threading.DispatcherPriority.Normal, cancellationToken);
810
811 if (cacheIdx % 50 == 0 || cacheIdx == totalFaces)
812 progressCallback?.Invoke(5 + (int)(5.0 * cacheIdx / Math.Max(1, totalFaces)), "InitializingFaceData");
813 }
814 }
815 }
816
817 return await Task.Run(() =>
818 {
819 BuildFaceAdjacencyMap(partBreps);
820 _contexts.Clear();
821 var partMeshes = new List<Mesh>();
822 ResultMeshes.Clear();
823 int faceOffset = 0;
824
825 for (int partIdx = 0; partIdx < partBreps.Count; partIdx++)
826 {
827 cancellationToken.ThrowIfCancellationRequested();
828 var brep = partBreps[partIdx];
829
830 double partBaseProgress = 10.0 + (partIdx * (80.0 / partBreps.Count));
831 double partProgressRange = 80.0 / partBreps.Count;
832
833 if (partBreps.Count == 1)
834 progressCallback?.Invoke((int)partBaseProgress, "ProcessingSinglePart");
835 else
836 progressCallback?.Invoke((int)partBaseProgress, $"ProcessingPart|{partIdx + 1}|{partBreps.Count}|");
837
838 _triangles.Clear();
839
840 // STEP 1 — tessellate this part into triangle soup
841 progressCallback?.Invoke((int)(partBaseProgress + partProgressRange * 0.05), "TessellatingBREP");
842 TessellateBRep(new List<Brep> { brep }, holeFaceIndices, flippedFaceNormals, faceOffset, cancellationToken);
843 faceOffset += brep.Faces?.Length ?? 0;
844 if (_triangles.Count == 0) continue;
845
846 var ctx = new SdfContext();
847
848 // STEP 2 — build voxel grid for this part
849 progressCallback?.Invoke((int)(partBaseProgress + partProgressRange * 0.15), "BuildingGrid");
850 BuildGrid(voxelSize, ctx, cancellationToken);
851
852 // STEP 3 — compute signed distance field
853 ComputeSDF(ctx, partBaseProgress + partProgressRange * 0.20, partProgressRange * 0.55, progressCallback, cancellationToken);
854 _contexts.Add(ctx);
855
856 // STEP 4 — EXTENDED MARCHING CUBES (EMC)
857 // Returns the generated mesh and the HashSet of protected sharp vertices
858 progressCallback?.Invoke((int)(partBaseProgress + partProgressRange * 0.80), "RunningMarchingCubes");
859 Mesh? mc = RunMarchingCubesEMC((float)OffsetDistance, ctx, out var protectedFeatures, cancellationToken);
860
861 if (mc != null && mc.Vertices != null && mc.Vertices.Length > 0)
862 {
863 // STEP 5 — smooth & fix normals
864 progressCallback?.Invoke((int)(partBaseProgress + partProgressRange * 0.90), "SmoothingShellMesh");
865 // Laplacian Smoothing avoids destroying the EMC sharp corners
866 int passes = 20;
867 if (int.TryParse(System.Configuration.ConfigurationManager.AppSettings["SdfShell_LaplacianPasses"], out int parsedPasses)) passes = parsedPasses;
868
869 float factor = 0.6f;
870 if (float.TryParse(System.Configuration.ConfigurationManager.AppSettings["SdfShell_LaplacianSmoothFactor"], System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out float parsedFactor)) factor = parsedFactor;
871
872
873 LaplacianSmooth(mc, passes, factor, ctx, (float)OffsetDistance, protectedFeatures, cancellationToken);

1123 private static void ClosestPointOnTriangle_Scalar(
1124 double px, double py, double pz,
1125 Point3D a, Point3D b, Point3D c,
1126 out double cpx, out double cpy, out double cpz,
1127 out double distSq)
1128 {
1129 double ax = a.X, ay = a.Y, az = a.Z;
1130 double bx = b.X, by = b.Y, bz = b.Z;
1131 double cx = c.X, cy = c.Y, cz = c.Z;
1132
1133 double abX = bx - ax, abY = by - ay, abZ = bz - az;
1134 double acX = cx - ax, acY = cy - ay, acZ = cz - az;
1135 double apX = px - ax, apY = py - ay, apZ = pz - az;
1136
1137 double d1 = abX * apX + abY * apY + abZ * apZ;
1138 double d2 = acX * apX + acY * apY + acZ * apZ;
1139 if (d1 <= 0 && d2 <= 0)
1140 {
1141 cpx = ax; cpy = ay; cpz = az;
1142 double dx = px - cpx, dy = py - cpy, dz = pz - cpz;
1143 distSq = dx * dx + dy * dy + dz * dz;
1144 return;
1145 }
1146
1147 double bpX = px - bx, bpY = py - by, bpZ = pz - bz;
1148 double d3 = abX * bpX + abY * bpY + abZ * bpZ;
1149 double d4 = acX * bpX + acY * bpY + acZ * bpZ;
1150 if (d3 >= 0 && d4 <= d3)
1151 {
1152 cpx = bx; cpy = by; cpz = bz;
1153 double dx = px - cpx, dy = py - cpy, dz = pz - cpz;
1154 distSq = dx * dx + dy * dy + dz * dz;
1155 return;
1156 }
1157
1158 double vc = d1 * d4 - d3 * d2;
1159 if (vc <= 0 && d1 >= 0 && d3 <= 0)
1160 {
1161 double v = d1 / (d1 - d3);
1162 cpx = ax + v * abX; cpy = ay + v * abY; cpz = az + v * abZ;
1163 double dx = px - cpx, dy = py - cpy, dz = pz - cpz;
1164 distSq = dx * dx + dy * dy + dz * dz;
1165 return;
1166 }
1167
1168 double cpX = px - cx, cpY = py - cy, cpZ = pz - cz;
1169 double d5 = abX * cpX + abY * cpY + abZ * cpZ;
1170 double d6 = acX * cpX + acY * cpY + acZ * cpZ;
1171 if (d6 >= 0 && d5 <= d6)
1172 {
1173 cpx = cx; cpy = cy; cpz = cz;
1174 double dx = px - cpx, dy = py - cpy, dz = pz - cpz;
1175 distSq = dx * dx + dy * dy + dz * dz;
1176 return;
1177 }
1178
1179 double vb = d5 * d2 - d1 * d6;
1180 if (vb <= 0 && d2 >= 0 && d6 <= 0)
1181 {
1182 double w = d2 / (d2 - d6);
1183 cpx = ax + w * acX; cpy = ay + w * acY; cpz = az + w * acZ;
1184 double dx = px - cpx, dy = py - cpy, dz = pz - cpz;
1185 distSq = dx * dx + dy * dy + dz * dz;
1186 return;
1187 }
1188
1189 double va = d3 * d6 - d5 * d4;
1190 if (va <= 0 && (d4 - d3) >= 0 && (d5 - d6) >= 0)
1191 {
1192 double w = (d4 - d3) / ((d4 - d3) + (d5 - d6));
1193 cpx = bx + w * (cx - bx);
1194 cpy = by + w * (cy - by);
1195 cpz = bz + w * (cz - bz);
1196 double dx = px - cpx, dy = py - cpy, dz = pz - cpz;
1197 distSq = dx * dx + dy * dy + dz * dz;
1198 return;
1199 }
1200
1201 double denom = 1.0 / (va + vb + vc);
1202 double v2 = vb * denom;
1203 double w2 = vc * denom;
1204 cpx = ax + v2 * abX + w2 * acX;
1205 cpy = ay + v2 * abY + w2 * acY;
1206 cpz = az + v2 * abZ + w2 * acZ;
1207 double ddx = px - cpx, ddy = py - cpy, ddz = pz - cpz;
1208 distSq = ddx * ddx + ddy * ddy + ddz * ddz;
1209 }

1217 private static bool RayTriangleIntersection(Point3D origin, Vector3D dir, Point3D v0, Point3D v1, Point3D v2, out double t)
1218 {
1219 t = 0;
1220 const double EPSILON = 1e-6;
1221
1222 double e1x = v1.X - v0.X, e1y = v1.Y - v0.Y, e1z = v1.Z - v0.Z;
1223 double e2x = v2.X - v0.X, e2y = v2.Y - v0.Y, e2z = v2.Z - v0.Z;
1224
1225 double hx = dir.Y * e2z - dir.Z * e2y;
1226 double hy = dir.Z * e2x - dir.X * e2z;
1227 double hz = dir.X * e2y - dir.Y * e2x;
1228
1229 double a = e1x * hx + e1y * hy + e1z * hz;
1230 if (a > -EPSILON && a < EPSILON) return false;
1231
1232 double f = 1.0 / a;
1233 double sx = origin.X - v0.X, sy = origin.Y - v0.Y, sz = origin.Z - v0.Z;
1234 double u = f * (sx * hx + sy * hy + sz * hz);
1235 if (u < 0.0 || u > 1.0) return false;
1236
1237 double qx = sy * e1z - sz * e1y;
1238 double qy = sz * e1x - sx * e1z;
1239 double qz = sx * e1y - sy * e1x;
1240
1241 double v = f * (dir.X * qx + dir.Y * qy + dir.Z * qz);
1242 if (v < 0.0 || u + v > 1.0) return false;
1243
1244 t = f * (e2x * qx + e2y * qy + e2z * qz);
1245 return t > EPSILON;
1246 }

1642 private static void DetermineVoxelSigns(double px, double py, double pz, TriangleBvh bvh, out float signA, out float signB)
1643 {
1644 // 1. Outside part bounding box: guaranteed outside both solids
1645 if (px < bvh.BoundsMinX || px > bvh.BoundsMaxX ||
1646 py < bvh.BoundsMinY || py > bvh.BoundsMaxY ||
1647 pz < bvh.BoundsMinZ || pz > bvh.BoundsMaxZ)
1648 {
1649 signA = 1.0f;
1650 signB = 1.0f;
1651 return;
1652 }
1653
1654 // 2. Fast path: Cast +X ray (1, 0, 0)
1655 bvh.CastRay(px, py, pz, 1.0, 0.0, 0.0, out int hitsA1, out int hitsB1, out bool edgeHitA1, out bool edgeHitB1);
1656 if (!edgeHitA1 && !edgeHitB1)
1657 {
1658 signA = (hitsA1 % 2 == 1) ? -1.0f : 1.0f;
1659 signB = (hitsB1 % 2 == 1) ? -1.0f : 1.0f;
1660 return;
1661 }
1662
1663 // 3. Fallback: Edge hit detected on +X ray, perform 6-ray parity voting
1664 int validVotesA = 0, votesInsideA = 0;
1665 int validVotesB = 0, votesInsideB = 0;
1666
1667 if (!edgeHitA1) { validVotesA++; if (hitsA1 % 2 == 1) votesInsideA++; }
1668 if (!edgeHitB1) { validVotesB++; if (hitsB1 % 2 == 1) votesInsideB++; }
1669
1670 // -X
1671 bvh.CastRay(px, py, pz, -1.0, 0.0, 0.0, out int hitsA, out int hitsB, out bool edgeA, out bool edgeB);
1672 if (!edgeA) { validVotesA++; if (hitsA % 2 == 1) votesInsideA++; }
1673 if (!edgeB) { validVotesB++; if (hitsB % 2 == 1) votesInsideB++; }
1674
1675 // +Y
1676 bvh.CastRay(px, py, pz, 0.0, 1.0, 0.0, out hitsA, out hitsB, out edgeA, out edgeB);
1677 if (!edgeA) { validVotesA++; if (hitsA % 2 == 1) votesInsideA++; }
1678 if (!edgeB) { validVotesB++; if (hitsB % 2 == 1) votesInsideB++; }
1679
1680 // -Y
1681 bvh.CastRay(px, py, pz, 0.0, -1.0, 0.0, out hitsA, out hitsB, out edgeA, out edgeB);
1682 if (!edgeA) { validVotesA++; if (hitsA % 2 == 1) votesInsideA++; }
1683 if (!edgeB) { validVotesB++; if (hitsB % 2 == 1) votesInsideB++; }
1684
1685 // +Z
1686 bvh.CastRay(px, py, pz, 0.0, 0.0, 1.0, out hitsA, out hitsB, out edgeA, out edgeB);
1687 if (!edgeA) { validVotesA++; if (hitsA % 2 == 1) votesInsideA++; }
1688 if (!edgeB) { validVotesB++; if (hitsB % 2 == 1) votesInsideB++; }
1689
1690 // -Z
1691 bvh.CastRay(px, py, pz, 0.0, 0.0, -1.0, out hitsA, out hitsB, out edgeA, out edgeB);
1692 if (!edgeA) { validVotesA++; if (hitsA % 2 == 1) votesInsideA++; }
1693 if (!edgeB) { validVotesB++; if (hitsB % 2 == 1) votesInsideB++; }
1694
1695 if (validVotesA == 0 || validVotesB == 0)
1696 {
1697 double dx = 0.314159, dy = 0.271828, dz = 0.161803;
1698 double len = Math.Sqrt(dx * dx + dy * dy + dz * dz);
1699 bvh.CastRay(px, py, pz, dx / len, dy / len, dz / len, out int slantA, out int slantB, out _, out _);
1700
1701 signA = (validVotesA == 0)
1702 ? ((slantA % 2 == 1) ? -1.0f : 1.0f)
1703 : ((votesInsideA * 2 > validVotesA) ? -1.0f : 1.0f);
1704
1705 signB = (validVotesB == 0)
1706 ? ((slantB % 2 == 1) ? -1.0f : 1.0f)
1707 : ((votesInsideB * 2 > validVotesB) ? -1.0f : 1.0f);
1708 return;
1709 }
1710
1711 signA = (votesInsideA * 2 > validVotesA) ? -1.0f : 1.0f;
1712 signB = (votesInsideB * 2 > validVotesB) ? -1.0f : 1.0f;
1713 }

1912 private Point3D ComputeSharpFeature(List<Point3D> points, List<Vector3D> normals, Point3D centroid, double voxelSize)
1913 {
1914 double lambda = 0.005;
1915
1916 double a00 = lambda, a01 = 0, a02 = 0;
1917 double a10 = 0, a11 = lambda, a12 = 0;
1918 double a20 = 0, a21 = 0, a22 = lambda;
1919 double b0 = lambda * centroid.X, b1 = lambda * centroid.Y, b2 = lambda * centroid.Z;
1920
1921 for (int i = 0; i < points.Count; i++)
1922 {
1923 var n = normals[i];
1924 var p = points[i];
1925
1926 a00 += n.X * n.X; a01 += n.X * n.Y; a02 += n.X * n.Z;
1927 a10 += n.Y * n.X; a11 += n.Y * n.Y; a12 += n.Y * n.Z;
1928 a20 += n.Z * n.X; a21 += n.Z * n.Y; a22 += n.Z * n.Z;
1929
1930 double dot = n.X * p.X + n.Y * p.Y + n.Z * p.Z;
1931 b0 += dot * n.X;
1932 b1 += dot * n.Y;
1933 b2 += dot * n.Z;
1934 }
1935
1936 double det = a00 * (a11 * a22 - a12 * a21) - a01 * (a10 * a22 - a12 * a20) + a02 * (a10 * a21 - a11 * a20);
1937
1938 if (Math.Abs(det) < 1e-9) return centroid;
1939
1940 double invDet = 1.0 / det;
1941 double i00 = (a11 * a22 - a12 * a21) * invDet;
1942 double i01 = -(a01 * a22 - a02 * a21) * invDet;
1943 double i02 = (a01 * a12 - a02 * a11) * invDet;
1944
1945 double i10 = -(a10 * a22 - a12 * a20) * invDet;
1946 double i11 = (a00 * a22 - a02 * a20) * invDet;
1947 double i12 = -(a00 * a12 - a02 * a10) * invDet;
1948
1949 double i20 = (a10 * a21 - a11 * a20) * invDet;
1950 double i21 = -(a00 * a21 - a01 * a20) * invDet;
1951 double i22 = (a00 * a11 - a01 * a10) * invDet;
1952
1953 return new Point3D(
1954 i00 * b0 + i01 * b1 + i02 * b2,
1955 i10 * b0 + i11 * b1 + i12 * b2,
1956 i20 * b0 + i21 * b1 + i22 * b2
1957 );
1958 }
