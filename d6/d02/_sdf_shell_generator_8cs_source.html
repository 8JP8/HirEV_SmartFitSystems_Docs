245 public double GetDistance(Point3D p)
246 {
247 if (Sdf == null || GridOrigin == null) return double.MaxValue;
248
249 double gx = (p.X - GridOrigin.X) / VoxelSize - 0.5;
250 double gy = (p.Y - GridOrigin.Y) / VoxelSize - 0.5;
251 double gz = (p.Z - GridOrigin.Z) / VoxelSize - 0.5;
252
253 int ix = (int)Math.Floor(gx);
254 int iy = (int)Math.Floor(gy);
255 int iz = (int)Math.Floor(gz);
256
257 if (ix < 0 || ix >= Nx - 1 || iy < 0 || iy >= Ny - 1 || iz < 0 || iz >= Nz - 1)
258 return double.MaxValue;
259
260 double tx = gx - ix;
261 double ty = gy - iy;
262 double tz = gz - iz;
263
264 float c000 = Sdf[ix + Nx * (iy + Ny * iz)];
265 float c100 = Sdf[ix + 1 + Nx * (iy + Ny * iz)];
266 float c010 = Sdf[ix + Nx * (iy + 1 + Ny * iz)];
267 float c110 = Sdf[ix + 1 + Nx * (iy + 1 + Ny * iz)];
268 float c001 = Sdf[ix + Nx * (iy + Ny * (iz + 1))];
269 float c101 = Sdf[ix + 1 + Nx * (iy + Ny * (iz + 1))];
270 float c011 = Sdf[ix + Nx * (iy + 1 + Ny * (iz + 1))];
271 float c111 = Sdf[ix + 1 + Nx * (iy + 1 + Ny * (iz + 1))];
272
273 double c00 = c000 * (1 - tx) + c100 * tx;
274 double c10 = c010 * (1 - tx) + c110 * tx;
275 double c01 = c001 * (1 - tx) + c101 * tx;
276 double c11 = c011 * (1 - tx) + c111 * tx;
277
278 double c0 = c00 * (1 - ty) + c10 * ty;
279 double c1 = c01 * (1 - ty) + c11 * ty;
280
281 return c0 * (1 - tz) + c1 * tz;
282 }

408 private static bool TryDecompressFloatArray(string? base64, int expectedCount, out float[] values, out string reason)
409 {
410 values = Array.Empty<float>();
411 reason = string.Empty;
412
413 if (string.IsNullOrWhiteSpace(base64))
414 {
415 reason = "compressed SDF data is missing";
416 return false;
417 }
418
419 try
420 {
421 var compressed = Convert.FromBase64String(base64);
422 using var input = new MemoryStream(compressed);
423 using var gzip = new GZipStream(input, CompressionMode.Decompress);
424 using var output = new MemoryStream();
425 gzip.CopyTo(output);
426
427 var bytes = output.ToArray();
428 int expectedBytes = checked(expectedCount * sizeof(float));
429 if (bytes.Length != expectedBytes)

434
435 values = new float[expectedCount];
436 Buffer.BlockCopy(bytes, 0, values, 0, bytes.Length);
437
438 if (values.Any(float.IsNaN) || values.Any(float.IsInfinity))
439 {
440 reason = "SDF data contains invalid floating point values";
441 return false;
442 }
443
444 return true;

659 public async Task<Mesh?> BuildAsync(
660 List<Brep> partBreps,
661 HashSet<int>? holeFaceIndices = null,
662 HashSet<int>? flippedFaceNormals = null,
663 double voxelSize = 1.5,
664 Action<int, string>? progressCallback = null,
665 System.Threading.CancellationToken cancellationToken = default)
666 {
667 // Capture UI context safely if available
668 var dispatcher = System.Windows.Application.Current?.Dispatcher;
669
670 ResultMesh = null;
671 _triangles.Clear();
672
673 if (partBreps == null || partBreps.Count == 0)
674 {
675 Debug.WriteLine("[SdfShell] No BReps provided.");
676 return null;
677 }
678
679 holeFaceIndices ??= new HashSet<int>();
680 _breps = partBreps;
681
682 progressCallback?.Invoke(5, "InitializingFaceData");
683
684 // Extract meshes safely - only jump to UI thread for the conversion itself
685 int totalFaces = partBreps.Sum(b => b.Faces?.Length ?? 0);
686 _faceMeshCache = new Mesh?[totalFaces];
687
688 int cacheIdx = 0;
689 foreach (var brep in partBreps)
690 {
691 cancellationToken.ThrowIfCancellationRequested();
692 if (brep.Faces == null) continue;
693 for (int i = 0; i < brep.Faces.Length; i++)
694 {
695 cancellationToken.ThrowIfCancellationRequested();
696 int localIdx = i;
697 int currentCacheIdx = cacheIdx++;
698 if (dispatcher != null)
699 {
700 await dispatcher.InvokeAsync(() =>
701 {
702 try
703 {
704 var face = brep.Faces[localIdx];
705 _faceMeshCache[currentCacheIdx] = face.Tessellation?.ConvertToMesh() ?? face.ConvertToMesh();
706 }
707 catch { }
708 }, System.Windows.Threading.DispatcherPriority.Background);
709 }
710
711 if (currentCacheIdx % 50 == 0) progressCallback?.Invoke(5 + (int)(5.0 * currentCacheIdx / totalFaces), "InitializingFaceData");
712 }
713 }
714
715 return await Task.Run(() =>
716 {
717 // Note: Adjacency map only handles the first part for now to ensure compatibility
718 BuildFaceAdjacencyMap(partBreps[0]);
719
720 _contexts.Clear();
721 var partMeshes = new List<Mesh>();
722 int faceOffset = 0;
723
724 for (int partIdx = 0; partIdx < partBreps.Count; partIdx++)
725 {
726 cancellationToken.ThrowIfCancellationRequested();
727 var brep = partBreps[partIdx];
728 int partNum = partIdx + 1;
729 int totalParts = partBreps.Count;
730 double partWeight = 80.0 / totalParts;
731 int baseProgress = 10 + (int)(partIdx * partWeight);
732
733 progressCallback?.Invoke(baseProgress, $"ProcessingPart|{partNum}|{totalParts}|");
734 System.Threading.Thread.Sleep(500);
735
736 _triangles.Clear();
737
738 // STEP 1 — tessellate this part into triangle soup
739 TessellateBRep(new List<Brep> { brep }, holeFaceIndices, flippedFaceNormals, faceOffset, cancellationToken);
740 faceOffset += brep.Faces?.Length ?? 0;
741
742 if (_triangles.Count == 0) continue;
743
744 var ctx = new SdfContext();
745
746 // STEP 2 — build voxel grid for this part
747 BuildGrid(voxelSize, ctx, cancellationToken);
748
749 // STEP 3 — compute signed distance field
750 ComputeSDF(ctx, progressCallback, cancellationToken);
751 _contexts.Add(ctx);
752
753 // STEP 4 — extract iso-surface at SDF == OffsetDistance
754 Mesh? mc = RunMarchingCubes((float)OffsetDistance, ctx, cancellationToken);
755
756 if (mc != null && mc.Vertices != null && mc.Vertices.Length > 0)
757 {
758 // STEP 5 — smooth & fix normals on the INDIVIDUAL mesh to avoid deforming disjoint parts
759 int passes = 120;
760 if (int.TryParse(System.Configuration.ConfigurationManager.AppSettings["SdfShell_LaplacianPasses"], out int parsedPasses)) passes = parsedPasses;
761
762 float factor = 0.6f;
763 if (float.TryParse(System.Configuration.ConfigurationManager.AppSettings["SdfShell_LaplacianSmoothFactor"], System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out float parsedFactor)) factor = parsedFactor;
764
765 // Use iterations with volume-preserving projection to avoid shrinking on cylinders
766 LaplacianSmooth(mc, passes, factor, ctx, (float)OffsetDistance, cancellationToken);
767 EnsureOutwardNormals(mc, ctx, cancellationToken);
768 mc.Regen(0.01);
769 partMeshes.Add(mc);
770 }
771 }
772
773 if (partMeshes.Count == 0)
774 {
775 Debug.WriteLine("[SdfShell] Marching Cubes produced no geometry.");
776 return null;
777 }
778
779 progressCallback?.Invoke(95, "MergingResults");
780 Mesh? combined = CombineMeshes(partMeshes);
781
782 ResultMesh = combined;
783 progressCallback?.Invoke(100, "ShellMeshComplete");
784 return combined;
785 });
786 }

999 private Point3D ClosestPointOnTriangle(Point3D p, Point3D a, Point3D b, Point3D c)
1000 {
1001 Vector3D ab = new Vector3D(a, b);
1002 Vector3D ac = new Vector3D(a, c);
1003 Vector3D ap = new Vector3D(a, p);
1004 double d1 = Vector3D.Dot(ab, ap);
1005 double d2 = Vector3D.Dot(ac, ap);
1006 if (d1 <= 0.0 && d2 <= 0.0) return a;
1007
1008 Vector3D bp = new Vector3D(b, p);
1009 double d3 = Vector3D.Dot(ab, bp);
1010 double d4 = Vector3D.Dot(ac, bp);
1011 if (d3 >= 0.0 && d4 <= d3) return b;
1012
1013 double vc = d1 * d4 - d3 * d2;
1014 if (vc <= 0.0 && d1 >= 0.0 && d3 <= 0.0)
1015 {
1016 double v = d1 / (d1 - d3);
1017 return a + v * ab;
1018 }
1019
1020 Vector3D cp = new Vector3D(c, p);
1021 double d5 = Vector3D.Dot(ab, cp);
1022 double d6 = Vector3D.Dot(ac, cp);
1023 if (d6 >= 0.0 && d5 <= d6) return c;
1024
1025 double vb = d5 * d2 - d1 * d6;
1026 if (vb <= 0.0 && d2 >= 0.0 && d6 <= 0.0)
1027 {
1028 double w = d2 / (d2 - d6);
1029 return a + w * ac;
1030 }
1031
1032 double va = d3 * d6 - d5 * d4;
1033 if (va <= 0.0 && (d4 - d3) >= 0.0 && (d5 - d6) >= 0.0)
1034 {
1035 double w = (d4 - d3) / ((d4 - d3) + (d5 - d6));
1036 return b + w * (new Vector3D(b, c));
1037 }
1038
1039 double denom = 1.0 / (va + vb + vc);
1040 double v2 = vb * denom;
1041 double w2 = vc * denom;
1042 return a + v2 * ab + w2 * ac;
1043 }

1055 private bool RayTriangleIntersection(Point3D origin, Vector3D dir, Point3D v0, Point3D v1, Point3D v2, out double t)
1056 {
1057 t = 0;
1058 const double EPSILON = 1e-6;
1059 Vector3D edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
1060 Vector3D edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
1061 Vector3D h = Vector3D.Cross(dir, edge2);
1062 double a = Vector3D.Dot(edge1, h);
1063
1064 if (a > -EPSILON && a < EPSILON) return false;
1065
1066 double f = 1.0 / a;
1067 Vector3D s = new Vector3D(origin.X - v0.X, origin.Y - v0.Y, origin.Z - v0.Z);
1068 double u = f * Vector3D.Dot(s, h);
1069
1070 if (u < 0.0 || u > 1.0) return false;
1071
1072 Vector3D q = Vector3D.Cross(s, edge1);
1073 double v = f * Vector3D.Dot(dir, q);
1074
1075 if (v < 0.0 || u + v > 1.0) return false;
1076
1077 t = f * Vector3D.Dot(edge2, q);
1078 return t > EPSILON;
1079 }

1556 private static void LaplacianSmooth(Mesh mesh, int iterations, float factor, SdfContext? ctx = null, float iso = 0f, System.Threading.CancellationToken cancellationToken = default)
1557 {
1558 if (mesh.Vertices == null || mesh.Triangles == null) return;
1559
1560 var adj = new List<int>[mesh.Vertices.Length];
1561 for (int i = 0; i < adj.Length; i++) adj[i] = new List<int>();
1562
1563 foreach (var tri in mesh.Triangles)
1564 {
1565 if (!adj[tri.V1].Contains(tri.V2)) adj[tri.V1].Add(tri.V2);
1566 if (!adj[tri.V1].Contains(tri.V3)) adj[tri.V1].Add(tri.V3);
1567 if (!adj[tri.V2].Contains(tri.V1)) adj[tri.V2].Add(tri.V1);
1568 if (!adj[tri.V2].Contains(tri.V3)) adj[tri.V2].Add(tri.V3);
1569 if (!adj[tri.V3].Contains(tri.V1)) adj[tri.V3].Add(tri.V1);
1570 if (!adj[tri.V3].Contains(tri.V2)) adj[tri.V3].Add(tri.V2);
1571 }
1572
1573 var v = mesh.Vertices;
1574 for (int iter = 0; iter < iterations; iter++)
1575 {
1576 cancellationToken.ThrowIfCancellationRequested();
1577 var nv = new Point3D[v.Length];
1578 for (int i = 0; i < v.Length; i++)
1579 {
1580 if (adj[i].Count == 0) { nv[i] = v[i]; continue; }
1581 double sx = 0, sy = 0, sz = 0;
1582 foreach (int n in adj[i]) { sx += v[n].X; sy += v[n].Y; sz += v[n].Z; }
1583 int cnt = adj[i].Count;
1584 Point3D smoothed = new Point3D(v[i].X + factor * (sx / cnt - v[i].X),
1585 v[i].Y + factor * (sy / cnt - v[i].Y),
1586 v[i].Z + factor * (sz / cnt - v[i].Z));
1587
1588 // Volume-preserving projection back to SDF iso-surface
1589 if (ctx != null)
1590 {
1591 double d = ctx.GetDistance(smoothed);
1592 if (d != double.MaxValue)
1593 {
1594 // Estimate gradient using central differences
1595 double eps = ctx.VoxelSize * 0.1;
1596 double dpx = ctx.GetDistance(new Point3D(smoothed.X + eps, smoothed.Y, smoothed.Z));
1597 double dmx = ctx.GetDistance(new Point3D(smoothed.X - eps, smoothed.Y, smoothed.Z));
1598 double dpy = ctx.GetDistance(new Point3D(smoothed.X, smoothed.Y + eps, smoothed.Z));
1599 double dmy = ctx.GetDistance(new Point3D(smoothed.X, smoothed.Y - eps, smoothed.Z));
1600 double dpz = ctx.GetDistance(new Point3D(smoothed.X, smoothed.Y, smoothed.Z + eps));
1601 double dmz = ctx.GetDistance(new Point3D(smoothed.X, smoothed.Y, smoothed.Z - eps));
1602
1603 if (dpx != double.MaxValue && dmx != double.MaxValue &&
1604 dpy != double.MaxValue && dmy != double.MaxValue &&
1605 dpz != double.MaxValue && dmz != double.MaxValue)
1606 {
1607 double dx = (dpx - dmx) / (2 * eps);
1608 double dy = (dpy - dmy) / (2 * eps);
1609 double dz = (dpz - dmz) / (2 * eps);
1610
1611 Vector3D grad = new Vector3D(dx, dy, dz);
1612 double len = grad.Length;
1613 if (len > 1e-6)
1614 {
1615 grad.Normalize();
1616 double diff = iso - d;
1617 smoothed = smoothed + grad * diff;
1618 }
1619 }
1620 }
1621 }
1622
1623 nv[i] = smoothed;
1624 }
1625 mesh.Vertices = nv;
1626 v = nv;
1627 }

1698 private static double PtTriDist(Point3D p, Triangle3D t)
1699 {
1700 Vector3D ab = new Vector3D(t.A, t.B), ac = new Vector3D(t.A, t.C), ap = new Vector3D(t.A, p);
1701 double d1 = Vector3D.Dot(ab, ap), d2 = Vector3D.Dot(ac, ap);
1702 if (d1 <= 0 && d2 <= 0) return p.DistanceTo(t.A);
1703 Vector3D bp = new Vector3D(t.B, p); double d3 = Vector3D.Dot(ab, bp), d4 = Vector3D.Dot(ac, bp);
1704 if (d3 >= 0 && d4 <= d3) return p.DistanceTo(t.B);
1705 double vc = d1 * d4 - d3 * d2;
1706 if (vc <= 0 && d1 >= 0 && d3 <= 0) return p.DistanceTo(t.A + (d1 / (d1 - d3)) * ab);
1707 Vector3D cp = new Vector3D(t.C, p); double d5 = Vector3D.Dot(ab, cp), d6 = Vector3D.Dot(ac, cp);
1708 if (d6 >= 0 && d5 <= d6) return p.DistanceTo(t.C);
1709 double vb = d5 * d2 - d1 * d6;
1710 if (vb <= 0 && d2 >= 0 && d6 <= 0) return p.DistanceTo(t.A + (d2 / (d2 - d6)) * ac);
1711 double va = d3 * d6 - d5 * d4;
1712 if (va <= 0 && (d4 - d3) >= 0 && (d5 - d6) >= 0) return p.DistanceTo(t.B + ((d4 - d3) / ((d4 - d3) + (d5 - d6))) * (new Vector3D(t.B, t.C)));
1713 double denom = 1.0 / (va + vb + vc);
1714 return p.DistanceTo(t.A + (vb * denom) * ab + (vc * denom) * ac);
