382 public List<ScanPath> GenerateSphericalScanPaths(
383 IEnumerable<CADFace> faces, Mesh partMesh, ScanPathOptions options)
384 {
385 options ??= BuildOptionsFromConfig();
386 var faceList = faces?.Where(f => !IsIgnoredFace(f)).ToList() ?? new List<CADFace>();
387 if (!faceList.Any())
388 return new List<ScanPath>();
389
390 bool hasMesh = partMesh?.Vertices != null && partMesh.Vertices.Length > 0;
391
392 Log("🚀 Spherical scan-path generation started");
393 Log($" {faceList.Count} faces | Fibonacci samples: {options.FibonacciSampleCount} | Mesh: {hasMesh}");
394 Log($" AvoidBottom: {options.AvoidBottomFace} ({options.BottomAvoidanceAngleDeg}°) | ScanBottom: {options.ScanBottomFace} | Holes: {options.ScanHoles && hasMesh}");
395 ReportProgress("Progress_ScanPathBoundingSphere", 5);
396
397 // ── 1. Bounding sphere ────────────────────────────────────────────────
398 // When a mesh is available use it for a tight sphere; otherwise fall
399 // back to the face control points so the mesh-less path still works.
400 if (hasMesh)
401 InitializeBoundingSphere(partMesh!);
402 else
403 InitializeBoundingSphereFromFaces(faceList);
404 double scannerSphereRadius = CalculateScannerSphereRadius(_sphereRadius, options);
405
406 Log($"📦 Bounding sphere C={Fmt(_sphereCenter)} r={_sphereRadius:F1} mm");
407 Log($"📡 Scanner sphere r={scannerSphereRadius:F1} mm");
408
409 // ── 2. Surface sample points (coverage targets) ───────────────────────
410 var surfaceSamples = ExtractSurfaceSamples(faceList);
411 if (!surfaceSamples.Any()) return new List<ScanPath>();
412 var surfacePoints = surfaceSamples.Select(s => s.Position).ToList();
413 Log($"📍 {surfaceSamples.Count} surface sample points");
414 ReportProgress("Progress_ScanPathSurfaceSamples", 15);
415
416 // ── 3. Candidate positions on Fibonacci sphere ────────────────────────
417 var candidates = GenerateFibonacciCandidates(
418 _sphereCenter, scannerSphereRadius, options);
419 Log($"SUCCESS: {candidates.Count} candidate positions after filtering");
420 ReportProgress("Progress_ScanPathCandidates", 25);
421
422 // ── 4. Greedy coverage selection ──────────────────────────────────────
423 var selected = SelectCoveragePositions(
424 candidates, surfaceSamples, partMesh, options);
425 Log($"SUCCESS: {selected.Count} positions selected by greedy set-cover");
426 ReportProgress("Progress_ScanPathCoverage", 50);
427
428 // ── 5. Hole axial scanning ────────────────────────────────────────────
429 if (options.ScanHoles && hasMesh)
430 {
431 var holes = DetectHoles(partMesh!);
432 Log($"🔩 {holes.Count} holes detected");
433 ReportProgress("Progress_ScanPathHoles", 60);
434 foreach (var hole in holes)
435 {
436 if (hole.Radius < options.MinHoleRadius) continue;
437 var holePoses = GenerateHoleScanPositions(hole, options);
438 selected.AddRange(holePoses);
439 Log($" Hole r={hole.Radius:F1} depth={hole.Depth:F1} → {holePoses.Count} positions");
440 }
441 }
442
443 if (!selected.Any()) return new List<ScanPath>();
444
445 // ── 6. TSP ordering on sphere ─────────────────────────────────────────
446 var ordered = OrderOnSphere(selected, _sphereCenter);
447 ReportProgress("Progress_ScanPathOrdering", 70);
448
449 // ── 7. Great-circle interpolation ─────────────────────────────────────
450 var sphere = new PartBoundingSphere(_sphereCenter, _sphereRadius, scannerSphereRadius, _partBottomZ, _partTopZ);
451 var finalPath = InterpolateGreatCirclePath(ordered, sphere, options);
452 ReportProgress("Progress_ScanPathInterpolating", 85);
453
454 Log($"🏁 Final path: {finalPath.Count} waypoints " +
455 $"({ordered.Count} measurement + {finalPath.Count - ordered.Count} arc)");
456
457 if (hasMesh && partMesh != null)
458 {
459 LogCoverage(ordered, surfacePoints, partMesh);
460 }
461
462 return finalPath.Any()
463 ? new List<ScanPath> { new ScanPath(finalPath.First().ParentFace!, finalPath) }
464 : new List<ScanPath>();
465 }

532 public List<ScanPath> GenerateMultiPartSphericalScanPaths(
533 IEnumerable<CADFace> allFaces,
534 Mesh? combinedMesh,
535 ScanPathOptions options,
536 IEnumerable<CADHole>? knownHoles = null)
537 {
538 var knownHoleList = knownHoles?.ToList() ?? new List<CADHole>();
539 options ??= BuildOptionsFromConfig(knownHoleList);
540 var faceList = allFaces?.Where(f => !IsIgnoredFace(f)).ToList() ?? new List<CADFace>();
541 if (!faceList.Any()) return new List<ScanPath>();
542
543 // ── Group faces by entity (= one CAD body / loaded file) ─────────────
544 var facesByEntity = faceList
545 .GroupBy(f => f.EntityIndex)
546 .OrderBy(g => g.Key)
547 .ToList();
548 ReportProgress("Progress_ScanPathPreparingParts", 0, facesByEntity.Count);
549
550 var partSpheresByEntity = facesByEntity.ToDictionary(
551 g => g.Key,
552 g => ComputePartSphere(g.ToList(), options));
553
554 // Build hole lookup: entityIndex -> holes on that body. FaceGlobalIndex can be
555 // missing or local/repeated in multi-part data, so fall back to the hole center
556 // against each part sphere instead of assigning every ambiguous hole to part 0.
557 var holesByEntity = BuildHolesByEntity(knownHoleList, faceList, partSpheresByEntity);
558
559 Log($"🏭 Multi-part scan: {facesByEntity.Count} part(s), knownHoles: {knownHoleList.Count}");
560
561 // ── Per-part planning ─────────────────────────────────────────────────
562 var partGroups = new List<PartScanGroup>();
563
564 foreach (var group in facesByEntity)
565 {
566 int entityIdx = group.Key;
567 var partFaces = group.ToList();
568 var partHoles = holesByEntity.TryGetValue(entityIdx, out var h) ? h : new List<CADHole>();
569 var sphere = partSpheresByEntity.TryGetValue(entityIdx, out var cachedSphere)
570 ? cachedSphere
571 : ComputePartSphere(partFaces, options);
572 int partNumber = partGroups.Count + 1;
573 double partBasePercent = facesByEntity.Count == 0
574 ? 0.0
575 : 5.0 + (partNumber - 1) * (75.0 / facesByEntity.Count);
576 ReportProgress("Progress_ScanPathPlanningPart", partBasePercent, partNumber, facesByEntity.Count);
577
578 Log($" Part {entityIdx}: {partFaces.Count} faces | sphere C={Fmt(sphere.Center)} R={sphere.ScannerRadius:F1}mm");
579
580 // Temporarily load instance bounding-sphere fields so helper methods work
581 _sphereCenter = sphere.Center;
582 _sphereRadius = sphere.PartRadius;
583 _partBottomZ = sphere.BottomZ;
584 _partTopZ = sphere.TopZ;
585
586
587 var exteriorPartFaces = partFaces.Where(f => f.IsExterior).ToList();
588 if (!exteriorPartFaces.Any())
589 exteriorPartFaces = partFaces;
590 var exteriorMesh = BuildExteriorMeshFromFaces(exteriorPartFaces);
591
592 // Surface sample points for this part
593 var surfSamples = ExtractSurfaceSamples(exteriorPartFaces);
594 var surfPts = surfSamples.Select(s => s.Position).ToList();
595 ReportProgress("Progress_ScanPathSurfaceSamples", partBasePercent + 10.0);
596
597 // Candidates on this part's Fibonacci sphere
598 var candidates = GenerateFibonacciCandidates(sphere.Center, sphere.ScannerRadius, options);
599 ReportProgress("Progress_ScanPathCandidates", partBasePercent + 20.0);
600
601 // Greedy coverage
602 var selected = SelectCoveragePositions(candidates, surfSamples, exteriorMesh, options);
603 ReportProgress("Progress_ScanPathCoverage", partBasePercent + 35.0);
604
605 // Inject known holes (using CADHole from the app's HoleDetector)
606 if (options.ScanHoles)
607 {
608 if (partHoles.Any())
609 {
610 foreach (var cadHole in partHoles)
611 {
612 double holeRadius = GetHoleRadius(cadHole);
613 if (holeRadius < options.MinHoleRadius)
614 {
615 Log($" Hole {cadHole.HoleId} skipped: radius {holeRadius:F2}mm below minimum {options.MinHoleRadius:F2}mm");
616 continue;
617 }
618
619 var det = CadHoleToDetectedHole(cadHole, sphere.Center);
620 var holePoses = GenerateHoleScanPositions(det, options);
621 CADFace? holeParentFace = partFaces.FirstOrDefault(f => cadHole.FaceGlobalIndices?.Contains(f.FaceGlobalIndex) == true)
622 ?? partFaces.FirstOrDefault();
623 if (holeParentFace != null)
624 holePoses = holePoses.Select(p => new ScanPoint(p.Position, p.Orientation, p.UpVector, holeParentFace, p.PointType)).ToList();
625 selected.AddRange(holePoses);
626 Log($" Hole {cadHole.HoleId} r={holeRadius:F1} → {holePoses.Count} poses");
627 }
628 }
629
630 // Fallback or supplementary: use the shell mesh of the current part
631 if (exteriorMesh != null)
632 {
633 var detectedHoles = DetectHoles(exteriorMesh);
634 if (detectedHoles.Any())
635 {
636 Log($" Detected {detectedHoles.Count} additional holes on part {entityIdx} shell mesh");
637 foreach (var hole in detectedHoles)
638 {
639 if (hole.Radius < options.MinHoleRadius) continue;
640 var holePoses = GenerateHoleScanPositions(hole, options);
641 CADFace? holeParentFace = partFaces.FirstOrDefault();
642 if (holeParentFace != null)
643 holePoses = holePoses.Select(p => new ScanPoint(p.Position, p.Orientation, p.UpVector, holeParentFace, p.PointType)).ToList();
644 selected.AddRange(holePoses);
645 Log($" Mesh Hole r={hole.Radius:F1} depth={hole.Depth:F1} → {holePoses.Count} positions");
646 }
647 }
648 }
649 }
650
651 if (!selected.Any()) continue;
652
653 // TSP ordering + SLERP interpolation for this part
654 var ordered = OrderOnSphere(selected, sphere.Center);
655 ReportProgress("Progress_ScanPathOrdering", partBasePercent + 50.0);
656 var interped = InterpolateGreatCirclePath(ordered, sphere, options);
657 ReportProgress("Progress_ScanPathInterpolating", partBasePercent + 65.0);
658
659 partGroups.Add(new PartScanGroup(entityIdx, sphere, interped, ordered));
660 }
661
662 if (!partGroups.Any()) return new List<ScanPath>();
663
664 // ── Single part: no bridging needed ───────────────────────────────────
665 if (partGroups.Count == 1)
666 {
667 var g = partGroups[0];
668 return g.Waypoints.Any()
669 ? new List<ScanPath> { new ScanPath(g.Waypoints.First().ParentFace!, g.Waypoints) }
670 : new List<ScanPath>();
671 }
672
673 // ── Multi-part: stitch groups with bridge arcs ───────────────────────
674 var fullPath = new List<ScanPoint>(partGroups[0].Waypoints);
675
676 for (int i = 0; i < partGroups.Count - 1; i++)
677 {
678 var partA = partGroups[i];
679 var partB = partGroups[i + 1];
680
681 var bridge = BuildPartBridge(
682 from: partA.Waypoints.Last(),
683 sphereA: partA.Sphere,
684 to: partB.Waypoints.First(),
685 sphereB: partB.Sphere,
686 options: options);
687 ReportProgress("Progress_ScanPathBridgingParts", 80.0 + (i * 15.0 / Math.Max(1, partGroups.Count - 1)), i + 1, i + 2);
688
689 fullPath.AddRange(bridge);
690 fullPath.AddRange(partB.Waypoints);
691
692 Log($"🌉 Bridge part{partA.EntityIndex}→part{partB.EntityIndex}: {bridge.Count} waypoints");
693 }
694
695 ReportProgress("Progress_ScanPathComplete", 100);
696 return new List<ScanPath> { new ScanPath(fullPath.First().ParentFace!, fullPath) };
697 }

703 private Dictionary<int, List<CADHole>> BuildHolesByEntity(
704 IReadOnlyCollection<CADHole> knownHoles,
705 List<CADFace> faceList,
706 Dictionary<int, PartBoundingSphere> partSpheresByEntity)
707 {
708 var holesByEntity = new Dictionary<int, List<CADHole>>();
709 if (knownHoles.Count == 0 || partSpheresByEntity.Count == 0)
710 return holesByEntity;
711
712 int mapped = 0;
713 int skipped = 0;
714
715 foreach (var hole in knownHoles)
716 {
717 int entityIndex = ResolveHoleEntityIndex(hole, faceList, partSpheresByEntity);
718 if (entityIndex < 0)
719 {
720 skipped++;
721 continue;
722 }
723
724 if (!holesByEntity.TryGetValue(entityIndex, out var holes))
725 {
726 holes = new List<CADHole>();
727 holesByEntity[entityIndex] = holes;
728 }
729
730 holes.Add(hole);
731 mapped++;
732 }
733
734 Log($" Known holes mapped to parts: {mapped}/{knownHoles.Count} mapped, {skipped} skipped");
735 foreach (var group in holesByEntity.OrderBy(kvp => kvp.Key))
736 Log($" Part {group.Key}: {group.Value.Count} known hole(s)");
737
738 return holesByEntity;
739 }

745 private int ResolveHoleEntityIndex(
746 CADHole hole,
747 List<CADFace> faceList,
748 Dictionary<int, PartBoundingSphere> partSpheresByEntity)
749 {
750 var matchedEntities = new List<int>();
751 if (hole.FaceGlobalIndices != null && hole.FaceGlobalIndices.Count > 0)
752 {
753 var faceIds = new HashSet<int>(hole.FaceGlobalIndices);
754 matchedEntities = faceList
755 .Where(f => faceIds.Contains(f.FaceGlobalIndex))
756 .Select(f => f.EntityIndex)
757 .Distinct()
758 .ToList();
759 }
760
761 var referencePoint = GetHoleReferencePoint(hole);
762 var globalBest = FindNearestPartSphere(referencePoint, partSpheresByEntity);
763 if (globalBest.entityIndex < 0)
764 return -1;
765
766 if (matchedEntities.Count > 0)
767 {
768 var matchedCandidates = partSpheresByEntity
769 .Where(kvp => matchedEntities.Contains(kvp.Key))
770 .ToDictionary(kvp => kvp.Key, kvp => kvp.Value);
771 var matchedBest = FindNearestPartSphere(referencePoint, matchedCandidates);
772
773 double tolerance = Math.Max(_config.HoleSpatialMatchToleranceMm, partSpheresByEntity[globalBest.entityIndex].PartRadius * _config.HoleSpatialMatchPartRadiusFactor);
774 if (matchedBest.entityIndex >= 0 && matchedBest.score <= globalBest.score + tolerance)
775 return matchedBest.entityIndex;
776
777 Log($" Hole {hole.HoleId} face-index match [{string.Join(",", matchedEntities)}] overridden by spatial part {globalBest.entityIndex}");
778 }
779
780 return globalBest.entityIndex;
781 }

789 private static (int entityIndex, double score) FindNearestPartSphere(
790 Point3D referencePoint,
791 Dictionary<int, PartBoundingSphere> partSpheresByEntity)
792 {
793 int bestEntity = -1;
794 double bestScore = double.MaxValue;
795
796 foreach (var candidate in partSpheresByEntity)
797 {
798 double distance = Distance(referencePoint, candidate.Value.Center);
799 double boundaryDistance = Math.Max(0.0, distance - Math.Max(1.0, candidate.Value.PartRadius));
800 if (boundaryDistance < bestScore)
801 {
802 bestScore = boundaryDistance;
803 bestEntity = candidate.Key;
804 }
805 }
806
807 return (bestEntity, bestScore);
808 }

838 private static double Distance(Point3D a, Point3D b)
839 {
840 double dx = a.X - b.X;

887 private static Point3D ConstrainPointToBottomAvoidance(Point3D pos, PartBoundingSphere sphere, ScanPathOptions options)
888 {
889 if (!options.AvoidBottomFace)
890 return pos;
891
892 double minElev = options.BottomAvoidanceAngleDeg * Math.PI / 180.0;
893 if (minElev <= 0.0)
894 return pos;
895
896 double currentElev = ElevationAboveBottomForSphere(pos, sphere);
897 if (currentElev >= minElev)
898 return pos;
899
900 // Clamping elevation to minElev while preserving azimuth/longitude
901 double dx = pos.X - sphere.Center.X;
902 double dy = pos.Y - sphere.Center.Y;
903 double phi = Math.Atan2(dy, dx);
904 double R = sphere.ScannerRadius;
905 double dZ = sphere.Center.Z - sphere.BottomZ;
906 double k = 1.0 / Math.Tan(minElev);
907
908 double A = k * k + 1.0;
909 double B = 2.0 * k * k * dZ;
910 double C = k * k * dZ * dZ - R * R;
911
912 double discriminant = B * B - 4.0 * A * C;
913 if (discriminant >= 0)
914 {
915 double zRel = (-B + Math.Sqrt(discriminant)) / (2.0 * A);
916 double hDist = Math.Sqrt(Math.Max(0.0, R * R - zRel * zRel));
917 return new Point3D(
918 sphere.Center.X + hDist * Math.Cos(phi),
919 sphere.Center.Y + hDist * Math.Sin(phi),
920 sphere.Center.Z + zRel);
921 }
922
923 return pos;
924 }

1207 private List<ScanPoint> GenerateFibonacciCandidates(
1208 Point3D center, double radius, ScanPathOptions options)
1209 {
1210 // Surface area amplification
1211 double baseRadius = Math.Max(1.0, options.FibonacciBaseRadius); // FibonacciSampleCount is calibrated for this orbital radius
1212 double areaRatio = (radius * radius) / (baseRadius * baseRadius);
1213 double minAmp = Math.Max(0.1, options.FibonacciMinAmplification);
1214 double maxAmp = Math.Max(minAmp, options.FibonacciMaxAmplification);
1215 double amplification = Math.Max(minAmp, Math.Min(maxAmp, areaRatio));
1216 int n = Math.Max(2, (int)(options.FibonacciSampleCount * amplification));
1217
1218 double minAngle = options.BottomAvoidanceAngleDeg * Math.PI / 180.0;
1219 bool avoidBot = options.AvoidBottomFace;
1220 bool scanBot = options.ScanBottomFace && !avoidBot;
1221
1222 const double GoldenAngle = Math.PI * (3.0 - 2.2360679774997896); // π(3−√5)
1223
1224 var candidates = new List<ScanPoint>(n);
1225
1226 for (int i = 0; i < n; i++)
1227 {
1228 // Fibonacci mapping to unit sphere
1229 double t = (double)i / (n - 1);
1230 double inclination = Math.Acos(1.0 - 2.0 * t); // 0 (north) → π (south)
1231 double azimuth = GoldenAngle * i;
1232
1233 double sinI = Math.Sin(inclination);
1234 double ux = sinI * Math.Cos(azimuth);
1235 double uy = sinI * Math.Sin(azimuth);
1236 double uz = Math.Cos(inclination);
1237
1238 var pos = new Point3D(
1239 center.X + ux * radius,
1240 center.Y + uy * radius,
1241 center.Z + uz * radius);
1242
1243 // ── Bottom-face policy ────────────────────────────────────────────
1244 if (avoidBot)
1245 {
1246 double elevRad = ElevationAngleAboveBottom(pos);
1247 if (elevRad < minAngle) continue; // below allowed elevation
1248 }
1249 else if (!scanBot)
1250 {
1251 // Neither avoid nor explicitly scan bottom → keep hemisphere only
1252 if (pos.Z < _sphereCenter.Z) continue;
1253 }
1254 // If scanBot is true we keep all positions including nadir.
1255
1256 // Orientation: point from sphere surface inward toward centre
1257 var orientation = new Vector3D(-ux, -uy, -uz); // already normalised
1258
1259 var fwd = orientation;
1260 var worldUp = Math.Abs(fwd.Z) < 0.99 ? Vector3D.AxisZ : Vector3D.AxisX;
1261 var right = Vector3D.Cross(worldUp, fwd);
1262 if (right.Length > 1e-6) right.Normalize(); else right = new Vector3D(1, 0, 0);
1263 var up0 = Vector3D.Cross(fwd, right);
1264 if (up0.Length > 1e-6) up0.Normalize(); else up0 = new Vector3D(0, 1, 0);
1265 var up90 = right;
1266
1267 candidates.Add(new ScanPoint(pos, orientation, up0, null));
1268 if (options.AddRolledCandidates)
1269 candidates.Add(new ScanPoint(pos, orientation, up90, null));
1270
1271 // Add lateral angled variations to better target downward/lateral faces
1272 if (options.AddAngledCandidates && uz < options.AngledCandidateMaxUnitZ) // Faces pointing sideways or downwards
1273 {
1274 // Pitch and Yaw variations using Rodrigues' rotation formula
1275 Func<Vector3D, Vector3D, double, Vector3D> rotateVector = (v, axis, a) =>
1276 {
1277 double cos = Math.Cos(a);
1278 double sin = Math.Sin(a);
1279 var cross = Vector3D.Cross(axis, v);
1280 double dot = Vector3D.Dot(axis, v);
1281 var res = new Vector3D(
1282 v.X * cos + cross.X * sin + axis.X * dot * (1 - cos),
1283 v.Y * cos + cross.Y * sin + axis.Y * dot * (1 - cos),
1284 v.Z * cos + cross.Z * sin + axis.Z * dot * (1 - cos)
1285 );
1286 res.Normalize();
1287 return res;
1288 };
1289
1290 foreach (double angleDeg in options.AngledCandidateAnglesDeg)
1291 {
1292 double angleRad = angleDeg * Math.PI / 180.0;
1293 var fwdR = rotateVector(fwd, right, angleRad);
1294 var fwdRN = rotateVector(fwd, right, -angleRad);
1295 var fwdU = rotateVector(fwd, up0, angleRad);
1296 var fwdUN = rotateVector(fwd, up0, -angleRad);
1297
1298 candidates.Add(new ScanPoint(pos, fwdR, up0, null));
1299 candidates.Add(new ScanPoint(pos, fwdRN, up0, null));
1300 candidates.Add(new ScanPoint(pos, fwdU, up90, null));
1301 candidates.Add(new ScanPoint(pos, fwdUN, up90, null));
1302 }
1303 }
1304 }
1305
1306 return candidates;
1307 }

1589 private FrustumBounds CalculateFrustumBoundingBox(Point3D scannerPos, Vector3D orientation, Vector3D upVector, ScannerProperties props)

1592 if (fwd.Length > 1e-6) fwd.Normalize(); else fwd = new Vector3D(0, 0, -1);

1594 var up = new Vector3D(upVector.X, upVector.Y, upVector.Z);
1595 if (up.Length > 1e-6) up.Normalize(); else up = new Vector3D(0, 1, 0);
1596
1597 var right = Vector3D.Cross(up, fwd);
1598 if (right.Length > 1e-6) right.Normalize(); else right = new Vector3D(1, 0, 0);
1599
1600 up = Vector3D.Cross(fwd, right);
1601 if (up.Length > 1e-6) up.Normalize();

1614 foreach (var dist in dists)
1615 {

1617 double halfHeight = (props.FovY / 2.0) * (dist / props.OptimalStandoff);

1621 var p2 = new Point3D(center.X + right.X * halfWidth - up.X * halfHeight, center.Y + right.Y * halfWidth - up.Y * halfHeight, center.Z + right.Z * halfWidth - up.Z * halfHeight);
1622 var p3 = new Point3D(center.X - right.X * halfWidth + up.X * halfHeight, center.Y - right.Y * halfWidth + up.Y * halfHeight, center.Z - right.Z * halfWidth + up.Z * halfHeight);
1623 var p4 = new Point3D(center.X - right.X * halfWidth - up.X * halfHeight, center.Y - right.Y * halfWidth - up.Y * halfHeight, center.Z - right.Z * halfWidth - up.Z * halfHeight);

1644 if (scannerPos.X > bounds.MaxX) bounds.MaxX = scannerPos.X;

1658 private bool IsPointInScannerFOV(
1659 Point3D scannerPos, Vector3D orientation, Vector3D upVector, Point3D point)
1660 {
1661 double dx = point.X - scannerPos.X;
1662 double dy = point.Y - scannerPos.Y;
1663 double dz = point.Z - scannerPos.Z;
1664 double dist = Math.Sqrt(dx * dx + dy * dy + dz * dz);
1665
1666 if (dist < _scannerProperties.MinStandoff ||
1667 dist > _scannerProperties.MaxStandoff)
1668 return false;
1669
1670 // Normalise direction to point
1671 double invD = 1.0 / dist;
1672 double tpX = dx * invD, tpY = dy * invD, tpZ = dz * invD;
1673
1674 // Must be in front of scanner
1675 double dot = orientation.X * tpX + orientation.Y * tpY + orientation.Z * tpZ;
1676 if (dot <= 0) return false;
1677
1678 // Build scanner-local frame (forward = orientation)
1679 var fwd = new Vector3D(orientation.X, orientation.Y, orientation.Z);
1680 if (fwd.Length > 1e-6) fwd.Normalize(); else fwd = new Vector3D(0, 0, -1);
1681
1682 var up = new Vector3D(upVector.X, upVector.Y, upVector.Z);
1683 if (up.Length > 1e-6) up.Normalize(); else up = new Vector3D(0, 1, 0);
1684
1685 var right = Vector3D.Cross(up, fwd);
1686 if (right.Length > 1e-6) right.Normalize(); else right = new Vector3D(1, 0, 0);
1687
1688 // Re-orthogonalize up vector
1689 up = Vector3D.Cross(fwd, right);
1690 if (up.Length > 1e-6) up.Normalize();
1691
1692 // Project onto scanner axes
1693 double toX = dx, toY = dy, toZ = dz;
1694 double projFwd = fwd.X * toX + fwd.Y * toY + fwd.Z * toZ;
1695 double projRight = right.X * toX + right.Y * toY + right.Z * toZ;
1696 double projUp = up.X * toX + up.Y * toY + up.Z * toZ;
1697
1698 if (projFwd <= 0) return false;
1699
1700 // Half-angle of FOV frustum
1701 double halfFovX = Math.Atan(_scannerProperties.FovX /
1702 (2.0 * _scannerProperties.OptimalStandoff));
1703 double halfFovY = Math.Atan(_scannerProperties.FovY /
1704 (2.0 * _scannerProperties.OptimalStandoff));
1705
1706 double angX = Math.Atan2(Math.Abs(projRight), projFwd);
1707 double angY = Math.Atan2(Math.Abs(projUp), projFwd);
1708
1709 return angX <= halfFovX && angY <= halfFovY;
1710 }

1766 private List<DetectedHole> DetectHoles(Mesh mesh)
1767 {
1768 var holes = new List<DetectedHole>();
1769 if (mesh?.Vertices == null || mesh.Triangles == null) return holes;
1770
1771 // ── Step 1: edge adjacency ────────────────────────────────────────────
1772 // Key = (min-idx, max-idx), Value = list of triangle indices sharing it
1773 var edgeMap = new Dictionary<(int, int), List<int>>();
1774
1775 for (int ti = 0; ti < mesh.Triangles.Length; ti++)
1776 {
1777 var tri = mesh.Triangles[ti];
1778 AddEdge(edgeMap, tri.V1, tri.V2, ti);
1779 AddEdge(edgeMap, tri.V2, tri.V3, ti);
1780 AddEdge(edgeMap, tri.V3, tri.V1, ti);
1781 }
1782
1783 // Boundary edges (used by exactly one triangle)
1784 var boundaryEdges = new HashSet<(int, int)>();
1785 var vertexNeighbours = new Dictionary<int, List<int>>();
1786
1787 foreach (var (edge, tris) in edgeMap)
1788 {
1789 if (tris.Count != 1) continue;
1790 boundaryEdges.Add(edge);
1791 AddNeighbour(vertexNeighbours, edge.Item1, edge.Item2);
1792 AddNeighbour(vertexNeighbours, edge.Item2, edge.Item1);
1793 }
1794
1795 if (boundaryEdges.Count == 0) return holes;
1796
1797 // ── Step 2: trace loops ────────────────────────────────────────────────
1798 var visited = new HashSet<int>();
1799 var loops = new List<List<int>>();
1800
1801 foreach (var (a, b) in boundaryEdges)
1802 {
1803 if (visited.Contains(a)) continue;
1804
1805 var loop = new List<int>();
1806 int current = a;
1807 int prev = -1;
1808
1809 for (int safety = 0; safety < mesh.Vertices.Length + 2; safety++)
1810 {
1811 if (visited.Contains(current)) { loop.Add(current); break; }
1812 loop.Add(current);
1813 visited.Add(current);
1814
1815 int next = -1;
1816 if (vertexNeighbours.TryGetValue(current, out var nbrs))
1817 {
1818 foreach (var nb in nbrs)
1819 {
1820 if (nb != prev && (!visited.Contains(nb) || nb == a))
1821 {
1822 next = nb;
1823 break;
1824 }
1825 }
1826 }
1827
1828 if (next < 0 || (next == a && loop.Count > 2)) break;
1829 prev = current;
1830 current = next;
1831 }
1832
1833 if (loop.Count >= 6) // meaningful loop needs several vertices
1834 loops.Add(loop);
1835 }
1836
1837 // ── Steps 3–5: classify loops ─────────────────────────────────────────
1838 foreach (var loop in loops)
1839 {
1840 var pts = loop.Select(idx => mesh.Vertices[idx]).ToList();
1841
1842 var centroid = new Point3D(
1843 pts.Average(p => p.X),
1844 pts.Average(p => p.Y),
1845 pts.Average(p => p.Z));
1846
1847 // Average radius
1848 double avgR = pts.Average(p => centroid.DistanceTo(p));
1849
1850 if (avgR < _config.MinHoleRadiusForDetection) continue;
1851
1852 // Radius variance – low value means nearly circular
1853 double varR = pts.Average(p => Math.Pow(centroid.DistanceTo(p) - avgR, 2));
1854 double cvR = Math.Sqrt(varR) / (avgR + 1e-9); // coefficient of variation
1855
1856 if (cvR > 0.25) continue; // not circular enough
1857
1858 // Best-fit normal via Newell's method
1859 var normal = NewellNormal(pts);
1860 if (normal.Length < 0.001) continue;
1861 normal.Normalize();
1862
1863 // Ensure normal points outward (away from part interior)
1864 var toCenter = new Vector3D(
1865 _sphereCenter.X - centroid.X,
1866 _sphereCenter.Y - centroid.Y,
1867 _sphereCenter.Z - centroid.Z);
1868 toCenter.Normalize();
1869 if (Vector3D.Dot(normal, toCenter) > 0)
1870 {
1871 normal = new Vector3D(-normal.X, -normal.Y, -normal.Z);
1872 }
1873 // 'normal' now points INTO the hole
1874
1875 // Estimate depth by raycasting along axis
1876 double depth = EstimateHoleDepth(centroid, normal, mesh);
1877 if (depth < avgR * 0.5) continue; // too shallow – probably not a bore
1878
1879 holes.Add(new DetectedHole
1880 {
1881 Center = centroid,
1882 Axis = normal,
1883 Radius = avgR,
1884 Depth = depth
1885 });
1886 }
1887
1888 return holes;
1889 }

1935 private static Vector3D NewellNormal(List<Point3D> pts)
1936 {
1937 double nx = 0, ny = 0, nz = 0;
1938 int n = pts.Count;
1939 for (int i = 0; i < n; i++)
1940 {
1941 var cur = pts[i];
1942 var next = pts[(i + 1) % n];
1943 nx += (cur.Y - next.Y) * (cur.Z + next.Z);
1944 ny += (cur.Z - next.Z) * (cur.X + next.X);
1945 nz += (cur.X - next.X) * (cur.Y + next.Y);
1946 }
1947 return new Vector3D(nx, ny, nz);

1977 private List<ScanPoint> GenerateHoleScanPositions(
1978 DetectedHole hole, ScanPathOptions options)
1979 {
1980 var positions = new List<ScanPoint>();
1981
1982 // Scanner aims INTO the hole (same direction as hole axis)
1983 var scannerFacing = new Vector3D(hole.Axis.X, hole.Axis.Y, hole.Axis.Z);
1984 scannerFacing.Normalize();
1985
1986 // Scanner sits ABOVE the hole opening (opposite to axis direction)
1987 // Allow diving in closer to the hole using HoleScanApproximationThreshold
1988 double standoff = Math.Max(_scannerProperties.MinStandoff, Math.Min(_scannerProperties.OptimalStandoff, options.HoleScanApproximationThreshold));
1989
1990 // Main axial position: directly above the hole, covering the full bore
1991 var mainPos = new Point3D(
1992 hole.Center.X - scannerFacing.X * standoff,
1993 hole.Center.Y - scannerFacing.Y * standoff,
1994 hole.Center.Z - scannerFacing.Z * standoff);
1995
1996 positions.Add(new ScanPoint(mainPos, scannerFacing, null, "HoleScan"));
1997
1998 // Additional rings at successive depths to catch the bore walls
1999 int rings = options.HoleScanRings;
2000 for (int i = 1; i <= rings; i++)
2001 {
2002 double t = (double)i / (rings + 1);
2003 double depthFraction = Math.Max(0.0, Math.Min(1.0, options.HoleScanDepthFraction));
2004 double depthAlongAxis = t * hole.Depth * depthFraction; // don't go past bottom
2005
2006 // Radial offset positions (N-sided polygon at this depth)
2007 int radialCount = Math.Max(1, options.HoleScanRadialCount);
2008 double radialOffset = hole.Radius * Math.Max(0.0, options.HoleScanRadialOffsetFactor); // lean in toward the wall
2009
2010 // Build a local frame perpendicular to the axis
2011 var axisVec = hole.Axis;
2012 var worldUp = Math.Abs(axisVec.Z) < 0.9 ? Vector3D.AxisZ : Vector3D.AxisX;
2013 var tangent1 = Vector3D.Cross(worldUp, axisVec); tangent1.Normalize();
2014 var tangent2 = Vector3D.Cross(axisVec, tangent1); tangent2.Normalize();
2015
2016 // Target point at this depth on the bore axis
2017 var depthCenter = new Point3D(
2018 hole.Center.X + scannerFacing.X * depthAlongAxis,
2019 hole.Center.Y + scannerFacing.Y * depthAlongAxis,
2020 hole.Center.Z + scannerFacing.Z * depthAlongAxis);
2021
2022 for (int k = 0; k < radialCount; k++)
2023 {
2024 double angle = 2.0 * Math.PI * k / radialCount;
2025 double uOff = radialOffset * Math.Cos(angle);
2026 double vOff = radialOffset * Math.Sin(angle);
2027
2028 // Scanner position: outside the hole (at standoff above entry)
2029 // but offset radially so it views the bore wall at this depth
2030 var scanPos = new Point3D(
2031 hole.Center.X - scannerFacing.X * standoff
2032 + tangent1.X * uOff + tangent2.X * vOff,
2033 hole.Center.Y - scannerFacing.Y * standoff
2034 + tangent1.Y * uOff + tangent2.Y * vOff,
2035 hole.Center.Z - scannerFacing.Z * standoff
2036 + tangent1.Z * uOff + tangent2.Z * vOff);
2037
2038 // Aim at the bore-wall target
2039 var aimDir = new Vector3D(
2040 depthCenter.X - scanPos.X,
2041 depthCenter.Y - scanPos.Y,
2042 depthCenter.Z - scanPos.Z);
2043 aimDir.Normalize();
2044
2045 positions.Add(new ScanPoint(scanPos, aimDir, null, "HoleScan"));
2046 }
2047 }
2048

2050 {

2054 .Where(p => ElevationAngleAboveBottom(p.Position) >= minElevationRad)
2055 .Where(p => p.Position.Z >= hole.Center.Z - Math.Max(0.0, options.HoleScanBottomAvoidanceZTolerance))

2063 }

2215 private static Vector3D Slerp(Vector3D a, Vector3D b, double t)
2216 {
2217 double cosO = Math.Max(-1.0, Math.Min(1.0,
2218 a.X * b.X + a.Y * b.Y + a.Z * b.Z));
2219
2220 if (cosO > 0.9999)
2221 {
2222 // Nearly identical – lerp + normalise
2223 var r = new Vector3D(
2224 a.X + (b.X - a.X) * t,
2225 a.Y + (b.Y - a.Y) * t,
2226 a.Z + (b.Z - a.Z) * t);
2227 r.Normalize();
2228 return r;
2229 }
2230
2231 if (cosO < -0.9999)
2232 {
2233 // Anti-parallel – build perpendicular halfway vector
2234 var perp = BuildPerpendicularVector(a);
2235 return Slerp(a, perp, t * 2.0 <= 1.0 ? t * 2.0 : 2.0 - t * 2.0);
2236 }
2237
2238 double omega = Math.Acos(cosO);
2239 double sinO = Math.Sin(omega);
2240 double w0 = Math.Sin((1.0 - t) * omega) / sinO;
2241 double w1 = Math.Sin(t * omega) / sinO;
2242
2243 return new Vector3D(
2244 a.X * w0 + b.X * w1,
2245 a.Y * w0 + b.Y * w1,
2246 a.Z * w0 + b.Z * w1);
2247 }

2257 var perp = Vector3D.Cross(v, candidate);
2258 perp.Normalize();
