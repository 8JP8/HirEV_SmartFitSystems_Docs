389 public List<ScanPath> GenerateSphericalScanPaths(
390 IEnumerable<CADFace> faces, Mesh partMesh, ScanPathOptions options)
391 {
392 options ??= BuildOptionsFromConfig();
393 var faceList = faces?.Where(f => !IsIgnoredFace(f)).ToList() ?? new List<CADFace>();
394 if (!faceList.Any())
395 return new List<ScanPath>();
396
397 bool hasMesh = partMesh?.Vertices != null && partMesh.Vertices.Length > 0;
398
399 Log("🚀 Spherical scan-path generation started");
400 Log($" {faceList.Count} faces | Fibonacci samples: {options.FibonacciSampleCount} | Mesh: {hasMesh}");
401 Log($" AvoidBottom: {options.AvoidBottomFace} ({options.BottomAvoidanceAngleDeg}°) | ScanBottom: {options.ScanBottomFace} | Holes: {options.ScanHoles && hasMesh}");
402 ReportProgress("Progress_ScanPathBoundingSphere", 5);
403
404 // ── 1. Bounding sphere ────────────────────────────────────────────────
405 // When a mesh is available use it for a tight sphere; otherwise fall
406 // back to the face control points so the mesh-less path still works.
407 if (hasMesh)
408 InitializeBoundingSphere(partMesh!);
409 else
410 InitializeBoundingSphereFromFaces(faceList);
411 double scannerSphereRadius = CalculateScannerSphereRadius(_sphereRadius, options);
412
413 Log($"📦 Bounding sphere C={Fmt(_sphereCenter)} r={_sphereRadius:F1} mm");
414 Log($"📡 Scanner sphere r={scannerSphereRadius:F1} mm");
415
416 // ── 2. Surface sample points (coverage targets) ───────────────────────
417 var surfaceSamples = ExtractSurfaceSamples(faceList);
418 if (!surfaceSamples.Any()) return new List<ScanPath>();
419 var surfacePoints = surfaceSamples.Select(s => s.Position).ToList();
420 Log($"📍 {surfaceSamples.Count} surface sample points");
421 ReportProgress("Progress_ScanPathSurfaceSamples", 15);
422
423 // ── 3. Candidate positions on Fibonacci sphere ────────────────────────
424 var candidates = GenerateFibonacciCandidates(
425 _sphereCenter, scannerSphereRadius, options);
426 Log($"SUCCESS: {candidates.Count} candidate positions after filtering");
427 ReportProgress("Progress_ScanPathCandidates", 25);
428
429 // ── 4. Greedy coverage selection ──────────────────────────────────────
430 var selected = SelectCoveragePositions(
431 candidates, surfaceSamples, partMesh, options);
432 Log($"SUCCESS: {selected.Count} positions selected by greedy set-cover");
433 ReportProgress("Progress_ScanPathCoverage", 50);
434
435 // ── 5. Hole axial scanning ────────────────────────────────────────────
436 if (options.ScanHoles && hasMesh)
437 {
438 var holes = DetectHoles(partMesh!);
439 Log($"🔩 {holes.Count} holes detected");
440 ReportProgress("Progress_ScanPathHoles", 60);
441 foreach (var hole in holes)
442 {
443 if (hole.Radius < options.MinHoleRadius) continue;
444 var holePoses = GenerateHoleScanPositions(hole, options);
445 selected.AddRange(holePoses);
446 Log($" Hole r={hole.Radius:F1} depth={hole.Depth:F1} → {holePoses.Count} positions");
447 }
448 }
449
450 if (!selected.Any()) return new List<ScanPath>();
451
452 // ── 6. TSP ordering on sphere ─────────────────────────────────────────
453 var ordered = OrderOnSphere(selected, _sphereCenter);
454 ReportProgress("Progress_ScanPathOrdering", 70);
455
456 // ── 7. Great-circle interpolation ─────────────────────────────────────
457 var sphere = new PartBoundingSphere(_sphereCenter, _sphereRadius, scannerSphereRadius, _partBottomZ, _partTopZ);
458 var finalPath = InterpolateGreatCirclePath(ordered, sphere, options);
459 ReportProgress("Progress_ScanPathInterpolating", 85);
460
461 Log($"🏁 Final path: {finalPath.Count} waypoints " +
462 $"({ordered.Count} measurement + {finalPath.Count - ordered.Count} arc)");
463
464 if (hasMesh && partMesh != null)
465 {
466 LogCoverage(ordered, surfacePoints, partMesh);
467 }
468
469 return finalPath.Any()
470 ? new List<ScanPath> { new ScanPath(finalPath.First().ParentFace!, finalPath) }
471 : new List<ScanPath>();
472 }

539 public List<ScanPath> GenerateMultiPartSphericalScanPaths(
540 IEnumerable<CADFace> allFaces,
541 Mesh? combinedMesh,
542 ScanPathOptions options,
543 IEnumerable<CADHole>? knownHoles = null)
544 {
545 var knownHoleList = knownHoles?.ToList() ?? new List<CADHole>();
546 options ??= BuildOptionsFromConfig(knownHoleList);
547 var faceList = allFaces?.Where(f => !IsIgnoredFace(f)).ToList() ?? new List<CADFace>();
548 if (!faceList.Any()) return new List<ScanPath>();
549
550 // ── Group faces by entity (= one CAD body / loaded file) ─────────────
551 var facesByEntity = faceList
552 .GroupBy(f => f.EntityIndex)
553 .OrderBy(g => g.Key)
554 .ToList();
555 ReportProgress("Progress_ScanPathPreparingParts", 0, facesByEntity.Count);
556
557 var partSpheresByEntity = facesByEntity.ToDictionary(
558 g => g.Key,
559 g => ComputePartSphere(g.ToList(), options));
560
561 // Build hole lookup: entityIndex -> holes on that body. FaceGlobalIndex can be
562 // missing or local/repeated in multi-part data, so fall back to the hole center
563 // against each part sphere instead of assigning every ambiguous hole to part 0.
564 var holesByEntity = BuildHolesByEntity(knownHoleList, faceList, partSpheresByEntity);
565
566 Log($"🏭 Multi-part scan: {facesByEntity.Count} part(s), knownHoles: {knownHoleList.Count}");
567
568 // ── Per-part planning ─────────────────────────────────────────────────
569 var partGroups = new List<PartScanGroup>();
570
571 foreach (var group in facesByEntity)
572 {
573 int entityIdx = group.Key;
574 var partFaces = group.ToList();
575 var partHoles = holesByEntity.TryGetValue(entityIdx, out var h) ? h : new List<CADHole>();
576 var sphere = partSpheresByEntity.TryGetValue(entityIdx, out var cachedSphere)
577 ? cachedSphere
578 : ComputePartSphere(partFaces, options);
579 int partNumber = partGroups.Count + 1;
580 double partBasePercent = facesByEntity.Count == 0
581 ? 0.0
582 : 5.0 + (partNumber - 1) * (75.0 / facesByEntity.Count);
583 ReportProgress("Progress_ScanPathPlanningPart", partBasePercent, partNumber, facesByEntity.Count);
584
585 Log($" Part {entityIdx}: {partFaces.Count} faces | sphere C={Fmt(sphere.Center)} R={sphere.ScannerRadius:F1}mm");
586
587 // Temporarily load instance bounding-sphere fields so helper methods work
588 _sphereCenter = sphere.Center;
589 _sphereRadius = sphere.PartRadius;
590 _partBottomZ = sphere.BottomZ;
591 _partTopZ = sphere.TopZ;
592
593
594 var exteriorPartFaces = partFaces.Where(f => f.IsExterior).ToList();
595 if (!exteriorPartFaces.Any())
596 exteriorPartFaces = partFaces;
597 var exteriorMesh = BuildExteriorMeshFromFaces(exteriorPartFaces);
598
599 // Surface sample points for this part
600 var surfSamples = ExtractSurfaceSamples(exteriorPartFaces);
601 var surfPts = surfSamples.Select(s => s.Position).ToList();
602 ReportProgress("Progress_ScanPathSurfaceSamples", partBasePercent + 10.0);
603
604 // Candidates on this part's Fibonacci sphere
605 var candidates = GenerateFibonacciCandidates(sphere.Center, sphere.ScannerRadius, options);
606 ReportProgress("Progress_ScanPathCandidates", partBasePercent + 20.0);
607
608 // Greedy coverage
609 var selected = SelectCoveragePositions(candidates, surfSamples, exteriorMesh, options);
610 ReportProgress("Progress_ScanPathCoverage", partBasePercent + 35.0);
611
612 // Inject known holes (using CADHole from the app's HoleDetector)
613 if (options.ScanHoles)
614 {
615 if (partHoles.Any())
616 {
617 foreach (var cadHole in partHoles)
618 {
619 double holeRadius = GetHoleRadius(cadHole);
620 if (holeRadius < options.MinHoleRadius)
621 {
622 Log($" Hole {cadHole.HoleId} skipped: radius {holeRadius:F2}mm below minimum {options.MinHoleRadius:F2}mm");
623 continue;
624 }
625
626 var det = CadHoleToDetectedHole(cadHole, sphere.Center);
627 var holePoses = GenerateHoleScanPositions(det, options);
628 CADFace? holeParentFace = partFaces.FirstOrDefault(f => cadHole.FaceGlobalIndices?.Contains(f.FaceGlobalIndex) == true)
629 ?? partFaces.FirstOrDefault();
630 if (holeParentFace != null)
631 holePoses = holePoses.Select(p => new ScanPoint(p.Position, p.Orientation, p.UpVector, holeParentFace, p.PointType)).ToList();
632 selected.AddRange(holePoses);
633 Log($" Hole {cadHole.HoleId} r={holeRadius:F1} → {holePoses.Count} poses");
634 }
635 }
636
637 // Fallback or supplementary: use the shell mesh of the current part
638 if (exteriorMesh != null)
639 {
640 var detectedHoles = DetectHoles(exteriorMesh);
641 if (detectedHoles.Any())
642 {
643 Log($" Detected {detectedHoles.Count} additional holes on part {entityIdx} shell mesh");
644 foreach (var hole in detectedHoles)
645 {
646 if (hole.Radius < options.MinHoleRadius) continue;
647 var holePoses = GenerateHoleScanPositions(hole, options);
648 CADFace? holeParentFace = partFaces.FirstOrDefault();
649 if (holeParentFace != null)
650 holePoses = holePoses.Select(p => new ScanPoint(p.Position, p.Orientation, p.UpVector, holeParentFace, p.PointType)).ToList();
651 selected.AddRange(holePoses);
652 Log($" Mesh Hole r={hole.Radius:F1} depth={hole.Depth:F1} → {holePoses.Count} positions");
653 }
654 }
655 }
656 }
657
658 if (!selected.Any()) continue;
659
660 // TSP ordering + SLERP interpolation for this part
661 var ordered = OrderOnSphere(selected, sphere.Center);
662 ReportProgress("Progress_ScanPathOrdering", partBasePercent + 50.0);
663 var interped = InterpolateGreatCirclePath(ordered, sphere, options);
664 ReportProgress("Progress_ScanPathInterpolating", partBasePercent + 65.0);
665
666 partGroups.Add(new PartScanGroup(entityIdx, sphere, interped, ordered));
667 }
668
669 if (!partGroups.Any()) return new List<ScanPath>();
670
671 // ── Single part: no bridging needed ───────────────────────────────────
672 if (partGroups.Count == 1)
673 {
674 var g = partGroups[0];
675 return g.Waypoints.Any()
676 ? new List<ScanPath> { new ScanPath(g.Waypoints.First().ParentFace!, g.Waypoints) }
677 : new List<ScanPath>();
678 }
679
680 // ── Multi-part: stitch groups with bridge arcs ───────────────────────
681 var fullPath = new List<ScanPoint>(partGroups[0].Waypoints);
682
683 for (int i = 0; i < partGroups.Count - 1; i++)
684 {
685 var partA = partGroups[i];
686 var partB = partGroups[i + 1];
687
688 var bridge = BuildPartBridge(
689 from: partA.Waypoints.Last(),
690 sphereA: partA.Sphere,
691 to: partB.Waypoints.First(),
692 sphereB: partB.Sphere,
693 options: options);
694 ReportProgress("Progress_ScanPathBridgingParts", 80.0 + (i * 15.0 / Math.Max(1, partGroups.Count - 1)), i + 1, i + 2);
695
696 fullPath.AddRange(bridge);
697 fullPath.AddRange(partB.Waypoints);
698
699 Log($"🌉 Bridge part{partA.EntityIndex}→part{partB.EntityIndex}: {bridge.Count} waypoints");
700 }
701
702 ReportProgress("Progress_ScanPathComplete", 100);
703 return new List<ScanPath> { new ScanPath(fullPath.First().ParentFace!, fullPath) };
704 }

710 private Dictionary<int, List<CADHole>> BuildHolesByEntity(
711 IReadOnlyCollection<CADHole> knownHoles,
712 List<CADFace> faceList,
713 Dictionary<int, PartBoundingSphere> partSpheresByEntity)
714 {
715 var holesByEntity = new Dictionary<int, List<CADHole>>();
716 if (knownHoles.Count == 0 || partSpheresByEntity.Count == 0)
717 return holesByEntity;
718
719 int mapped = 0;
720 int skipped = 0;
721
722 foreach (var hole in knownHoles)
723 {
724 int entityIndex = ResolveHoleEntityIndex(hole, faceList, partSpheresByEntity);
725 if (entityIndex < 0)
726 {
727 skipped++;
728 continue;
729 }
730
731 if (!holesByEntity.TryGetValue(entityIndex, out var holes))
732 {
733 holes = new List<CADHole>();
734 holesByEntity[entityIndex] = holes;
735 }
736
737 holes.Add(hole);
738 mapped++;
739 }
740
741 Log($" Known holes mapped to parts: {mapped}/{knownHoles.Count} mapped, {skipped} skipped");
742 foreach (var group in holesByEntity.OrderBy(kvp => kvp.Key))
743 Log($" Part {group.Key}: {group.Value.Count} known hole(s)");
744
745 return holesByEntity;
746 }

752 private int ResolveHoleEntityIndex(
753 CADHole hole,
754 List<CADFace> faceList,
755 Dictionary<int, PartBoundingSphere> partSpheresByEntity)
756 {
757 var matchedEntities = new List<int>();
758 if (hole.FaceGlobalIndices != null && hole.FaceGlobalIndices.Count > 0)
759 {
760 var faceIds = new HashSet<int>(hole.FaceGlobalIndices);
761 matchedEntities = faceList
762 .Where(f => faceIds.Contains(f.FaceGlobalIndex))
763 .Select(f => f.EntityIndex)
764 .Distinct()
765 .ToList();
766 }
767
768 var referencePoint = GetHoleReferencePoint(hole);
769 var globalBest = FindNearestPartSphere(referencePoint, partSpheresByEntity);
770 if (globalBest.entityIndex < 0)
771 return -1;
772
773 if (matchedEntities.Count > 0)
774 {
775 var matchedCandidates = partSpheresByEntity
776 .Where(kvp => matchedEntities.Contains(kvp.Key))
777 .ToDictionary(kvp => kvp.Key, kvp => kvp.Value);
778 var matchedBest = FindNearestPartSphere(referencePoint, matchedCandidates);
779
780 double tolerance = Math.Max(_config.HoleSpatialMatchToleranceMm, partSpheresByEntity[globalBest.entityIndex].PartRadius * _config.HoleSpatialMatchPartRadiusFactor);
781 if (matchedBest.entityIndex >= 0 && matchedBest.score <= globalBest.score + tolerance)
782 return matchedBest.entityIndex;
783
784 Log($" Hole {hole.HoleId} face-index match [{string.Join(",", matchedEntities)}] overridden by spatial part {globalBest.entityIndex}");
785 }
786
787 return globalBest.entityIndex;
788 }

796 private static (int entityIndex, double score) FindNearestPartSphere(
797 Point3D referencePoint,
798 Dictionary<int, PartBoundingSphere> partSpheresByEntity)
799 {
800 int bestEntity = -1;
801 double bestScore = double.MaxValue;
802
803 foreach (var candidate in partSpheresByEntity)
804 {
805 double distance = Distance(referencePoint, candidate.Value.Center);
806 double boundaryDistance = Math.Max(0.0, distance - Math.Max(1.0, candidate.Value.PartRadius));
807 if (boundaryDistance < bestScore)
808 {
809 bestScore = boundaryDistance;
810 bestEntity = candidate.Key;
811 }
812 }
813
814 return (bestEntity, bestScore);
815 }

822 private static Point3D GetHoleReferencePoint(CADHole hole)
823 {
824 if (hole.TopCenterPoint != null && hole.BottomCenterPoint != null)
825 {
826 var top = hole.TopCenterPoint;
827 var bottom = hole.BottomCenterPoint;
828 return new Point3D(
829 (top.X + bottom.X) * 0.5,
830 (top.Y + bottom.Y) * 0.5,
831 (top.Z + bottom.Z) * 0.5);
832 }
833
834 return hole.TopCenterPoint

894 private static Point3D ConstrainPointToBottomAvoidance(Point3D pos, PartBoundingSphere sphere, ScanPathOptions options)
895 {
896 if (!options.AvoidBottomFace)
897 return pos;
898
899 Point3D result = pos;
900
901 double minElev = options.BottomAvoidanceAngleDeg * Math.PI / 180.0;
902 if (minElev > 0.0)
903 {
904 double currentElev = ElevationAboveBottomForSphere(result, sphere);
905 if (currentElev < minElev)
906 {
907 // Clamping elevation to minElev while preserving azimuth/longitude
908 double dx = result.X - sphere.Center.X;
909 double dy = result.Y - sphere.Center.Y;
910 double phi = Math.Atan2(dy, dx);
911 double R = sphere.ScannerRadius;
912 double dZ = sphere.Center.Z - sphere.BottomZ;
913 double k = 1.0 / Math.Tan(minElev);
914
915 double A = k * k + 1.0;
916 double B = 2.0 * k * k * dZ;
917 double C = k * k * dZ * dZ - R * R;
918
919 double discriminant = B * B - 4.0 * A * C;
920 if (discriminant >= 0)
921 {
922 double zRel = (-B + Math.Sqrt(discriminant)) / (2.0 * A);
923 double hDist = Math.Sqrt(Math.Max(0.0, R * R - zRel * zRel));
924 result = new Point3D(
925 sphere.Center.X + hDist * Math.Cos(phi),
926 sphere.Center.Y + hDist * Math.Sin(phi),
927 sphere.Center.Z + zRel);
928 }
929 }
930 }
931
932 double minZ = sphere.BottomZ + options.BottomAvoidanceHeight;
933 if (result.Z < minZ)
934 {
935 // Clamp directly to minZ height on the sphere
936 double zRel = minZ - sphere.Center.Z;
937 double R = sphere.ScannerRadius;
938 if (Math.Abs(zRel) <= R)
939 {
940 double hDist = Math.Sqrt(Math.Max(0.0, R * R - zRel * zRel));
941 double dx = result.X - sphere.Center.X;
942 double dy = result.Y - sphere.Center.Y;
943 double phi = Math.Atan2(dy, dx);
944
945 result = new Point3D(
946 sphere.Center.X + hDist * Math.Cos(phi),
947 sphere.Center.Y + hDist * Math.Sin(phi),
948 minZ);
949 }
950 else if (zRel > R) // MinZ is entirely above the sphere
951 {
952 result = new Point3D(sphere.Center.X, sphere.Center.Y, sphere.Center.Z + R);
953 }
954 }
955
956 return result;
957 }

1240 private List<ScanPoint> GenerateFibonacciCandidates(
1241 Point3D center, double radius, ScanPathOptions options)
1242 {
1243 // Surface area amplification
1244 double baseRadius = Math.Max(1.0, options.FibonacciBaseRadius); // FibonacciSampleCount is calibrated for this orbital radius
1245 double areaRatio = (radius * radius) / (baseRadius * baseRadius);
1246 double minAmp = Math.Max(0.1, options.FibonacciMinAmplification);
1247 double maxAmp = Math.Max(minAmp, options.FibonacciMaxAmplification);
1248 double amplification = Math.Max(minAmp, Math.Min(maxAmp, areaRatio));
1249 int n = Math.Max(2, (int)(options.FibonacciSampleCount * amplification));
1250
1251 double minAngle = options.BottomAvoidanceAngleDeg * Math.PI / 180.0;
1252 bool avoidBot = options.AvoidBottomFace;
1253 bool scanBot = options.ScanBottomFace && !avoidBot;
1254
1255 const double GoldenAngle = Math.PI * (3.0 - 2.2360679774997896); // π(3−√5)
1256
1257 var candidates = new List<ScanPoint>(n);
1258
1259 for (int i = 0; i < n; i++)
1260 {
1261 // Fibonacci mapping to unit sphere
1262 double t = (double)i / (n - 1);
1263 double inclination = Math.Acos(1.0 - 2.0 * t); // 0 (north) → π (south)
1264 double azimuth = GoldenAngle * i;
1265
1266 double sinI = Math.Sin(inclination);
1267 double ux = sinI * Math.Cos(azimuth);
1268 double uy = sinI * Math.Sin(azimuth);
1269 double uz = Math.Cos(inclination);
1270
1271 var pos = new Point3D(
1272 center.X + ux * radius,
1273 center.Y + uy * radius,
1274 center.Z + uz * radius);
1275
1276 // ── Bottom-face policy ────────────────────────────────────────────
1277 if (avoidBot)
1278 {
1279 double elevRad = ElevationAngleAboveBottom(pos);
1280 if (elevRad < minAngle) continue; // below allowed elevation
1281 }
1282 else if (!scanBot)
1283 {
1284 // Neither avoid nor explicitly scan bottom → keep hemisphere only
1285 if (pos.Z < _sphereCenter.Z) continue;
1286 }
1287 // If scanBot is true we keep all positions including nadir.
1288
1289 // Orientation: point from sphere surface inward toward centre
1290 var orientation = new Vector3D(-ux, -uy, -uz); // already normalised
1291
1292 var fwd = orientation;
1293 var worldUp = Math.Abs(fwd.Z) < 0.99 ? Vector3D.AxisZ : Vector3D.AxisX;
1294 var right = Vector3D.Cross(worldUp, fwd);
1295 if (right.Length > 1e-6) right.Normalize(); else right = new Vector3D(1, 0, 0);
1296 var up0 = Vector3D.Cross(fwd, right);
1297 if (up0.Length > 1e-6) up0.Normalize(); else up0 = new Vector3D(0, 1, 0);
1298 var up90 = right;
1299
1300 candidates.Add(new ScanPoint(pos, orientation, up0, null));
1301 if (options.AddRolledCandidates)
1302 candidates.Add(new ScanPoint(pos, orientation, up90, null));
1303
1304 // Add lateral angled variations to better target downward/lateral faces
1305 if (options.AddAngledCandidates && uz < options.AngledCandidateMaxUnitZ) // Faces pointing sideways or downwards
1306 {
1307 // Pitch and Yaw variations using Rodrigues' rotation formula
1308 Func<Vector3D, Vector3D, double, Vector3D> rotateVector = (v, axis, a) =>
1309 {
1310 double cos = Math.Cos(a);
1311 double sin = Math.Sin(a);
1312 var cross = Vector3D.Cross(axis, v);
1313 double dot = Vector3D.Dot(axis, v);
1314 var res = new Vector3D(
1315 v.X * cos + cross.X * sin + axis.X * dot * (1 - cos),
1316 v.Y * cos + cross.Y * sin + axis.Y * dot * (1 - cos),
1317 v.Z * cos + cross.Z * sin + axis.Z * dot * (1 - cos)
1318 );
1319 res.Normalize();
1320 return res;
1321 };
1322
1323 foreach (double angleDeg in options.AngledCandidateAnglesDeg)
1324 {
1325 double angleRad = angleDeg * Math.PI / 180.0;
1326 var fwdR = rotateVector(fwd, right, angleRad);
1327 var fwdRN = rotateVector(fwd, right, -angleRad);
1328 var fwdU = rotateVector(fwd, up0, angleRad);
1329 var fwdUN = rotateVector(fwd, up0, -angleRad);
1330
1331 candidates.Add(new ScanPoint(pos, fwdR, up0, null));
1332 candidates.Add(new ScanPoint(pos, fwdRN, up0, null));
1333 candidates.Add(new ScanPoint(pos, fwdU, up90, null));
1334 candidates.Add(new ScanPoint(pos, fwdUN, up90, null));
1335 }
1336 }
1337 }
1338
1339 return candidates;
1340 }

1487 if (up.Length > 1e-6) up.Normalize(); else up = new Vector3D(0, 1, 0);

1492 if (up.Length > 1e-6) up.Normalize();

1512 continue;

1622 private FrustumBounds CalculateFrustumBoundingBox(Point3D scannerPos, Vector3D orientation, Vector3D upVector, ScannerProperties props)
1623 {
1624 var fwd = new Vector3D(orientation.X, orientation.Y, orientation.Z);
1625 if (fwd.Length > 1e-6) fwd.Normalize(); else fwd = new Vector3D(0, 0, -1);
1626
1627 var up = new Vector3D(upVector.X, upVector.Y, upVector.Z);
1628 if (up.Length > 1e-6) up.Normalize(); else up = new Vector3D(0, 1, 0);
1629
1630 var right = Vector3D.Cross(up, fwd);
1631 if (right.Length > 1e-6) right.Normalize(); else right = new Vector3D(1, 0, 0);
1632
1633 up = Vector3D.Cross(fwd, right);
1634 if (up.Length > 1e-6) up.Normalize();
1635
1636 var bounds = new FrustumBounds
1637 {
1638 MinX = double.MaxValue,
1639 MaxX = double.MinValue,
1640 MinY = double.MaxValue,
1641 MaxY = double.MinValue,
1642 MinZ = double.MaxValue,
1643 MaxZ = double.MinValue
1644 };
1645
1646 double[] dists = { props.MinStandoff, props.MaxStandoff };
1647 foreach (var dist in dists)
1648 {
1649 double halfWidth = (props.FovX / 2.0) * (dist / props.OptimalStandoff);
1650 double halfHeight = (props.FovY / 2.0) * (dist / props.OptimalStandoff);
1651
1652 var center = new Point3D(scannerPos.X + fwd.X * dist, scannerPos.Y + fwd.Y * dist, scannerPos.Z + fwd.Z * dist);
1653 var p1 = new Point3D(center.X + right.X * halfWidth + up.X * halfHeight, center.Y + right.Y * halfWidth + up.Y * halfHeight, center.Z + right.Z * halfWidth + up.Z * halfHeight);
1654 var p2 = new Point3D(center.X + right.X * halfWidth - up.X * halfHeight, center.Y + right.Y * halfWidth - up.Y * halfHeight, center.Z + right.Z * halfWidth - up.Z * halfHeight);
1655 var p3 = new Point3D(center.X - right.X * halfWidth + up.X * halfHeight, center.Y - right.Y * halfWidth + up.Y * halfHeight, center.Z - right.Z * halfWidth + up.Z * halfHeight);
1656 var p4 = new Point3D(center.X - right.X * halfWidth - up.X * halfHeight, center.Y - right.Y * halfWidth - up.Y * halfHeight, center.Z - right.Z * halfWidth - up.Z * halfHeight);
1657
1658 var pts = new[] { p1, p2, p3, p4 };
1659 for (int i = 0; i < 4; i++)
1660 {
1661 var p = pts[i];
1662 if (p.X < bounds.MinX) bounds.MinX = p.X;
1663 if (p.X > bounds.MaxX) bounds.MaxX = p.X;
1664 if (p.Y < bounds.MinY) bounds.MinY = p.Y;
1665 if (p.Y > bounds.MaxY) bounds.MaxY = p.Y;
1666 if (p.Z < bounds.MinZ) bounds.MinZ = p.Z;
1667 if (p.Z > bounds.MaxZ) bounds.MaxZ = p.Z;
1668 }
1669 }
1670
1671 // Apply a small safety margin
1672 bounds.MinX -= 1.0; bounds.MaxX += 1.0;
1673 bounds.MinY -= 1.0; bounds.MaxY += 1.0;
1674 bounds.MinZ -= 1.0; bounds.MaxZ += 1.0;
1675
1676 if (scannerPos.X < bounds.MinX) bounds.MinX = scannerPos.X;
1677 if (scannerPos.X > bounds.MaxX) bounds.MaxX = scannerPos.X;
1678 if (scannerPos.Y < bounds.MinY) bounds.MinY = scannerPos.Y;
1679 if (scannerPos.Y > bounds.MaxY) bounds.MaxY = scannerPos.Y;
1680 if (scannerPos.Z < bounds.MinZ) bounds.MinZ = scannerPos.Z;
1681 if (scannerPos.Z > bounds.MaxZ) bounds.MaxZ = scannerPos.Z;
1682
1683 return bounds;

1691 private bool IsPointInScannerFOV(
1692 Point3D scannerPos, Vector3D orientation, Vector3D upVector, Point3D point)
1693 {
1694 double dx = point.X - scannerPos.X;
1695 double dy = point.Y - scannerPos.Y;
1696 double dz = point.Z - scannerPos.Z;
1697 double dist = Math.Sqrt(dx * dx + dy * dy + dz * dz);
1698
1699 if (dist < _scannerProperties.MinStandoff ||
1700 dist > _scannerProperties.MaxStandoff)
1701 return false;
1702
1703 // Normalise direction to point
1704 double invD = 1.0 / dist;
1705 double tpX = dx * invD, tpY = dy * invD, tpZ = dz * invD;
1706
1707 // Must be in front of scanner
1708 double dot = orientation.X * tpX + orientation.Y * tpY + orientation.Z * tpZ;
1709 if (dot <= 0) return false;
1710
1711 // Build scanner-local frame (forward = orientation)
1712 var fwd = new Vector3D(orientation.X, orientation.Y, orientation.Z);
1713 if (fwd.Length > 1e-6) fwd.Normalize(); else fwd = new Vector3D(0, 0, -1);
1714
1715 var up = new Vector3D(upVector.X, upVector.Y, upVector.Z);
1716 if (up.Length > 1e-6) up.Normalize(); else up = new Vector3D(0, 1, 0);
1717
1718 var right = Vector3D.Cross(up, fwd);
1719 if (right.Length > 1e-6) right.Normalize(); else right = new Vector3D(1, 0, 0);
1720
1721 // Re-orthogonalize up vector
1722 up = Vector3D.Cross(fwd, right);
1723 if (up.Length > 1e-6) up.Normalize();
1724
1725 // Project onto scanner axes
1726 double toX = dx, toY = dy, toZ = dz;
1727 double projFwd = fwd.X * toX + fwd.Y * toY + fwd.Z * toZ;
1728 double projRight = right.X * toX + right.Y * toY + right.Z * toZ;
1729 double projUp = up.X * toX + up.Y * toY + up.Z * toZ;
1730
1731 if (projFwd <= 0) return false;
1732
1733 // Half-angle of FOV frustum
1734 double halfFovX = Math.Atan(_scannerProperties.FovX /
1735 (2.0 * _scannerProperties.OptimalStandoff));
1736 double halfFovY = Math.Atan(_scannerProperties.FovY /
1737 (2.0 * _scannerProperties.OptimalStandoff));
1738
1739 double angX = Math.Atan2(Math.Abs(projRight), projFwd);
1740 double angY = Math.Atan2(Math.Abs(projUp), projFwd);
1741
1742 return angX <= halfFovX && angY <= halfFovY;
1743 }

1752 {

1756 {
1757 double dx = target.X - scannerPos.X;

1799 private List<DetectedHole> DetectHoles(Mesh mesh)
1800 {
1801 var holes = new List<DetectedHole>();
1802 if (mesh?.Vertices == null || mesh.Triangles == null) return holes;
1803
1804 // ── Step 1: edge adjacency ────────────────────────────────────────────
1805 // Key = (min-idx, max-idx), Value = list of triangle indices sharing it
1806 var edgeMap = new Dictionary<(int, int), List<int>>();
1807
1808 for (int ti = 0; ti < mesh.Triangles.Length; ti++)
1809 {
1810 var tri = mesh.Triangles[ti];
1811 AddEdge(edgeMap, tri.V1, tri.V2, ti);
1812 AddEdge(edgeMap, tri.V2, tri.V3, ti);
1813 AddEdge(edgeMap, tri.V3, tri.V1, ti);
1814 }
1815
1816 // Boundary edges (used by exactly one triangle)
1817 var boundaryEdges = new HashSet<(int, int)>();
1818 var vertexNeighbours = new Dictionary<int, List<int>>();
1819
1820 foreach (var (edge, tris) in edgeMap)
1821 {
1822 if (tris.Count != 1) continue;
1823 boundaryEdges.Add(edge);
1824 AddNeighbour(vertexNeighbours, edge.Item1, edge.Item2);
1825 AddNeighbour(vertexNeighbours, edge.Item2, edge.Item1);
1826 }
1827
1828 if (boundaryEdges.Count == 0) return holes;
1829
1830 // ── Step 2: trace loops ────────────────────────────────────────────────
1831 var visited = new HashSet<int>();
1832 var loops = new List<List<int>>();
1833
1834 foreach (var (a, b) in boundaryEdges)
1835 {
1836 if (visited.Contains(a)) continue;
1837
1838 var loop = new List<int>();
1839 int current = a;
1840 int prev = -1;
1841
1842 for (int safety = 0; safety < mesh.Vertices.Length + 2; safety++)
1843 {
1844 if (visited.Contains(current)) { loop.Add(current); break; }
1845 loop.Add(current);
1846 visited.Add(current);
1847
1848 int next = -1;
1849 if (vertexNeighbours.TryGetValue(current, out var nbrs))
1850 {
1851 foreach (var nb in nbrs)
1852 {
1853 if (nb != prev && (!visited.Contains(nb) || nb == a))
1854 {
1855 next = nb;
1856 break;
1857 }
1858 }
1859 }
1860
1861 if (next < 0 || (next == a && loop.Count > 2)) break;
1862 prev = current;
1863 current = next;
1864 }
1865
1866 if (loop.Count >= 6) // meaningful loop needs several vertices
1867 loops.Add(loop);
1868 }
1869
1870 // ── Steps 3–5: classify loops ─────────────────────────────────────────
1871 foreach (var loop in loops)
1872 {
1873 var pts = loop.Select(idx => mesh.Vertices[idx]).ToList();
1874
1875 var centroid = new Point3D(
1876 pts.Average(p => p.X),
1877 pts.Average(p => p.Y),
1878 pts.Average(p => p.Z));
1879
1880 // Average radius
1881 double avgR = pts.Average(p => centroid.DistanceTo(p));
1882
1883 if (avgR < _config.MinHoleRadiusForDetection) continue;
1884
1885 // Radius variance – low value means nearly circular
1886 double varR = pts.Average(p => Math.Pow(centroid.DistanceTo(p) - avgR, 2));
1887 double cvR = Math.Sqrt(varR) / (avgR + 1e-9); // coefficient of variation
1888
1889 if (cvR > 0.25) continue; // not circular enough
1890
1891 // Best-fit normal via Newell's method
1892 var normal = NewellNormal(pts);
1893 if (normal.Length < 0.001) continue;
1894 normal.Normalize();
1895
1896 // Ensure normal points outward (away from part interior)
1897 var toCenter = new Vector3D(
1898 _sphereCenter.X - centroid.X,
1899 _sphereCenter.Y - centroid.Y,
1900 _sphereCenter.Z - centroid.Z);
1901 toCenter.Normalize();
1902 if (Vector3D.Dot(normal, toCenter) > 0)
1903 {
1904 normal = new Vector3D(-normal.X, -normal.Y, -normal.Z);
1905 }
1906 // 'normal' now points INTO the hole
1907
1908 // Estimate depth by raycasting along axis
1909 double depth = EstimateHoleDepth(centroid, normal, mesh);
1910 if (depth < avgR * 0.5) continue; // too shallow – probably not a bore
1911
1912 holes.Add(new DetectedHole
1913 {
1914 Center = centroid,
1915 Axis = normal,
1916 Radius = avgR,
1917 Depth = depth
1918 });
1919 }
1920
1921 return holes;
1922 }

1968 private static Vector3D NewellNormal(List<Point3D> pts)
1969 {
1970 double nx = 0, ny = 0, nz = 0;
1971 int n = pts.Count;
1972 for (int i = 0; i < n; i++)
1973 {
1974 var cur = pts[i];
1975 var next = pts[(i + 1) % n];
1976 nx += (cur.Y - next.Y) * (cur.Z + next.Z);
1977 ny += (cur.Z - next.Z) * (cur.X + next.X);
1978 nz += (cur.X - next.X) * (cur.Y + next.Y);
1979 }
1980 return new Vector3D(nx, ny, nz);

2037 double depthAlongAxis = t * hole.Depth * depthFraction; // don't go past bottom
2038

2040 int radialCount = Math.Max(1, options.HoleScanRadialCount);

2052 hole.Center.Y + scannerFacing.Y * depthAlongAxis,
2053 hole.Center.Z + scannerFacing.Z * depthAlongAxis);

2248 private static Vector3D Slerp(Vector3D a, Vector3D b, double t)
2249 {
2250 double cosO = Math.Max(-1.0, Math.Min(1.0,
2251 a.X * b.X + a.Y * b.Y + a.Z * b.Z));
2252
2253 if (cosO > 0.9999)
2254 {
2255 // Nearly identical – lerp + normalise
2256 var r = new Vector3D(
2257 a.X + (b.X - a.X) * t,
2258 a.Y + (b.Y - a.Y) * t,
2259 a.Z + (b.Z - a.Z) * t);
2260 r.Normalize();
2261 return r;
2262 }
2263
2264 if (cosO < -0.9999)
