133 public void CalculateFovAngles()
134 {
135 if (OptimalStandoff > 0)
136 {
137 FovAngleHorizontal = 2.0 * System.Math.Atan(FovX / (2.0 * OptimalStandoff)) * (180.0 / System.Math.PI);
138 FovAngleVertical = 2.0 * System.Math.Atan(FovY / (2.0 * OptimalStandoff)) * (180.0 / System.Math.PI);
139 }

153 public static Dictionary<string, ScannerProperties> Load(string filePath)
154 {
155 var presets = new Dictionary<string, ScannerProperties>();
156 if (!File.Exists(filePath))
157 {
158 return presets;
159 }
160
161 try
162 {
163 var doc = new XmlDocument();
164 doc.Load(filePath);
165
166 var nodes = doc.SelectNodes("//Scanner");
167 if (nodes == null) return presets;
168
169 foreach (XmlNode node in nodes)
170 {
171 var nameAttribute = node.Attributes?["name"];
172 if (nameAttribute == null) continue;
173
174 var props = new ScannerProperties
175 {
176 Name = nameAttribute.Value,
177 FovX = GetDouble(node, "FovX"),
178 FovY = GetDouble(node, "FovY"),
179 OptimalStandoff = GetDouble(node, "OptimalStandoff"),
180 MinStandoff = GetDouble(node, "MinStandoff"),
181 MaxStandoff = GetDouble(node, "MaxStandoff"),
182 AspectRatio = GetDouble(node, "AspectRatio"),
183 RaycastingTolerance = GetDouble(node, "RaycastingTolerance", 0.5),
184 ModelFile = node.SelectSingleNode("ModelFile")?.InnerText ?? "",
185
186 // Sensor transformation settings
187 SensorOffsetX = GetDouble(node, "SensorOffsetX", 0.0),
188 SensorOffsetY = GetDouble(node, "SensorOffsetY", 0.0),
189 SensorOffsetZ = GetDouble(node, "SensorOffsetZ", 0.0),
190 SensorRotationX = GetDouble(node, "SensorRotationX", 0.0),
191 SensorRotationY = GetDouble(node, "SensorRotationY", 0.0),
192 SensorRotationZ = GetDouble(node, "SensorRotationZ", 0.0),
193
194 // FOV angle and sensor dimensions
195 FovAngleHorizontal = GetDouble(node, "FovAngleHorizontal", 0.0),
196 FovAngleVertical = GetDouble(node, "FovAngleVertical", 0.0),
197 SensorWidth = GetDouble(node, "SensorWidth", 20.0),
198 SensorHeight = GetDouble(node, "SensorHeight", 15.0)
199 };
200
201 // Calculate FOV angles if not explicitly set
202 if (props.FovAngleHorizontal == 0.0 || props.FovAngleVertical == 0.0)
203 {
204 props.CalculateFovAngles();
205 }
206
207 presets[props.Name] = props;
208 }
209 }
210 catch (Exception ex)
211 {
212 System.Diagnostics.Debug.WriteLine($"❌ Error loading scanner configurations: {ex.Message}");
213
214 // Rename corrupt file so it doesn't cause persistent crashes
215 try
216 {
217 string corruptPath = filePath + ".corrupt";
218 if (File.Exists(corruptPath)) File.Delete(corruptPath);
219 File.Move(filePath, corruptPath);
220 System.Diagnostics.Debug.WriteLine($"⚠️ Corrupt file renamed to: {corruptPath}");
221 }
222 catch (Exception moveEx)
223 {
224 System.Diagnostics.Debug.WriteLine($"❌ Could not rename corrupt file: {moveEx.Message}");
225 }
226
227 return new Dictionary<string, ScannerProperties>(); // Return empty to trigger regeneration
228 }
229
230 return presets;
231 }

240 private static double GetDouble(XmlNode node, string elementName, double defaultValue = 0.0)
241 {
242 var element = node.SelectSingleNode(elementName);
243 if (element != null && double.TryParse(element.InnerText, NumberStyles.Any, CultureInfo.InvariantCulture, out double value))
244 {
245 return value;
246 }
247 return defaultValue;
248 }
249 }
