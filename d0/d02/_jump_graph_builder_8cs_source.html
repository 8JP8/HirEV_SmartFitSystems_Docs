77 public Point3D? FindSameFaceIntermediatePoint(JumpNode nodeA, JumpNode nodeB)
78 {
79 if (_shellGenerator == null || _shellGenerator.ResultMesh == null)
80 return null;
81
82 Mesh mesh = _shellGenerator.ResultMesh;
83
84 if (mesh.Vertices == null || mesh.Vertices.Length == 0)
85 return null;
86
87 var startPt = nodeA.ApproachPoint;
88 var endPt = nodeB.ApproachPoint;
89 var midpoint = new Point3D(
90 (startPt.X + endPt.X) / 2,
91 (startPt.Y + endPt.Y) / 2,
92 (startPt.Z + endPt.Z) / 2);
93
94 // Find nearest mesh vertex to the midpoint that is on the same parent face
95 Point3D? bestCandidate = null;
96 double bestScore = double.MaxValue;
97
98 foreach (var vertex in mesh.Vertices)
99 {
100 // Score: distance from midpoint
101 double distFromMid = vertex.DistanceTo(midpoint);
102 double distFromStart = vertex.DistanceTo(startPt);
103 double distFromEnd = vertex.DistanceTo(endPt);
104
105 // Skip if too close to either endpoint
106 if (distFromStart < 2.0 || distFromEnd < 2.0)
107 continue;
108
109 // Check if paths through this vertex are valid (same-face logic)
110 // For SDF mode, we just check if the vertex belongs to the same parent face index
111 var faceIdx = _shellGenerator.GetFaceIndexAtPoint(vertex);
112 if (faceIdx != nodeA.ParentFaceIndex)
113 continue;
114
115 if (distFromMid < bestScore)
116 {
117 bestScore = distFromMid;
118 bestCandidate = vertex;
119 }
120 }
121
122 return bestCandidate;
123 }

128 public bool ValidateSameFacePath(Point3D p1, Point3D p2)
129 {
130 if (_shellGenerator == null) return true;
131
132 // Sample points along the segment and check if any fall inside the part
133 int samples = 5;
134 for (int i = 1; i < samples; i++)
135 {
136 double t = (double)i / samples;
137 var sample = new Point3D(
138 p1.X + t * (p2.X - p1.X),
139 p1.Y + t * (p2.Y - p1.Y),
140 p1.Z + t * (p2.Z - p1.Z));
141
142 if (_shellGenerator.IsPointInsidePart(sample))
143 {
144 return false; // Collision detected
145 }
146 }
147
148 return true;
149 }
