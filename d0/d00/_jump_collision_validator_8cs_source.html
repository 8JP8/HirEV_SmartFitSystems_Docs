44 public bool ValidateJump(Point3D from, Point3D to, Brep originalPart)
45 {
46 if (!_enableValidation)
47 return true; // Skip validation for performance
48
49 if (originalPart == null)
50 return true; // No part to check against
51
52 // Sample points along jump line
53 for (int i = 1; i < _sampleCount; i++)
54 {
55 double t = (double)i / _sampleCount;
56 var samplePoint = new Point3D(
57 from.X + (to.X - from.X) * t,
58 from.Y + (to.Y - from.Y) * t,
59 from.Z + (to.Z - from.Z) * t);
60
61 // Check if sample point is inside part
62 if (IsPointInsidePart(samplePoint, originalPart))
63 {
64 Debug.WriteLine($"⚠️ JumpCollisionValidator: Collision detected at t={t:F2}");
65 return false; // COLLISION DETECTED
66 }
67 }
68
69 return true; // No collision
70 }

81 public Point3D? TryOffsetJumpOutward(
82 Point3D from,
83 Point3D to,
84 Brep originalPart,
85 Vector3D offsetDirection)
86 {
87 if (!_enableValidation)
88 {
89 // Just return midpoint if validation disabled
90 return new Point3D(
91 (from.X + to.X) / 2,
92 (from.Y + to.Y) / 2,
93 (from.Z + to.Z) / 2);
94 }
95
96 // Normalize offset direction
97 if (offsetDirection.Length < 0.0001)
98 offsetDirection = Vector3D.AxisZ;
99 else
100 offsetDirection.Normalize();
101
102 for (int attempt = 1; attempt <= _maxOffsetAttempts; attempt++)
103 {
104 double offset = _offsetIncrement * attempt;
105
106 var offsetFrom = new Point3D(
107 from.X + offsetDirection.X * offset,
108 from.Y + offsetDirection.Y * offset,
109 from.Z + offsetDirection.Z * offset);
110
111 var offsetTo = new Point3D(
112 to.X + offsetDirection.X * offset,
113 to.Y + offsetDirection.Y * offset,
114 to.Z + offsetDirection.Z * offset);
115
116 if (ValidateJump(offsetFrom, offsetTo, originalPart))
117 {
118 // Return midpoint of collision-free path
119 return new Point3D(
120 (offsetFrom.X + offsetTo.X) / 2,
121 (offsetFrom.Y + offsetTo.Y) / 2,
122 (offsetFrom.Z + offsetTo.Z) / 2);
123 }
124 }
125
126 Debug.WriteLine($"⚠️ JumpCollisionValidator: Could not find collision-free jump after {_maxOffsetAttempts} attempts");
127 return null; // Could not find collision-free jump
128 }

134 private bool IsPointInsidePart(Point3D point, Brep part)
135 {
136 try
137 {
138 // Quick bounding box check
139 var bbox = part.BoxMin;
140 var bboxMax = part.BoxMax;
141
142 if (bbox == null || bboxMax == null)
143 return false;
144
145 // If outside bounding box, definitely not inside part
146 if (point.X < bbox.X || point.X > bboxMax.X ||
147 point.Y < bbox.Y || point.Y > bboxMax.Y ||
148 point.Z < bbox.Z || point.Z > bboxMax.Z)
149 {
150 return false;
151 }
152
153 // Point is inside bounding box - use raycast for more accurate check
154 // Cast ray in +Z direction, count intersections
155 // Odd count = inside, even count = outside
156 var ray = new Segment3D(point, new Point3D(point.X, point.Y, point.Z + 10000));
157
158 int intersections = 0;
159 foreach (var face in part.Faces)
160 {
161 if (face?.Tessellation == null) continue;
162
163 var mesh = face.Tessellation.ConvertToMesh();
164 if (mesh?.Triangles == null) continue;
165
166 foreach (var tri in mesh.Triangles)
167 {
168 if (tri.V1 >= mesh.Vertices.Length ||
169 tri.V2 >= mesh.Vertices.Length ||
170 tri.V3 >= mesh.Vertices.Length)
171 continue;
172
173 var v0 = mesh.Vertices[tri.V1];
174 var v1 = mesh.Vertices[tri.V2];
175 var v2 = mesh.Vertices[tri.V3];
176
177 if (RayIntersectsTriangle(point, Vector3D.AxisZ, v0, v1, v2))
178 {
179 intersections++;
180 }
181 }
182 }
183
184 return (intersections % 2) == 1; // Odd = inside
185 }
186 catch (Exception ex)
187 {
188 Debug.WriteLine($"⚠️ JumpCollisionValidator.IsPointInsidePart error: {ex.Message}");
189 return false; // Assume safe if error
190 }
191 }

196 private bool RayIntersectsTriangle(Point3D origin, Vector3D direction, Point3D v0, Point3D v1, Point3D v2)
197 {
198 const double EPSILON = 0.0000001;
199
200 var edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
201 var edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
202
203 var h = Vector3D.Cross(direction, edge2);
204 double a = edge1.X * h.X + edge1.Y * h.Y + edge1.Z * h.Z;
205
206 if (a > -EPSILON && a < EPSILON)
207 return false;
208
209 double f = 1.0 / a;
210 var s = new Vector3D(origin.X - v0.X, origin.Y - v0.Y, origin.Z - v0.Z);
211 double u = f * (s.X * h.X + s.Y * h.Y + s.Z * h.Z);
212
213 if (u < 0.0 || u > 1.0)
214 return false;
215
216 var q = Vector3D.Cross(s, edge1);
217 double v = f * (direction.X * q.X + direction.Y * q.Y + direction.Z * q.Z);
218
219 if (v < 0.0 || u + v > 1.0)
220 return false;
221
222 double t = f * (edge2.X * q.X + edge2.Y * q.Y + edge2.Z * q.Z);
223
224 return t > EPSILON; // Ray intersection
225 }
