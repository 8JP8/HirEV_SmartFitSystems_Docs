134 public (double AngleA, double AngleB) GetEffectiveWristAngles()
135 {
136 // Explicit angles override approach vector
137 if (Math.Abs(WristAngleA) > 0.001 || Math.Abs(WristAngleB) > 0.001)
138 return (WristAngleA, WristAngleB);
139
140 // Derive from approach vector
141 double nx = TipApproachX;
142 double ny = TipApproachY;
143 double nz = TipApproachZ;
144 double len = Math.Sqrt(nx * nx + ny * ny + nz * nz);
145 if (len < 1e-9)
146 return (0.0, 0.0);
147
148 nx /= len; ny /= len; nz /= len;
149
150 // A = angle from -Z axis: cos(A) = -nz → A = acos(clamp(-nz, -1, 1))
151 double angleA = Math.Acos(Math.Max(-1.0, Math.Min(1.0, -nz))) * 180.0 / Math.PI;
152
153 // B = atan2 of lateral components
154 double angleB = (Math.Abs(nx) < 1e-9 && Math.Abs(ny) < 1e-9)
155 ? 0.0
156 : Math.Atan2(ny, nx) * 180.0 / Math.PI;
157
158 return (angleA, angleB);
159 }
