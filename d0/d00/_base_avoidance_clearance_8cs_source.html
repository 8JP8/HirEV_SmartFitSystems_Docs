144 public static bool IsUnsafe(Point3D point, Vector3D vector, BaseAvoidanceConfig config, double baseZ, bool isFaceNormal = false)
145 {
146 // Points genuinely below the physical base can never be reached safely.
147 // A point exactly at base height is intentionally evaluated by the
148 // configured angular rule: SafeAngleDeg applies at height zero and then
149 // tapers to zero at SafeHeight.
150 if (point.Z < baseZ - ZTolerance)
151 return true;
152
153 if (!config.Enabled)
154 return false;
155
156 double localHeight = point.Z - baseZ;
157 return IsLocalClearanceUnsafe(localHeight, vector, config);
158 }

284 public static double GetMinimumZ(Vector3D approachVector, BaseAvoidanceConfig config, double baseZ)
285 {
286 if (!config.Enabled || config.SafeHeight <= 0.0 || config.SafeAngleDeg <= 0.0)
287 return baseZ;
288
289 if (IsDownwardOrLevel(approachVector))
290 return baseZ + config.SafeHeight;
291
292 double angleToBasePlane = GetAngleToBasePlaneDeg(approachVector);
293 if (angleToBasePlane >= config.SafeAngleDeg - 1e-9)
294 return baseZ;
295
296 double clearanceRatio = 1.0 - (angleToBasePlane / Math.Max(1e-9, config.SafeAngleDeg));
297 return baseZ + (config.SafeHeight * Math.Max(0.0, Math.Min(1.0, clearanceRatio)));
298 }

306 public static double GetInclinationFromVerticalDeg(Vector3D approachVector, double fallbackAngleDeg)
307 {
308 if (approachVector.Length < 1e-9)
309 return fallbackAngleDeg;
310
311 var normalized = new Vector3D(approachVector.X, approachVector.Y, approachVector.Z);
312 normalized.Normalize();
313 double absZ = Math.Abs(normalized.Z);
314 absZ = Math.Max(0.0, Math.Min(1.0, absZ));
315 return Math.Acos(absZ) * 180.0 / Math.PI;
316 }

323 public static double GetAngleToBasePlaneDeg(Vector3D approachVector)
324 {
325 if (approachVector.Length < 1e-9)
326 return 0.0;
327
328 var normalized = new Vector3D(approachVector.X, approachVector.Y, approachVector.Z);
329 normalized.Normalize();
330 double absZ = Math.Abs(normalized.Z);
331 absZ = Math.Max(0.0, Math.Min(1.0, absZ));
332 return Math.Asin(absZ) * 180.0 / Math.PI;
333 }

342 private static bool IsLocalClearanceUnsafe(double localHeight, Vector3D approachVector, BaseAvoidanceConfig config)
343 {
344 if (!config.Enabled || config.SafeHeight <= 0.0 || config.SafeAngleDeg <= 0.0)
345 return false;
346
347 if (localHeight >= config.SafeHeight - ZTolerance)
348 return false;
349
350 // The vector points from the measurement target to its associated
351 // approach point. Below the clearance plane it may not point downward;
352 // horizontal motion becomes progressively acceptable as the measurement
353 // height approaches the configured plane.
354 if (approachVector.Length < 1e-9 || approachVector.Z < -ZTolerance)
355 return true;
356
357 double clampedHeight = Math.Max(0.0, Math.Min(config.SafeHeight, localHeight));
358 double requiredMinAngle = config.SafeAngleDeg * (1.0 - (clampedHeight / config.SafeHeight));
359 requiredMinAngle = Math.Max(0.0, Math.Min(config.SafeAngleDeg, requiredMinAngle));
360 return GetAngleToBasePlaneDeg(approachVector) < requiredMinAngle - 1e-9;
361 }
