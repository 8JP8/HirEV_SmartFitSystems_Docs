141 public void OK()
142 {
143 if (Password != null)
144 {
145 if (Tries < Max_Tries)
146 {
147 Tries++;
148
149 (bool isCorrect, string message) = SecurePasswordManager.Access(Password); // Retrieve the password
150 switch (message)
151 {
152 case "Access Blocked: The password file has been deleted (Contact the Administrator.)":
153 message = GetXMLElementValue(DocLanguage, "ErrorMessage_NoPasswordFile") ?? message;
154 break;
155 case "Access Denied: Incorrect Password":
156 message = GetXMLElementValue(DocLanguage, "ErrorMessage_IncorrectPassword") ?? message;
157 break;
158 case "Admin Access Granted":
159 message = GetXMLElementValue(DocLanguage, "AdminAccessGranted") ?? message;
160 break;
161 }
162 PasswordCheck_Message = message;
163 if (isCorrect) { PasswordCheck_MessageColor = Brushes.Green; DialogResult = true; return; }
164 else { PasswordCheck_MessageColor = Brushes.Red; }
165 }
166 else DialogResult = false;
167 }
168 else
169 {
170 PasswordCheck_Message = GetXMLElementValue(DocLanguage, "ErrorMessage_NoPassword") ?? "Error: Enter a Password";
171 PasswordCheck_MessageColor = Brushes.Red;
172 }
173 }
174
177 private void Cancel()
178 {
179 DialogResult = false; // Close the dialog without a result
180 }

184 public void ChangePassword()
185 {
186 if (Password != null && NewPassword != null)
187 {
188 if (Tries < Max_Tries)
189 {
190 Tries++;
191
192 (bool isCorrect, string message) = SecurePasswordManager.Access(Password); // Validate current password
193 switch (message)
194 {
195 case "Access Blocked: The password file has been deleted (Contact the Administrator.)":
196 PasswordCheck_Message = GetXMLElementValue(DocLanguage, "ErrorMessage_NoPasswordFile") ?? message;
197 PasswordCheck_MessageColor = Brushes.Red;
198 return;
199 case "Access Denied: Incorrect Password":
200 PasswordCheck_Message = GetXMLElementValue(DocLanguage, "ErrorMessage_IncorrectPassword") ?? message;
201 PasswordCheck_MessageColor = Brushes.Red;
202 return;
203 case "Admin Access Granted":
204 PasswordCheck_Message = GetXMLElementValue(DocLanguage, "AdminAccessGranted") ?? message;
205 PasswordCheck_MessageColor = Brushes.Green;
206 break;
207 }
208
209 // Call the static method to change the password
210 string changeMessage = SecurePasswordManager.ChangePassword(Password, NewPassword);
211 PasswordCheck_MessageColor = changeMessage.Contains("successfully") ? Brushes.Green : Brushes.Red;
212 PasswordCheck_Message = changeMessage.Contains("successfully") ? GetXMLElementValue(DocLanguage, "ErrorMessage_PasswordChangeSuccessful") : GetXMLElementValue(DocLanguage, "ErrorMessage_PasswordChangeFailed");
213 }
214 else
215 {
216 PasswordCheck_Message = GetXMLElementValue(DocLanguage, "ErrorMessage_MaxTriesReached") ?? "Access Denied: Maximum attempts reached.";
217 PasswordCheck_MessageColor = Brushes.Red;
218 DialogResult = false;
219 }
220 }
221 else
222 {
223 PasswordCheck_Message = GetXMLElementValue(DocLanguage, "ErrorMessage_NoPassword") ?? "Error: Enter a Password";
224 PasswordCheck_MessageColor = Brushes.Red;
225 }
226 }
