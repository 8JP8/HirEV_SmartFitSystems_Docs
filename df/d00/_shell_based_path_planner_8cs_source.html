164 public List<ApproachMeasurementPair> GenerateApproachPoints(
165 List<(Point3D point, Vector3D normal, CADFace face, int faceIndex)> measurementPoints,
166 Action<int, string>? progressCallback = null,
167 System.Threading.CancellationToken cancellationToken = default,
168 ISet<int>? explicitHoleFaceIndices = null)
169 {
170 progressCallback?.Invoke(0, "Progress_GeneratingApproachPoints");
171 var input = measurementPoints.Select(m => (m.point, m.normal, m.face)).ToList();
172 var audit = measurementPoints
173 .GroupBy(item => item.face.FaceGlobalIndex)
174 .ToDictionary(
175 group => group.Key,
176 group => new ApproachGenerationAudit
177 {
178 FaceGlobalIndex = group.Key,
179 InputCount = group.Count()
180 });
181 var pairs = _approachGenerator.GenerateApproachPoints(input, new HashSet<int>(), new HashSet<int>(), cancellationToken: cancellationToken);
182 var markedHoleFaces = explicitHoleFaceIndices ?? new HashSet<int>();
183
184 // Cylindrical and conical geometry alone does not make a feature a hole.
185 // The caller's explicit face classification is the single source of truth
186 // for the special hole sequence and the base-avoidance exemption.
187 foreach (var pair in pairs)
188 {
189 pair.IsHoleMeasurement = markedHoleFaces.Contains(pair.ParentFace.FaceGlobalIndex);
190 if (audit.TryGetValue(pair.ParentFace.FaceGlobalIndex, out var faceAudit))
191 faceAudit.GeneratedCount++;
192 }
193
194 var validPairs = new List<ApproachMeasurementPair>();
195 _pointToNormalMap.Clear();
196
197 // Threshold for inner collision: half of the offset distance.
198 double minAllowedSdf = _approachOffsetDistance / 2.0;
199
200 int total = pairs.Count;
201 for (int i = 0; i < total; i++)
202 {
203 cancellationToken.ThrowIfCancellationRequested();
204 var pair = pairs[i];
205 if (i % 10 == 0 || i == total - 1)
206 {
207 int perc = (int)((double)i / total * 100);
208 progressCallback?.Invoke(perc, "Progress_CheckingApproachPointCollisions");
209 }
210
211 // Check if the generated approach point is too close to the part or inside it.
212 double sdfDist = _shellGenerator.GetDistance(pair.ApproachPoint);
213
214 bool isHole = IsHolePair(pair);
215
216 // Regular surface approaches must remain in the outer half of the
217 // offset shell. Explicit hole points are planned independently on
218 // their axis and are the only exemption.
219 if (!isHole &&
220 ((sdfDist != double.MaxValue && sdfDist < minAllowedSdf) ||
221 IsInsidePartSolid(pair.ApproachPoint, pair.ParentFace.EntityIndex, cancellationToken)))
222 {
223 string ignoredFormat = GetXMLElement(DocLanguage, "PathManager_IgnoredApproachPoint", " ⚠️ Ignored approach point for face {0}: SDF dist {1:F2} < {2:F2} (potential collision)");
224 Debug.WriteLine(string.Format(ignoredFormat, pair.ParentFace.FaceGlobalIndex, sdfDist, minAllowedSdf));
225 if (audit.TryGetValue(pair.ParentFace.FaceGlobalIndex, out var sdfAudit))
226 sdfAudit.SdfRejectedCount++;
227 continue;
228 }
229
230 // Hole features are deliberately exempt: their entry/axis/radial/exit
231 // sequence is planned independently and must not be removed by the
232 // bottom-face rule.
233 if (IsPairBelowBaseAvoidanceClearance(pair))
234 {
235 Debug.WriteLine(
236 $" Ignored approach/measurement pair for face {pair.ParentFace.FaceGlobalIndex}: " +
237 "base avoidance clearance failed.");
238 if (audit.TryGetValue(pair.ParentFace.FaceGlobalIndex, out var baseAudit))
239 baseAudit.BaseRejectedCount++;
240 continue;
241 }
242
243 if (!_pointToNormalMap.ContainsKey(pair.ApproachPoint))
244 _pointToNormalMap[pair.ApproachPoint] = pair.Normal;
245 validPairs.Add(pair);
246 if (audit.TryGetValue(pair.ParentFace.FaceGlobalIndex, out var acceptedAudit))
247 acceptedAudit.AcceptedCount++;
248 }
249 LastApproachGenerationAudit = audit;
250 return validPairs;
251 }

279 public (Point3D ShellTop, Point3D AirTop) GetSafeTopPoints(int entityIndex, List<ApproachMeasurementPair>? allPairs = null, System.Threading.CancellationToken cancellationToken = default)
280 {
281 Point3D shellTop = Point3D.Origin;
282 double cx = 0, cy = 0;
283 double maxZ = double.MinValue;
284 bool foundPoints = false;
285
286 if (allPairs != null)
287 {
288 var points = allPairs
289 .Where(p =>
290 p.ParentFace.EntityIndex == entityIndex &&
291 !IsHolePair(p) &&
292 IsOuterHalfShellSafe(p.ApproachPoint, entityIndex, cancellationToken))
293 .Select(p => p.ApproachPoint)
294 .ToList();
295 if (points.Count > 0)
296 {
297 shellTop = points.OrderByDescending(p => p.Z).First();
298 cx = points.Average(p => p.X);
299 cy = points.Average(p => p.Y);
300 maxZ = points.Max(p => p.Z);
301 foundPoints = true;
302 }
303 }
304
305 if (_entityIndexToBrepMap != null && _entityIndexToBrepMap.TryGetValue(entityIndex, out var targetBrep))
306 {
307 if (targetBrep.BoxMin == null || targetBrep.BoxMax == null) targetBrep.Regen(0.01);
308 var bMin = targetBrep.BoxMin;
309 var bMax = targetBrep.BoxMax;
310 if (bMin != null && bMax != null)
311 {
312 var airTop = new Point3D((bMin.X + bMax.X) * 0.5, (bMin.Y + bMax.Y) * 0.5, bMax.Z + _approachOffsetDistance * 2);
313 if (!foundPoints) shellTop = new Point3D(airTop.X, airTop.Y, bMax.Z);
314 return (shellTop, airTop);
315 }
316 }
317
318 foreach (var brep in _partBreps)
319 {
320 if (brep.BoxMin == null || brep.BoxMax == null) brep.Regen(0.01);
321 var bMin = brep.BoxMin;
322 var bMax = brep.BoxMax;
323 if (bMin != null && bMax != null)
324 {
325 if (!foundPoints || (cx >= bMin.X - 1 && cx <= bMax.X + 1 &&
326 cy >= bMin.Y - 1 && cy <= bMax.Y + 1))
327 {
328 var airTop = new Point3D((bMin.X + bMax.X) * 0.5, (bMin.Y + bMax.Y) * 0.5, bMax.Z + _approachOffsetDistance * 2);
329 if (!foundPoints) shellTop = new Point3D(airTop.X, airTop.Y, bMax.Z);
330 return (shellTop, airTop);
331 }
332 }
333 }
334
335 var fallbackAir = new Point3D(cx, cy, maxZ + _approachOffsetDistance * 2);
336 if (!foundPoints) shellTop = new Point3D(cx, cy, maxZ);
337 return (shellTop, fallbackAir);
338 }

347 public Point3D GetClosestPartShellPoint(int entityIndex, Point3D airTop, List<ApproachMeasurementPair> allPairs, System.Threading.CancellationToken cancellationToken = default)
348 {
349 var partPairs = allPairs
350 .Where(p =>
351 p.ParentFace.EntityIndex == entityIndex &&
352 !IsHolePair(p) &&
353 IsOuterHalfShellSafe(p.ApproachPoint, entityIndex, cancellationToken))
354 .ToList();
355 if (partPairs.Count > 0)
356 {
357 Point3D closest = partPairs[0].ApproachPoint;
358 double minDist = closest.DistanceTo(airTop);
359 foreach (var pair in partPairs)
360 {
361 double dist = pair.ApproachPoint.DistanceTo(airTop);
362 if (dist < minDist)
363 {
364 minDist = dist;
365 closest = pair.ApproachPoint;
366 }
367 }
368 return closest;
369 }
370 Point3D? projectedShellPoint = _shellGenerator.FindNearestShellPoint(airTop);
371 if (projectedShellPoint != null && IsOuterHalfShellSafe(projectedShellPoint, entityIndex, cancellationToken))
372 return projectedShellPoint;
373
374 // Fallback to ShellTop
375 var tops = GetSafeTopPoints(entityIndex, allPairs);
376 return tops.ShellTop;
377 }

388 public List<(Point3D Point, Vector3D Normal, int EntityIndex)> GeneratePath(
389 List<ApproachMeasurementPair> pairs,
390 Dictionary<CADFace, Color> colorGroups,
391 HashSet<int> holeFaceIndices,
392 Action<int, int, string>? progressCallback = null,
393 System.Threading.CancellationToken cancellationToken = default,
394 IReadOnlyCollection<CADHole>? explicitHoles = null)
395 {
396 if (pairs == null || pairs.Count == 0) return new List<(Point3D, Vector3D, int)>();
397
398 progressCallback?.Invoke(5, 100, "Initializing path data...");
399
400 var pathPoints = new List<(Point3D Point, Vector3D Normal, int EntityIndex)>();
401
402 // PHASE 1: Auto-detect holes
403 progressCallback?.Invoke(15, 100, "Detecting hole features...");
404 var holes = BuildExplicitHoleFeatures(pairs, holeFaceIndices, explicitHoles);
405
406 // PHASE 2: Identify bridge faces (between holes)
407 progressCallback?.Invoke(25, 100, "Analyzing topological bridges...");
408 var separatingFaces = new HashSet<int>();
409 var specialPlanarFaces = new Dictionary<int, HoleFeature>();
410
411 var allPlanarFaces = pairs.Where(p => p.ParentFace.FaceType == FaceTypeEnum.Planar)
412 .Select(p => p.ParentFace)
413 .Distinct()
414 .ToList();
415
416 foreach (var face in allPlanarFaces)
417 {
418 cancellationToken.ThrowIfCancellationRequested();
419 var faceCenter = EyeshotGeometryUtils.GetFaceCenter(face);
420 var faceNormal = EyeshotGeometryUtils.GetFaceAxis(face);
421
422 HoleFeature? holeAbove = null;
423 foreach (var hole in holes)
424 {
425 if (hole.Axis == null || hole.Center == null) continue;
426 double distToAxis = DistancePointToLine(faceCenter, hole.Center, hole.Axis);
427 double alignmentDot = Math.Abs(EyeshotGeometryUtils.DotProduct(faceNormal, hole.Axis));
428
429 if (distToAxis < hole.Radius * 1.8 && alignmentDot > 0.85)
430 {
431 double distToTop = faceCenter.DistanceTo(hole.TopCenterPoint ?? Point3D.Origin);
432 double distToBottom = faceCenter.DistanceTo(hole.BottomCenterPoint ?? Point3D.Origin);
433
434 if (distToTop < 6.0 || distToBottom < 6.0)
435 {
436 Vector3D faceToHole = new Vector3D(faceCenter, hole.Center);
437 if (EyeshotGeometryUtils.DotProduct(faceToHole, faceNormal) > 0.5)
438 {
439 holeAbove = hole;
440 break;
441 }
442 }
443 }
444 }
445
446 if (holeAbove != null)
447 {
448 separatingFaces.Add(face.FaceGlobalIndex);
449 specialPlanarFaces[face.FaceGlobalIndex] = holeAbove;
450 }
451 }
452
453 // PHASE 3: Keep radial hole measurements away from every bore limit.
454 // Internal limits between coaxial constituents receive the larger bridge
455 // clearance. The explicit hole feature itself is never removed, so its
456 // shell-to-axis access route remains available even when no radial sample
457 // can be retained safely.
458 progressCallback?.Invoke(40, 100, "Filtering measurement points...");
459 foreach (var hole in holes)
460 {
461 cancellationToken.ThrowIfCancellationRequested();
462 int originalPairCount = hole.Pairs.Count;
463 int removedPairCount = FilterHolePairsByEndClearance(
464 hole,
465 pairs,
466 separatingFaces);
467
468 if (removedPairCount > 0)
469 {
470 _holeRoutingLog?.Invoke(
471 string.Format(
472 GetXMLElement(
473 DocLanguage,
474 "PathManager_HoleRadialClearanceFiltered",
475 "Hole {0}: retained {1}/{2} radial measurements after end/bridge clearance filtering; axis access was preserved."),
476 hole.HoleId,
477 hole.Pairs.Count,
478 originalPairCount),
479 hole.Pairs.Count == 0);
480 }
481 }
482
483 // PHASE 4: Group SURFACE pairs by color
484 progressCallback?.Invoke(55, 100, "Grouping by color...");
485 var surfacePairs = pairs.Where(p => !IsHolePair(p) && !separatingFaces.Contains(p.ParentFace.FaceGlobalIndex)).ToList();
486
487 var atomicChains = new Dictionary<Color, List<List<ApproachMeasurementPair>>>();
488 var unignoredSpecialPairs = pairs.Where(p => specialPlanarFaces.ContainsKey(p.ParentFace.FaceGlobalIndex) && !p.ParentFace.IsIgnored).ToList();
489
490 foreach (var group in unignoredSpecialPairs.GroupBy(p => p.ParentFace.FaceGlobalIndex))
491 {
492 cancellationToken.ThrowIfCancellationRequested();
493 var face = group.First().ParentFace;
494 var holeAbove = specialPlanarFaces[face.FaceGlobalIndex];
495 var faceNormal = EyeshotGeometryUtils.GetFaceAxis(face);
496
497 var topCenter = holeAbove.TopCenterPoint ?? (holeAbove.Center ?? Point3D.Origin);
498 var bottomCenter = holeAbove.BottomCenterPoint ?? (holeAbove.Center ?? Point3D.Origin);
499
500 if (bottomCenter.Z > topCenter.Z)
501 {
502 var temp = topCenter;
503 topCenter = bottomCenter;
504 bottomCenter = temp;
505 }
506
507 var axis = holeAbove.Axis ?? Vector3D.AxisZ;
508 var topToBottom = new Vector3D(topCenter, bottomCenter);
509 if (topToBottom.Length > 0.001 && Vector3D.Dot(axis, topToBottom) < 0.0)
510 axis = axis * -1.0;
511 if (axis.Z > 0)
512 axis = axis * -1.0;
513
514 var (topShell, _, _, _) = FindHoleAxisShellIntersections(topCenter, bottomCenter, axis);
515 var holeTopAxis = topShell;
516 var center = holeAbove.Center ?? Point3D.Origin;
517
518 Color faceColor = colorGroups.TryGetValue(face, out var c) ? c : Color.Gray;
519 if (!atomicChains.ContainsKey(faceColor)) atomicChains[faceColor] = new List<List<ApproachMeasurementPair>>();
520
521 // Find distances to axis for all points in this face
522 var pointDistances = group.Select(p => new { Point = p, Dist = DistancePointToLine(p.MeasurementPoint, center, axis) }).ToList();
523 if (!pointDistances.Any()) continue;
524
525 // Only keep the inner circumference (ignore the outer ring adjacent to the bigger hole wall)
526 double minDist = pointDistances.Min(pd => pd.Dist);
527 var innerPoints = pointDistances.Where(pd => pd.Dist < minDist + 2.0).Select(pd => pd.Point).ToList();
528
529 if (!innerPoints.Any()) continue;
530
531 // Single atomic sequence for all points in the inner ring
532 var chain = new List<ApproachMeasurementPair>();
533
534 // 1. Entry at top of bigger hole axis
535 chain.Add(new ApproachMeasurementPair(holeTopAxis, holeTopAxis, faceNormal, face, true));
536
537 // 2. Move down to center of hole (closest point on axis to the first approach point)
538 var firstPt = innerPoints.First().MeasurementPoint;
539 double t = Vector3D.Dot(new Vector3D(center, firstPt), axis);
540 Point3D axisPt = center + axis * t;
541 chain.Add(new ApproachMeasurementPair(axisPt, axisPt, faceNormal, face, true));
542
543 // 3. Measure all inner points
544 foreach (var p in innerPoints)
545 {
546 // Ensure approach points are offset in the normal direction (outward from face)
547 p.ApproachPoint = p.MeasurementPoint + faceNormal * 2.0;
548 chain.Add(p);
549 }
550
551 // 4. Exit back to axis center
552 var lastPt = innerPoints.Last().MeasurementPoint;
553 double tLast = Vector3D.Dot(new Vector3D(center, lastPt), axis);
554 Point3D axisPtLast = center + axis * tLast;
555 chain.Add(new ApproachMeasurementPair(axisPtLast, axisPtLast, faceNormal, face, true));
556
557 // 5. Exit back out of hole
558 chain.Add(new ApproachMeasurementPair(holeTopAxis, holeTopAxis, faceNormal, face, true));
559
560 atomicChains[faceColor].Add(chain);
561 }
562
563 progressCallback?.Invoke(70, 100, "Optimizing path sequence...");
564 var groupedPairs = GroupByColor(surfacePairs, colorGroups);
565 Point3D? lastApproach = null;
566
567 int globalLastEntityIndex = -1;
568 int globalLastFaceIndex = -1;
569
570 foreach (var color in colorGroups.Values.Distinct())
571 {
572 cancellationToken.ThrowIfCancellationRequested();
573 var standardPairs = surfacePairs.Where(p => colorGroups.TryGetValue(p.ParentFace, out var c) && c == color).ToList();
574 int lastEntityIndex = globalLastEntityIndex;
575
576 var pairsByPart = standardPairs.GroupBy(p => p.ParentFace.EntityIndex).ToList();
577
578 foreach (var partGroup in pairsByPart)
579 {
580 int currentEntityIndex = partGroup.Key;
581
582 foreach (var current in OrderByProximity(partGroup.ToList()))
583 {
584 cancellationToken.ThrowIfCancellationRequested();
585 // If jumping between different parts, insert safe top points
586 if (lastApproach != null && lastEntityIndex != -1 && currentEntityIndex != lastEntityIndex && _partBreps.Count > 1)
587 {
588 var lastTops = GetSafeTopPoints(lastEntityIndex, pairs);
589 var currentTops = GetSafeTopPoints(currentEntityIndex, pairs);
590
591 Point3D closestShellPtA = GetClosestPartShellPoint(lastEntityIndex, lastTops.AirTop, pairs);
592 Point3D closestShellPtB = GetClosestPartShellPoint(currentEntityIndex, currentTops.AirTop, pairs);
593
594 // 1. Move to closest shell point on A
595 ProcessTransition(lastApproach, closestShellPtA, Vector3D.AxisZ, lastEntityIndex, pathPoints, cancellationToken);
596 pathPoints.Add((closestShellPtA, Vector3D.AxisZ, lastEntityIndex));
597 pathPoints.Add((lastTops.AirTop, Vector3D.AxisZ, lastEntityIndex));
598
599 // 2. Move to Top Point of the current part
600 pathPoints.Add((currentTops.AirTop, Vector3D.AxisZ, currentEntityIndex));
601 pathPoints.Add((closestShellPtB, Vector3D.AxisZ, currentEntityIndex));
602
603 // 3. Update lastApproach so ProcessPair transitions from closestShellPtB
604 lastApproach = closestShellPtB;
605 globalLastFaceIndex = -1;
606 }
607
608 ProcessPair(current, ref lastApproach, ref globalLastFaceIndex, pathPoints, cancellationToken);
609 lastEntityIndex = currentEntityIndex;
610 globalLastEntityIndex = currentEntityIndex;
611 }
612 }
613
614 if (atomicChains.TryGetValue(color, out var chains))
615 {
616 foreach (var chain in chains)
617 {
618 for (int i = 0; i < chain.Count; i++)
619 {
620 cancellationToken.ThrowIfCancellationRequested();
621 var current = chain[i];
622 int currentEntityIndex = current.ParentFace.EntityIndex;
623
624 // Handle part jumps even when entering atomic chains (e.g. holes)
625 if (i == 0 && lastApproach != null && lastEntityIndex != -1 && currentEntityIndex != lastEntityIndex && _partBreps.Count > 1)
626 {
627 var lastTops = GetSafeTopPoints(lastEntityIndex, pairs);
628 var currentTops = GetSafeTopPoints(currentEntityIndex, pairs);
629
630 Point3D closestShellPtA = GetClosestPartShellPoint(lastEntityIndex, lastTops.AirTop, pairs);
631 Point3D closestShellPtB = GetClosestPartShellPoint(currentEntityIndex, currentTops.AirTop, pairs);
632
633 ProcessTransition(lastApproach, closestShellPtA, Vector3D.AxisZ, lastEntityIndex, pathPoints, cancellationToken);
634 pathPoints.Add((closestShellPtA, Vector3D.AxisZ, lastEntityIndex));
635 pathPoints.Add((lastTops.AirTop, Vector3D.AxisZ, lastEntityIndex));
636 pathPoints.Add((currentTops.AirTop, Vector3D.AxisZ, currentEntityIndex));
637 pathPoints.Add((closestShellPtB, Vector3D.AxisZ, currentEntityIndex));
638
639 lastApproach = closestShellPtB;
640 globalLastFaceIndex = -1;
641 }
642
643 if (i == 0) ProcessPair(current, ref lastApproach, ref globalLastFaceIndex, pathPoints, cancellationToken);
644 else
645 {
646 if (current.IsHoleMeasurement && current.MeasurementPoint.Equals(current.ApproachPoint))
647 pathPoints.Add((current.MeasurementPoint, current.Normal, currentEntityIndex));
648 else
649 {
650 pathPoints.Add((current.ApproachPoint, current.Normal, currentEntityIndex));
651 pathPoints.Add((current.MeasurementPoint, current.Normal, currentEntityIndex));
652 pathPoints.Add((current.ApproachPoint, current.Normal, currentEntityIndex));
653 }
654 lastApproach = current.ApproachPoint;
655 }
656
657 lastEntityIndex = currentEntityIndex;
658 globalLastEntityIndex = currentEntityIndex;
659 }
660 }
661 }
662 }
663
664 // 2. Process explicit hole features. Access and transit are constrained
665 // to the actual CADHole axis; only the radial measurement excursion may
666 // leave that line.
667 var holesByPart = holes
668 .GroupBy(hole => hole.EntityIndex)
669 .ToList();
670 int holeLastEntityIndex = globalLastEntityIndex;
671
672 foreach (var partHoles in holesByPart)
673 {
674 int currentEntityIndex = partHoles.Key;
675
676 foreach (var hole in partHoles)
677 {
678 cancellationToken.ThrowIfCancellationRequested();
679 var axis = NormalizeOrDefault(hole.Axis ?? Vector3D.AxisZ, Vector3D.AxisZ);
680 var topCenter = hole.TopCenterPoint ?? (hole.Center ?? Point3D.Origin);
681 var bottomCenter = hole.BottomCenterPoint ?? (hole.Center ?? Point3D.Origin);
682
683 // Ensure topCenter is physically above bottomCenter in Z
684 if (bottomCenter.Z > topCenter.Z)
685 {
686 var temp = topCenter;
687 topCenter = bottomCenter;
688 bottomCenter = temp;
689 }
690
691 // CADHole.AxisDirection is defined from top to bottom. Correct
692 // imported data that disagrees with the actual endpoints.
693 var topToBottom = new Vector3D(topCenter, bottomCenter);
694 if (topToBottom.Length > 0.001 && Vector3D.Dot(axis, topToBottom) < 0.0)
695 axis = axis * -1.0;
696
697 // Ensure axis always points generally downwards
698 if (axis.Z > 0)
699 axis = axis * -1.0;
700
701 bool bottomAccessAllowed = !_baseAvoidanceConfig.Enabled;
702 var (topShell, bottomShell, usedTopFallback, usedBottomFallback) =
703 FindHoleAxisShellIntersections(topCenter, bottomCenter, axis);
704
705 if (usedTopFallback || (bottomAccessAllowed && usedBottomFallback))
706 {
707 _holeRoutingLog?.Invoke(
708 string.Format(
709 GetXMLElement(DocLanguage, "PathManager_HoleAxisIntersectionFallback",
710 "Hole {0}: an exact shell-axis intersection was unavailable; an on-axis clearance fallback was used."),
711 hole.HoleId),
712 true);
713 }
714
715 // If jumping between different parts to process a hole, insert safe top points
716 if (lastApproach != null && holeLastEntityIndex != -1 && currentEntityIndex != holeLastEntityIndex && _partBreps.Count > 1)
717 {
718 var lastTops = GetSafeTopPoints(holeLastEntityIndex, pairs);
719 var currentTops = GetSafeTopPoints(currentEntityIndex, pairs);
720
721 Point3D closestShellPtA = GetClosestPartShellPoint(holeLastEntityIndex, lastTops.AirTop, pairs);
722 Point3D closestShellPtB = GetClosestPartShellPoint(currentEntityIndex, currentTops.AirTop, pairs);
723
724 ProcessTransition(lastApproach, closestShellPtA, Vector3D.AxisZ, holeLastEntityIndex, pathPoints, cancellationToken);
725 pathPoints.Add((closestShellPtA, Vector3D.AxisZ, holeLastEntityIndex));
726 pathPoints.Add((lastTops.AirTop, Vector3D.AxisZ, holeLastEntityIndex));
727 pathPoints.Add((currentTops.AirTop, Vector3D.AxisZ, currentEntityIndex));
728 pathPoints.Add((closestShellPtB, Vector3D.AxisZ, currentEntityIndex));
729
730 lastApproach = closestShellPtB;
731 }
732
733 var candidate = SelectHoleRoute(
734 hole,
735 topShell,
736 bottomShell,
737 axis,
738 lastApproach,
739 bottomAccessAllowed);
740
741 if (candidate.CollisionCount > 0)
742 {
743 _holeRoutingLog?.Invoke(
744 string.Format(
745 GetXMLElement(DocLanguage, "PathManager_HoleRouteCollisionFallback",
746 "Hole {0}: all axis-only routes collide; retained the best route with {1} collision(s)."),
747 hole.HoleId,
748 candidate.CollisionCount),
749 true);
750 }
751 else
752 {
753 _holeRoutingLog?.Invoke(
754 string.Format(
755 GetXMLElement(DocLanguage, "PathManager_HoleRouteSelected",
756 "Hole {0}: selected {1} axis route."),
757 hole.HoleId,
758 candidate.Name),
759 false);
760 }
761
762 ProcessTransition(lastApproach, candidate.Entry, Vector3D.AxisZ, currentEntityIndex, pathPoints, cancellationToken);
763 foreach (var point in candidate.Points)
764 AddVerticalHoleWaypoint(pathPoints, point, currentEntityIndex);
765
766 lastApproach = candidate.Exit;
767 holeLastEntityIndex = currentEntityIndex;
768 }
769 }
770
771 // Move to safe top point after the very last measurement
772 if (lastApproach != null && pairs.Count > 0)
773 {
774 int finalEntityIndex = holeLastEntityIndex != -1 ? holeLastEntityIndex : globalLastEntityIndex;
775 if (finalEntityIndex == -1) finalEntityIndex = pairs.Last().ParentFace.EntityIndex;
776
777 var tops = GetSafeTopPoints(finalEntityIndex, pairs);
778 Point3D closestShellPt = GetClosestPartShellPoint(finalEntityIndex, tops.AirTop, pairs);
779 ProcessTransition(lastApproach, closestShellPt, Vector3D.AxisZ, finalEntityIndex, pathPoints, cancellationToken);
780 pathPoints.Add((closestShellPt, Vector3D.AxisZ, finalEntityIndex));
781 if (_partBreps.Count > 1)
782 {
783 pathPoints.Add((tops.AirTop, Vector3D.AxisZ, finalEntityIndex));
784 lastApproach = tops.AirTop;
785 }
786 else
787 {
788 lastApproach = closestShellPt;
789 }
790 }
791
792 int purgedWaypoints = PurgeUnsafeGeneratedWaypoints(pathPoints, pairs, cancellationToken);
793 if (purgedWaypoints > 0)
794 {
795 string purgeMessage = string.Format(
796 GetXMLElement(
797 DocLanguage,
798 "PathManager_PurgedUnsafeGeneratedWaypoints",
799 "Purged {0} generated waypoint(s) inside the inner half-shell."),
800 purgedWaypoints);
801 Debug.WriteLine($" {purgeMessage}");
802 _holeRoutingLog?.Invoke(purgeMessage, true);
803 }
804
805 return pathPoints;

870 public List<Point3D> PlanShellRoute(Point3D start, Point3D end, System.Threading.CancellationToken cancellationToken = default)
871 {
872 if (_macroNavigator != null)
873 {
874 var macroRoute = _macroNavigator.FindShortestPath(start, end, cancellationToken: cancellationToken);
875 if (macroRoute != null && macroRoute.Count > 0)
876 return macroRoute;
877 }
878
879 if (_navigator != null)
880 {
881 var meshRoute = _navigator.FindShortestPath(start, end, cancellationToken: cancellationToken);
882 if (meshRoute != null && meshRoute.Count > 0)
883 return meshRoute;
884 }
885
886 // FALLBACK: Sample direct line and project onto SDF shell surface
887 var fallbackRoute = new List<Point3D> { start };
888 double dist = start.DistanceTo(end);
889 double maxInterp = 2.0;
890 if (double.TryParse(System.Configuration.ConfigurationManager.AppSettings["MaxInterpolationDistanceInNonPlanarSurfaces"], System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out double parsedVal))
891 maxInterp = parsedVal;
892 int samples = (int)Math.Ceiling(dist / maxInterp); // Use configured interpolation step
893
894 if (samples > 1)
895 {
896 for (int i = 1; i < samples; i++)
897 {
898 cancellationToken.ThrowIfCancellationRequested();
899 double t = (double)i / samples;
900 Point3D interp = new Point3D(
901 start.X + (end.X - start.X) * t,
902 start.Y + (end.Y - start.Y) * t,
903 start.Z + (end.Z - start.Z) * t);
904
905 Point3D? projected = _shellGenerator.FindNearestShellPoint(interp);
906 if (projected != null)
907 {
908 if (fallbackRoute.Last().DistanceTo(projected) > 1.0)
909 fallbackRoute.Add(projected);
910 }
911 }
912 }
913 fallbackRoute.Add(end);
914
915 return fallbackRoute;
916 }

1096 private bool TryGetNearestPartSurfaceNormal(Point3D point, int entityIndex, Vector3D fallbackNormal, out Vector3D normal)
1097 {
1098 normal = fallbackNormal;
1099 if (!TryGetTargetBrep(entityIndex, out var brep) || brep?.Faces == null)
1100 return false;
1101
1102 double bestDistance = double.MaxValue;
1103 Point3D bestSurfacePoint = Point3D.Origin;
1104 Vector3D bestNormal = fallbackNormal;
1105 bool found = false;
1106
1107 for (int faceIndex = 0; faceIndex < brep.Faces.Length; faceIndex++)
1108 {
1109 var face = brep.Faces[faceIndex];
1110 var mesh = GetFaceMesh(entityIndex, faceIndex, face);
1111 if (mesh?.Vertices == null || mesh.Triangles == null)
1112 continue;
1113
1114 foreach (var triangle in mesh.Triangles)
1115 {
1116 if (triangle.V1 >= mesh.Vertices.Length ||
1117 triangle.V2 >= mesh.Vertices.Length ||
1118 triangle.V3 >= mesh.Vertices.Length)
1119 {
1120 continue;
1121 }
1122
1123 var a = mesh.Vertices[triangle.V1];
1124 var b = mesh.Vertices[triangle.V2];
1125 var c = mesh.Vertices[triangle.V3];
1126 var closest = EyeshotGeometryUtils.ClosestPointOnTriangle(point, a, b, c);
1127 double distance = point.DistanceTo(closest);
1128 if (distance >= bestDistance)
1129 continue;
1130
1131 var triangleNormal = Vector3D.Cross(new Vector3D(a, b), new Vector3D(a, c));
1132 if (triangleNormal.Length <= 0.0001)
1133 continue;
1134
1135 triangleNormal.Normalize();
1136 bestDistance = distance;
1137 bestSurfacePoint = closest;
1138 bestNormal = triangleNormal;
1139 found = true;
1140 }
1141 }
1142
1143 if (!found)
1144 return false;
1145
1146 var towardShell = new Vector3D(bestSurfacePoint, point);
1147 if (towardShell.Length > 0.0001)
1148 {
1149 if (Vector3D.Dot(bestNormal, towardShell) < 0)
1150 bestNormal = bestNormal * -1.0;
1151 }
1152 else if (Vector3D.Dot(bestNormal, fallbackNormal) < 0)
1153 {
1154 bestNormal = bestNormal * -1.0;
1155 }
1156
1157 normal = NormalizeOrDefault(bestNormal, fallbackNormal);
1158 return true;
1159 }

1259 private List<HoleFeature> BuildExplicitHoleFeatures(
1260 List<ApproachMeasurementPair> pairs,
1261 HashSet<int> holeFaceIndices,
1262 IReadOnlyCollection<CADHole>? explicitHoles)
1263 {
1264 foreach (var pair in pairs)
1265 pair.IsHoleMeasurement = holeFaceIndices.Contains(pair.ParentFace.FaceGlobalIndex);
1266
1267 var features = new List<HoleFeature>();
1268 var claimedFaces = new HashSet<int>();
1269
1270 foreach (var cadHole in explicitHoles ?? Array.Empty<CADHole>())
1271 {
1272 if (cadHole == null || cadHole.IsIgnored || cadHole.FaceGlobalIndices == null)
1273 continue;
1274
1275 var holePairs = pairs
1276 .Where(pair => cadHole.FaceGlobalIndices.Contains(pair.ParentFace.FaceGlobalIndex))
1277 .ToList();
1278 if (holePairs.Count == 0)
1279 continue;
1280
1281 foreach (var faceIndex in cadHole.FaceGlobalIndices)
1282 claimedFaces.Add(faceIndex);
1283
1284 Vector3D axis = cadHole.AxisDirection != null && cadHole.AxisDirection.Length > 0.0001
1285 ? new Vector3D(cadHole.AxisDirection.X, cadHole.AxisDirection.Y, cadHole.AxisDirection.Z)
1286 : ExtractHoleAxis(holePairs[0].ParentFace);
1287 axis = NormalizeOrDefault(axis, Vector3D.AxisZ);
1288
1289 Point3D center = CalculateFaceCentroid(holePairs[0].ParentFace);
1290 double depth = !double.IsNaN(cadHole.Depth) && cadHole.Depth > 0.001
1291 ? cadHole.Depth
1292 : CalculateHoleDepth(holePairs[0].ParentFace);
1293 Point3D top = cadHole.TopCenterPoint != null
1294 ? (Point3D)cadHole.TopCenterPoint.Clone()
1295 : center - axis * (depth * 0.5);
1296 Point3D bottom = cadHole.BottomCenterPoint != null
1297 ? (Point3D)cadHole.BottomCenterPoint.Clone()
1298 : center + axis * (depth * 0.5);
1299
1300 var topToBottom = new Vector3D(top, bottom);
1301 if (topToBottom.Length > 0.001)
1302 {
1303 if (Vector3D.Dot(axis, topToBottom) < 0.0)
1304 axis = axis * -1.0;
1305 depth = topToBottom.Length;
1306 center = new Point3D(
1307 (top.X + bottom.X) * 0.5,
1308 (top.Y + bottom.Y) * 0.5,
1309 (top.Z + bottom.Z) * 0.5);
1310 }
1311
1312 double radius = new[] { cadHole.TopDiameter, cadHole.BottomDiameter }
1313 .Where(value => !double.IsNaN(value) && value > 0.0)
1314 .DefaultIfEmpty(ExtractHoleRadius(holePairs[0].ParentFace) * 2.0)
1315 .Max() * 0.5;
1316
1317 features.Add(new HoleFeature
1318 {
1319 HoleId = cadHole.HoleId,
1320 EntityIndex = holePairs[0].ParentFace.EntityIndex,
1321 Center = center,
1322 Axis = axis,
1323 Radius = radius,
1324 Depth = depth,
1325 ActualTopCenterPoint = top,
1326 ActualBottomCenterPoint = bottom,
1327 Pairs = holePairs,
1328 SourceHoles = new List<CADHole> { cadHole }
1329 });
1330 }
1331
1332 // Compatibility for a marked face whose CADHole metadata is incomplete.
1333 // It remains explicit because membership still comes from holeFaceIndices.
1334 foreach (var group in pairs
1335 .Where(pair => IsHolePair(pair) && !claimedFaces.Contains(pair.ParentFace.FaceGlobalIndex))
1336 .GroupBy(pair => pair.ParentFace.FaceGlobalIndex))
1337 {
1338 var face = group.First().ParentFace;
1339 var axis = NormalizeOrDefault(ExtractHoleAxis(face), Vector3D.AxisZ);
1340 var center = CalculateFaceCentroid(face);
1341 double depth = CalculateHoleDepth(face);
1342 features.Add(new HoleFeature
1343 {
1344 HoleId = group.Key,
1345 EntityIndex = face.EntityIndex,
1346 Center = center,
1347 Axis = axis,
1348 Radius = ExtractHoleRadius(face),
1349 Depth = depth,
1350 ActualTopCenterPoint = center - axis * (depth * 0.5),
1351 ActualBottomCenterPoint = center + axis * (depth * 0.5),
1352 Pairs = group.ToList()
1353 });
1354 }
1355
1356 return MergeCoaxialHoleFeatures(features);
1357 }

1359 private static List<HoleFeature> MergeCoaxialHoleFeatures(List<HoleFeature> source)
1360 {

1362 var merged = new List<HoleFeature>();
1363
1364 while (remaining.Count > 0)
1365 {
1366 var component = new List<HoleFeature> { remaining[0] };
1367 remaining.RemoveAt(0);

1370 do

1372 added = false;

1374 {

1376 continue;

1378 remaining.RemoveAt(index);

1390 private static bool AreHoleFeaturesCoaxial(HoleFeature first, HoleFeature second)
1391 {
1392 if (first.Axis == null || second.Axis == null || first.Center == null || second.Center == null)
1393 return false;
1394 if (first.EntityIndex < 0 || first.EntityIndex != second.EntityIndex)
1395 return false;
1396
1397 var firstAxis = NormalizeOrDefault(first.Axis, Vector3D.AxisZ);
1398 var secondAxis = NormalizeOrDefault(second.Axis, Vector3D.AxisZ);
1399 if (Math.Abs(Vector3D.Dot(firstAxis, secondAxis)) < 0.99)
1400 return false;
1401
1402 return DistancePointToLine(second.Center, first.Center, firstAxis) <= 2.0 &&
1403 DistancePointToLine(first.Center, second.Center, secondAxis) <= 2.0;

1406 private static HoleFeature MergeHoleFeatureComponent(List<HoleFeature> component)
1407 {
1408 if (component.Count == 1)
1409 return component[0];
1410
1411 var seed = component[0];
1412 var axis = NormalizeOrDefault(seed.Axis ?? Vector3D.AxisZ, Vector3D.AxisZ);
1413 var origin = seed.TopCenterPoint ?? seed.Center ?? Point3D.Origin;
1414 var endpoints = component
1415 .SelectMany(feature => new[]
1416 {
1417 feature.TopCenterPoint ?? feature.Center ?? Point3D.Origin,
1418 feature.BottomCenterPoint ?? feature.Center ?? Point3D.Origin
1419 })
1420 .ToList();
1421
1422 var seedDirection = new Vector3D(
1423 seed.TopCenterPoint ?? origin,
1424 seed.BottomCenterPoint ?? origin);
1425 if (seedDirection.Length > 0.001 && Vector3D.Dot(axis, seedDirection) < 0.0)
1426 axis = axis * -1.0;
1427
1428 double minT = endpoints.Min(point => Vector3D.Dot(new Vector3D(origin, point), axis));

1829 private static bool TryRayTriangleIntersection(
1830 Point3D origin,
1831 Vector3D direction,
1832 Point3D a,
1833 Point3D b,
1834 Point3D c,
1835 out double t)
1836 {
1837 const double epsilon = 1e-9;
1838 t = 0.0;
1839 var edge1 = new Vector3D(a, b);
1840 var edge2 = new Vector3D(a, c);
1841 var h = Vector3D.Cross(direction, edge2);
1842 double determinant = Vector3D.Dot(edge1, h);
1843 if (Math.Abs(determinant) < epsilon)
1844 return false;
1845
1846 double inverse = 1.0 / determinant;
1847 var s = new Vector3D(a, origin);
1848 double u = inverse * Vector3D.Dot(s, h);
1849 if (u < -epsilon || u > 1.0 + epsilon)
1850 return false;
1851
1852 var q = Vector3D.Cross(s, edge1);
1853 double v = inverse * Vector3D.Dot(direction, q);
1854 if (v < -epsilon || u + v > 1.0 + epsilon)
1855 return false;
1856
1857 t = inverse * Vector3D.Dot(edge2, q);
1858 return t > epsilon;
1859 }
