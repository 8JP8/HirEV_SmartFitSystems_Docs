87 public List<ApproachMeasurementPair> GenerateApproachPoints(
88 List<(Point3D point, Vector3D normal, CADFace face, int faceIndex)> measurementPoints,
89 Action<int, string>? progressCallback = null,
90 System.Threading.CancellationToken cancellationToken = default)
91 {
92 progressCallback?.Invoke(0, "Progress_GeneratingApproachPoints");
93 var input = measurementPoints.Select(m => (m.point, m.normal, m.face)).ToList();
94 var pairs = _approachGenerator.GenerateApproachPoints(input, new HashSet<int>(), new HashSet<int>(), cancellationToken: cancellationToken);
95
96 var validPairs = new List<ApproachMeasurementPair>();
97 _pointToNormalMap.Clear();
98
99 // Threshold for inner collision: half of the offset distance.
100 double minAllowedSdf = _approachOffsetDistance / 2.0;
101
102 int total = pairs.Count;
103 for (int i = 0; i < total; i++)
104 {
105 cancellationToken.ThrowIfCancellationRequested();
106 var pair = pairs[i];
107 if (i % 10 == 0 || i == total - 1)
108 {
109 int perc = (int)((double)i / total * 100);
110 progressCallback?.Invoke(perc, "Progress_CheckingApproachPointCollisions");
111 }
112
113 // Check if the generated approach point is too close to the part or inside it.
114 double sdfDist = _shellGenerator.GetDistance(pair.ApproachPoint);
115
116 bool isHole = pair.ParentFace.FaceType == FaceTypeEnum.Cylindrical || pair.ParentFace.FaceType == FaceTypeEnum.Conical;
117 bool isBridge = pair.ParentFace.IsBridgeBetweenHoles;
118
119 // 1. Regular surface points too close to part -> skip
120 if (!isHole && !isBridge && sdfDist != double.MaxValue && sdfDist < minAllowedSdf)
121 {
122 string ignoredFormat = GetXMLElement(DocLanguage, "PathManager_IgnoredApproachPoint", " ⚠️ Ignored approach point for face {0}: SDF dist {1:F2} < {2:F2} (potential collision)");
123 Debug.WriteLine(string.Format(ignoredFormat, pair.ParentFace.FaceGlobalIndex, sdfDist, minAllowedSdf));
124 continue;
125 }
126
127 // 2. Any point physically INSIDE the part solid (solid BREP check) -> push it out along normal
128 // This call can be slow for complex models
129 if (_shellGenerator.IsPointInsidePart(pair.ApproachPoint))
130 {
131 // Move away from wall along normal (for holes normal points inward, so this moves to center)
132 pair.ApproachPoint = pair.ApproachPoint + pair.Normal * 4.0;
133 string pushedFormat = GetXMLElement(DocLanguage, "PathManager_PushedOutApproachPoint", " 🚀 Pushed out approach point for face {0} (was inside part)");
134 Debug.WriteLine(string.Format(pushedFormat, pair.ParentFace.FaceGlobalIndex));
135 }
136
137 // 3. Ensure a minimum clearance from the surface even for holes to avoid "green points on surface"
138 if (sdfDist != double.MaxValue && sdfDist < 1.5)
139 {
140 pair.ApproachPoint = pair.ApproachPoint + pair.Normal * 2.5;
141 }
142
143 if (!_pointToNormalMap.ContainsKey(pair.ApproachPoint))
144 _pointToNormalMap[pair.ApproachPoint] = pair.Normal;
145 validPairs.Add(pair);
146 }
147 return validPairs;
148 }

156 public (Point3D ShellTop, Point3D AirTop) GetSafeTopPoints(int entityIndex, List<ApproachMeasurementPair>? allPairs = null)
157 {
158 Point3D shellTop = Point3D.Origin;
159 double cx = 0, cy = 0;
160 double maxZ = double.MinValue;
161 bool foundPoints = false;
162
163 if (allPairs != null)
164 {
165 var points = allPairs.Where(p => p.ParentFace.EntityIndex == entityIndex).Select(p => p.ApproachPoint).ToList();
166 if (points.Count > 0)
167 {
168 shellTop = points.OrderByDescending(p => p.Z).First();
169 cx = points.Average(p => p.X);
170 cy = points.Average(p => p.Y);
171 maxZ = points.Max(p => p.Z);
172 foundPoints = true;
173 }
174 }
175
176 if (_entityIndexToBrepMap != null && _entityIndexToBrepMap.TryGetValue(entityIndex, out var targetBrep))
177 {
178 if (targetBrep.BoxMin == null || targetBrep.BoxMax == null) targetBrep.Regen(0.01);
179 var bMin = targetBrep.BoxMin;
180 var bMax = targetBrep.BoxMax;
181 if (bMin != null && bMax != null)
182 {
183 var airTop = new Point3D((bMin.X + bMax.X) * 0.5, (bMin.Y + bMax.Y) * 0.5, bMax.Z + _approachOffsetDistance * 2);
184 if (!foundPoints) shellTop = new Point3D(airTop.X, airTop.Y, bMax.Z);
185 return (shellTop, airTop);
186 }
187 }
188
189 foreach (var brep in _partBreps)
190 {
191 if (brep.BoxMin == null || brep.BoxMax == null) brep.Regen(0.01);
192 var bMin = brep.BoxMin;
193 var bMax = brep.BoxMax;
194 if (bMin != null && bMax != null)
195 {
196 if (!foundPoints || (cx >= bMin.X - 1 && cx <= bMax.X + 1 &&
197 cy >= bMin.Y - 1 && cy <= bMax.Y + 1))
198 {
199 var airTop = new Point3D((bMin.X + bMax.X) * 0.5, (bMin.Y + bMax.Y) * 0.5, bMax.Z + _approachOffsetDistance * 2);
200 if (!foundPoints) shellTop = new Point3D(airTop.X, airTop.Y, bMax.Z);
201 return (shellTop, airTop);
202 }
203 }
204 }
205
206 var fallbackAir = new Point3D(cx, cy, maxZ + _approachOffsetDistance * 2);
207 if (!foundPoints) shellTop = new Point3D(cx, cy, maxZ);
208 return (shellTop, fallbackAir);
209 }

218 public Point3D GetClosestPartShellPoint(int entityIndex, Point3D airTop, List<ApproachMeasurementPair> allPairs)
219 {
220 var partPairs = allPairs.Where(p => p.ParentFace.EntityIndex == entityIndex).ToList();
221 if (partPairs.Count > 0)
222 {
223 Point3D closest = partPairs[0].ApproachPoint;
224 double minDist = closest.DistanceTo(airTop);
225 foreach (var pair in partPairs)
226 {
227 double dist = pair.ApproachPoint.DistanceTo(airTop);
228 if (dist < minDist)
229 {
230 minDist = dist;
231 closest = pair.ApproachPoint;
232 }
233 }
234 return closest;
235 }
236 // Fallback to ShellTop
237 var tops = GetSafeTopPoints(entityIndex, allPairs);
238 return tops.ShellTop;
239 }

250 public List<(Point3D Point, Vector3D Normal, int EntityIndex)> GeneratePath(
251 List<ApproachMeasurementPair> pairs,
252 Dictionary<CADFace, Color> colorGroups,
253 HashSet<int> holeFaceIndices,
254 Action<int, int, string>? progressCallback = null,
255 System.Threading.CancellationToken cancellationToken = default)
256 {
257 if (pairs == null || pairs.Count == 0) return new List<(Point3D, Vector3D, int)>();
258
259 progressCallback?.Invoke(5, 100, "Initializing path data...");
260
261 var pathPoints = new List<(Point3D Point, Vector3D Normal, int EntityIndex)>();
262
263 // PHASE 1: Auto-detect holes
264 progressCallback?.Invoke(15, 100, "Detecting hole features...");
265 var holes = AutoDetectHoles(pairs, holeFaceIndices);
266
267 // PHASE 2: Identify bridge faces (between holes)
268 progressCallback?.Invoke(25, 100, "Analyzing topological bridges...");
269 var separatingFaces = new HashSet<int>();
270 var specialPlanarFaces = new Dictionary<int, HoleFeature>();
271
272 var allPlanarFaces = pairs.Where(p => p.ParentFace.FaceType == FaceTypeEnum.Planar)
273 .Select(p => p.ParentFace)
274 .Distinct()
275 .ToList();
276
277 foreach (var face in allPlanarFaces)
278 {
279 cancellationToken.ThrowIfCancellationRequested();
280 var faceCenter = EyeshotGeometryUtils.GetFaceCenter(face);
281 var faceNormal = EyeshotGeometryUtils.GetFaceAxis(face);
282
283 HoleFeature? holeAbove = null;
284 foreach (var hole in holes)
285 {
286 if (hole.Axis == null || hole.Center == null) continue;
287 double distToAxis = DistancePointToLine(faceCenter, hole.Center, hole.Axis);
288 double alignmentDot = Math.Abs(EyeshotGeometryUtils.DotProduct(faceNormal, hole.Axis));
289
290 if (distToAxis < hole.Radius * 1.8 && alignmentDot > 0.85)
291 {
292 double distToTop = faceCenter.DistanceTo(hole.TopCenterPoint ?? Point3D.Origin);
293 double distToBottom = faceCenter.DistanceTo(hole.BottomCenterPoint ?? Point3D.Origin);
294
295 if (distToTop < 6.0 || distToBottom < 6.0)
296 {
297 Vector3D faceToHole = new Vector3D(faceCenter, hole.Center);
298 if (EyeshotGeometryUtils.DotProduct(faceToHole, faceNormal) > 0.5)
299 {
300 holeAbove = hole;
301 break;
302 }
303 }
304 }
305 }
306
307 if (holeAbove != null)
308 {
309 separatingFaces.Add(face.FaceGlobalIndex);
310 specialPlanarFaces[face.FaceGlobalIndex] = holeAbove;
311 }
312 }
313
314 // PHASE 3: Filter hole pairs
315 progressCallback?.Invoke(40, 100, "Filtering measurement points...");
316 foreach (var hole in holes)
317 {
318 cancellationToken.ThrowIfCancellationRequested();
319 var top = hole.TopCenterPoint ?? Point3D.Origin;
320 var bottom = hole.BottomCenterPoint ?? Point3D.Origin;
321 bool bridgeAtBottom = false;
322 bool bridgeAtTop = false;
323
324 foreach (var fIdx in separatingFaces)
325 {
326 var pair = pairs.FirstOrDefault(p => p.ParentFace.FaceGlobalIndex == fIdx);
327 if (pair == null) continue;
328 var faceCenter = EyeshotGeometryUtils.GetFaceCenter(pair.ParentFace);
329 var axis = hole.Axis ?? Vector3D.AxisZ;
330
331 double distZToFaceBottom = Math.Abs(Vector3D.Dot(new Vector3D(bottom, faceCenter), axis));
332 double distZToFaceTop = Math.Abs(Vector3D.Dot(new Vector3D(top, faceCenter), axis));
333
334 if (distZToFaceBottom < 4.0) bridgeAtBottom = true;
335 if (distZToFaceTop < 4.0) bridgeAtTop = true;
336 }
337
338 hole.Pairs = hole.Pairs.Where(p =>
339 {
340 var axis = hole.Axis ?? Vector3D.AxisZ;
341
342 double distZToTop = Math.Abs(Vector3D.Dot(new Vector3D(top, p.MeasurementPoint), axis));
343 double distZToBottom = Math.Abs(Vector3D.Dot(new Vector3D(bottom, p.MeasurementPoint), axis));
344
345 if (bridgeAtBottom && distZToBottom < 10.0) return false; // Larger clearance for bridge faces
346 if (bridgeAtTop && distZToTop < 10.0) return false;
347
348 return distZToTop > 4.0 && distZToBottom > 4.0; // Standard edge clearance
349 }).ToList();
350 }
351
352 // Remove any holes that had all their pairs filtered out to avoid First() exceptions later
353 holes.RemoveAll(h => h.Pairs == null || h.Pairs.Count == 0);
354
355 // PHASE 4: Group SURFACE pairs by color
356 progressCallback?.Invoke(55, 100, "Grouping by color...");
357 var surfacePairs = pairs.Where(p => !holeFaceIndices.Contains(p.ParentFace.FaceGlobalIndex) && !separatingFaces.Contains(p.ParentFace.FaceGlobalIndex)).ToList();
358
359 var atomicChains = new Dictionary<Color, List<List<ApproachMeasurementPair>>>();
360 var unignoredSpecialPairs = pairs.Where(p => specialPlanarFaces.ContainsKey(p.ParentFace.FaceGlobalIndex) && !p.ParentFace.IsIgnored).ToList();
361
362 foreach (var group in unignoredSpecialPairs.GroupBy(p => p.ParentFace.FaceGlobalIndex))
363 {
364 cancellationToken.ThrowIfCancellationRequested();
365 var face = group.First().ParentFace;
366 var holeAbove = specialPlanarFaces[face.FaceGlobalIndex];
367 var faceNormal = EyeshotGeometryUtils.GetFaceAxis(face);
368 var holeTopAxis = holeAbove.TopCenterPoint ?? (holeAbove.Center ?? Point3D.Origin);
369 var axis = holeAbove.Axis ?? Vector3D.AxisZ;
370 var center = holeAbove.Center ?? Point3D.Origin;
371
372 Color faceColor = colorGroups.TryGetValue(face, out var c) ? c : Color.Gray;
373 if (!atomicChains.ContainsKey(faceColor)) atomicChains[faceColor] = new List<List<ApproachMeasurementPair>>();
374
375 // Find distances to axis for all points in this face
376 var pointDistances = group.Select(p => new { Point = p, Dist = DistancePointToLine(p.MeasurementPoint, center, axis) }).ToList();
377 if (!pointDistances.Any()) continue;
378
379 // Only keep the inner circumference (ignore the outer ring adjacent to the bigger hole wall)
380 double minDist = pointDistances.Min(pd => pd.Dist);
381 var innerPoints = pointDistances.Where(pd => pd.Dist < minDist + 2.0).Select(pd => pd.Point).ToList();
382
383 if (!innerPoints.Any()) continue;
384
385 // Single atomic sequence for all points in the inner ring
386 var chain = new List<ApproachMeasurementPair>();
387
388 // 1. Entry at top of bigger hole axis
389 chain.Add(new ApproachMeasurementPair(holeTopAxis, holeTopAxis, faceNormal, face, true));
390
391 // 2. Move down to center of hole (closest point on axis to the first approach point)
392 var firstPt = innerPoints.First().MeasurementPoint;
393 double t = Vector3D.Dot(new Vector3D(center, firstPt), axis);
394 Point3D axisPt = center + axis * t;
395 chain.Add(new ApproachMeasurementPair(axisPt, axisPt, faceNormal, face, true));
396
397 // 3. Measure all inner points
398 foreach (var p in innerPoints)
399 {
400 // Ensure approach points are offset in the normal direction (outward from face)
401 p.ApproachPoint = p.MeasurementPoint + faceNormal * 2.0;
402 chain.Add(p);
403 }
404
405 // 4. Exit back to axis center
406 var lastPt = innerPoints.Last().MeasurementPoint;
407 double tLast = Vector3D.Dot(new Vector3D(center, lastPt), axis);
408 Point3D axisPtLast = center + axis * tLast;
409 chain.Add(new ApproachMeasurementPair(axisPtLast, axisPtLast, faceNormal, face, true));
410
411 // 5. Exit back out of hole
412 chain.Add(new ApproachMeasurementPair(holeTopAxis, holeTopAxis, faceNormal, face, true));
413
414 atomicChains[faceColor].Add(chain);
415 }
416
417 progressCallback?.Invoke(70, 100, "Optimizing path sequence...");
418 var groupedPairs = GroupByColor(surfacePairs, colorGroups);
419 Point3D? lastApproach = null;
420
421 int globalLastEntityIndex = -1;
422
423 foreach (var color in colorGroups.Values.Distinct())
424 {
425 cancellationToken.ThrowIfCancellationRequested();
426 var standardPairs = surfacePairs.Where(p => colorGroups.TryGetValue(p.ParentFace, out var c) && c == color).ToList();
427 int lastEntityIndex = globalLastEntityIndex;
428
429 var pairsByPart = standardPairs.GroupBy(p => p.ParentFace.EntityIndex).ToList();
430
431 foreach (var partGroup in pairsByPart)
432 {
433 int currentEntityIndex = partGroup.Key;
434
435 foreach (var current in OrderByProximity(partGroup.ToList()))
436 {
437 cancellationToken.ThrowIfCancellationRequested();
438 // If jumping between different parts, insert safe top points
439 if (lastApproach != null && lastEntityIndex != -1 && currentEntityIndex != lastEntityIndex && _partBreps.Count > 1)
440 {
441 var lastTops = GetSafeTopPoints(lastEntityIndex, pairs);
442 var currentTops = GetSafeTopPoints(currentEntityIndex, pairs);
443
444 Point3D closestShellPtA = GetClosestPartShellPoint(lastEntityIndex, lastTops.AirTop, pairs);
445 Point3D closestShellPtB = GetClosestPartShellPoint(currentEntityIndex, currentTops.AirTop, pairs);
446
447 // 1. Move to closest shell point on A
448 ProcessTransition(lastApproach, closestShellPtA, Vector3D.AxisZ, lastEntityIndex, pathPoints, cancellationToken);
449 pathPoints.Add((closestShellPtA, Vector3D.AxisZ, lastEntityIndex));
450 pathPoints.Add((lastTops.AirTop, Vector3D.AxisZ, lastEntityIndex));
451
452 // 2. Move to Top Point of the current part
453 pathPoints.Add((currentTops.AirTop, Vector3D.AxisZ, currentEntityIndex));
454 pathPoints.Add((closestShellPtB, Vector3D.AxisZ, currentEntityIndex));
455
456 // 3. Update lastApproach so ProcessPair transitions from closestShellPtB
457 lastApproach = closestShellPtB;
458 }
459
460 ProcessPair(current, ref lastApproach, pathPoints, cancellationToken);
461 lastEntityIndex = currentEntityIndex;
462 globalLastEntityIndex = currentEntityIndex;
463 }
464 }
465
466 if (atomicChains.TryGetValue(color, out var chains))
467 {
468 foreach (var chain in chains)
469 {
470 for (int i = 0; i < chain.Count; i++)
471 {
472 cancellationToken.ThrowIfCancellationRequested();
473 var current = chain[i];
474 int currentEntityIndex = current.ParentFace.EntityIndex;
475
476 // Handle part jumps even when entering atomic chains (e.g. holes)
477 if (i == 0 && lastApproach != null && lastEntityIndex != -1 && currentEntityIndex != lastEntityIndex && _partBreps.Count > 1)
478 {
479 var lastTops = GetSafeTopPoints(lastEntityIndex, pairs);
480 var currentTops = GetSafeTopPoints(currentEntityIndex, pairs);
481
482 Point3D closestShellPtA = GetClosestPartShellPoint(lastEntityIndex, lastTops.AirTop, pairs);
483 Point3D closestShellPtB = GetClosestPartShellPoint(currentEntityIndex, currentTops.AirTop, pairs);
484
485 ProcessTransition(lastApproach, closestShellPtA, Vector3D.AxisZ, lastEntityIndex, pathPoints, cancellationToken);
486 pathPoints.Add((closestShellPtA, Vector3D.AxisZ, lastEntityIndex));
487 pathPoints.Add((lastTops.AirTop, Vector3D.AxisZ, lastEntityIndex));
488 pathPoints.Add((currentTops.AirTop, Vector3D.AxisZ, currentEntityIndex));
489 pathPoints.Add((closestShellPtB, Vector3D.AxisZ, currentEntityIndex));
490
491 lastApproach = closestShellPtB;
492 }
493
494 if (i == 0) ProcessPair(current, ref lastApproach, pathPoints, cancellationToken);
495 else
496 {
497 if (current.IsHoleMeasurement && current.MeasurementPoint.Equals(current.ApproachPoint))
498 pathPoints.Add((current.MeasurementPoint, current.Normal, currentEntityIndex));
499 else
500 {
501 pathPoints.Add((current.ApproachPoint, current.Normal, currentEntityIndex));
502 pathPoints.Add((current.MeasurementPoint, current.Normal, currentEntityIndex));
503 pathPoints.Add((current.ApproachPoint, current.Normal, currentEntityIndex));
504 }
505 lastApproach = current.ApproachPoint;
506 }
507
508 lastEntityIndex = currentEntityIndex;
509 globalLastEntityIndex = currentEntityIndex;
510 }
511 }
512 }
513 }
514
515 // 2. Process Hole Features (already handles Safe points)
516 var holesByPart = holes.GroupBy(h => h.Pairs.First().ParentFace.EntityIndex).ToList();
517 int holeLastEntityIndex = globalLastEntityIndex;
518
519 foreach (var partHoles in holesByPart)
520 {
521 int currentEntityIndex = partHoles.Key;
522
523 foreach (var hole in partHoles)
524 {
525 cancellationToken.ThrowIfCancellationRequested();
526 var (topCenter, _) = GetHoleAxisPoints(hole);
527 var axis = hole.Axis ?? Vector3D.AxisZ;
528 var outVec = new Vector3D(hole.Center ?? Point3D.Origin, topCenter);
529 if (Vector3D.Dot(outVec, axis) < 0) axis = axis * -1.0;
530 axis.Normalize();
531
532 Point3D holeSafePoint = topCenter + axis * _approachOffsetDistance;
533 var shellEntry = FindClosestAdjacentShellPoint(topCenter) ?? holeSafePoint;
534
535 // If jumping between different parts to process a hole, insert safe top points
536 if (lastApproach != null && holeLastEntityIndex != -1 && currentEntityIndex != holeLastEntityIndex && _partBreps.Count > 1)
537 {
538 var lastTops = GetSafeTopPoints(holeLastEntityIndex, pairs);
539 var currentTops = GetSafeTopPoints(currentEntityIndex, pairs);
540
541 Point3D closestShellPtA = GetClosestPartShellPoint(holeLastEntityIndex, lastTops.AirTop, pairs);
542 Point3D closestShellPtB = GetClosestPartShellPoint(currentEntityIndex, currentTops.AirTop, pairs);
543
544 ProcessTransition(lastApproach, closestShellPtA, Vector3D.AxisZ, holeLastEntityIndex, pathPoints, cancellationToken);
545 pathPoints.Add((closestShellPtA, Vector3D.AxisZ, holeLastEntityIndex));
546 pathPoints.Add((lastTops.AirTop, Vector3D.AxisZ, holeLastEntityIndex));
547 pathPoints.Add((currentTops.AirTop, Vector3D.AxisZ, currentEntityIndex));
548 pathPoints.Add((closestShellPtB, Vector3D.AxisZ, currentEntityIndex));
549
550 lastApproach = closestShellPtB;
551 }
552
553 ProcessTransition(lastApproach, shellEntry, axis * -1.0, currentEntityIndex, pathPoints, cancellationToken);
554 pathPoints.Add((shellEntry, axis * -1.0, currentEntityIndex));
555 pathPoints.Add((holeSafePoint, axis * -1.0, currentEntityIndex));
556
557 foreach (var pair in hole.Pairs.OrderByDescending(p => Vector3D.Dot(new Vector3D(hole.Center ?? Point3D.Origin, p.MeasurementPoint), axis)))
558 {
559 double t = Vector3D.Dot(new Vector3D(hole.Center ?? Point3D.Origin, pair.MeasurementPoint), axis);
560 Point3D axisPt = (hole.Center ?? Point3D.Origin) + axis * t;
561 pathPoints.Add((axisPt, pair.Normal, currentEntityIndex));
562 pathPoints.Add((pair.MeasurementPoint, pair.Normal, currentEntityIndex));
563 pathPoints.Add((axisPt, pair.Normal, currentEntityIndex));
564 }
565
566 pathPoints.Add((holeSafePoint, axis * -1.0, currentEntityIndex));
567 pathPoints.Add((shellEntry, axis * -1.0, currentEntityIndex));
568 lastApproach = shellEntry;
569 holeLastEntityIndex = currentEntityIndex;
570 }
571 }
572
573 // Move to safe top point after the very last measurement
574 if (lastApproach != null && pairs.Count > 0)
575 {
576 int finalEntityIndex = holeLastEntityIndex != -1 ? holeLastEntityIndex : globalLastEntityIndex;
577 if (finalEntityIndex == -1) finalEntityIndex = pairs.Last().ParentFace.EntityIndex;
578
579 var tops = GetSafeTopPoints(finalEntityIndex, pairs);
580 Point3D closestShellPt = GetClosestPartShellPoint(finalEntityIndex, tops.AirTop, pairs);
581 ProcessTransition(lastApproach, closestShellPt, Vector3D.AxisZ, finalEntityIndex, pathPoints, cancellationToken);
582 pathPoints.Add((closestShellPt, Vector3D.AxisZ, finalEntityIndex));
583 if (_partBreps.Count > 1)
584 {
585 pathPoints.Add((tops.AirTop, Vector3D.AxisZ, finalEntityIndex));
586 lastApproach = tops.AirTop;
587 }
588 else
589 {
590 lastApproach = closestShellPt;
591 }
592 }
593
594 return pathPoints;
595 }

618 private void ProcessPair(ApproachMeasurementPair current, ref Point3D? lastApproach, List<(Point3D, Vector3D, int)> pathPoints, System.Threading.CancellationToken cancellationToken = default)
619 {
620 int entityIndex = current.ParentFace.EntityIndex;
621 ProcessTransition(lastApproach, current.ApproachPoint, current.Normal, entityIndex, pathPoints, cancellationToken);
622 if (current.IsHoleMeasurement && current.MeasurementPoint.Equals(current.ApproachPoint))
623 pathPoints.Add((current.MeasurementPoint, current.Normal, entityIndex));
624 else
625 {
626 pathPoints.Add((current.ApproachPoint, current.Normal, entityIndex));
627 pathPoints.Add((current.MeasurementPoint, current.Normal, entityIndex));

642 private void ProcessTransition(Point3D? from, Point3D to, Vector3D fallbackNormal, int entityIndex, List<(Point3D, Vector3D, int)> pathPoints, System.Threading.CancellationToken cancellationToken = default)
643 {
644 if (from != null && !from.Equals(to))
645 {

709 private List<ApproachMeasurementPair> OrderByProximity(List<ApproachMeasurementPair> pairs)
710 {
711 if (pairs.Count <= 1) return pairs;
712 var ordered = new List<ApproachMeasurementPair>();
713 var remaining = new List<ApproachMeasurementPair>(pairs);
714 var current = remaining[0];
715 ordered.Add(current); remaining.RemoveAt(0);
716 while (remaining.Count > 0)
717 {
718 var next = remaining.OrderBy(r => current.ApproachPoint.DistanceTo(r.ApproachPoint)).First();
719 ordered.Add(next); remaining.Remove(next); current = next;
720 }
721 return ordered;

764 private double CalculateHoleDepth(CADFace face)
765 {
766 if (face.ControlPoints == null || face.ControlPoints.Count < 2) return 10.0;
767 var axis = ExtractHoleAxis(face);
768 var pts = face.ControlPoints.Select(p => (Point3D)p).ToList();
769 double minProj = pts.Min(p => Vector3D.Dot(new Vector3D(pts[0], p), axis));
770 double maxProj = pts.Max(p => Vector3D.Dot(new Vector3D(pts[0], p), axis));
771 return Math.Max(1.0, maxProj - minProj);
772 }
