167 public static (List<CADHole> holes, List<CADFace> nonHoleFaces) DetectHoles(List<CADFace> allFaces, Action<int, int>? progressCallback = null, System.Threading.CancellationToken cancellationToken = default)
168 {
169 var holes = new List<CADHole>();
170 var processedFaceIndices = new HashSet<int>();
171
172 foreach (var face in allFaces)
173 {
174 face.IsBridgeBetweenHoles = false;
175 }
176
177 progressCallback?.Invoke(10, 0);
178
179 var cylindricalFaces = allFaces
180 .Where(f => f.FaceType == FaceTypeEnum.Cylindrical || f.FaceType == FaceTypeEnum.Conical)
181 .ToList();
182
183 progressCallback?.Invoke(30, 0);
184
185 var holeGroups = GroupFacesIntoHoles(cylindricalFaces, allFaces, cancellationToken);
186
187 progressCallback?.Invoke(50, 0);
188
189 int holeId = 1;
190 int totalGroups = holeGroups.Count;
191 int processedGroups = 0;
192
193 foreach (var group in holeGroups)
194 {
195 cancellationToken.ThrowIfCancellationRequested();
196
197 if (group.Count >= 2)
198 {
199 var hole = CreateHoleFromFaces(group, holeId, allFaces);
200 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
201 {
202 holes.Add(hole);
203 holeId++;
204 foreach (var f in group)
205 processedFaceIndices.Add(f.FaceGlobalIndex);
206 }
207 }
208 else if (group.Count == 1 && ENABLE_SINGLE_FACE_JOINED_HOLES)
209 {
210 var face = group[0];
211 if (IsSingleFaceJoinedHoleCandidate(face, cylindricalFaces, allFaces))
212 {
213 var hole = CreateHoleFromSingleFace(face, holeId, allFaces);
214 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
215 {
216 holes.Add(hole);
217 holeId++;
218 processedFaceIndices.Add(face.FaceGlobalIndex);
219 }
220 }
221 }
222
223 processedGroups++;
224 if (progressCallback != null && totalGroups > 0)
225 {
226 int percentage = 50 + (int)((processedGroups * 50.0) / totalGroups);
227 progressCallback(percentage, holes.Count);
228 }
229 }
230
231 // Post-process: Identify bridge faces for all detected holes
232 foreach (var hole in holes)
233 {
234 cancellationToken.ThrowIfCancellationRequested();
235 IdentifyBridgeFaces(hole, allFaces, cancellationToken);
236 }
237
238 // Post-process: Mark planar separator faces between cylindrical halves as bridge
239 // faces. When AreCylindersSeparatedByPlanarFaces prevents two half-cylinders from
240 // being grouped, the planar faces that close the sides of those halves are never
241 // path-planned. They must be marked IsBridgeBetweenHoles so the downstream
242 // pipeline auto-ignores them just like any other bridge face.
243 MarkSeparatorFacesAsBridge(cylindricalFaces, allFaces, processedFaceIndices, cancellationToken);
244
245 // Post-process: Detect concave isolated single-face holes.
246 // This runs strictly AFTER the original bridge detection and is not allowed to
247 // change any IsBridgeBetweenHoles result produced by that pipeline.
248 if (ENABLE_CONCAVE_ISOLATED_SINGLE_FACE_HOLES)
249 {
250 var bridgeStateBeforeConcaveDetection = allFaces.ToDictionary(
251 face => face.FaceGlobalIndex,
252 face => face.IsBridgeBetweenHoles);
253
254 foreach (var face in cylindricalFaces)
255 {
256 cancellationToken.ThrowIfCancellationRequested();
257
258 if (processedFaceIndices.Contains(face.FaceGlobalIndex)) continue;
259 if (face.IsBridgeBetweenHoles) continue;
260
261 if (!IsConcaveIsolatedSingleFaceHoleCandidate(face, allFaces))
262 continue;
263
264 var hole = CreateHoleFromSingleFace(face, holeId, allFaces);
265 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
266 {
267 holes.Add(hole);
268 holeId++;
269 processedFaceIndices.Add(face.FaceGlobalIndex);
270 }
271 }
272
273 // Restore the bridge flags exactly as they were before this isolated
274 // single-face fallback. This guarantees that enabling concave detection
275 // cannot create, remove or reclassify bridge faces.
276 foreach (var face in allFaces)
277 {
278 if (bridgeStateBeforeConcaveDetection.TryGetValue(
279 face.FaceGlobalIndex,
280 out bool wasBridge))
281 {
282 face.IsBridgeBetweenHoles = wasBridge;
283 }
284 }
285 }
286
287 var nonHoleFaces = allFaces
288 .Where(f => !processedFaceIndices.Contains(f.FaceGlobalIndex))
289 .ToList();
290
291 DetectNestedHoles(holes);
292
293 if (GetConfigBool("HoleDetection_AutoIgnoreObstructedByLarger", false))
294 {
295 var baseAvoidance = Utilities.Algorithms.BaseAvoidanceClearance.EnableForIgnoredBottomFace(
296 Utilities.Algorithms.BaseAvoidanceClearance.LoadConfig(),
297 Utilities.Algorithms.BaseAvoidanceFaceClassifier.HasIgnoredBottomFace(allFaces));
298 bool avoidBottomFace = baseAvoidance.Enabled ||
299 (bool.TryParse(System.Configuration.ConfigurationManager.AppSettings["Scanner_AvoidBottomFace"], out bool s) && s);
300 AutoIgnoreTopUnreachableHoles(holes, allFaces, avoidBottomFace);
301 }
302
303 progressCallback?.Invoke(100, holes.Count);
304
305 return (holes, nonHoleFaces);
306 }

349 {
350 if (face == null || (face.FaceType != FaceTypeEnum.Cylindrical && face.FaceType != FaceTypeEnum.Conical))
351 return null!;
352
353 if (!TryGetCylinderFitInfo(face, out var fit))
354 return null!;
355
356 var bottomCenter = PointOnAxis(fit.AxisPoint, fit.Axis, fit.Range.Min);
357 var topCenter = PointOnAxis(fit.AxisPoint, fit.Axis, fit.Range.Max);
358
359 var hole = new CADHole
360 {
361 HoleId = holeId,
362 HoleColor = face.FaceColor,
363 FaceGlobalIndices = new List<int> { face.FaceGlobalIndex },

365 BottomCenterPoint = bottomCenter

369 if (hole.Depth < MIN_HOLE_DEPTH_TOLERANCE) return null!;

374 hole.TopDiameter = diameter;
375 hole.BottomDiameter = diameter;
376 hole.HoleStandard = IdentifyStandardHoleType(diameter);
377 hole.GeometricInfo = $"{hole.HoleStandard} - Depth: {hole.Depth:F2}mm, Ø{diameter:F2}";
378

484 public static string IdentifyStandardHoleType(double diameter)
485 {
486 if (double.IsNaN(diameter) || diameter <= 0)
487 return "Unknown";
488
489 var closestMatch = StandardHoleDiameters
490 .Select(kvp => new { kvp.Key, kvp.Value, Diff = Math.Abs(kvp.Value - diameter) })
491 .OrderBy(x => x.Diff)
492 .FirstOrDefault();
493
494 if (closestMatch != null && closestMatch.Diff <= DIAMETER_TOLERANCE)
495 return closestMatch.Key;
496
497 var clearanceMatch = StandardHoleDiameters
498 .Select(kvp => new
499 {
500 Standard = kvp.Key + " Clearance",
501 NominalDiameter = double.Parse(kvp.Key.Substring(1)),
502 TapDiameter = kvp.Value,
503 Diff = Math.Abs(diameter - (double.Parse(kvp.Key.Substring(1)) + 0.5))
504 })
505 .OrderBy(x => x.Diff)
506 .FirstOrDefault();
507
508 if (clearanceMatch != null && clearanceMatch.Diff <= DIAMETER_TOLERANCE)
509 return clearanceMatch.Standard;
510
511 return $"Ø{diameter:F2}mm";
512 }

517 private static List<List<CADFace>> GroupFacesIntoHoles(List<CADFace> cylindricalFaces, List<CADFace> allFaces, System.Threading.CancellationToken cancellationToken = default)
518 {
519 var groups = new List<List<CADFace>>();
520 var processed = new HashSet<int>();
521
522 foreach (var face in cylindricalFaces)
523 {
524 cancellationToken.ThrowIfCancellationRequested();
525 if (processed.Contains(face.FaceGlobalIndex)) continue;
526
527 var group = new List<CADFace>();
528 var queue = new Queue<CADFace>();
529 queue.Enqueue(face);
530 processed.Add(face.FaceGlobalIndex);
531
532 while (queue.Count > 0)
533 {
534 cancellationToken.ThrowIfCancellationRequested();
535 var current = queue.Dequeue();
536 group.Add(current);
537
538 double radius1 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(current);
539 if (radius1 <= 0) continue;
540
541 foreach (var other in cylindricalFaces)
542 {
543 if (processed.Contains(other.FaceGlobalIndex)) continue;
544
545 double radius2 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(other);
546 if (radius2 <= 0 || Math.Abs(radius1 - radius2) > RADIAL_TOLERANCE)
547 continue;
548
549 if (AreCylindersSeparatedByPlanarFaces(current, other, allFaces))
550 continue;
551
552 // Check if faces are companions (split halves) or stacked segments
553 if (AreFacesPairedForHole(current, other) || AreFacesStacked(current, other))
554 {
555 processed.Add(other.FaceGlobalIndex);
556 queue.Enqueue(other);
557 }
558 }
559 }
560
561 if (group.Count > 0)
562 groups.Add(group);
563 }
564
565 return groups;
566 }

571 private static bool AreFacesStacked(CADFace face1, CADFace face2)
572 {
573 var pts1 = GetFaceAnalysisPoints(face1);
574 var pts2 = GetFaceAnalysisPoints(face2);
575
576 if (pts1.Count == 0 || pts2.Count == 0)
577 return false;
578
579 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
580 return false;
581
582 if (AreCylinderAxesSeparated(face1, face2, axis, radius, out _))
583 return false;
584
585 var combinedPoints = pts1.Concat(pts2).ToList();
586 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
587 return false;
588
589 var range1 = GetAxialRange(pts1, axisPoint, axis);
590 var range2 = GetAxialRange(pts2, axisPoint, axis);
591 if (!range1.HasValue || !range2.HasValue)
592 return false;
593
594 double gap = GetAxialGap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
595
596 // They are stacked if they are coaxial and their projected extents touch or are
597 // close enough to jump over a real separating bridge face.
598 return gap >= -BRIDGE_AXIAL_TOLERANCE && gap < MAX_JUMP_GAP;
599 }

605 private static bool AreFacesPairedForHole(CADFace face1, CADFace face2)
606 {
607 var pts1 = GetFaceAnalysisPoints(face1);
608 var pts2 = GetFaceAnalysisPoints(face2);
609
610 if (pts1.Count == 0 || pts2.Count == 0)
611 return false;
612
613 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
614 return false;
615
616 if (AreCylinderAxesSeparated(face1, face2, axis, radius, out _))
617 return false;
618
619 double range1Length = GetProjectedRangeLength(pts1, axis);
620 double range2Length = GetProjectedRangeLength(pts2, axis);
621 double maxAxialRange = Math.Max(range1Length, range2Length);
622 double adaptiveAxialTolerance = Math.Max(Z_DEVIATION_TOLERANCE, maxAxialRange * 0.15);
623
624 // Complementary split cylinder faces may have centroids on opposite sides of
625 // the bore. Check shared seam points before any centroid-based rejection.
626 if (CountPointMatches(pts1, pts2, axis, SHARED_VERTEX_TOLERANCE * 2.0, adaptiveAxialTolerance) >= SEAM_MATCHES_REQUIRED)
627 return true;
628
629 var combinedPoints = pts1.Concat(pts2).ToList();
630 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
631 return false;
632
633 var range1 = GetAxialRange(pts1, axisPoint, axis);
634 var range2 = GetAxialRange(pts2, axisPoint, axis);
635 if (!range1.HasValue || !range2.HasValue)
636 return false;
637
638 double overlap = GetAxialOverlap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
639 if (overlap < Math.Min(range1.Value.Length, range2.Value.Length) * 0.10 &&
640 overlap < MIN_HOLE_DEPTH_TOLERANCE)
641 {
642 return false;
643 }
644
645 if (HasCompleteCircularCoverage(combinedPoints, axisPoint, axis, MIN_CIRCULAR_COVERAGE_RADIANS))
646 return true;
647
648 int matchedPoints = CountPointMatches(pts1, pts2, axis, RADIAL_TOLERANCE, adaptiveAxialTolerance);
649 double similarity = (double)matchedPoints / Math.Max(pts1.Count, pts2.Count);
650
651 // Lower threshold for curved surfaces (70% vs 90%)
652 double threshold = maxAxialRange > MIN_HOLE_DEPTH_TOLERANCE ? 0.70 : 0.85;
653
654 return similarity >= threshold;
655 }

660 private static bool HaveSimilarControlPointsWithDeviation(List<Point4D> pts1, List<Point4D> pts2)
661 {
662 // Convert to CADFace-like structure for the enhanced check
663 // This is a simplified adapter - the main grouping now uses AreFacesPairedForHole
664 if (pts1.Count == 0 || pts2.Count == 0) return false;
665
666 int matched = 0;
667 double zRange = Math.Max(
668 pts1.Max(p => p.Z) - pts1.Min(p => p.Z),
669 pts2.Max(p => p.Z) - pts2.Min(p => p.Z)
670 );
671
672 double dynamicZTolerance = Math.Max(Z_DEVIATION_TOLERANCE, zRange * 0.15);
673
674 foreach (var p1 in pts1)
675 {
676 if (pts2.Any(p2 =>
677 Math.Abs(p1.X - p2.X) < SHARED_VERTEX_TOLERANCE &&
678 Math.Abs(p1.Y - p2.Y) < SHARED_VERTEX_TOLERANCE &&
679 Math.Abs(p1.Z - p2.Z) < dynamicZTolerance))
680 {
681 matched++;
682 }
683 }
684
685 double similarity = (double)matched / Math.Max(pts1.Count, pts2.Count);
686 return similarity >= 0.70; // lowered threshold for curved surfaces
687 }

697 private static CADHole CreateHoleFromFaces(List<CADFace> faces, int holeId, List<CADFace> allFaces, bool isManual = false)
698 {
699 if (faces.Count < 2) return null!;
700
701 var allControlPoints = faces.SelectMany(GetFaceAnalysisPoints).ToList();
702
703 if (allControlPoints.Count == 0) return null!;
704
705 var axis = GetDominantFaceAxis(faces);
706 double avgRadius = faces
707 .Select(EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo)
708 .Where(r => r > 0)
709 .DefaultIfEmpty(0)
710 .Average();
711
712 Point3D bottomCenter;
713 Point3D topCenter;
714 CADFace topFace;
715 CADFace bottomFace;
716
717 if (avgRadius > 0 &&
718 TryFitCylinderAxisPoint(allControlPoints, axis, avgRadius, out var axisPoint, out _, out _))
719 {
720 var range = GetAxialRange(allControlPoints, axisPoint, axis);
721 if (!range.HasValue || range.Value.Length < MIN_HOLE_DEPTH_TOLERANCE)
722 return null!;
723
724 bottomCenter = PointOnAxis(axisPoint, axis, range.Value.Min);
725 topCenter = PointOnAxis(axisPoint, axis, range.Value.Max);
726
727 topFace = faces.OrderByDescending(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
728 bottomFace = faces.OrderBy(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
729 }
730 else
731 {
732 var sortedPoints = allControlPoints.OrderBy(p => p.Z).ToList();
733 double zMin = sortedPoints.First().Z;
734 double zMax = sortedPoints.Last().Z;
735
736 if (Math.Abs(zMax - zMin) < MIN_HOLE_DEPTH_TOLERANCE)
737 return null!;
738
739 var finalBottomPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMin) < SHARED_VERTEX_TOLERANCE * 2).ToList();
740 var finalTopPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMax) < SHARED_VERTEX_TOLERANCE * 2).ToList();
741
742 bottomCenter = GetCentroid(finalBottomPoints);
743 topCenter = GetCentroid(finalTopPoints);
744 topFace = faces.OrderByDescending(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
745 bottomFace = faces.OrderBy(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
746 }
747
748 var hole = new CADHole
749 {
750 HoleId = holeId,
751 HoleColor = faces[0].FaceColor,
752 FaceGlobalIndices = faces.Select(f => f.FaceGlobalIndex).ToList(),
753 TopCenterPoint = topCenter,
754 BottomCenterPoint = bottomCenter
755 };
756
757 hole.Depth = EyeshotGeometryUtils.DistanceBetweenPoints(hole.BottomCenterPoint, hole.TopCenterPoint);
758
759 var axisVector = hole.TopCenterPoint.AsVector - hole.BottomCenterPoint.AsVector;
760 axisVector.Normalize();
761 hole.AxisDirection = axisVector;
762
763 if (!isManual && !IsHoleOpen(hole.BottomCenterPoint, hole.TopCenterPoint, faces, allFaces, hole.AxisDirection))
764 return null!;
765
766 if (hole.Depth < MIN_HOLE_DEPTH_TOLERANCE) return null!;
767
768 // Determine diameters from the extreme faces in the stack
769 hole.TopDiameter = GetFaceDiameter(topFace);
770 hole.BottomDiameter = GetFaceDiameter(bottomFace);
771 double avgDiameter = (hole.TopDiameter + hole.BottomDiameter) / 2.0;
772 hole.HoleStandard = IdentifyStandardHoleType(avgDiameter);
773 hole.GeometricInfo = $"{hole.HoleStandard} - Depth: {hole.Depth:F2}mm, Ø{avgDiameter:F2}";
774
775 return hole;
776 }

883 private static bool IsConcaveIsolatedSingleFaceHoleCandidate(CADFace face, List<CADFace> allFaces)
884 {
885 // Keep this fallback limited to true cylindrical faces. Conical faces are much
886 // more likely to be chamfers and remain handled only by the original pipeline.
887 if (face == null || face.FaceType != FaceTypeEnum.Cylindrical)
888 return false;
889
890 if (!TryGetCylinderFitInfo(face, out var fit))
891 return false;
892
893 // Only inward-facing cylindrical walls are concave holes. Exterior convex
894 // cylinders carry a negative value and are rejected before any U-shape checks.
895 if (!HasInwardFacingCylinderNormal(face))
896 return false;
897
898 var cylinderPoints = GetCylinderAnalysisPoints(face);
899 if (cylinderPoints.Count < 4)
900 return false;
901
902 if (!HasCompleteCircularCoverage(
903 cylinderPoints,
904 fit.AxisPoint,
905 fit.Axis,
906 MIN_CONCAVE_ISOLATED_HOLE_COVERAGE_RADIANS))
907 {
908 return false;
909 }
910
911 BuildPerpendicularBasis(fit.Axis, out var u, out var v);
912
913 double GetAngle(Point3D point)
914 {
915 double axial = ProjectOnAxis(point, fit.AxisPoint, fit.Axis);
916 var centerAtPoint = PointOnAxis(fit.AxisPoint, fit.Axis, axial);
917 var radial = new Vector3D(centerAtPoint, point);
918 return Math.Atan2(Vector3D.Dot(radial, v), Vector3D.Dot(radial, u));
919 }
920
921 var sideWallAngularCenters = new List<double>();
922
923 foreach (var planarFace in allFaces.Where(f => f.FaceType == FaceTypeEnum.Planar))
924 {
925 var planePoints = GetFaceAnalysisPoints(planarFace);
926 if (planePoints.Count < 3)
927 continue;
928
929 // A real U-slot side wall is parallel to the cylinder axis. Upper/lower
930 // planar faces are perpendicular to that axis and are rejected here before
931 // any adjacency or angular test, preventing the previous upper-face false
932 // positives.
933 var planeNormal = NormalizeAxis(EyeshotGeometryUtils.GetFaceAxis(planarFace));
934 if (Math.Abs(Vector3D.Dot(planeNormal, fit.Axis)) > 0.35)
935 continue;
936
937 var sharedPoints = GetConcaveSideWallSeamPoints(face, planarFace, fit);
938 if (sharedPoints.Count < 2)
939 continue;
940
941 var sharedAxialRange = GetAxialRange(sharedPoints, fit.AxisPoint, fit.Axis);
942 if (!sharedAxialRange.HasValue)
943 continue;
944
945 // Do not require 60% of the fitted depth: imported periodic cylinders can
946 // expose only the seam end vertices even when the entire wall is adjacent.
947 // A quarter of the depth is enough to reject a cap touching only one end,
948 // while still accepting the real U-shaped cylindrical face in this model.
949 double minimumSeamDepth = Math.Max(
950 MIN_HOLE_DEPTH_TOLERANCE,
951 fit.Range.Length * 0.25);
952
953 if (sharedAxialRange.Value.Length < minimumSeamDepth)
954 continue;
955
956 // The side wall must continue away from the cylindrical axis. Checking the
957 // farthest point is more reliable than checking the centroid of an irregular
958 // U-shaped planar wall, whose centroid may remain close to the bore radius.
959 double maxPlanarRadialDistance = planePoints.Max(point =>
960 DistancePointToAxis(point, fit.AxisPoint, fit.Axis));
961
962 if (maxPlanarRadialDistance <= fit.Radius + Math.Max(SHARED_VERTEX_TOLERANCE * 4.0, 0.05))
963 continue;
964
965 var seamAngles = sharedPoints
966 .Select(GetAngle)
967 .OrderBy(angle => angle)
968 .ToList();
969
970 double largestGap = 0;
971 for (int i = 0; i < seamAngles.Count; i++)
972 {
973 double current = seamAngles[i];
974 double next = i == seamAngles.Count - 1
975 ? seamAngles[0] + 2.0 * Math.PI
976 : seamAngles[i + 1];
977
978 largestGap = Math.Max(largestGap, next - current);
979 }
980
981 double angularSpan = 2.0 * Math.PI - largestGap;
982 if (angularSpan > MAX_CONCAVE_SIDE_WALL_ANGULAR_SPAN_RADIANS)
983 continue;
984
985 double meanSin = seamAngles.Average(angle => Math.Sin(angle));
986 double meanCos = seamAngles.Average(angle => Math.Cos(angle));
987 sideWallAngularCenters.Add(Math.Atan2(meanSin, meanCos));
988 }
989
990 if (sideWallAngularCenters.Count < 2)
991 return false;
992
993 // Two different side walls must flank the curved U face. This prevents a single
994 // nearby planar face from being enough to classify an exterior cylinder/fillet.
995 for (int i = 0; i < sideWallAngularCenters.Count; i++)
996 {
997 for (int j = i + 1; j < sideWallAngularCenters.Count; j++)
998 {
999 double difference =
1000 Math.Abs(sideWallAngularCenters[j] - sideWallAngularCenters[i]);
1001
1002 if (difference > Math.PI)
1003 difference = 2.0 * Math.PI - difference;
1004
1005 if (difference >= MIN_CONCAVE_SIDE_WALL_SEPARATION_RADIANS)
1006 return true;
1007 }
1008 }
1009
1010 return false;
1011 }

1044 CADFace cylinderFace,
1045 CADFace planarFace,
1046 CylinderFitInfo fit)
1047 {

1049 if (sharedPoints.Count == 0)
1050 return sharedPoints;

1053 SHARED_VERTEX_TOLERANCE * 8.0,
1054 Math.Min(0.15, Math.Max(0.05, CYLINDER_FIT_TOLERANCE * 0.15)));
1055

1057 // endpoints from the planar face by checking the fitted cylinder itself.

1198 CADFace planarFace,
1199 CADFace cylFace1,
1200 CADFace cylFace2,
1201 CylinderFitInfo fit1,
1202 CylinderFitInfo fit2)
1203 {

1206 return false;

1209 if (Math.Abs(Vector3D.Dot(planeNormal, fit1.Axis)) > 0.35)
1210 return false;

1619 if (axialDistance > axialTolerance)

1655 private static bool TryFitCylinderAxisPoint(List<Point3D> points, Vector3D axis, double expectedRadius, out Point3D axisPoint, out double fittedRadius, out double rmsError)
1656 {
1657 axisPoint = new Point3D(0, 0, 0);
1658 fittedRadius = 0;
1659 rmsError = double.MaxValue;
1660
1661 var fitPoints = DistinctPoints(points);
1662 if (fitPoints.Count < 3)
1663 return false;
1664
1665 axis = NormalizeAxis(axis);
1666 var origin = GetCentroid(fitPoints);
1667 BuildPerpendicularBasis(axis, out var u, out var v);
1668
1669 var normal = new double[3, 3];
1670 var rhs = new double[3];
1671
1672 foreach (var point in fitPoints)
1673 {
1674 var fromOrigin = new Vector3D(origin, point);
1675 double x = Vector3D.Dot(fromOrigin, u);
1676 double y = Vector3D.Dot(fromOrigin, v);
1677 double[] row = { x, y, 1.0 };
1678 double b = -(x * x + y * y);
1679
1680 for (int r = 0; r < 3; r++)
1681 {
1682 rhs[r] += row[r] * b;
1683 for (int c = 0; c < 3; c++)
1684 normal[r, c] += row[r] * row[c];
1685 }
1686 }
1687
1688 if (!TrySolve3x3(normal, rhs, out var solution))
1689 return false;
1690
1691 double centerX = -solution[0] / 2.0;
1692 double centerY = -solution[1] / 2.0;
1693 double radiusSquared = centerX * centerX + centerY * centerY - solution[2];
1694 if (radiusSquared <= 1e-9)
1695 return false;
1696
1697 fittedRadius = Math.Sqrt(radiusSquared);
1698 axisPoint = new Point3D(
1699 origin.X + u.X * centerX + v.X * centerY,
1700 origin.Y + u.Y * centerX + v.Y * centerY,
1701 origin.Z + u.Z * centerX + v.Z * centerY);
1702
1703 double sumSquaredError = 0;
1704 foreach (var point in fitPoints)
1705 {
1706 double radialDistance = DistancePointToAxis(point, axisPoint, axis);
1707 double error = radialDistance - fittedRadius;
1708 sumSquaredError += error * error;
1709 }
1710
1711 rmsError = Math.Sqrt(sumSquaredError / fitPoints.Count);
1712 double referenceRadius = expectedRadius > 0 ? expectedRadius : fittedRadius;
1713 double fitTolerance = Math.Max(CYLINDER_FIT_TOLERANCE, referenceRadius * 0.20);
1714
1715 if (expectedRadius > 0 && Math.Abs(fittedRadius - expectedRadius) > fitTolerance)
1716 return false;
1717
1718 return rmsError <= fitTolerance;
1719 }

1728 private static bool TrySolve3x3(double[,] matrix, double[] rhs, out double[] solution)
1729 {
1730 solution = new double[3];
1731 var augmented = new double[3, 4];
1732
1733 for (int r = 0; r < 3; r++)
1734 {
1735 for (int c = 0; c < 3; c++)
1736 augmented[r, c] = matrix[r, c];
1737 augmented[r, 3] = rhs[r];
1738 }
1739
1740 for (int pivot = 0; pivot < 3; pivot++)
1741 {
1742 int bestRow = pivot;
1743 double bestValue = Math.Abs(augmented[pivot, pivot]);
1744
1745 for (int r = pivot + 1; r < 3; r++)
1746 {
1747 double value = Math.Abs(augmented[r, pivot]);
1748 if (value > bestValue)
1749 {
1750 bestValue = value;
1751 bestRow = r;
1752 }
1753 }
1754
1755 if (bestValue < 1e-12)
1756 return false;
1757
1758 if (bestRow != pivot)
1759 {
1760 for (int c = pivot; c < 4; c++)
1761 {
1762 (augmented[pivot, c], augmented[bestRow, c]) = (augmented[bestRow, c], augmented[pivot, c]);
1763 }
1764 }
1765
1766 double divisor = augmented[pivot, pivot];
1767 for (int c = pivot; c < 4; c++)
1768 augmented[pivot, c] /= divisor;
1769
1770 for (int r = 0; r < 3; r++)
1771 {
1772 if (r == pivot) continue;
1773
1774 double factor = augmented[r, pivot];
1775 for (int c = pivot; c < 4; c++)
1776 augmented[r, c] -= factor * augmented[pivot, c];
1777 }
1778 }
1779
1780 solution[0] = augmented[0, 3];
1781 solution[1] = augmented[1, 3];
1782 solution[2] = augmented[2, 3];
1783 return true;
1784 }

1794 private static bool HasCompleteCircularCoverage(List<Point3D> points, Point3D axisPoint, Vector3D axis, double minCoverageRadians)
1795 {
1796 var distinct = DistinctPoints(points);
1797 if (distinct.Count < 4)
1798 return false;
1799
1800 BuildPerpendicularBasis(axis, out var u, out var v);
1801 var angles = new List<double>();
1802
1803 foreach (var point in distinct)
1804 {
1805 double axial = ProjectOnAxis(point, axisPoint, axis);
1806 var centerAtPoint = PointOnAxis(axisPoint, axis, axial);
1807 var radial = new Vector3D(centerAtPoint, point);
1808 if (radial.Length < RADIAL_TOLERANCE)
1809 continue;
1810
1811 double x = Vector3D.Dot(radial, u);
1812 double y = Vector3D.Dot(radial, v);
1813 angles.Add(Math.Atan2(y, x));
1814 }
1815
1816 if (angles.Count < 4)
1817 return false;
1818
1819 angles.Sort();
1820 double largestGap = 0;
1821
1822 for (int i = 0; i < angles.Count; i++)
1823 {
1824 double current = angles[i];
1825 double next = i == angles.Count - 1 ? angles[0] + 2.0 * Math.PI : angles[i + 1];
1826 largestGap = Math.Max(largestGap, next - current);
1827 }
1828
1829 double coverage = 2.0 * Math.PI - largestGap;
1830 return coverage >= minCoverageRadians;
1831 }

2035 private static bool TryGetCoaxialCylinderRange(CADFace cyl, Point3D axisPoint, Vector3D axis, out AxialRange range)
2036 {

2190 private static void IdentifyBridgeFaces(CADHole hole, List<CADFace> allFaces, System.Threading.CancellationToken cancellationToken = default)
2191 {

2195 return;
2196
2197 var holeAxis = NormalizeAxis(hole.AxisDirection);
2198 double holeLength = EyeshotGeometryUtils.DistanceBetweenPoints(hole.BottomCenterPoint, hole.TopCenterPoint);
2199 if (holeLength < MIN_HOLE_DEPTH_TOLERANCE)
2200 return;
2201
2202 foreach (var face in planarFaces)
2203 {
2204 cancellationToken.ThrowIfCancellationRequested();
2205 // Skip if already marked as bridge
2206 if (face.IsBridgeBetweenHoles) continue;

2209 if (TryGetPlanarFaceAxisPoint(face, hole.BottomCenterPoint, holeAxis, out var faceAxisPoint, out var axialPosition))
2210 {
2211 if (axialPosition >= -BRIDGE_AXIAL_TOLERANCE && axialPosition <= holeLength + BRIDGE_AXIAL_TOLERANCE)
2212 {

2217 {

2219 if (IsCircularBridgeFaceCandidate(face, faceAxisPoint, holeAxis, referenceDiameter))
2220 {
2221 face.IsBridgeBetweenHoles = true;
2222 continue;
2223 }
2224 }
2225 }
2226 }
2227
2228 // --- Secondary (shoulder/step) path: permissive centroid-to-axis-line distance check ---
2229 // Catches large annular "shoulder" faces (e.g. counterbore steps) whose edge points
2230 // extend beyond the IsBridgeFaceRadiallyBounded limit but whose centroid IS
2231 // geometrically near the hole axis. This mirrors the heuristic used in
2232 // ShellBasedPathPlanner.GeneratePath() Phase 2.
2233 var facePoints = GetFaceAnalysisPoints(face);
2234 if (facePoints.Count < 3) continue;
2235
2236 var faceNormal = NormalizeAxis(EyeshotGeometryUtils.GetFaceAxis(face));
2237 if (!EyeshotGeometryUtils.AreVectorsParallel(faceNormal, holeAxis, PARALLEL_AXIS_TOLERANCE)) continue;
2238
2239 var faceCentroid = GetCentroid(facePoints);
2240
2241 // Face centroid must be within 1.8× the hole radius of the hole axis line
2242 double holeRadius = Math.Max(hole.TopDiameter, hole.BottomDiameter) / 2.0;
2243 if (holeRadius <= 0 || double.IsNaN(holeRadius)) continue;
2244
2245 double distToAxis = DistancePointToAxis(faceCentroid, hole.BottomCenterPoint!, holeAxis);
2246 if (distToAxis > holeRadius * 1.8) continue;
2247
2248 // Face centroid must be axially near the top or bottom of the hole (within 6mm or 10% of hole length)
2249 double shoulderAxialPos = ProjectOnAxis(faceCentroid, hole.BottomCenterPoint!, holeAxis);
2250 double nearTopDist = Math.Abs(shoulderAxialPos - holeLength);
2251 double nearBotDist = Math.Abs(shoulderAxialPos);
2252 double axialMargin = Math.Max(6.0, holeLength * 0.10);
2253 if (nearTopDist > axialMargin && nearBotDist > axialMargin) continue;
2254
2255 // The hole center must be "above" the face normal (face looks toward the hole interior)
2256 var holeCenter = new Point3D(
2257 (hole.TopCenterPoint!.X + hole.BottomCenterPoint!.X) * 0.5,
2258 (hole.TopCenterPoint!.Y + hole.BottomCenterPoint!.Y) * 0.5,
2259 (hole.TopCenterPoint!.Z + hole.BottomCenterPoint!.Z) * 0.5);
2260 var faceToHole = new Vector3D(faceCentroid, holeCenter);

2262 if (faceToHoleDot <= 0.5 && faceToHoleDot >= -0.5) continue; // must face toward or away from hole, not be orthogonal
2263
2264 // Final check: there must be a coaxial cylinder on both sides of this face's axial position
2265 var shoulderAxisPoint = PointOnAxis(hole.BottomCenterPoint!, holeAxis, shoulderAxialPos);
2266 bool hasCylBelowShoulder = HasCoaxialCylinderTouchingSide(shoulderAxisPoint, holeAxis, cylindricalFaces, -1, null, out _);
2267 bool hasCylAboveShoulder = HasCoaxialCylinderTouchingSide(shoulderAxisPoint, holeAxis, cylindricalFaces, 1, null, out _);
2268 if (!hasCylBelowShoulder || !hasCylAboveShoulder) continue;
2269
2270 face.IsBridgeBetweenHoles = true;
2271 }

2625 private static void EvaluateTopReachabilityGroup(List<HoleReachabilityInfo> group, Vector3D approachAxis, List<Models.CADFace> allFaces, bool avoidBottomFace)
2626 {
2627 if (!avoidBottomFace || group.Count == 0)
2628 return;
2629
2630 var sorted = group.OrderByDescending(info => info.TopProjection).ToList();
2631
2632 double bottleneckDiameter = double.PositiveInfinity;
2633 double channelBottomProjection = double.PositiveInfinity;
2634 bool hasTopAccess = true;
2635 const double axialConnectionTolerance = 2.0;
2636
2637 for (int i = 0; i < sorted.Count; i++)
2638 {
2639 var info = sorted[i];
2640
2641 if (!hasTopAccess)
2642 {
2643 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2644 continue;
2645 }
2646
2647 if (i == 0 &&
2648 IsPhysicalTopEndClosed(info.Hole, info.TopEnd, approachAxis, allFaces))
2649 {
2650 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2651 hasTopAccess = false;
2652 continue;
2653 }
2654
2655 // A lower section that does not meet the channel above is not reachable
2656 // from the top while the underside is unavailable.
2657 if (i > 0 &&
2658 info.TopProjection < channelBottomProjection - axialConnectionTolerance)
2659 {
2660 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2661 hasTopAccess = false;
2662 continue;
2663 }
2664
2665 // The tool must fit through every preceding bore. If this feature is
2666 // wider than that bottleneck, its walls are unreachable. Keep tracking
2667 // its passage diameter, however: a still-smaller hole farther down may
2668 // remain reachable through this larger void.
2669 if (info.RequiredDiameter > bottleneckDiameter + OBSTRUCTION_DIAMETER_TOLERANCE)
2670 {
2671 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2672 }
2673
2674 bottleneckDiameter = Math.Min(bottleneckDiameter, info.PassageDiameter);
2675 channelBottomProjection = Math.Min(channelBottomProjection, info.BottomProjection);
2676 }

2691

2693 if (double.IsNaN(holeDiameter) || holeDiameter <= 0) holeDiameter = 1.0;
2694 double holeRadius = holeDiameter / 2.0;
2695

2697 {

2704 double maxRadius = double.MinValue;
2705

2709 if (dist < minRadius) minRadius = dist;
2710 if (dist > maxRadius) maxRadius = dist;

2714
2715 // 1. If maxRadius <= holeRadius + margin: It's a CAP exactly matching the hole. -> Blocked

2723 // 3. If minRadius <= holeRadius + margin AND maxRadius > holeRadius + margin:
2724 // It's an annulus (SHELF) like the bottom of a counterbore. It has a hole in the middle. -> NOT Blocked
2725 // So we do NOT return true here, continue checking other faces.
2726 }
2727 }
2728 return false;
