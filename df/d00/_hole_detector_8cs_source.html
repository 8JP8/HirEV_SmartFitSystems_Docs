140 public static (List<CADHole> holes, List<CADFace> nonHoleFaces) DetectHoles(List<CADFace> allFaces, Action<int, int>? progressCallback = null, System.Threading.CancellationToken cancellationToken = default)
141 {
142 var holes = new List<CADHole>();
143 var processedFaceIndices = new HashSet<int>();
144
145 foreach (var face in allFaces)
146 {
147 face.IsBridgeBetweenHoles = false;
148 }
149
150 progressCallback?.Invoke(10, 0);
151
152 var cylindricalFaces = allFaces
153 .Where(f => f.FaceType == FaceTypeEnum.Cylindrical || f.FaceType == FaceTypeEnum.Conical)
154 .ToList();
155
156 progressCallback?.Invoke(30, 0);
157
158 var holeGroups = GroupFacesIntoHoles(cylindricalFaces, cancellationToken);
159
160 progressCallback?.Invoke(50, 0);
161
162 int holeId = 1;
163 int totalGroups = holeGroups.Count;
164 int processedGroups = 0;
165
166 foreach (var group in holeGroups)
167 {
168 cancellationToken.ThrowIfCancellationRequested();
169
170 if (group.Count >= 2)
171 {
172 var hole = CreateHoleFromFaces(group, holeId, allFaces);
173 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
174 {
175 holes.Add(hole);
176 holeId++;
177 foreach (var f in group)
178 processedFaceIndices.Add(f.FaceGlobalIndex);
179 }
180 }
181
182 processedGroups++;
183 if (progressCallback != null && totalGroups > 0)
184 {
185 int percentage = 50 + (int)((processedGroups * 50.0) / totalGroups);
186 progressCallback(percentage, holes.Count);
187 }
188 }
189
190 // Post-process: Identify bridge faces for all detected holes
191 foreach (var hole in holes)
192 {
193 cancellationToken.ThrowIfCancellationRequested();
194 IdentifyBridgeFaces(hole, allFaces, cancellationToken);
195 }
196
197 var nonHoleFaces = allFaces
198 .Where(f => !processedFaceIndices.Contains(f.FaceGlobalIndex))
199 .ToList();
200
201 progressCallback?.Invoke(100, holes.Count);
202
203 return (holes, nonHoleFaces);
204 }

387 public static string IdentifyStandardHoleType(double diameter)
388 {
389 if (double.IsNaN(diameter) || diameter <= 0)
390 return "Unknown";
391
392 var closestMatch = StandardHoleDiameters
393 .Select(kvp => new { kvp.Key, kvp.Value, Diff = Math.Abs(kvp.Value - diameter) })
394 .OrderBy(x => x.Diff)
395 .FirstOrDefault();
396
397 if (closestMatch != null && closestMatch.Diff <= DIAMETER_TOLERANCE)
398 return closestMatch.Key;
399
400 var clearanceMatch = StandardHoleDiameters
401 .Select(kvp => new
402 {
403 Standard = kvp.Key + " Clearance",
404 NominalDiameter = double.Parse(kvp.Key.Substring(1)),
405 TapDiameter = kvp.Value,
406 Diff = Math.Abs(diameter - (double.Parse(kvp.Key.Substring(1)) + 0.5))
407 })
408 .OrderBy(x => x.Diff)
409 .FirstOrDefault();
410
411 if (clearanceMatch != null && clearanceMatch.Diff <= DIAMETER_TOLERANCE)
412 return clearanceMatch.Standard;
413
414 return $"Ø{diameter:F2}mm";
415 }

420 private static List<List<CADFace>> GroupFacesIntoHoles(List<CADFace> cylindricalFaces, System.Threading.CancellationToken cancellationToken = default)
421 {
422 var groups = new List<List<CADFace>>();
423 var processed = new HashSet<int>();
424
425 foreach (var face in cylindricalFaces)
426 {
427 cancellationToken.ThrowIfCancellationRequested();
428 if (processed.Contains(face.FaceGlobalIndex)) continue;
429
430 var group = new List<CADFace>();
431 var queue = new Queue<CADFace>();
432 queue.Enqueue(face);
433 processed.Add(face.FaceGlobalIndex);
434
435 while (queue.Count > 0)
436 {
437 cancellationToken.ThrowIfCancellationRequested();
438 var current = queue.Dequeue();
439 group.Add(current);
440
441 double radius1 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(current);
442 if (radius1 <= 0) continue;
443
444 foreach (var other in cylindricalFaces)
445 {
446 if (processed.Contains(other.FaceGlobalIndex)) continue;
447
448 double radius2 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(other);
449 if (radius2 <= 0 || Math.Abs(radius1 - radius2) > RADIAL_TOLERANCE)
450 continue;
451
452 // Check if faces are companions (split halves) or stacked segments
453 if (AreFacesPairedForHole(current, other) || AreFacesStacked(current, other))
454 {
455 processed.Add(other.FaceGlobalIndex);
456 queue.Enqueue(other);
457 }
458 }
459 }
460
461 if (group.Count > 0)
462 groups.Add(group);
463 }
464
465 return groups;
466 }

471 private static bool AreFacesStacked(CADFace face1, CADFace face2)
472 {
473 var pts1 = GetFaceAnalysisPoints(face1);
474 var pts2 = GetFaceAnalysisPoints(face2);
475
476 if (pts1.Count == 0 || pts2.Count == 0)
477 return false;
478
479 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
480 return false;
481
482 var combinedPoints = pts1.Concat(pts2).ToList();
483 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
484 return false;
485
486 var range1 = GetAxialRange(pts1, axisPoint, axis);
487 var range2 = GetAxialRange(pts2, axisPoint, axis);
488 if (!range1.HasValue || !range2.HasValue)
489 return false;
490
491 double gap = GetAxialGap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
492
493 // They are stacked if they are coaxial and their projected extents touch or are
494 // close enough to jump over a real separating bridge face.
495 return gap >= -BRIDGE_AXIAL_TOLERANCE && gap < MAX_JUMP_GAP;
496 }

502 private static bool AreFacesPairedForHole(CADFace face1, CADFace face2)
503 {
504 var pts1 = GetFaceAnalysisPoints(face1);
505 var pts2 = GetFaceAnalysisPoints(face2);
506
507 if (pts1.Count == 0 || pts2.Count == 0)
508 return false;
509
510 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
511 return false;
512
513 double range1Length = GetProjectedRangeLength(pts1, axis);
514 double range2Length = GetProjectedRangeLength(pts2, axis);
515 double maxAxialRange = Math.Max(range1Length, range2Length);
516 double adaptiveAxialTolerance = Math.Max(Z_DEVIATION_TOLERANCE, maxAxialRange * 0.15);
517
518 // Complementary split cylinder faces may have centroids on opposite sides of
519 // the bore. Check shared seam points before any centroid-based rejection.
520 if (CountPointMatches(pts1, pts2, axis, SHARED_VERTEX_TOLERANCE * 2.0, adaptiveAxialTolerance) >= SEAM_MATCHES_REQUIRED)
521 return true;
522
523 var combinedPoints = pts1.Concat(pts2).ToList();
524 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
525 return false;
526
527 var range1 = GetAxialRange(pts1, axisPoint, axis);
528 var range2 = GetAxialRange(pts2, axisPoint, axis);
529 if (!range1.HasValue || !range2.HasValue)
530 return false;
531
532 double overlap = GetAxialOverlap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
533 if (overlap < Math.Min(range1.Value.Length, range2.Value.Length) * 0.10 &&
534 overlap < MIN_HOLE_DEPTH_TOLERANCE)
535 {
536 return false;
537 }
538
539 if (HasCompleteCircularCoverage(combinedPoints, axisPoint, axis, MIN_CIRCULAR_COVERAGE_RADIANS))
540 return true;
541
542 int matchedPoints = CountPointMatches(pts1, pts2, axis, RADIAL_TOLERANCE, adaptiveAxialTolerance);
543 double similarity = (double)matchedPoints / Math.Max(pts1.Count, pts2.Count);
544
545 // Lower threshold for curved surfaces (70% vs 90%)
546 double threshold = maxAxialRange > MIN_HOLE_DEPTH_TOLERANCE ? 0.70 : 0.85;
547
548 return similarity >= threshold;
549 }

554 private static bool HaveSimilarControlPointsWithDeviation(List<Point4D> pts1, List<Point4D> pts2)
555 {
556 // Convert to CADFace-like structure for the enhanced check
557 // This is a simplified adapter - the main grouping now uses AreFacesPairedForHole
558 if (pts1.Count == 0 || pts2.Count == 0) return false;
559
560 int matched = 0;
561 double zRange = Math.Max(
562 pts1.Max(p => p.Z) - pts1.Min(p => p.Z),
563 pts2.Max(p => p.Z) - pts2.Min(p => p.Z)
564 );
565
566 double dynamicZTolerance = Math.Max(Z_DEVIATION_TOLERANCE, zRange * 0.15);
567
568 foreach (var p1 in pts1)
569 {
570 if (pts2.Any(p2 =>
571 Math.Abs(p1.X - p2.X) < SHARED_VERTEX_TOLERANCE &&
572 Math.Abs(p1.Y - p2.Y) < SHARED_VERTEX_TOLERANCE &&
573 Math.Abs(p1.Z - p2.Z) < dynamicZTolerance))
574 {
575 matched++;
576 }
577 }
578
579 double similarity = (double)matched / Math.Max(pts1.Count, pts2.Count);
580 return similarity >= 0.70; // lowered threshold for curved surfaces
581 }

591 private static CADHole CreateHoleFromFaces(List<CADFace> faces, int holeId, List<CADFace> allFaces, bool isManual = false)
592 {
593 if (faces.Count < 2) return null!;
594
595 var allControlPoints = faces.SelectMany(GetFaceAnalysisPoints).ToList();
596
597 if (allControlPoints.Count == 0) return null!;
598
599 var axis = GetDominantFaceAxis(faces);
600 double avgRadius = faces
601 .Select(EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo)
602 .Where(r => r > 0)
603 .DefaultIfEmpty(0)
604 .Average();
605
606 Point3D bottomCenter;
607 Point3D topCenter;
608 CADFace topFace;
609 CADFace bottomFace;
610
611 if (avgRadius > 0 &&
612 TryFitCylinderAxisPoint(allControlPoints, axis, avgRadius, out var axisPoint, out _, out _))
613 {
614 var range = GetAxialRange(allControlPoints, axisPoint, axis);
615 if (!range.HasValue || range.Value.Length < MIN_HOLE_DEPTH_TOLERANCE)
616 return null!;
617
618 bottomCenter = PointOnAxis(axisPoint, axis, range.Value.Min);
619 topCenter = PointOnAxis(axisPoint, axis, range.Value.Max);
620
621 topFace = faces.OrderByDescending(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
622 bottomFace = faces.OrderBy(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
623 }
624 else
625 {
626 var sortedPoints = allControlPoints.OrderBy(p => p.Z).ToList();
627 double zMin = sortedPoints.First().Z;
628 double zMax = sortedPoints.Last().Z;
629
630 if (Math.Abs(zMax - zMin) < MIN_HOLE_DEPTH_TOLERANCE)
631 return null!;
632
633 var finalBottomPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMin) < SHARED_VERTEX_TOLERANCE * 2).ToList();
634 var finalTopPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMax) < SHARED_VERTEX_TOLERANCE * 2).ToList();
635
636 bottomCenter = GetCentroid(finalBottomPoints);
637 topCenter = GetCentroid(finalTopPoints);
638 topFace = faces.OrderByDescending(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
639 bottomFace = faces.OrderBy(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
640 }
641
642 var hole = new CADHole
643 {
644 HoleId = holeId,
645 HoleColor = faces[0].FaceColor,
646 FaceGlobalIndices = faces.Select(f => f.FaceGlobalIndex).ToList(),
647 TopCenterPoint = topCenter,
648 BottomCenterPoint = bottomCenter
649 };
650
651 hole.Depth = EyeshotGeometryUtils.DistanceBetweenPoints(hole.BottomCenterPoint, hole.TopCenterPoint);
652
653 var axisVector = hole.TopCenterPoint.AsVector - hole.BottomCenterPoint.AsVector;
654 axisVector.Normalize();
655 hole.AxisDirection = axisVector;
656
657 if (!isManual && !IsHoleOpen(hole.BottomCenterPoint, hole.TopCenterPoint, faces, allFaces, hole.AxisDirection))
658 return null!;
659
660 if (hole.Depth < MIN_HOLE_DEPTH_TOLERANCE) return null!;
661
662 // Determine diameters from the extreme faces in the stack
663 hole.TopDiameter = GetFaceDiameter(topFace);
664 hole.BottomDiameter = GetFaceDiameter(bottomFace);
665 double avgDiameter = (hole.TopDiameter + hole.BottomDiameter) / 2.0;
666 hole.HoleStandard = IdentifyStandardHoleType(avgDiameter);
667 hole.GeometricInfo = $"{hole.HoleStandard} - Depth: {hole.Depth:F2}mm, Ø{avgDiameter:F2}";
668
669 return hole;
670 }

959 private static bool ArePointsNearAlongAxis(Point3D p1, Point3D p2, Vector3D axis, double radialTolerance, double axialTolerance)
960 {
961 var delta = new Vector3D(p1, p2);
962 double axialDistance = Math.Abs(Vector3D.Dot(delta, axis));
963 if (axialDistance > axialTolerance)

999 private static bool TryFitCylinderAxisPoint(List<Point3D> points, Vector3D axis, double expectedRadius, out Point3D axisPoint, out double fittedRadius, out double rmsError)
1000 {
1001 axisPoint = new Point3D(0, 0, 0);
1002 fittedRadius = 0;
1003 rmsError = double.MaxValue;
1004
1005 var fitPoints = DistinctPoints(points);
1006 if (fitPoints.Count < 3)
1007 return false;
1008
1009 axis = NormalizeAxis(axis);
1010 var origin = GetCentroid(fitPoints);
1011 BuildPerpendicularBasis(axis, out var u, out var v);
1012
1013 var normal = new double[3, 3];
1014 var rhs = new double[3];
1015
1016 foreach (var point in fitPoints)
1017 {
1018 var fromOrigin = new Vector3D(origin, point);
1019 double x = Vector3D.Dot(fromOrigin, u);
1020 double y = Vector3D.Dot(fromOrigin, v);
1021 double[] row = { x, y, 1.0 };
1022 double b = -(x * x + y * y);
1023
1024 for (int r = 0; r < 3; r++)
1025 {
1026 rhs[r] += row[r] * b;
1027 for (int c = 0; c < 3; c++)
1028 normal[r, c] += row[r] * row[c];
1029 }
1030 }
1031
1032 if (!TrySolve3x3(normal, rhs, out var solution))
1033 return false;
1034
1035 double centerX = -solution[0] / 2.0;
1036 double centerY = -solution[1] / 2.0;
1037 double radiusSquared = centerX * centerX + centerY * centerY - solution[2];
1038 if (radiusSquared <= 1e-9)
1039 return false;
1040
1041 fittedRadius = Math.Sqrt(radiusSquared);
1042 axisPoint = new Point3D(
1043 origin.X + u.X * centerX + v.X * centerY,
1044 origin.Y + u.Y * centerX + v.Y * centerY,
1045 origin.Z + u.Z * centerX + v.Z * centerY);
1046
1047 double sumSquaredError = 0;
1048 foreach (var point in fitPoints)
1049 {
1050 double radialDistance = DistancePointToAxis(point, axisPoint, axis);
1051 double error = radialDistance - fittedRadius;
1052 sumSquaredError += error * error;
1053 }
1054
1055 rmsError = Math.Sqrt(sumSquaredError / fitPoints.Count);
1056 double referenceRadius = expectedRadius > 0 ? expectedRadius : fittedRadius;
1057 double fitTolerance = Math.Max(CYLINDER_FIT_TOLERANCE, referenceRadius * 0.20);
1058
1059 if (expectedRadius > 0 && Math.Abs(fittedRadius - expectedRadius) > fitTolerance)
1060 return false;
1061
1062 return rmsError <= fitTolerance;
1063 }

1072 private static bool TrySolve3x3(double[,] matrix, double[] rhs, out double[] solution)
1073 {
1074 solution = new double[3];
1075 var augmented = new double[3, 4];
1076
1077 for (int r = 0; r < 3; r++)
1078 {
1079 for (int c = 0; c < 3; c++)
1080 augmented[r, c] = matrix[r, c];
1081 augmented[r, 3] = rhs[r];
1082 }
1083
1084 for (int pivot = 0; pivot < 3; pivot++)
1085 {
1086 int bestRow = pivot;
1087 double bestValue = Math.Abs(augmented[pivot, pivot]);
1088
1089 for (int r = pivot + 1; r < 3; r++)
1090 {
1091 double value = Math.Abs(augmented[r, pivot]);
1092 if (value > bestValue)
1093 {
1094 bestValue = value;
1095 bestRow = r;
1096 }
1097 }
1098
1099 if (bestValue < 1e-12)
1100 return false;
1101
1102 if (bestRow != pivot)
1103 {
1104 for (int c = pivot; c < 4; c++)
1105 {
1106 (augmented[pivot, c], augmented[bestRow, c]) = (augmented[bestRow, c], augmented[pivot, c]);
1107 }
1108 }
1109
1110 double divisor = augmented[pivot, pivot];
1111 for (int c = pivot; c < 4; c++)
1112 augmented[pivot, c] /= divisor;
1113
1114 for (int r = 0; r < 3; r++)
1115 {
1116 if (r == pivot) continue;
1117
1118 double factor = augmented[r, pivot];
1119 for (int c = pivot; c < 4; c++)
1120 augmented[r, c] -= factor * augmented[pivot, c];
1121 }
1122 }
1123
1124 solution[0] = augmented[0, 3];
1125 solution[1] = augmented[1, 3];
1126 solution[2] = augmented[2, 3];
1127 return true;
1128 }

1138 private static bool HasCompleteCircularCoverage(List<Point3D> points, Point3D axisPoint, Vector3D axis, double minCoverageRadians)
1139 {
1140 var distinct = DistinctPoints(points);
1141 if (distinct.Count < 4)
1142 return false;
1143
1144 BuildPerpendicularBasis(axis, out var u, out var v);
1145 var angles = new List<double>();
1146
1147 foreach (var point in distinct)
1148 {
1149 double axial = ProjectOnAxis(point, axisPoint, axis);
1150 var centerAtPoint = PointOnAxis(axisPoint, axis, axial);
1151 var radial = new Vector3D(centerAtPoint, point);
1152 if (radial.Length < RADIAL_TOLERANCE)
1153 continue;
1154
1155 double x = Vector3D.Dot(radial, u);
1156 double y = Vector3D.Dot(radial, v);
1157 angles.Add(Math.Atan2(y, x));
1158 }
1159
1160 if (angles.Count < 4)
1161 return false;
1162
1163 angles.Sort();
1164 double largestGap = 0;
1165
1166 for (int i = 0; i < angles.Count; i++)
1167 {
1168 double current = angles[i];
1169 double next = i == angles.Count - 1 ? angles[0] + 2.0 * Math.PI : angles[i + 1];
1170 largestGap = Math.Max(largestGap, next - current);
1171 }
1172
1173 double coverage = 2.0 * Math.PI - largestGap;
1174 return coverage >= minCoverageRadians;
1175 }
