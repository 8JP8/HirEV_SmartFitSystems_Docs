161 public static (List<CADHole> holes, List<CADFace> nonHoleFaces) DetectHoles(List<CADFace> allFaces, Action<int, int>? progressCallback = null, System.Threading.CancellationToken cancellationToken = default)
162 {
163 var holes = new List<CADHole>();
164 var processedFaceIndices = new HashSet<int>();
165
166 foreach (var face in allFaces)
167 {
168 face.IsBridgeBetweenHoles = false;
169 }
170
171 progressCallback?.Invoke(10, 0);
172
173 var cylindricalFaces = allFaces
174 .Where(f => f.FaceType == FaceTypeEnum.Cylindrical || f.FaceType == FaceTypeEnum.Conical)
175 .ToList();
176
177 progressCallback?.Invoke(30, 0);
178
179 var holeGroups = GroupFacesIntoHoles(cylindricalFaces, allFaces, cancellationToken);
180
181 progressCallback?.Invoke(50, 0);
182
183 int holeId = 1;
184 int totalGroups = holeGroups.Count;
185 int processedGroups = 0;
186
187 foreach (var group in holeGroups)
188 {
189 cancellationToken.ThrowIfCancellationRequested();
190
191 if (group.Count >= 2)
192 {
193 var hole = CreateHoleFromFaces(group, holeId, allFaces);
194 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
195 {
196 holes.Add(hole);
197 holeId++;
198 foreach (var f in group)
199 processedFaceIndices.Add(f.FaceGlobalIndex);
200 }
201 }
202 else if (group.Count == 1 && ENABLE_SINGLE_FACE_JOINED_HOLES)
203 {
204 var face = group[0];
205 if (IsSingleFaceJoinedHoleCandidate(face, cylindricalFaces, allFaces))
206 {
207 var hole = CreateHoleFromSingleFace(face, holeId, allFaces);
208 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
209 {
210 holes.Add(hole);
211 holeId++;
212 processedFaceIndices.Add(face.FaceGlobalIndex);
213 }
214 }
215 }
216
217 processedGroups++;
218 if (progressCallback != null && totalGroups > 0)
219 {
220 int percentage = 50 + (int)((processedGroups * 50.0) / totalGroups);
221 progressCallback(percentage, holes.Count);
222 }
223 }
224
225 // Post-process: Identify bridge faces for all detected holes
226 foreach (var hole in holes)
227 {
228 cancellationToken.ThrowIfCancellationRequested();
229 IdentifyBridgeFaces(hole, allFaces, cancellationToken);
230 }
231
232 var nonHoleFaces = allFaces
233 .Where(f => !processedFaceIndices.Contains(f.FaceGlobalIndex))
234 .ToList();
235
236 progressCallback?.Invoke(100, holes.Count);
237
238 return (holes, nonHoleFaces);
239 }

421 public static string IdentifyStandardHoleType(double diameter)
422 {
423 if (double.IsNaN(diameter) || diameter <= 0)
424 return "Unknown";
425
426 var closestMatch = StandardHoleDiameters
427 .Select(kvp => new { kvp.Key, kvp.Value, Diff = Math.Abs(kvp.Value - diameter) })
428 .OrderBy(x => x.Diff)
429 .FirstOrDefault();
430
431 if (closestMatch != null && closestMatch.Diff <= DIAMETER_TOLERANCE)
432 return closestMatch.Key;
433
434 var clearanceMatch = StandardHoleDiameters
435 .Select(kvp => new
436 {
437 Standard = kvp.Key + " Clearance",
438 NominalDiameter = double.Parse(kvp.Key.Substring(1)),
439 TapDiameter = kvp.Value,
440 Diff = Math.Abs(diameter - (double.Parse(kvp.Key.Substring(1)) + 0.5))
441 })
442 .OrderBy(x => x.Diff)
443 .FirstOrDefault();
444
445 if (clearanceMatch != null && clearanceMatch.Diff <= DIAMETER_TOLERANCE)
446 return clearanceMatch.Standard;
447
448 return $"Ø{diameter:F2}mm";
449 }

454 private static List<List<CADFace>> GroupFacesIntoHoles(List<CADFace> cylindricalFaces, List<CADFace> allFaces, System.Threading.CancellationToken cancellationToken = default)
455 {
456 var groups = new List<List<CADFace>>();
457 var processed = new HashSet<int>();
458
459 foreach (var face in cylindricalFaces)
460 {
461 cancellationToken.ThrowIfCancellationRequested();
462 if (processed.Contains(face.FaceGlobalIndex)) continue;
463
464 var group = new List<CADFace>();
465 var queue = new Queue<CADFace>();
466 queue.Enqueue(face);
467 processed.Add(face.FaceGlobalIndex);
468
469 while (queue.Count > 0)
470 {
471 cancellationToken.ThrowIfCancellationRequested();
472 var current = queue.Dequeue();
473 group.Add(current);
474
475 double radius1 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(current);
476 if (radius1 <= 0) continue;
477
478 foreach (var other in cylindricalFaces)
479 {
480 if (processed.Contains(other.FaceGlobalIndex)) continue;
481
482 double radius2 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(other);
483 if (radius2 <= 0 || Math.Abs(radius1 - radius2) > RADIAL_TOLERANCE)
484 continue;
485
486 if (AreCylindersSeparatedByPlanarFaces(current, other, allFaces))
487 continue;
488
489 // Check if faces are companions (split halves) or stacked segments
490 if (AreFacesPairedForHole(current, other) || AreFacesStacked(current, other))
491 {
492 processed.Add(other.FaceGlobalIndex);
493 queue.Enqueue(other);
494 }
495 }
496 }
497
498 if (group.Count > 0)
499 groups.Add(group);
500 }
501
502 return groups;
503 }

508 private static bool AreFacesStacked(CADFace face1, CADFace face2)
509 {
510 var pts1 = GetFaceAnalysisPoints(face1);
511 var pts2 = GetFaceAnalysisPoints(face2);
512
513 if (pts1.Count == 0 || pts2.Count == 0)
514 return false;
515
516 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
517 return false;
518
519 if (AreCylinderAxesSeparated(face1, face2, axis, radius, out _))
520 return false;
521
522 var combinedPoints = pts1.Concat(pts2).ToList();
523 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
524 return false;
525
526 var range1 = GetAxialRange(pts1, axisPoint, axis);
527 var range2 = GetAxialRange(pts2, axisPoint, axis);
528 if (!range1.HasValue || !range2.HasValue)
529 return false;
530
531 double gap = GetAxialGap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
532
533 // They are stacked if they are coaxial and their projected extents touch or are
534 // close enough to jump over a real separating bridge face.
535 return gap >= -BRIDGE_AXIAL_TOLERANCE && gap < MAX_JUMP_GAP;
536 }

542 private static bool AreFacesPairedForHole(CADFace face1, CADFace face2)
543 {
544 var pts1 = GetFaceAnalysisPoints(face1);
545 var pts2 = GetFaceAnalysisPoints(face2);
546
547 if (pts1.Count == 0 || pts2.Count == 0)
548 return false;
549
550 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
551 return false;
552
553 if (AreCylinderAxesSeparated(face1, face2, axis, radius, out _))
554 return false;
555
556 double range1Length = GetProjectedRangeLength(pts1, axis);
557 double range2Length = GetProjectedRangeLength(pts2, axis);
558 double maxAxialRange = Math.Max(range1Length, range2Length);
559 double adaptiveAxialTolerance = Math.Max(Z_DEVIATION_TOLERANCE, maxAxialRange * 0.15);
560
561 // Complementary split cylinder faces may have centroids on opposite sides of
562 // the bore. Check shared seam points before any centroid-based rejection.
563 if (CountPointMatches(pts1, pts2, axis, SHARED_VERTEX_TOLERANCE * 2.0, adaptiveAxialTolerance) >= SEAM_MATCHES_REQUIRED)
564 return true;
565
566 var combinedPoints = pts1.Concat(pts2).ToList();
567 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
568 return false;
569
570 var range1 = GetAxialRange(pts1, axisPoint, axis);
571 var range2 = GetAxialRange(pts2, axisPoint, axis);
572 if (!range1.HasValue || !range2.HasValue)
573 return false;
574
575 double overlap = GetAxialOverlap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
576 if (overlap < Math.Min(range1.Value.Length, range2.Value.Length) * 0.10 &&
577 overlap < MIN_HOLE_DEPTH_TOLERANCE)
578 {
579 return false;
580 }
581
582 if (HasCompleteCircularCoverage(combinedPoints, axisPoint, axis, MIN_CIRCULAR_COVERAGE_RADIANS))
583 return true;
584
585 int matchedPoints = CountPointMatches(pts1, pts2, axis, RADIAL_TOLERANCE, adaptiveAxialTolerance);
586 double similarity = (double)matchedPoints / Math.Max(pts1.Count, pts2.Count);
587
588 // Lower threshold for curved surfaces (70% vs 90%)
589 double threshold = maxAxialRange > MIN_HOLE_DEPTH_TOLERANCE ? 0.70 : 0.85;
590
591 return similarity >= threshold;
592 }

597 private static bool HaveSimilarControlPointsWithDeviation(List<Point4D> pts1, List<Point4D> pts2)
598 {
599 // Convert to CADFace-like structure for the enhanced check
600 // This is a simplified adapter - the main grouping now uses AreFacesPairedForHole
601 if (pts1.Count == 0 || pts2.Count == 0) return false;
602
603 int matched = 0;
604 double zRange = Math.Max(
605 pts1.Max(p => p.Z) - pts1.Min(p => p.Z),
606 pts2.Max(p => p.Z) - pts2.Min(p => p.Z)
607 );
608
609 double dynamicZTolerance = Math.Max(Z_DEVIATION_TOLERANCE, zRange * 0.15);
610
611 foreach (var p1 in pts1)
612 {
613 if (pts2.Any(p2 =>
614 Math.Abs(p1.X - p2.X) < SHARED_VERTEX_TOLERANCE &&
615 Math.Abs(p1.Y - p2.Y) < SHARED_VERTEX_TOLERANCE &&
616 Math.Abs(p1.Z - p2.Z) < dynamicZTolerance))
617 {
618 matched++;
619 }
620 }
621
622 double similarity = (double)matched / Math.Max(pts1.Count, pts2.Count);
623 return similarity >= 0.70; // lowered threshold for curved surfaces
624 }

634 private static CADHole CreateHoleFromFaces(List<CADFace> faces, int holeId, List<CADFace> allFaces, bool isManual = false)
635 {
636 if (faces.Count < 2) return null!;
637
638 var allControlPoints = faces.SelectMany(GetFaceAnalysisPoints).ToList();
639
640 if (allControlPoints.Count == 0) return null!;
641
642 var axis = GetDominantFaceAxis(faces);
643 double avgRadius = faces
644 .Select(EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo)
645 .Where(r => r > 0)
646 .DefaultIfEmpty(0)
647 .Average();
648
649 Point3D bottomCenter;
650 Point3D topCenter;
651 CADFace topFace;
652 CADFace bottomFace;
653
654 if (avgRadius > 0 &&
655 TryFitCylinderAxisPoint(allControlPoints, axis, avgRadius, out var axisPoint, out _, out _))
656 {
657 var range = GetAxialRange(allControlPoints, axisPoint, axis);
658 if (!range.HasValue || range.Value.Length < MIN_HOLE_DEPTH_TOLERANCE)
659 return null!;
660
661 bottomCenter = PointOnAxis(axisPoint, axis, range.Value.Min);
662 topCenter = PointOnAxis(axisPoint, axis, range.Value.Max);
663
664 topFace = faces.OrderByDescending(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
665 bottomFace = faces.OrderBy(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
666 }
667 else
668 {
669 var sortedPoints = allControlPoints.OrderBy(p => p.Z).ToList();
670 double zMin = sortedPoints.First().Z;
671 double zMax = sortedPoints.Last().Z;
672
673 if (Math.Abs(zMax - zMin) < MIN_HOLE_DEPTH_TOLERANCE)
674 return null!;
675
676 var finalBottomPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMin) < SHARED_VERTEX_TOLERANCE * 2).ToList();
677 var finalTopPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMax) < SHARED_VERTEX_TOLERANCE * 2).ToList();
678
679 bottomCenter = GetCentroid(finalBottomPoints);
680 topCenter = GetCentroid(finalTopPoints);
681 topFace = faces.OrderByDescending(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
682 bottomFace = faces.OrderBy(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
683 }
684
685 var hole = new CADHole
686 {
687 HoleId = holeId,
688 HoleColor = faces[0].FaceColor,
689 FaceGlobalIndices = faces.Select(f => f.FaceGlobalIndex).ToList(),
690 TopCenterPoint = topCenter,
691 BottomCenterPoint = bottomCenter
692 };
693
694 hole.Depth = EyeshotGeometryUtils.DistanceBetweenPoints(hole.BottomCenterPoint, hole.TopCenterPoint);
695
696 var axisVector = hole.TopCenterPoint.AsVector - hole.BottomCenterPoint.AsVector;
697 axisVector.Normalize();
698 hole.AxisDirection = axisVector;
699
700 if (!isManual && !IsHoleOpen(hole.BottomCenterPoint, hole.TopCenterPoint, faces, allFaces, hole.AxisDirection))
701 return null!;
702
703 if (hole.Depth < MIN_HOLE_DEPTH_TOLERANCE) return null!;
704
705 // Determine diameters from the extreme faces in the stack
706 hole.TopDiameter = GetFaceDiameter(topFace);
707 hole.BottomDiameter = GetFaceDiameter(bottomFace);
708 double avgDiameter = (hole.TopDiameter + hole.BottomDiameter) / 2.0;
709 hole.HoleStandard = IdentifyStandardHoleType(avgDiameter);
710 hole.GeometricInfo = $"{hole.HoleStandard} - Depth: {hole.Depth:F2}mm, Ø{avgDiameter:F2}";
711
712 return hole;
713 }

785 private static bool AreCylindersSeparatedByPlanarFaces(CADFace face1, CADFace face2, List<CADFace> allFaces)
786 {

788 return false;
789
790 if (!AreCylinderAxesSeparated(face1, face2, axis, radius, out var axisDistance))
791 return false;

794 if (diameter <= 0)
795 return false;

798 if (axisDistance > maxAxisDistance)
799 return false;
800
801 if (!TryGetCylinderFitInfo(face1, out var fit1) || !TryGetCylinderFitInfo(face2, out var fit2))
802 return false;
803
804 int separatorCount = allFaces
805 .Where(f => f.FaceType == FaceTypeEnum.Planar)
806 .Count(f => IsPlanarSeparatorBetweenCylinders(f, face1, face2, fit1, fit2));
807

811 private static bool AreCylinderAxesSeparated(CADFace face1, CADFace face2, Vector3D axis, double radius, out double axisDistance)
812 {
813 axisDistance = 0;
814
815 if (!TryGetCylinderFitInfo(face1, out var fit1) || !TryGetCylinderFitInfo(face2, out var fit2))
816 return false;
817
818 axisDistance = DistancePointToAxis(fit2.AxisPoint, fit1.AxisPoint, axis);
819 double diameter = radius > 0 ? radius * 2.0 : Math.Max(fit1.Radius, fit2.Radius) * 2.0;
820 double minAxisDistance = Math.Max(
821 Math.Max(CYLINDER_FIT_TOLERANCE, RADIAL_TOLERANCE * BRIDGE_RADIAL_MULTIPLIER),
822 diameter * SEPARATED_CYLINDER_MIN_AXIS_OFFSET_DIAMETER_MULTIPLIER);
823
824 return axisDistance > minAxisDistance;
825 }

827 private static bool IsPlanarSeparatorBetweenCylinders(
828 CADFace planarFace,
829 CADFace cylFace1,
830 CADFace cylFace2,
831 CylinderFitInfo fit1,
832 CylinderFitInfo fit2)
833 {
834 var planePoints = GetFaceAnalysisPoints(planarFace);
835 if (planePoints.Count < 3)
836 return false;
837
838 var planeNormal = NormalizeAxis(EyeshotGeometryUtils.GetFaceAxis(planarFace));
839 if (Math.Abs(Vector3D.Dot(planeNormal, fit1.Axis)) > 0.35)
840 return false;

842 var axisOffset = GetPerpendicularOffset(fit1.AxisPoint, fit2.AxisPoint, fit1.Axis);

1285 private static bool TryFitCylinderAxisPoint(List<Point3D> points, Vector3D axis, double expectedRadius, out Point3D axisPoint, out double fittedRadius, out double rmsError)
1286 {
1287 axisPoint = new Point3D(0, 0, 0);
1288 fittedRadius = 0;
1289 rmsError = double.MaxValue;
1290
1291 var fitPoints = DistinctPoints(points);
1292 if (fitPoints.Count < 3)
1293 return false;
1294
1295 axis = NormalizeAxis(axis);
1296 var origin = GetCentroid(fitPoints);
1297 BuildPerpendicularBasis(axis, out var u, out var v);
1298
1299 var normal = new double[3, 3];
1300 var rhs = new double[3];
1301
1302 foreach (var point in fitPoints)
1303 {
1304 var fromOrigin = new Vector3D(origin, point);
1305 double x = Vector3D.Dot(fromOrigin, u);
1306 double y = Vector3D.Dot(fromOrigin, v);
1307 double[] row = { x, y, 1.0 };
1308 double b = -(x * x + y * y);
1309
1310 for (int r = 0; r < 3; r++)
1311 {
1312 rhs[r] += row[r] * b;
1313 for (int c = 0; c < 3; c++)
1314 normal[r, c] += row[r] * row[c];
1315 }
1316 }
1317
1318 if (!TrySolve3x3(normal, rhs, out var solution))
1319 return false;
1320
1321 double centerX = -solution[0] / 2.0;
1322 double centerY = -solution[1] / 2.0;
1323 double radiusSquared = centerX * centerX + centerY * centerY - solution[2];
1324 if (radiusSquared <= 1e-9)
1325 return false;
1326
1327 fittedRadius = Math.Sqrt(radiusSquared);
1328 axisPoint = new Point3D(
1329 origin.X + u.X * centerX + v.X * centerY,
1330 origin.Y + u.Y * centerX + v.Y * centerY,
1331 origin.Z + u.Z * centerX + v.Z * centerY);
1332
1333 double sumSquaredError = 0;
1334 foreach (var point in fitPoints)
1335 {
1336 double radialDistance = DistancePointToAxis(point, axisPoint, axis);
1337 double error = radialDistance - fittedRadius;
1338 sumSquaredError += error * error;
1339 }
1340
1341 rmsError = Math.Sqrt(sumSquaredError / fitPoints.Count);
1342 double referenceRadius = expectedRadius > 0 ? expectedRadius : fittedRadius;
1343 double fitTolerance = Math.Max(CYLINDER_FIT_TOLERANCE, referenceRadius * 0.20);
1344
1345 if (expectedRadius > 0 && Math.Abs(fittedRadius - expectedRadius) > fitTolerance)
1346 return false;
1347
1348 return rmsError <= fitTolerance;
1349 }

1358 private static bool TrySolve3x3(double[,] matrix, double[] rhs, out double[] solution)
1359 {
1360 solution = new double[3];
1361 var augmented = new double[3, 4];
1362
1363 for (int r = 0; r < 3; r++)
1364 {
1365 for (int c = 0; c < 3; c++)
1366 augmented[r, c] = matrix[r, c];
1367 augmented[r, 3] = rhs[r];
1368 }
1369
1370 for (int pivot = 0; pivot < 3; pivot++)
1371 {
1372 int bestRow = pivot;
1373 double bestValue = Math.Abs(augmented[pivot, pivot]);
1374
1375 for (int r = pivot + 1; r < 3; r++)
1376 {
1377 double value = Math.Abs(augmented[r, pivot]);
1378 if (value > bestValue)
1379 {
1380 bestValue = value;
1381 bestRow = r;
1382 }
1383 }
1384
1385 if (bestValue < 1e-12)
1386 return false;
1387
1388 if (bestRow != pivot)
1389 {
1390 for (int c = pivot; c < 4; c++)
1391 {
1392 (augmented[pivot, c], augmented[bestRow, c]) = (augmented[bestRow, c], augmented[pivot, c]);
1393 }
1394 }
1395
1396 double divisor = augmented[pivot, pivot];
1397 for (int c = pivot; c < 4; c++)
1398 augmented[pivot, c] /= divisor;
1399
1400 for (int r = 0; r < 3; r++)
1401 {
1402 if (r == pivot) continue;
1403
1404 double factor = augmented[r, pivot];
1405 for (int c = pivot; c < 4; c++)
1406 augmented[r, c] -= factor * augmented[pivot, c];
1407 }
1408 }
1409
1410 solution[0] = augmented[0, 3];
1411 solution[1] = augmented[1, 3];
1412 solution[2] = augmented[2, 3];
1413 return true;
1414 }

1424 private static bool HasCompleteCircularCoverage(List<Point3D> points, Point3D axisPoint, Vector3D axis, double minCoverageRadians)
1425 {
1426 var distinct = DistinctPoints(points);
1427 if (distinct.Count < 4)
1428 return false;
1429
1430 BuildPerpendicularBasis(axis, out var u, out var v);
1431 var angles = new List<double>();
1432
1433 foreach (var point in distinct)
1434 {
1435 double axial = ProjectOnAxis(point, axisPoint, axis);
1436 var centerAtPoint = PointOnAxis(axisPoint, axis, axial);
1437 var radial = new Vector3D(centerAtPoint, point);
1438 if (radial.Length < RADIAL_TOLERANCE)
1439 continue;
1440
1441 double x = Vector3D.Dot(radial, u);
1442 double y = Vector3D.Dot(radial, v);
1443 angles.Add(Math.Atan2(y, x));
1444 }
1445
1446 if (angles.Count < 4)
1447 return false;
1448
1449 angles.Sort();
1450 double largestGap = 0;
1451
1452 for (int i = 0; i < angles.Count; i++)
1453 {
1454 double current = angles[i];
1455 double next = i == angles.Count - 1 ? angles[0] + 2.0 * Math.PI : angles[i + 1];
1456 largestGap = Math.Max(largestGap, next - current);
1457 }
1458
1459 double coverage = 2.0 * Math.PI - largestGap;
1460 return coverage >= minCoverageRadians;
1461 }

1805 private static void IdentifyBridgeFaces(CADHole hole, List<CADFace> allFaces, System.Threading.CancellationToken cancellationToken = default)
1806 {
1807 var planarFaces = allFaces.Where(f => f.FaceType == FaceTypeEnum.Planar).ToList();
1808 var cylindricalFaces = allFaces.Where(f => f.FaceType == FaceTypeEnum.Cylindrical || f.FaceType == FaceTypeEnum.Conical).ToList();
1809 if (hole.BottomCenterPoint == null || hole.TopCenterPoint == null || hole.AxisDirection == null)
1810 return;
1811
1812 var holeAxis = NormalizeAxis(hole.AxisDirection);
1813 double holeLength = EyeshotGeometryUtils.DistanceBetweenPoints(hole.BottomCenterPoint, hole.TopCenterPoint);
1814 if (holeLength < MIN_HOLE_DEPTH_TOLERANCE)
1815 return;
1816
1817 foreach (var face in planarFaces)
1818 {
1819 cancellationToken.ThrowIfCancellationRequested();
1820 // Skip if already marked as bridge
1821 if (face.IsBridgeBetweenHoles) continue;
1822
1823 if (!TryGetPlanarFaceAxisPoint(face, hole.BottomCenterPoint, holeAxis, out var faceAxisPoint, out var axialPosition))
1824 continue;
1825
1826 if (axialPosition < -BRIDGE_AXIAL_TOLERANCE || axialPosition > holeLength + BRIDGE_AXIAL_TOLERANCE)
1827 continue;
1828
1829 bool hasCylBelow = HasCoaxialCylinderTouchingSide(faceAxisPoint, holeAxis, cylindricalFaces, -1, null, out var cylBelow);
1830 bool hasCylAbove = HasCoaxialCylinderTouchingSide(faceAxisPoint, holeAxis, cylindricalFaces, 1, null, out var cylAbove);
1831
1832 if (hasCylBelow && hasCylAbove)
1833 {
1834 double referenceDiameter = GetBridgeReferenceDiameter(hole, new CADFace?[] { cylBelow, cylAbove });
1835 if (IsCircularBridgeFaceCandidate(face, faceAxisPoint, holeAxis, referenceDiameter))
1836 {
1837 face.IsBridgeBetweenHoles = true;
1838 }
1839 }
1840 }
