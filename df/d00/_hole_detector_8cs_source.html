167 public static (List<CADHole> holes, List<CADFace> nonHoleFaces) DetectHoles(List<CADFace> allFaces, Action<int, int>? progressCallback = null, System.Threading.CancellationToken cancellationToken = default)
168 {
169 var holes = new List<CADHole>();
170 var processedFaceIndices = new HashSet<int>();
171
172 foreach (var face in allFaces)
173 {
174 face.IsBridgeBetweenHoles = false;
175 }
176
177 progressCallback?.Invoke(10, 0);
178
179 var cylindricalFaces = allFaces
180 .Where(f => f.FaceType == FaceTypeEnum.Cylindrical || f.FaceType == FaceTypeEnum.Conical)
181 .ToList();
182
183 progressCallback?.Invoke(30, 0);
184
185 var holeGroups = GroupFacesIntoHoles(cylindricalFaces, allFaces, cancellationToken);
186
187 progressCallback?.Invoke(50, 0);
188
189 int holeId = 1;
190 int totalGroups = holeGroups.Count;
191 int processedGroups = 0;
192
193 foreach (var group in holeGroups)
194 {
195 cancellationToken.ThrowIfCancellationRequested();
196
197 if (group.Count >= 2)
198 {
199 var hole = CreateHoleFromFaces(group, holeId, allFaces);
200 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
201 {
202 holes.Add(hole);
203 holeId++;
204 foreach (var f in group)
205 processedFaceIndices.Add(f.FaceGlobalIndex);
206 }
207 }
208 else if (group.Count == 1 && ENABLE_SINGLE_FACE_JOINED_HOLES)
209 {
210 var face = group[0];
211 if (IsSingleFaceJoinedHoleCandidate(face, cylindricalFaces, allFaces))
212 {
213 var hole = CreateHoleFromSingleFace(face, holeId, allFaces);
214 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
215 {
216 holes.Add(hole);
217 holeId++;
218 processedFaceIndices.Add(face.FaceGlobalIndex);
219 }
220 }
221 }
222
223 processedGroups++;
224 if (progressCallback != null && totalGroups > 0)
225 {
226 int percentage = 50 + (int)((processedGroups * 50.0) / totalGroups);
227 progressCallback(percentage, holes.Count);
228 }
229 }
230
231 // Post-process: Identify bridge faces for all detected holes
232 foreach (var hole in holes)
233 {
234 cancellationToken.ThrowIfCancellationRequested();
235 IdentifyBridgeFaces(hole, allFaces, cancellationToken);
236 }
237
238 // Post-process: Detect concave isolated single-face holes.
239 // This runs strictly AFTER the original bridge detection and is not allowed to
240 // change any IsBridgeBetweenHoles result produced by that pipeline.
241 if (ENABLE_CONCAVE_ISOLATED_SINGLE_FACE_HOLES)
242 {
243 var bridgeStateBeforeConcaveDetection = allFaces.ToDictionary(
244 face => face.FaceGlobalIndex,
245 face => face.IsBridgeBetweenHoles);
246
247 foreach (var face in cylindricalFaces)
248 {
249 cancellationToken.ThrowIfCancellationRequested();
250
251 if (processedFaceIndices.Contains(face.FaceGlobalIndex)) continue;
252 if (face.IsBridgeBetweenHoles) continue;
253
254 if (!IsConcaveIsolatedSingleFaceHoleCandidate(face, allFaces))
255 continue;
256
257 var hole = CreateHoleFromSingleFace(face, holeId, allFaces);
258 if (hole != null && hole.Depth > MIN_HOLE_DEPTH_TOLERANCE)
259 {
260 holes.Add(hole);
261 holeId++;
262 processedFaceIndices.Add(face.FaceGlobalIndex);
263 }
264 }
265
266 // Restore the bridge flags exactly as they were before this isolated
267 // single-face fallback. This guarantees that enabling concave detection
268 // cannot create, remove or reclassify bridge faces.
269 foreach (var face in allFaces)
270 {
271 if (bridgeStateBeforeConcaveDetection.TryGetValue(
272 face.FaceGlobalIndex,
273 out bool wasBridge))
274 {
275 face.IsBridgeBetweenHoles = wasBridge;
276 }
277 }
278 }
279
280 var nonHoleFaces = allFaces
281 .Where(f => !processedFaceIndices.Contains(f.FaceGlobalIndex))
282 .ToList();
283
284 DetectNestedHoles(holes);
285
286 if (GetConfigBool("HoleDetection_AutoIgnoreObstructedByLarger", false))
287 {
288 var baseAvoidance = Utilities.Algorithms.BaseAvoidanceClearance.EnableForIgnoredBottomFace(
289 Utilities.Algorithms.BaseAvoidanceClearance.LoadConfig(),
290 Utilities.Algorithms.BaseAvoidanceFaceClassifier.HasIgnoredBottomFace(allFaces));
291 bool avoidBottomFace = baseAvoidance.Enabled ||
292 (bool.TryParse(System.Configuration.ConfigurationManager.AppSettings["Scanner_AvoidBottomFace"], out bool s) && s);
293 AutoIgnoreTopUnreachableHoles(holes, allFaces, avoidBottomFace);
294 }
295
296 progressCallback?.Invoke(100, holes.Count);
297
298 return (holes, nonHoleFaces);
299 }

477 public static string IdentifyStandardHoleType(double diameter)
478 {
479 if (double.IsNaN(diameter) || diameter <= 0)
480 return "Unknown";
481
482 var closestMatch = StandardHoleDiameters
483 .Select(kvp => new { kvp.Key, kvp.Value, Diff = Math.Abs(kvp.Value - diameter) })
484 .OrderBy(x => x.Diff)
485 .FirstOrDefault();
486
487 if (closestMatch != null && closestMatch.Diff <= DIAMETER_TOLERANCE)
488 return closestMatch.Key;
489
490 var clearanceMatch = StandardHoleDiameters
491 .Select(kvp => new
492 {
493 Standard = kvp.Key + " Clearance",
494 NominalDiameter = double.Parse(kvp.Key.Substring(1)),
495 TapDiameter = kvp.Value,
496 Diff = Math.Abs(diameter - (double.Parse(kvp.Key.Substring(1)) + 0.5))
497 })
498 .OrderBy(x => x.Diff)
499 .FirstOrDefault();
500
501 if (clearanceMatch != null && clearanceMatch.Diff <= DIAMETER_TOLERANCE)
502 return clearanceMatch.Standard;
503
504 return $"Ø{diameter:F2}mm";
505 }

510 private static List<List<CADFace>> GroupFacesIntoHoles(List<CADFace> cylindricalFaces, List<CADFace> allFaces, System.Threading.CancellationToken cancellationToken = default)
511 {
512 var groups = new List<List<CADFace>>();
513 var processed = new HashSet<int>();
514
515 foreach (var face in cylindricalFaces)
516 {
517 cancellationToken.ThrowIfCancellationRequested();
518 if (processed.Contains(face.FaceGlobalIndex)) continue;
519
520 var group = new List<CADFace>();
521 var queue = new Queue<CADFace>();
522 queue.Enqueue(face);
523 processed.Add(face.FaceGlobalIndex);
524
525 while (queue.Count > 0)
526 {
527 cancellationToken.ThrowIfCancellationRequested();
528 var current = queue.Dequeue();
529 group.Add(current);
530
531 double radius1 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(current);
532 if (radius1 <= 0) continue;
533
534 foreach (var other in cylindricalFaces)
535 {
536 if (processed.Contains(other.FaceGlobalIndex)) continue;
537
538 double radius2 = EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo(other);
539 if (radius2 <= 0 || Math.Abs(radius1 - radius2) > RADIAL_TOLERANCE)
540 continue;
541
542 if (AreCylindersSeparatedByPlanarFaces(current, other, allFaces))
543 continue;
544
545 // Check if faces are companions (split halves) or stacked segments
546 if (AreFacesPairedForHole(current, other) || AreFacesStacked(current, other))
547 {
548 processed.Add(other.FaceGlobalIndex);
549 queue.Enqueue(other);
550 }
551 }
552 }
553
554 if (group.Count > 0)
555 groups.Add(group);
556 }
557
558 return groups;
559 }

564 private static bool AreFacesStacked(CADFace face1, CADFace face2)
565 {
566 var pts1 = GetFaceAnalysisPoints(face1);
567 var pts2 = GetFaceAnalysisPoints(face2);
568
569 if (pts1.Count == 0 || pts2.Count == 0)
570 return false;
571
572 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
573 return false;
574
575 if (AreCylinderAxesSeparated(face1, face2, axis, radius, out _))
576 return false;
577
578 var combinedPoints = pts1.Concat(pts2).ToList();
579 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
580 return false;
581
582 var range1 = GetAxialRange(pts1, axisPoint, axis);
583 var range2 = GetAxialRange(pts2, axisPoint, axis);
584 if (!range1.HasValue || !range2.HasValue)
585 return false;
586
587 double gap = GetAxialGap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
588
589 // They are stacked if they are coaxial and their projected extents touch or are
590 // close enough to jump over a real separating bridge face.
591 return gap >= -BRIDGE_AXIAL_TOLERANCE && gap < MAX_JUMP_GAP;
592 }

598 private static bool AreFacesPairedForHole(CADFace face1, CADFace face2)
599 {
600 var pts1 = GetFaceAnalysisPoints(face1);
601 var pts2 = GetFaceAnalysisPoints(face2);
602
603 if (pts1.Count == 0 || pts2.Count == 0)
604 return false;
605
606 if (!TryGetCompatibleCylinderAxis(face1, face2, out var axis, out var radius))
607 return false;
608
609 if (AreCylinderAxesSeparated(face1, face2, axis, radius, out _))
610 return false;
611
612 double range1Length = GetProjectedRangeLength(pts1, axis);
613 double range2Length = GetProjectedRangeLength(pts2, axis);
614 double maxAxialRange = Math.Max(range1Length, range2Length);
615 double adaptiveAxialTolerance = Math.Max(Z_DEVIATION_TOLERANCE, maxAxialRange * 0.15);
616
617 // Complementary split cylinder faces may have centroids on opposite sides of
618 // the bore. Check shared seam points before any centroid-based rejection.
619 if (CountPointMatches(pts1, pts2, axis, SHARED_VERTEX_TOLERANCE * 2.0, adaptiveAxialTolerance) >= SEAM_MATCHES_REQUIRED)
620 return true;
621
622 var combinedPoints = pts1.Concat(pts2).ToList();
623 if (!TryFitCylinderAxisPoint(combinedPoints, axis, radius, out var axisPoint, out _, out _))
624 return false;
625
626 var range1 = GetAxialRange(pts1, axisPoint, axis);
627 var range2 = GetAxialRange(pts2, axisPoint, axis);
628 if (!range1.HasValue || !range2.HasValue)
629 return false;
630
631 double overlap = GetAxialOverlap(range1.Value.Min, range1.Value.Max, range2.Value.Min, range2.Value.Max);
632 if (overlap < Math.Min(range1.Value.Length, range2.Value.Length) * 0.10 &&
633 overlap < MIN_HOLE_DEPTH_TOLERANCE)
634 {
635 return false;
636 }
637
638 if (HasCompleteCircularCoverage(combinedPoints, axisPoint, axis, MIN_CIRCULAR_COVERAGE_RADIANS))
639 return true;
640
641 int matchedPoints = CountPointMatches(pts1, pts2, axis, RADIAL_TOLERANCE, adaptiveAxialTolerance);
642 double similarity = (double)matchedPoints / Math.Max(pts1.Count, pts2.Count);
643
644 // Lower threshold for curved surfaces (70% vs 90%)
645 double threshold = maxAxialRange > MIN_HOLE_DEPTH_TOLERANCE ? 0.70 : 0.85;
646
647 return similarity >= threshold;
648 }

653 private static bool HaveSimilarControlPointsWithDeviation(List<Point4D> pts1, List<Point4D> pts2)
654 {
655 // Convert to CADFace-like structure for the enhanced check
656 // This is a simplified adapter - the main grouping now uses AreFacesPairedForHole
657 if (pts1.Count == 0 || pts2.Count == 0) return false;
658
659 int matched = 0;
660 double zRange = Math.Max(
661 pts1.Max(p => p.Z) - pts1.Min(p => p.Z),
662 pts2.Max(p => p.Z) - pts2.Min(p => p.Z)
663 );
664
665 double dynamicZTolerance = Math.Max(Z_DEVIATION_TOLERANCE, zRange * 0.15);
666
667 foreach (var p1 in pts1)
668 {
669 if (pts2.Any(p2 =>
670 Math.Abs(p1.X - p2.X) < SHARED_VERTEX_TOLERANCE &&
671 Math.Abs(p1.Y - p2.Y) < SHARED_VERTEX_TOLERANCE &&
672 Math.Abs(p1.Z - p2.Z) < dynamicZTolerance))
673 {
674 matched++;
675 }
676 }
677
678 double similarity = (double)matched / Math.Max(pts1.Count, pts2.Count);
679 return similarity >= 0.70; // lowered threshold for curved surfaces
680 }

690 private static CADHole CreateHoleFromFaces(List<CADFace> faces, int holeId, List<CADFace> allFaces, bool isManual = false)
691 {
692 if (faces.Count < 2) return null!;
693
694 var allControlPoints = faces.SelectMany(GetFaceAnalysisPoints).ToList();
695
696 if (allControlPoints.Count == 0) return null!;
697
698 var axis = GetDominantFaceAxis(faces);
699 double avgRadius = faces
700 .Select(EyeshotGeometryUtils.GetFaceRadiusFromGeometricInfo)
701 .Where(r => r > 0)
702 .DefaultIfEmpty(0)
703 .Average();
704
705 Point3D bottomCenter;
706 Point3D topCenter;
707 CADFace topFace;
708 CADFace bottomFace;
709
710 if (avgRadius > 0 &&
711 TryFitCylinderAxisPoint(allControlPoints, axis, avgRadius, out var axisPoint, out _, out _))
712 {
713 var range = GetAxialRange(allControlPoints, axisPoint, axis);
714 if (!range.HasValue || range.Value.Length < MIN_HOLE_DEPTH_TOLERANCE)
715 return null!;
716
717 bottomCenter = PointOnAxis(axisPoint, axis, range.Value.Min);
718 topCenter = PointOnAxis(axisPoint, axis, range.Value.Max);
719
720 topFace = faces.OrderByDescending(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
721 bottomFace = faces.OrderBy(f => GetAverageProjection(GetFaceAnalysisPoints(f), axisPoint, axis)).First();
722 }
723 else
724 {
725 var sortedPoints = allControlPoints.OrderBy(p => p.Z).ToList();
726 double zMin = sortedPoints.First().Z;
727 double zMax = sortedPoints.Last().Z;
728
729 if (Math.Abs(zMax - zMin) < MIN_HOLE_DEPTH_TOLERANCE)
730 return null!;
731
732 var finalBottomPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMin) < SHARED_VERTEX_TOLERANCE * 2).ToList();
733 var finalTopPoints = sortedPoints.Where(p => Math.Abs(p.Z - zMax) < SHARED_VERTEX_TOLERANCE * 2).ToList();
734
735 bottomCenter = GetCentroid(finalBottomPoints);
736 topCenter = GetCentroid(finalTopPoints);
737 topFace = faces.OrderByDescending(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
738 bottomFace = faces.OrderBy(f => f.ControlPoints.Any() ? f.ControlPoints.Average(p => p.Z) : 0).First();
739 }
740
741 var hole = new CADHole
742 {
743 HoleId = holeId,
744 HoleColor = faces[0].FaceColor,
745 FaceGlobalIndices = faces.Select(f => f.FaceGlobalIndex).ToList(),
746 TopCenterPoint = topCenter,
747 BottomCenterPoint = bottomCenter
748 };
749
750 hole.Depth = EyeshotGeometryUtils.DistanceBetweenPoints(hole.BottomCenterPoint, hole.TopCenterPoint);
751
752 var axisVector = hole.TopCenterPoint.AsVector - hole.BottomCenterPoint.AsVector;
753 axisVector.Normalize();
754 hole.AxisDirection = axisVector;
755
756 if (!isManual && !IsHoleOpen(hole.BottomCenterPoint, hole.TopCenterPoint, faces, allFaces, hole.AxisDirection))
757 return null!;
758
759 if (hole.Depth < MIN_HOLE_DEPTH_TOLERANCE) return null!;
760
761 // Determine diameters from the extreme faces in the stack
762 hole.TopDiameter = GetFaceDiameter(topFace);
763 hole.BottomDiameter = GetFaceDiameter(bottomFace);
764 double avgDiameter = (hole.TopDiameter + hole.BottomDiameter) / 2.0;
765 hole.HoleStandard = IdentifyStandardHoleType(avgDiameter);
766 hole.GeometricInfo = $"{hole.HoleStandard} - Depth: {hole.Depth:F2}mm, Ø{avgDiameter:F2}";
767
768 return hole;
769 }

876 private static bool IsConcaveIsolatedSingleFaceHoleCandidate(CADFace face, List<CADFace> allFaces)
877 {
878 // Keep this fallback limited to true cylindrical faces. Conical faces are much
879 // more likely to be chamfers and remain handled only by the original pipeline.
880 if (face == null || face.FaceType != FaceTypeEnum.Cylindrical)
881 return false;
882
883 if (!TryGetCylinderFitInfo(face, out var fit))
884 return false;
885
886 // Only inward-facing cylindrical walls are concave holes. Exterior convex
887 // cylinders carry a negative value and are rejected before any U-shape checks.
888 if (!HasInwardFacingCylinderNormal(face))
889 return false;
890
891 var cylinderPoints = GetCylinderAnalysisPoints(face);
892 if (cylinderPoints.Count < 4)
893 return false;
894
895 if (!HasCompleteCircularCoverage(
896 cylinderPoints,
897 fit.AxisPoint,
898 fit.Axis,
899 MIN_CONCAVE_ISOLATED_HOLE_COVERAGE_RADIANS))
900 {
901 return false;
902 }
903
904 BuildPerpendicularBasis(fit.Axis, out var u, out var v);
905
906 double GetAngle(Point3D point)
907 {
908 double axial = ProjectOnAxis(point, fit.AxisPoint, fit.Axis);
909 var centerAtPoint = PointOnAxis(fit.AxisPoint, fit.Axis, axial);
910 var radial = new Vector3D(centerAtPoint, point);
911 return Math.Atan2(Vector3D.Dot(radial, v), Vector3D.Dot(radial, u));
912 }
913
914 var sideWallAngularCenters = new List<double>();
915
916 foreach (var planarFace in allFaces.Where(f => f.FaceType == FaceTypeEnum.Planar))
917 {
918 var planePoints = GetFaceAnalysisPoints(planarFace);
919 if (planePoints.Count < 3)
920 continue;
921
922 // A real U-slot side wall is parallel to the cylinder axis. Upper/lower
923 // planar faces are perpendicular to that axis and are rejected here before
924 // any adjacency or angular test, preventing the previous upper-face false
925 // positives.
926 var planeNormal = NormalizeAxis(EyeshotGeometryUtils.GetFaceAxis(planarFace));
927 if (Math.Abs(Vector3D.Dot(planeNormal, fit.Axis)) > 0.35)
928 continue;
929
930 var sharedPoints = GetConcaveSideWallSeamPoints(face, planarFace, fit);
931 if (sharedPoints.Count < 2)
932 continue;
933
934 var sharedAxialRange = GetAxialRange(sharedPoints, fit.AxisPoint, fit.Axis);
935 if (!sharedAxialRange.HasValue)
936 continue;
937
938 // Do not require 60% of the fitted depth: imported periodic cylinders can
939 // expose only the seam end vertices even when the entire wall is adjacent.
940 // A quarter of the depth is enough to reject a cap touching only one end,
941 // while still accepting the real U-shaped cylindrical face in this model.
942 double minimumSeamDepth = Math.Max(
943 MIN_HOLE_DEPTH_TOLERANCE,
944 fit.Range.Length * 0.25);
945
946 if (sharedAxialRange.Value.Length < minimumSeamDepth)
947 continue;
948
949 // The side wall must continue away from the cylindrical axis. Checking the
950 // farthest point is more reliable than checking the centroid of an irregular
951 // U-shaped planar wall, whose centroid may remain close to the bore radius.
952 double maxPlanarRadialDistance = planePoints.Max(point =>
953 DistancePointToAxis(point, fit.AxisPoint, fit.Axis));
954
955 if (maxPlanarRadialDistance <= fit.Radius + Math.Max(SHARED_VERTEX_TOLERANCE * 4.0, 0.05))
956 continue;
957
958 var seamAngles = sharedPoints
959 .Select(GetAngle)
960 .OrderBy(angle => angle)
961 .ToList();
962
963 double largestGap = 0;
964 for (int i = 0; i < seamAngles.Count; i++)
965 {
966 double current = seamAngles[i];
967 double next = i == seamAngles.Count - 1
968 ? seamAngles[0] + 2.0 * Math.PI
969 : seamAngles[i + 1];
970
971 largestGap = Math.Max(largestGap, next - current);
972 }
973
974 double angularSpan = 2.0 * Math.PI - largestGap;
975 if (angularSpan > MAX_CONCAVE_SIDE_WALL_ANGULAR_SPAN_RADIANS)
976 continue;
977
978 double meanSin = seamAngles.Average(angle => Math.Sin(angle));
979 double meanCos = seamAngles.Average(angle => Math.Cos(angle));
980 sideWallAngularCenters.Add(Math.Atan2(meanSin, meanCos));
981 }
982
983 if (sideWallAngularCenters.Count < 2)
984 return false;
985
986 // Two different side walls must flank the curved U face. This prevents a single
987 // nearby planar face from being enough to classify an exterior cylinder/fillet.
988 for (int i = 0; i < sideWallAngularCenters.Count; i++)
989 {
990 for (int j = i + 1; j < sideWallAngularCenters.Count; j++)
991 {
992 double difference =
993 Math.Abs(sideWallAngularCenters[j] - sideWallAngularCenters[i]);
994
995 if (difference > Math.PI)
996 difference = 2.0 * Math.PI - difference;
997
998 if (difference >= MIN_CONCAVE_SIDE_WALL_SEPARATION_RADIANS)
999 return true;
1000 }
1001 }
1002
1003 return false;
1004 }

1036 private static List<Point3D> GetConcaveSideWallSeamPoints(
1037 CADFace cylinderFace,
1038 CADFace planarFace,
1039 CylinderFitInfo fit)
1040 {
1041 var sharedPoints = GetTopologicallySharedBoundaryPoints(cylinderFace, planarFace);
1042 if (sharedPoints.Count == 0)
1043 return sharedPoints;
1044
1045 double surfaceTolerance = Math.Max(
1046 SHARED_VERTEX_TOLERANCE * 8.0,
1047 Math.Min(0.15, Math.Max(0.05, CYLINDER_FIT_TOLERANCE * 0.15)));
1048
1049 // Once real boundary contact has been established, recover duplicated seam
1050 // endpoints from the planar face by checking the fitted cylinder itself.
1051 foreach (var point in GetBoundaryPoints(planarFace))
1052 {
1053 double axial = ProjectOnAxis(point, fit.AxisPoint, fit.Axis);
1054 if (axial < fit.Range.Min - surfaceTolerance ||
1055 axial > fit.Range.Max + surfaceTolerance)
1056 {
1057 continue;
1058 }
1059
1060 double radial = DistancePointToAxis(point, fit.AxisPoint, fit.Axis);
1061 if (Math.Abs(radial - fit.Radius) <= surfaceTolerance)
1062 sharedPoints.Add(point);
1063 }
1064
1065 return DistinctPoints(sharedPoints);
1066 }

1190 private static bool IsPlanarSeparatorBetweenCylinders(
1191 CADFace planarFace,
1192 CADFace cylFace1,
1193 CADFace cylFace2,
1194 CylinderFitInfo fit1,
1195 CylinderFitInfo fit2)
1196 {
1197 var planePoints = GetFaceAnalysisPoints(planarFace);
1198 if (planePoints.Count < 3)
1199 return false;
1200
1201 var planeNormal = NormalizeAxis(EyeshotGeometryUtils.GetFaceAxis(planarFace));
1202 if (Math.Abs(Vector3D.Dot(planeNormal, fit1.Axis)) > 0.35)
1203 return false;
1204
1205 var axisOffset = GetPerpendicularOffset(fit1.AxisPoint, fit2.AxisPoint, fit1.Axis);
1206 double axisDistance = axisOffset.Length;
1207 if (axisDistance <= CYLINDER_FIT_TOLERANCE)
1208 return false;
1209
1210 axisOffset.Normalize();
1211 var planeCentroid = GetCentroid(planePoints);
1212 double alongCenters = Vector3D.Dot(new Vector3D(fit1.AxisPoint, planeCentroid), axisOffset);
1213 if (alongCenters < -SEPARATOR_FACE_TOUCH_TOLERANCE || alongCenters > axisDistance + SEPARATOR_FACE_TOUCH_TOLERANCE)
1214 return false;
1215
1216 var cylinder1Points = GetCylinderAnalysisPoints(cylFace1);
1217 var cylinder2Points = GetCylinderAnalysisPoints(cylFace2);
1218 if (!HasAxialOverlapWithCylinder(planePoints, fit1, fit1.AxisPoint, fit1.Axis) ||
1219 !HasAxialOverlapWithCylinder(planePoints, fit2, fit1.AxisPoint, fit1.Axis))
1220 {
1221 return false;
1222 }
1223
1224 return HasSharedPoints(planePoints, cylinder1Points, SEPARATOR_FACE_TOUCH_TOLERANCE) &&
1225 HasSharedPoints(planePoints, cylinder2Points, SEPARATOR_FACE_TOUCH_TOLERANCE);
1226 }

1648 private static bool TryFitCylinderAxisPoint(List<Point3D> points, Vector3D axis, double expectedRadius, out Point3D axisPoint, out double fittedRadius, out double rmsError)
1649 {
1650 axisPoint = new Point3D(0, 0, 0);
1651 fittedRadius = 0;
1652 rmsError = double.MaxValue;
1653
1654 var fitPoints = DistinctPoints(points);
1655 if (fitPoints.Count < 3)
1656 return false;
1657
1658 axis = NormalizeAxis(axis);
1659 var origin = GetCentroid(fitPoints);
1660 BuildPerpendicularBasis(axis, out var u, out var v);
1661
1662 var normal = new double[3, 3];
1663 var rhs = new double[3];
1664
1665 foreach (var point in fitPoints)
1666 {
1667 var fromOrigin = new Vector3D(origin, point);
1668 double x = Vector3D.Dot(fromOrigin, u);
1669 double y = Vector3D.Dot(fromOrigin, v);
1670 double[] row = { x, y, 1.0 };
1671 double b = -(x * x + y * y);
1672
1673 for (int r = 0; r < 3; r++)
1674 {
1675 rhs[r] += row[r] * b;
1676 for (int c = 0; c < 3; c++)
1677 normal[r, c] += row[r] * row[c];
1678 }
1679 }
1680
1681 if (!TrySolve3x3(normal, rhs, out var solution))
1682 return false;
1683
1684 double centerX = -solution[0] / 2.0;
1685 double centerY = -solution[1] / 2.0;
1686 double radiusSquared = centerX * centerX + centerY * centerY - solution[2];
1687 if (radiusSquared <= 1e-9)
1688 return false;
1689
1690 fittedRadius = Math.Sqrt(radiusSquared);
1691 axisPoint = new Point3D(
1692 origin.X + u.X * centerX + v.X * centerY,
1693 origin.Y + u.Y * centerX + v.Y * centerY,
1694 origin.Z + u.Z * centerX + v.Z * centerY);
1695
1696 double sumSquaredError = 0;
1697 foreach (var point in fitPoints)
1698 {
1699 double radialDistance = DistancePointToAxis(point, axisPoint, axis);
1700 double error = radialDistance - fittedRadius;
1701 sumSquaredError += error * error;
1702 }
1703
1704 rmsError = Math.Sqrt(sumSquaredError / fitPoints.Count);
1705 double referenceRadius = expectedRadius > 0 ? expectedRadius : fittedRadius;
1706 double fitTolerance = Math.Max(CYLINDER_FIT_TOLERANCE, referenceRadius * 0.20);
1707
1708 if (expectedRadius > 0 && Math.Abs(fittedRadius - expectedRadius) > fitTolerance)
1709 return false;
1710
1711 return rmsError <= fitTolerance;
1712 }

1721 private static bool TrySolve3x3(double[,] matrix, double[] rhs, out double[] solution)
1722 {
1723 solution = new double[3];
1724 var augmented = new double[3, 4];
1725
1726 for (int r = 0; r < 3; r++)
1727 {
1728 for (int c = 0; c < 3; c++)
1729 augmented[r, c] = matrix[r, c];
1730 augmented[r, 3] = rhs[r];
1731 }
1732
1733 for (int pivot = 0; pivot < 3; pivot++)
1734 {
1735 int bestRow = pivot;
1736 double bestValue = Math.Abs(augmented[pivot, pivot]);
1737
1738 for (int r = pivot + 1; r < 3; r++)
1739 {
1740 double value = Math.Abs(augmented[r, pivot]);
1741 if (value > bestValue)
1742 {
1743 bestValue = value;
1744 bestRow = r;
1745 }
1746 }
1747
1748 if (bestValue < 1e-12)
1749 return false;
1750
1751 if (bestRow != pivot)
1752 {
1753 for (int c = pivot; c < 4; c++)
1754 {
1755 (augmented[pivot, c], augmented[bestRow, c]) = (augmented[bestRow, c], augmented[pivot, c]);
1756 }
1757 }
1758
1759 double divisor = augmented[pivot, pivot];
1760 for (int c = pivot; c < 4; c++)
1761 augmented[pivot, c] /= divisor;
1762
1763 for (int r = 0; r < 3; r++)
1764 {
1765 if (r == pivot) continue;
1766
1767 double factor = augmented[r, pivot];
1768 for (int c = pivot; c < 4; c++)
1769 augmented[r, c] -= factor * augmented[pivot, c];
1770 }
1771 }
1772
1773 solution[0] = augmented[0, 3];
1774 solution[1] = augmented[1, 3];
1775 solution[2] = augmented[2, 3];
1776 return true;
1777 }

1787 private static bool HasCompleteCircularCoverage(List<Point3D> points, Point3D axisPoint, Vector3D axis, double minCoverageRadians)
1788 {
1789 var distinct = DistinctPoints(points);
1790 if (distinct.Count < 4)
1791 return false;
1792
1793 BuildPerpendicularBasis(axis, out var u, out var v);
1794 var angles = new List<double>();
1795
1796 foreach (var point in distinct)
1797 {
1798 double axial = ProjectOnAxis(point, axisPoint, axis);
1799 var centerAtPoint = PointOnAxis(axisPoint, axis, axial);
1800 var radial = new Vector3D(centerAtPoint, point);
1801 if (radial.Length < RADIAL_TOLERANCE)
1802 continue;
1803
1804 double x = Vector3D.Dot(radial, u);
1805 double y = Vector3D.Dot(radial, v);
1806 angles.Add(Math.Atan2(y, x));
1807 }
1808
1809 if (angles.Count < 4)
1810 return false;
1811
1812 angles.Sort();
1813 double largestGap = 0;
1814
1815 for (int i = 0; i < angles.Count; i++)
1816 {
1817 double current = angles[i];
1818 double next = i == angles.Count - 1 ? angles[0] + 2.0 * Math.PI : angles[i + 1];
1819 largestGap = Math.Max(largestGap, next - current);
1820 }
1821
1822 double coverage = 2.0 * Math.PI - largestGap;
1823 return coverage >= minCoverageRadians;
1824 }

2183 private static void IdentifyBridgeFaces(CADHole hole, List<CADFace> allFaces, System.Threading.CancellationToken cancellationToken = default)

2188 return;

2190 var holeAxis = NormalizeAxis(hole.AxisDirection);
2191 double holeLength = EyeshotGeometryUtils.DistanceBetweenPoints(hole.BottomCenterPoint, hole.TopCenterPoint);
2192 if (holeLength < MIN_HOLE_DEPTH_TOLERANCE)
2193 return;

2197 cancellationToken.ThrowIfCancellationRequested();

2202 if (TryGetPlanarFaceAxisPoint(face, hole.BottomCenterPoint, holeAxis, out var faceAxisPoint, out var axialPosition))
2203 {

2210 {

2212 if (IsCircularBridgeFaceCandidate(face, faceAxisPoint, holeAxis, referenceDiameter))

2215 continue;
2216 }
2217 }
2218 }
2219 }

2222 // Catches large annular "shoulder" faces (e.g. counterbore steps) whose edge points
2223 // extend beyond the IsBridgeFaceRadiallyBounded limit but whose centroid IS
2224 // geometrically near the hole axis. This mirrors the heuristic used in
2225 // ShellBasedPathPlanner.GeneratePath() Phase 2.
2226 var facePoints = GetFaceAnalysisPoints(face);
2227 if (facePoints.Count < 3) continue;
2228
2229 var faceNormal = NormalizeAxis(EyeshotGeometryUtils.GetFaceAxis(face));
2230 if (!EyeshotGeometryUtils.AreVectorsParallel(faceNormal, holeAxis, PARALLEL_AXIS_TOLERANCE)) continue;
2231
2232 var faceCentroid = GetCentroid(facePoints);
2233
2234 // Face centroid must be within 1.8× the hole radius of the hole axis line
2235 double holeRadius = Math.Max(hole.TopDiameter, hole.BottomDiameter) / 2.0;
2236 if (holeRadius <= 0 || double.IsNaN(holeRadius)) continue;
2237
2238 double distToAxis = DistancePointToAxis(faceCentroid, hole.BottomCenterPoint!, holeAxis);
2239 if (distToAxis > holeRadius * 1.8) continue;
2240
2241 // Face centroid must be axially near the top or bottom of the hole (within 6mm or 10% of hole length)
2242 double shoulderAxialPos = ProjectOnAxis(faceCentroid, hole.BottomCenterPoint!, holeAxis);
2243 double nearTopDist = Math.Abs(shoulderAxialPos - holeLength);
2244 double nearBotDist = Math.Abs(shoulderAxialPos);
2245 double axialMargin = Math.Max(6.0, holeLength * 0.10);
2246 if (nearTopDist > axialMargin && nearBotDist > axialMargin) continue;
2247
2248 // The hole center must be "above" the face normal (face looks toward the hole interior)
2249 var holeCenter = new Point3D(
2250 (hole.TopCenterPoint!.X + hole.BottomCenterPoint!.X) * 0.5,
2251 (hole.TopCenterPoint!.Y + hole.BottomCenterPoint!.Y) * 0.5,
2252 (hole.TopCenterPoint!.Z + hole.BottomCenterPoint!.Z) * 0.5);
2253 var faceToHole = new Vector3D(faceCentroid, holeCenter);
2254 double faceToHoleDot = Vector3D.Dot(faceToHole, faceNormal);
2255 if (faceToHoleDot <= 0.5 && faceToHoleDot >= -0.5) continue; // must face toward or away from hole, not be orthogonal
2256
2257 // Final check: there must be a coaxial cylinder on both sides of this face's axial position
2258 var shoulderAxisPoint = PointOnAxis(hole.BottomCenterPoint!, holeAxis, shoulderAxialPos);
2259 bool hasCylBelowShoulder = HasCoaxialCylinderTouchingSide(shoulderAxisPoint, holeAxis, cylindricalFaces, -1, null, out _);
2260 bool hasCylAboveShoulder = HasCoaxialCylinderTouchingSide(shoulderAxisPoint, holeAxis, cylindricalFaces, 1, null, out _);
2261 if (!hasCylBelowShoulder || !hasCylAboveShoulder) continue;

2263 face.IsBridgeBetweenHoles = true;
2264 }

2267 private static void DetectNestedHoles(List<CADHole> holes)
2268 {
2269 foreach (var h1 in holes)
2270 {
2271 if (h1.AxisDirection == null || h1.TopCenterPoint == null || h1.BottomCenterPoint == null)
2272 continue;
2273
2274 foreach (var h2 in holes)
2275 {
2276 if (h1.HoleId == h2.HoleId)
2277 continue;
2278
2279 if (h2.AxisDirection == null || h2.TopCenterPoint == null || h2.BottomCenterPoint == null)
2280 continue;

2282 // Check if axes are parallel (dot product near 1 or -1)
2283 double dot = Vector3D.Dot(h1.AxisDirection, h2.AxisDirection);
2284 if (Math.Abs(dot) < 0.99)
2285 continue;

2288 Vector3D vToH2 = new Vector3D(h1.TopCenterPoint, h2.TopCenterPoint);

2290 if (cross.Length > 1.0) // 1mm tolerance for coaxiality
2291 continue;

2294 // 1. h1 TopDiameter must be larger than h2 TopDiameter
2295 if (h1.TopDiameter <= h2.TopDiameter)
2296 continue;
2297
2298 // 2. h1 must be "above" h2 or encapsulate it.
2299 // Let's project centers along the shared vertical axis (assuming Z points down is negative)
2300 // Wait, TopCenterPoint is not guaranteed to be higher Z. Let's just use Z coordinate if vertical.
2301 if (Math.Abs(h1.AxisDirection.Z) > 0.9)
2302 {
2303 double h1TopZ = Math.Max(h1.TopCenterPoint.Z, h1.BottomCenterPoint.Z);
2304 double h1BotZ = Math.Min(h1.TopCenterPoint.Z, h1.BottomCenterPoint.Z);
2305 double h2TopZ = Math.Max(h2.TopCenterPoint.Z, h2.BottomCenterPoint.Z);
2306 double h2BotZ = Math.Min(h2.TopCenterPoint.Z, h2.BottomCenterPoint.Z);

2309 // Counterbore: h1 is above h2. (h1BotZ is near h2TopZ, or h1 encapsulates h2)
2310 if (h1TopZ >= h2TopZ - 1.0)
2311 {

2313 h2.IsNested = true;

2408 {

2411 return false;

2413 var holeAxis = new Vector3D(hole.AxisDirection.X, hole.AxisDirection.Y, hole.AxisDirection.Z);
2414 if (holeAxis.Length < 1e-9)
2415 return false;
2416
2417 holeAxis.Normalize();

2419 return false;
2420
2421 var firstEnd = hole.TopCenterPoint!;
2422 var secondEnd = hole.BottomCenterPoint!;

2424 double secondProj = Vector3D.Dot(new Vector3D(Point3D.Origin, secondEnd), approachAxis);

2434 double.IsNaN(passageDiameter) ||
2435 double.IsInfinity(passageDiameter) ||
2436 passageDiameter <= 0.0)
2437 {
2438 return false;

2441 info = new HoleReachabilityInfo
2442 {
2443 Hole = hole,
2444 TopEnd = topEnd,
2445 BottomEnd = bottomEnd,
2446 TopProjection = Math.Max(firstProj, secondProj),
2447 BottomProjection = Math.Min(firstProj, secondProj),
2448 RequiredDiameter = requiredDiameter,
2449 PassageDiameter = passageDiameter

2529 private static void EvaluateTopReachabilityGroup(List<HoleReachabilityInfo> group, Vector3D approachAxis, List<Models.CADFace> allFaces, bool avoidBottomFace)
2530 {
2531 if (!avoidBottomFace || group.Count == 0)
2532 return;
2533
2534 var sorted = group.OrderByDescending(info => info.TopProjection).ToList();
2535
2536 double bottleneckDiameter = double.PositiveInfinity;
2537 double channelBottomProjection = double.PositiveInfinity;
2538 bool hasTopAccess = true;
2539 const double axialConnectionTolerance = 2.0;
2540
2541 for (int i = 0; i < sorted.Count; i++)
2542 {
2543 var info = sorted[i];
2544
2545 if (!hasTopAccess)
2546 {
2547 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2548 continue;
2549 }
2550
2551 if (i == 0 &&
2552 IsPhysicalTopEndClosed(info.Hole, info.TopEnd, approachAxis, allFaces))
2553 {
2554 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2555 hasTopAccess = false;
2556 continue;
2557 }
2558
2559 // A lower section that does not meet the channel above is not reachable
2560 // from the top while the underside is unavailable.
2561 if (i > 0 &&
2562 info.TopProjection < channelBottomProjection - axialConnectionTolerance)
2563 {
2564 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2565 hasTopAccess = false;
2566 continue;
2567 }
2568
2569 // The tool must fit through every preceding bore. If this feature is
2570 // wider than that bottleneck, its walls are unreachable. Keep tracking
2571 // its passage diameter, however: a still-smaller hole farther down may
2572 // remain reachable through this larger void.
2573 if (info.RequiredDiameter > bottleneckDiameter + OBSTRUCTION_DIAMETER_TOLERANCE)
2574 {
2575 MarkHoleAndFacesIgnored(info.Hole, allFaces);
2576 }
2577
2578 bottleneckDiameter = Math.Min(bottleneckDiameter, info.PassageDiameter);
2579 channelBottomProjection = Math.Min(channelBottomProjection, info.BottomProjection);
2580 }
2581 }

2592 private static bool IsPhysicalTopEndClosed(Models.CADHole hole, Point3D endPoint, Vector3D axis, List<Models.CADFace> allFaces)
2593 {
2594 var planarFaces = allFaces.Where(f => f.FaceType == Models.FaceTypeEnum.Planar && !f.IsIgnored).ToList();
2595
2596 double holeDiameter = hole.TopDiameter > 0 ? hole.TopDiameter : hole.BottomDiameter;
2597 if (double.IsNaN(holeDiameter) || holeDiameter <= 0) holeDiameter = 1.0;
2598 double holeRadius = holeDiameter / 2.0;
2599
2600 foreach (var face in planarFaces)
2601 {
2602 if (IsPlanarFaceAtAxisPoint(face, endPoint, axis, 0.5))
2603 {
2604 var points = GetFaceAnalysisPoints(face);
2605 if (points.Count == 0) continue;
2606
2607 double minRadius = double.MaxValue;
2608 double maxRadius = double.MinValue;
2609
2610 foreach (var point in points)
2611 {
2612 double dist = DistancePointToAxis(point, endPoint, axis);
2613 if (dist < minRadius) minRadius = dist;
2614 if (dist > maxRadius) maxRadius = dist;
2615 }
2616
2617 double radialMargin = 1.0; // 1mm margin for radius matching
2618
2619 // 1. If maxRadius <= holeRadius + margin: It's a CAP exactly matching the hole. -> Blocked
2620 if (maxRadius <= holeRadius + radialMargin)
2621 return true;
2622
2623 // 2. If minRadius > holeRadius + margin: It's a PLATE covering the hole (edges are far). -> Blocked
2624 if (minRadius > holeRadius + radialMargin)
2625 return true;
2626
2627 // 3. If minRadius <= holeRadius + margin AND maxRadius > holeRadius + margin:
2628 // It's an annulus (SHELF) like the bottom of a counterbore. It has a hole in the middle. -> NOT Blocked
2629 // So we do NOT return true here, continue checking other faces.
2630 }
2631 }
2632 return false;
2633 }
