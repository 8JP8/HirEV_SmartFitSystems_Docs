193 public static Vector3D GetFaceNormalApprox(Brep.Face? face)
194 {
195 try
196 {
197 var list = new List<Point3D>();
198 if (face?.Parametric != null)
199 {
200 foreach (var surf in face.Parametric)
201 {
202 if (surf == null) continue;
203
204 if (surf.Vertices == null) surf.Regen(0.01);
205 if (surf.Vertices != null) list.AddRange(surf.Vertices);
206 }
207 }
208
209 if (list.Count < 3)
210 return new Vector3D(0, 0, 0);
211
212 // Find three non-collinear points (try a few combinations if necessary)
213 Point3D a = list[0];
214 int idxB = 1;
215 for (int i = 1; i < list.Count; i++)
216 {
217 if (!PointsAreEqual(a, list[i])) { idxB = i; break; }
218 }
219 Point3D b = list[idxB];
220
221 bool found = false;
222 for (int i = idxB + 1; i < list.Count && !found; i++)
223 {
224 Point3D c = list[i];
225 var vx1 = b.X - a.X; var vy1 = b.Y - a.Y; var vz1 = b.Z - a.Z;
226 var vx2 = c.X - a.X; var vy2 = c.Y - a.Y; var vz2 = c.Z - a.Z;
227 // cross product
228 var cx = vy1 * vz2 - vz1 * vy2;
229 var cy = vz1 * vx2 - vx1 * vz2;
230 var cz = vx1 * vy2 - vy1 * vx2;
231 double clen = Math.Sqrt(cx * cx + cy * cy + cz * cz);
232 if (clen > 1e-9)
233 {
234 // Normalize and return
235 return new Vector3D(cx / clen, cy / clen, cz / clen);
236 }
237 }
238
239 // Fallback try with the chosen three points
240 Point3D cFallback = list.Count > idxB + 1 ? list[idxB + 1] : list[idxB];
241 var v1x = b.X - a.X; var v1y = b.Y - a.Y; var v1z = b.Z - a.Z;
242 var v2x = cFallback.X - a.X; var v2y = cFallback.Y - a.Y; var v2z = cFallback.Z - a.Z;
243 var rx = v1y * v2z - v1z * v2y;
244 var ry = v1z * v2x - v1x * v2z;
245 var rz = v1x * v2y - v1y * v2x;
246 double rlen = Math.Sqrt(rx * rx + ry * ry + rz * rz);
247 if (rlen < 1e-9)
248 return new Vector3D(0, 0, 0);
249
250 return new Vector3D(rx / rlen, ry / rlen, rz / rlen);
251 }
252 catch
253 {
254 return new Vector3D(0, 0, 0);
255 }
256 }

352 public static bool RayTriangleIntersect(Point3D rayStart, Vector3D rayDirection, Point3D v1, Point3D v2, Point3D v3, out Point3D intersection)
353 {
354 intersection = Point3D.Origin;
355
356 try
357 {
358 Vector3D edge1 = new Vector3D(v2.X - v1.X, v2.Y - v1.Y, v2.Z - v1.Z);
359 Vector3D edge2 = new Vector3D(v3.X - v1.X, v3.Y - v1.Y, v3.Z - v1.Z);
360
361 Vector3D h = CrossProduct(rayDirection, edge2);
362 double a = DotProduct(edge1, h);
363
364 if (a > -1e-9 && a < 1e-9) // Ray parallel to triangle
365 return false;
366
367 double f = 1.0 / a;
368 Vector3D s = new Vector3D(rayStart.X - v1.X, rayStart.Y - v1.Y, rayStart.Z - v1.Z);
369 double u = f * DotProduct(s, h);
370
371 if (u < 0.0 || u > 1.0)
372 return false;
373
374 Vector3D q = CrossProduct(s, edge1);
375 double v = f * DotProduct(rayDirection, q);
376
377 if (v < 0.0 || u + v > 1.0)
378 return false;
379
380 double t = f * DotProduct(edge2, q);
381
382 if (t > 1e-9) // Ray intersection (t must be positive)
383 {
384 intersection = new Point3D(
385 rayStart.X + rayDirection.X * t,
386 rayStart.Y + rayDirection.Y * t,
387 rayStart.Z + rayDirection.Z * t
388 );
389 return true;
390 }
391
392 return false;
393 }
394 catch
395 {
396 return false;
397 }

412 public static bool IntersectWithMesh(Mesh mesh, Point3D rayStart, Vector3D rayDirection, out Point3D intersection)
413 {
414 intersection = Point3D.Origin;
415
416 try
417 {
418 if (mesh.Triangles == null || mesh.Vertices == null)
419 return false;
420
421 double closestDistance = double.MaxValue;
422 bool foundHit = false;
423
424 for (int i = 0; i < mesh.Triangles.Length; i++)
425 {
426 var triangle = mesh.Triangles[i];
427 Point3D v1 = mesh.Vertices[triangle.V1];
428 Point3D v2 = mesh.Vertices[triangle.V2];
429 Point3D v3 = mesh.Vertices[triangle.V3];
430
431 Point3D hitPoint;
432 if (RayTriangleIntersect(rayStart, rayDirection, v1, v2, v3, out hitPoint))
433 {
434 double distance = DistanceBetweenPoints(rayStart, hitPoint);
435 if (distance < closestDistance)
436 {
437 closestDistance = distance;
438 intersection = hitPoint;
439 foundHit = true;
440 }
441 }
442 }
443
444 return foundHit;
445 }
446 catch
447 {
448 return false;
449 }

634 public static Point3D ClosestPointOnTriangle(Point3D p, Point3D a, Point3D b, Point3D c)
635 {
636 // Check if P in vertex region outside A
637 Vector3D ab = new Vector3D(b.X - a.X, b.Y - a.Y, b.Z - a.Z);
638 Vector3D ac = new Vector3D(c.X - a.X, c.Y - a.Y, c.Z - a.Z);
639 Vector3D ap = new Vector3D(p.X - a.X, p.Y - a.Y, p.Z - a.Z);
640
641 double d1 = DotProduct(ab, ap);
642 double d2 = DotProduct(ac, ap);
643 if (d1 <= 0.0 && d2 <= 0.0) return a;
644
645 // Check if P in vertex region outside B
646 Vector3D bp = new Vector3D(p.X - b.X, p.Y - b.Y, p.Z - b.Z);
647 double d3 = DotProduct(ab, bp);
648 double d4 = DotProduct(ac, bp);
649 if (d3 >= 0.0 && d4 <= d3) return b;
650
651 // Check if P in edge region of AB
652 double vc = d1 * d4 - d3 * d2;
653 if (vc <= 0.0 && d1 >= 0.0 && d3 <= 0.0)
654 {
655 double v = d1 / (d1 - d3);
656 return new Point3D(a.X + v * ab.X, a.Y + v * ab.Y, a.Z + v * ab.Z);
657 }
658
659 // Check if P in vertex region outside C
660 Vector3D cp = new Vector3D(p.X - c.X, p.Y - c.Y, p.Z - c.Z);
661 double d5 = DotProduct(ab, cp);
662 double d6 = DotProduct(ac, cp);
663 if (d6 >= 0.0 && d5 <= d6) return c;
664
665 // Check if P in edge region of AC
666 double vb = d5 * d2 - d1 * d6;
667 if (vb <= 0.0 && d2 >= 0.0 && d6 <= 0.0)
668 {
669 double w = d2 / (d2 - d6);
670 return new Point3D(a.X + w * ac.X, a.Y + w * ac.Y, a.Z + w * ac.Z);
671 }
672
673 // Check if P in edge region of BC
674 double va = d3 * d6 - d5 * d4;
675 if (va <= 0.0 && (d4 - d3) >= 0.0 && (d5 - d6) >= 0.0)
676 {
677 double w = (d4 - d3) / ((d4 - d3) + (d5 - d6));
678 return new Point3D(b.X + w * (c.X - b.X), b.Y + w * (c.Y - b.Y), b.Z + w * (c.Z - b.Z));
679 }
680
681 // P inside face region
682 double denom = 1.0 / (va + vb + vc);
683 double v2 = vb * denom;
684 double w2 = vc * denom;
685 return new Point3D(a.X + ab.X * v2 + ac.X * w2, a.Y + ab.Y * v2 + ac.Y * w2, a.Z + ab.Z * v2 + ac.Z * w2);
686 }

797 public static List<Point3D> GenerateGridPointsOnFaceSurface(Brep brep, int faceIndex, double gridSpacing, double marginFromEdge, int maxPoints, List<Point3D>? existingControlPoints = null)
798 {
799 var points = new List<Point3D>();
800
801 if (brep == null || faceIndex < 0 || faceIndex >= brep.Faces.Length)
802 return points;
803
804 try
805 {
806 var face = brep.Faces[faceIndex];
807 var mesh = face.ConvertToMesh();
808
809 if (mesh == null || mesh.Triangles == null || mesh.Vertices == null)
810 return points;
811
812 if (mesh.Triangles.Length == 0 || mesh.Vertices.Length == 0)
813 return points;
814
815 // Get bounding box of the mesh
816 var boxMin = mesh.BoxMin;
817 var boxMax = mesh.BoxMax;
818 if (boxMin == null || boxMax == null)
819 return points;
820
821 double minX = boxMin.X;
822 double minY = boxMin.Y;
823 double minZ = boxMin.Z;
824 double maxX = boxMax.X;
825 double maxY = boxMax.Y;
826 double maxZ = boxMax.Z;
827
828 System.Diagnostics.Debug.WriteLine($" 📦 BBox: ({minX:F1},{minY:F1},{minZ:F1}) to ({maxX:F1},{maxY:F1},{maxZ:F1})");
829
830 // Generate a 2D grid in the dominant plane
831 // Determine which plane has the largest area
832 double sizeX = maxX - minX;
833 double sizeY = maxY - minY;
834 double sizeZ = maxZ - minZ;
835
836 int gridPointsAdded = 0;
837 int gridPointsSkippedNearControl = 0;
838 int gridPointsSkippedOutside = 0;
839 int gridPointsSkippedMargin = 0;
840
841 // Create grid in the two largest dimensions
842 if (sizeX >= sizeY && sizeX >= sizeZ)
843 {
844 // X is largest, use YZ plane grid
845 for (double y = minY + marginFromEdge; y <= maxY - marginFromEdge; y += gridSpacing)
846 {
847 for (double z = minZ + marginFromEdge; z <= maxZ - marginFromEdge; z += gridSpacing)
848 {
849 if (points.Count >= maxPoints) break;
850
851 // Create a point and project it onto the surface
852 Point3D testPoint = new Point3D((minX + maxX) / 2, y, z);
853 if (TryProjectPointToMesh(testPoint, mesh, out Point3D projectedPoint, out double distance))
854 {
855 // Check if too close to existing control points
856 if (existingControlPoints != null && IsPointNearExisting(projectedPoint, existingControlPoints, gridSpacing))
857 {
858 gridPointsSkippedNearControl++;
859 continue;
860 }
861
862 // Check if point is too close to mesh edges
863 if (IsPointTooCloseToEdge(projectedPoint, mesh, marginFromEdge))
864 {
865 gridPointsSkippedMargin++;
866 continue;
867 }
868
869 points.Add(projectedPoint);
870 gridPointsAdded++;
871 }
872 else
873 {
874 gridPointsSkippedOutside++;
875 }
876 }
877 if (points.Count >= maxPoints) break;
878 }
879 }
880 else if (sizeY >= sizeX && sizeY >= sizeZ)
881 {
882 // Y is largest, use XZ plane grid
883 for (double x = minX + marginFromEdge; x <= maxX - marginFromEdge; x += gridSpacing)
884 {
885 for (double z = minZ + marginFromEdge; z <= maxZ - marginFromEdge; z += gridSpacing)
886 {
887 if (points.Count >= maxPoints) break;
888
889 Point3D testPoint = new Point3D(x, (minY + maxY) / 2, z);
890 if (TryProjectPointToMesh(testPoint, mesh, out Point3D projectedPoint, out double distance))
891 {
892 if (existingControlPoints != null && IsPointNearExisting(projectedPoint, existingControlPoints, gridSpacing))
893 {
894 gridPointsSkippedNearControl++;
895 continue;
896 }
897
898 if (IsPointTooCloseToEdge(projectedPoint, mesh, marginFromEdge))
899 {
900 gridPointsSkippedMargin++;
901 continue;
902 }
903
904 points.Add(projectedPoint);
905 gridPointsAdded++;
906 }
907 else
908 {
909 gridPointsSkippedOutside++;
910 }
911 }
912 if (points.Count >= maxPoints) break;
913 }
914 }
915 else
916 {
917 // Z is largest, use XY plane grid
918 for (double x = minX + marginFromEdge; x <= maxX - marginFromEdge; x += gridSpacing)
919 {
920 for (double y = minY + marginFromEdge; y <= maxY - marginFromEdge; y += gridSpacing)
921 {
922 if (points.Count >= maxPoints) break;
923
924 Point3D testPoint = new Point3D(x, y, (minZ + maxZ) / 2);
925 if (TryProjectPointToMesh(testPoint, mesh, out Point3D projectedPoint, out double distance))
926 {
927 if (existingControlPoints != null && IsPointNearExisting(projectedPoint, existingControlPoints, gridSpacing))
928 {
929 gridPointsSkippedNearControl++;
930 continue;
931 }
932
933 if (IsPointTooCloseToEdge(projectedPoint, mesh, marginFromEdge))
934 {
935 gridPointsSkippedMargin++;
936 continue;
937 }
938
939 points.Add(projectedPoint);
940 gridPointsAdded++;
941 }
942 else
943 {
944 gridPointsSkippedOutside++;
945 }
946 }
947 if (points.Count >= maxPoints) break;
948 }
949 }
950
951 System.Diagnostics.Debug.WriteLine($" ✅ Grid generation: Added {gridPointsAdded}, Skipped {gridPointsSkippedNearControl} (near control pts), {gridPointsSkippedOutside} (outside), {gridPointsSkippedMargin} (edge margin)");
952 }

1073 private static double CalculateTriangleArea(Point3D v1, Point3D v2, Point3D v3)
1074 {
1075 // Using Heron's formula
1076 double a = DistanceBetweenPoints(v1, v2);
1077 double b = DistanceBetweenPoints(v2, v3);
1078 double c = DistanceBetweenPoints(v3, v1);
1079 double s = (a + b + c) / 2.0; // semi-perimeter
1080 double area = Math.Sqrt(Math.Max(0, s * (s - a) * (s - b) * (s - c)));
1081 return area;
1082 }

1096 private static double DistancePointToLineSegment(Point3D p, Point3D lineStart, Point3D lineEnd)
1097 {
1098 double segmentLengthSquared = DistanceBetweenPoints(lineStart, lineEnd);
1099 segmentLengthSquared *= segmentLengthSquared;
1100
1101 if (segmentLengthSquared < 1e-10)
1102 return DistanceBetweenPoints(p, lineStart);
1103
1104 // Calculate projection parameter t
1105 double t = ((p.X - lineStart.X) * (lineEnd.X - lineStart.X) +
1106 (p.Y - lineStart.Y) * (lineEnd.Y - lineStart.Y) +
1107 (p.Z - lineStart.Z) * (lineEnd.Z - lineStart.Z)) / segmentLengthSquared;
1108
1109 t = Math.Max(0, Math.Min(1, t)); // Clamp to [0, 1]
1110
1111 // Find closest point on segment
1112 double closestX = lineStart.X + t * (lineEnd.X - lineStart.X);
1113 double closestY = lineStart.Y + t * (lineEnd.Y - lineStart.Y);
1114 double closestZ = lineStart.Z + t * (lineEnd.Z - lineStart.Z);
1115
1116 Point3D closest = new Point3D(closestX, closestY, closestZ);
1117 return DistanceBetweenPoints(p, closest);
1118 }

1160 public static Vector3D GetFaceAxis(CADFace face)
1161 {
1162 try
1163 {
1164 if (face == null) return new Vector3D(0, 0, 1);
1165
1166 // Try to infer from GeometricInfo (example: "Axis: (0.00, 0.00, 1.00)")
1167 if (!string.IsNullOrEmpty(face.GeometricInfo))

1170 face.GeometricInfo,
1171 @"Axis:\s*\‍(([-0-9.+Ee]+),\s*([-0-9.+Ee]+),\s*([-0-9.+Ee]+)\‍)",
1172 System.Text.RegularExpressions.RegexOptions.IgnoreCase
1173 );
1174
1175 if (match.Success)
1176 {
1177 if (double.TryParse(match.Groups[1].Value, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out double x) &&
1178 double.TryParse(match.Groups[2].Value, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out double y) &&
1179 double.TryParse(match.Groups[3].Value, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out double z))
1180 {
1181 var axis = new Vector3D(x, y, z);
1182 NormalizeVector(ref axis);
1183 return axis;
1184 }
1185 }
1186 }
1187
1188 // Fallback for planar faces: approximate using keypoints
1189 if (face.KeyPoints != null && face.KeyPoints.Count >= 3)
1190 {
1191 var a = face.KeyPoints[0];
1192 var b = face.KeyPoints[1];
1193 var c = face.KeyPoints[2];
1194
1195 var ab = new Vector3D(b.X - a.X, b.Y - a.Y, b.Z - a.Z);
1196 var ac = new Vector3D(c.X - a.X, c.Y - a.Y, c.Z - a.Z);
1197 var normal = CrossProduct(ab, ac);
1198 NormalizeVector(ref normal);
1199 return normal;
1200 }
1201
1202 // Default fallback: Z-axis
1203 return new Vector3D(0, 0, 1);
1204 }
1205 catch
1206 {
1207 return new Vector3D(0, 0, 1);
1208 }
