203 public static Vector3D GetFaceNormalApprox(Brep.Face? face)
204 {
205 try
206 {
207 var list = new List<Point3D>();
208 if (face?.Parametric != null)
209 {
210 foreach (var surf in face.Parametric)
211 {
212 if (surf == null) continue;
213
214 if (surf.Vertices == null) surf.Regen(0.01);
215 if (surf.Vertices != null) list.AddRange(surf.Vertices);
216 }
217 }
218
219 if (list.Count < 3)
220 return new Vector3D(0, 0, 0);
221
222 // Find three non-collinear points (try a few combinations if necessary)
223 Point3D a = list[0];
224 int idxB = 1;
225 for (int i = 1; i < list.Count; i++)
226 {
227 if (!PointsAreEqual(a, list[i])) { idxB = i; break; }
228 }
229 Point3D b = list[idxB];
230
231 bool found = false;
232 for (int i = idxB + 1; i < list.Count && !found; i++)
233 {
234 Point3D c = list[i];
235 var vx1 = b.X - a.X; var vy1 = b.Y - a.Y; var vz1 = b.Z - a.Z;
236 var vx2 = c.X - a.X; var vy2 = c.Y - a.Y; var vz2 = c.Z - a.Z;
237 // cross product
238 var cx = vy1 * vz2 - vz1 * vy2;
239 var cy = vz1 * vx2 - vx1 * vz2;
240 var cz = vx1 * vy2 - vy1 * vx2;
241 double clen = Math.Sqrt(cx * cx + cy * cy + cz * cz);
242 if (clen > 1e-9)
243 {
244 // Normalize and return
245 return new Vector3D(cx / clen, cy / clen, cz / clen);
246 }
247 }
248
249 // Fallback try with the chosen three points
250 Point3D cFallback = list.Count > idxB + 1 ? list[idxB + 1] : list[idxB];
251 var v1x = b.X - a.X; var v1y = b.Y - a.Y; var v1z = b.Z - a.Z;
252 var v2x = cFallback.X - a.X; var v2y = cFallback.Y - a.Y; var v2z = cFallback.Z - a.Z;
253 var rx = v1y * v2z - v1z * v2y;
254 var ry = v1z * v2x - v1x * v2z;
255 var rz = v1x * v2y - v1y * v2x;
256 double rlen = Math.Sqrt(rx * rx + ry * ry + rz * rz);
257 if (rlen < 1e-9)
258 return new Vector3D(0, 0, 0);
259
260 return new Vector3D(rx / rlen, ry / rlen, rz / rlen);
261 }
262 catch
263 {
264 return new Vector3D(0, 0, 0);
265 }
266 }

362 public static bool RayTriangleIntersect(Point3D rayStart, Vector3D rayDirection, Point3D v1, Point3D v2, Point3D v3, out Point3D intersection)
363 {
364 intersection = Point3D.Origin;
365
366 try
367 {
368 Vector3D edge1 = new Vector3D(v2.X - v1.X, v2.Y - v1.Y, v2.Z - v1.Z);
369 Vector3D edge2 = new Vector3D(v3.X - v1.X, v3.Y - v1.Y, v3.Z - v1.Z);
370
371 Vector3D h = CrossProduct(rayDirection, edge2);
372 double a = DotProduct(edge1, h);
373
374 if (a > -1e-9 && a < 1e-9) // Ray parallel to triangle
375 return false;
376
377 double f = 1.0 / a;
378 Vector3D s = new Vector3D(rayStart.X - v1.X, rayStart.Y - v1.Y, rayStart.Z - v1.Z);
379 double u = f * DotProduct(s, h);
380
381 if (u < 0.0 || u > 1.0)
382 return false;
383
384 Vector3D q = CrossProduct(s, edge1);
385 double v = f * DotProduct(rayDirection, q);
386
387 if (v < 0.0 || u + v > 1.0)
388 return false;
389
390 double t = f * DotProduct(edge2, q);
391
392 if (t > 1e-9) // Ray intersection (t must be positive)
393 {
394 intersection = new Point3D(
395 rayStart.X + rayDirection.X * t,
396 rayStart.Y + rayDirection.Y * t,
397 rayStart.Z + rayDirection.Z * t
398 );
399 return true;
400 }
401
402 return false;
403 }
404 catch
405 {
406 return false;
407 }

422 public static bool IntersectWithMesh(Mesh mesh, Point3D rayStart, Vector3D rayDirection, out Point3D intersection)
423 {
424 intersection = Point3D.Origin;
425
426 try
427 {
428 if (mesh.Triangles == null || mesh.Vertices == null)
429 return false;
430
431 double closestDistance = double.MaxValue;
432 bool foundHit = false;
433
434 for (int i = 0; i < mesh.Triangles.Length; i++)
435 {
436 var triangle = mesh.Triangles[i];
437 Point3D v1 = mesh.Vertices[triangle.V1];
438 Point3D v2 = mesh.Vertices[triangle.V2];
439 Point3D v3 = mesh.Vertices[triangle.V3];

441 Point3D hitPoint;
442 if (RayTriangleIntersect(rayStart, rayDirection, v1, v2, v3, out hitPoint))
443 {
444 double distance = DistanceBetweenPoints(rayStart, hitPoint);
445 if (distance < closestDistance)
446 {
447 closestDistance = distance;
448 intersection = hitPoint;
449 foundHit = true;
450 }
451 }
452 }
453
454 return foundHit;
455 }

644 public static Point3D ClosestPointOnTriangle(Point3D p, Point3D a, Point3D b, Point3D c)
645 {
646 // Check if P in vertex region outside A
647 Vector3D ab = new Vector3D(b.X - a.X, b.Y - a.Y, b.Z - a.Z);
648 Vector3D ac = new Vector3D(c.X - a.X, c.Y - a.Y, c.Z - a.Z);
649 Vector3D ap = new Vector3D(p.X - a.X, p.Y - a.Y, p.Z - a.Z);
650
651 double d1 = DotProduct(ab, ap);
652 double d2 = DotProduct(ac, ap);
653 if (d1 <= 0.0 && d2 <= 0.0) return a;
654
655 // Check if P in vertex region outside B
656 Vector3D bp = new Vector3D(p.X - b.X, p.Y - b.Y, p.Z - b.Z);
657 double d3 = DotProduct(ab, bp);
658 double d4 = DotProduct(ac, bp);
659 if (d3 >= 0.0 && d4 <= d3) return b;
660
661 // Check if P in edge region of AB
662 double vc = d1 * d4 - d3 * d2;
663 if (vc <= 0.0 && d1 >= 0.0 && d3 <= 0.0)
664 {
665 double v = d1 / (d1 - d3);
666 return new Point3D(a.X + v * ab.X, a.Y + v * ab.Y, a.Z + v * ab.Z);
667 }
668
669 // Check if P in vertex region outside C
670 Vector3D cp = new Vector3D(p.X - c.X, p.Y - c.Y, p.Z - c.Z);
671 double d5 = DotProduct(ab, cp);
672 double d6 = DotProduct(ac, cp);
673 if (d6 >= 0.0 && d5 <= d6) return c;
674
675 // Check if P in edge region of AC
676 double vb = d5 * d2 - d1 * d6;
677 if (vb <= 0.0 && d2 >= 0.0 && d6 <= 0.0)
678 {
679 double w = d2 / (d2 - d6);
680 return new Point3D(a.X + w * ac.X, a.Y + w * ac.Y, a.Z + w * ac.Z);
681 }
682
683 // Check if P in edge region of BC
684 double va = d3 * d6 - d5 * d4;
685 if (va <= 0.0 && (d4 - d3) >= 0.0 && (d5 - d6) >= 0.0)
686 {
687 double w = (d4 - d3) / ((d4 - d3) + (d5 - d6));
688 return new Point3D(b.X + w * (c.X - b.X), b.Y + w * (c.Y - b.Y), b.Z + w * (c.Z - b.Z));
689 }
690
691 // P inside face region
692 double denom = 1.0 / (va + vb + vc);
693 double v2 = vb * denom;
694 double w2 = vc * denom;
695 return new Point3D(a.X + ab.X * v2 + ac.X * w2, a.Y + ab.Y * v2 + ac.Y * w2, a.Z + ab.Z * v2 + ac.Z * w2);
696 }

736 public static bool TryProjectPointToExactFaceSurface(
737 Point3D point,
738 Brep brep,
739 int faceIndex,
740 out Point3D projectedPoint,
741 out Vector3D surfaceNormal)
742 {
743 projectedPoint = point;
744 surfaceNormal = Vector3D.AxisZ;
745
746 if (point == null || brep == null || faceIndex < 0 || faceIndex >= brep.Faces.Length)
747 return false;
748
749 try
750 {
751 var face = brep.Faces[faceIndex];
752 var mesh = face.ConvertToMesh();
753 Point3D meshSeed = mesh != null
754 ? ProjectPointToMesh(point, mesh)
755 : point;
756
757 Point3D? bestPoint = null;
758 Vector3D? bestNormal = null;
759 double bestDistance = double.MaxValue;
760
761 if (face.Parametric != null)
762 {
763 foreach (var surface in face.Parametric)
764 {
765 if (surface == null)
766 continue;
767
768 foreach (var seed in new[] { point, meshSeed })
769 {
770 try
771 {
772 surface.ClosestPointTo(seed, out double u, out double v);
773 var uv = new Point2D(u, v);
774 Point3D candidate = surface.PointAt(u, v);
775 if (!IsFiniteCoordinate(candidate))
776 continue;
777
778 bool insideTrim = true;
779 if (surface.IsTrimmed)
780 {
781 insideTrim = surface.PolyRegion?.IsPointInside(uv, 1e-7) == true;
782 if (!insideTrim && mesh != null)
783 {
784 // Points on a trimming contour are not always reported as
785 // polygon-interior. Accept them only when the face mesh confirms
786 // that the candidate belongs to this trimmed patch.
787 Point3D trimWitness = ProjectPointToMesh(candidate, mesh);
788 insideTrim = candidate.DistanceTo(trimWitness) <= 0.5;
789 }
790 }
791
792 if (!insideTrim)
793 continue;
794
795 double distance = point.DistanceTo(candidate);
796 if (distance >= bestDistance)
797 continue;
798
799 Vector3D normal;
800 try
801 {
802 normal = face.Normal(candidate);
803 }
804 catch
805 {
806 normal = surface.NormalAt(u, v);
807 }
808
809 if (normal == null || normal.Length <= 1e-8)
810 normal = Vector3D.AxisZ;
811 else
812 normal.Normalize();
813
814 bestPoint = candidate;
815 bestNormal = normal;
816 bestDistance = distance;
817 }
818 catch
819 {
820 // Try the next parametric patch/seed.
821 }
822 }
823 }
824 }
825
826 if (bestPoint == null)
827 return false;
828
829 projectedPoint = bestPoint;
830 surfaceNormal = bestNormal ?? Vector3D.AxisZ;
831 return true;
832 }
833 catch
834 {
835 return false;
836 }
837 }

936 public static List<Point3D> GenerateGridPointsOnFaceSurface(Brep brep, int faceIndex, double gridSpacing, double marginFromEdge, int maxPoints, List<Point3D>? existingControlPoints = null)
937 {
938 var points = new List<Point3D>();
939
940 if (brep == null || faceIndex < 0 || faceIndex >= brep.Faces.Length)
941 return points;
942
943 try
944 {
945 var face = brep.Faces[faceIndex];
946 var mesh = face.ConvertToMesh();
947
948 if (mesh == null || mesh.Triangles == null || mesh.Vertices == null)
949 return points;
950
951 if (mesh.Triangles.Length == 0 || mesh.Vertices.Length == 0)
952 return points;
953
954 // Get bounding box of the mesh
955 var boxMin = mesh.BoxMin;
956 var boxMax = mesh.BoxMax;
957 if (boxMin == null || boxMax == null)
958 return points;
959
960 double minX = boxMin.X;
961 double minY = boxMin.Y;
962 double minZ = boxMin.Z;
963 double maxX = boxMax.X;
964 double maxY = boxMax.Y;
965 double maxZ = boxMax.Z;
966
967 System.Diagnostics.Debug.WriteLine($" 📦 BBox: ({minX:F1},{minY:F1},{minZ:F1}) to ({maxX:F1},{maxY:F1},{maxZ:F1})");
968
969 // Generate a 2D grid in the dominant plane
970 // Determine which plane has the largest area
971 double sizeX = maxX - minX;
972 double sizeY = maxY - minY;
973 double sizeZ = maxZ - minZ;
974
975 int gridPointsAdded = 0;
976 int gridPointsSkippedNearControl = 0;
977 int gridPointsSkippedOutside = 0;
978 int gridPointsSkippedMargin = 0;
979
980 // Create grid in the two largest dimensions
981 if (sizeX >= sizeY && sizeX >= sizeZ)
982 {

1140 foreach (var tri in mesh.Triangles)

1173 private static bool IsPointTooCloseToEdge(Point3D point, devDept.Eyeshot.Entities.Mesh mesh, double margin)
1174 {
1175 if (margin <= 0)
1176 return false;
1177
1178 // Find the triangle containing or closest to this point
1179 double minDistToEdge = double.MaxValue;
1180
1181 foreach (var tri in mesh.Triangles)
1182 {
1183 Point3D v1 = mesh.Vertices[tri.V1];
1184 Point3D v2 = mesh.Vertices[tri.V2];
1185 Point3D v3 = mesh.Vertices[tri.V3];
1186
1187 // Check if point is inside or very close to this triangle
1188 double d1 = DistancePointToLineSegment(point, v1, v2);
1189 double d2 = DistancePointToLineSegment(point, v2, v3);
1190 double d3 = DistancePointToLineSegment(point, v3, v1);
1191
1192 double closestEdgeDist = Math.Min(d1, Math.Min(d2, d3));
1193
1194 if (closestEdgeDist < minDistToEdge)
1195 minDistToEdge = closestEdgeDist;
1196 }
1197

1212 private static double CalculateTriangleArea(Point3D v1, Point3D v2, Point3D v3)
1213 {
1214 // Using Heron's formula
1215 double a = DistanceBetweenPoints(v1, v2);
1216 double b = DistanceBetweenPoints(v2, v3);
1217 double c = DistanceBetweenPoints(v3, v1);
1218 double s = (a + b + c) / 2.0; // semi-perimeter
1219 double area = Math.Sqrt(Math.Max(0, s * (s - a) * (s - b) * (s - c)));
1220 return area;
1221 }

1235 private static double DistancePointToLineSegment(Point3D p, Point3D lineStart, Point3D lineEnd)
1236 {
1237 double segmentLengthSquared = DistanceBetweenPoints(lineStart, lineEnd);
1238 segmentLengthSquared *= segmentLengthSquared;
1239
1240 if (segmentLengthSquared < 1e-10)
1241 return DistanceBetweenPoints(p, lineStart);
1242
1243 // Calculate projection parameter t
1244 double t = ((p.X - lineStart.X) * (lineEnd.X - lineStart.X) +
1245 (p.Y - lineStart.Y) * (lineEnd.Y - lineStart.Y) +
1246 (p.Z - lineStart.Z) * (lineEnd.Z - lineStart.Z)) / segmentLengthSquared;
1247
1248 t = Math.Max(0, Math.Min(1, t)); // Clamp to [0, 1]
1249
1250 // Find closest point on segment
1251 double closestX = lineStart.X + t * (lineEnd.X - lineStart.X);
1252 double closestY = lineStart.Y + t * (lineEnd.Y - lineStart.Y);
1253 double closestZ = lineStart.Z + t * (lineEnd.Z - lineStart.Z);
1254
1255 Point3D closest = new Point3D(closestX, closestY, closestZ);
1256 return DistanceBetweenPoints(p, closest);
1257 }

1299 public static Vector3D GetFaceAxis(CADFace face)
1300 {
1301 try
1302 {
1303 if (face == null) return new Vector3D(0, 0, 1);
1304
1305 // Try to infer from GeometricInfo (example: "Axis: (0.00, 0.00, 1.00)")
1306 if (!string.IsNullOrEmpty(face.GeometricInfo))
1307 {

1309 face.GeometricInfo,
1310 @"Axis:\s*\‍(([-0-9.+Ee]+),\s*([-0-9.+Ee]+),\s*([-0-9.+Ee]+)\‍)",
1311 System.Text.RegularExpressions.RegexOptions.IgnoreCase
1312 );
1313
1314 if (match.Success)
1315 {

1317 double.TryParse(match.Groups[2].Value, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out double y) &&
1318 double.TryParse(match.Groups[3].Value, System.Globalization.NumberStyles.Any, System.Globalization.CultureInfo.InvariantCulture, out double z))
1319 {
1320 var axis = new Vector3D(x, y, z);
1321 NormalizeVector(ref axis);
1322 return axis;
1323 }
1324 }
1325 }
1326
1327 // Fallback for planar faces: approximate using keypoints
1328 if (face.KeyPoints != null && face.KeyPoints.Count >= 3)
1329 {
1330 var a = face.KeyPoints[0];

1337 NormalizeVector(ref normal);
1338 return normal;
1339 }
1340
1341 // Default fallback: Z-axis
1342 return new Vector3D(0, 0, 1);

1400 public static bool AreVectorsParallel(Vector3D v1, Vector3D v2, double tolerance = 1e-6)
