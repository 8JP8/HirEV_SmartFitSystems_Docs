57 public CollisionCheckResult ValidatePathClearsPartVolume(List<Point3D> pathPoints)
58 {
59 var result = new CollisionCheckResult { IsCollisionFree = true };
60
61 if (pathPoints == null || pathPoints.Count < 2)
62 {
63 result.Message = "Path too short to validate";
64 return result;
65 }
66
67 if (_partMesh?.Triangles == null || _partMesh?.Vertices == null)
68 {
69 result.Message = "No part mesh available for collision check";
70 return result;
71 }
72
73 Debug.WriteLine($"\n🔍 Validating path ({pathPoints.Count} points) for collisions...");
74
75 for (int i = 0; i < pathPoints.Count - 1; i++)
76 {
77 var segStart = pathPoints[i];
78 var segEnd = pathPoints[i + 1];
79
80 var intersection = CheckSegmentMeshIntersection(segStart, segEnd, _partMesh);
81
82 if (intersection != null)
83 {
84 result.IsCollisionFree = false;
85 result.CollisionPoints.Add(intersection);
86 result.CollisionSegmentIndices.Add(i);
87 }
88 }
89
90 if (result.IsCollisionFree)
91 {
92 result.Message = "Path is collision-free";
93 Debug.WriteLine($" ✅ {result.Message}");
94 }
95 else
96 {
97 result.Message = $"Found {result.CollisionPoints.Count} collision(s)";
98 Debug.WriteLine($" ⚠️ {result.Message}");
99 }
100
101 return result;
102 }

111 private Point3D? CheckSegmentMeshIntersection(Point3D start, Point3D end, Mesh mesh)
112 {
113 if (mesh?.Vertices == null || mesh?.Triangles == null)
114 return null;
115
116 Vector3D direction = new Vector3D(end.X - start.X, end.Y - start.Y, end.Z - start.Z);
117 double segmentLength = direction.Length;
118
119 if (segmentLength < 0.0001)
120 return null;
121
122 direction.Normalize();
123
124 Point3D? closestHit = null;
125 double closestDist = segmentLength + _collisionTolerance;
126
127 foreach (var triangle in mesh.Triangles)
128 {
129 if (triangle.V1 >= mesh.Vertices.Length ||
130 triangle.V2 >= mesh.Vertices.Length ||
131 triangle.V3 >= mesh.Vertices.Length)
132 continue;
133
134 var v0 = mesh.Vertices[triangle.V1];
135 var v1 = mesh.Vertices[triangle.V2];
136 var v2 = mesh.Vertices[triangle.V3];
137
138 var hit = RayTriangleIntersection(start, direction, v0, v1, v2);
139 if (hit != null)
140 {
141 double dist = start.DistanceTo(hit);
142 // Only count as collision if within segment bounds
143 if (dist > _collisionTolerance && dist < segmentLength - _collisionTolerance && dist < closestDist)
144 {
145 closestDist = dist;
146 closestHit = hit;
147 }
148 }
149 }
150
151 return closestHit;
152 }

157 private static Point3D? RayTriangleIntersection(Point3D rayOrigin, Vector3D rayDir, Point3D v0, Point3D v1, Point3D v2)
158 {
159 const double EPSILON = 0.0000001;
160
161 var edge1 = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
162 var edge2 = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
163 var h = Vector3D.Cross(rayDir, edge2);
164 var a = Vector3D.Dot(edge1, h);
165
166 if (a > -EPSILON && a < EPSILON)
167 return null;
168
169 double f = 1.0 / a;
170 var s = new Vector3D(rayOrigin.X - v0.X, rayOrigin.Y - v0.Y, rayOrigin.Z - v0.Z);
171 double u = f * Vector3D.Dot(s, h);
172
173 if (u < 0.0 || u > 1.0)
174 return null;
175
176 var q = Vector3D.Cross(s, edge1);
177 double v = f * Vector3D.Dot(rayDir, q);
178
179 if (v < 0.0 || u + v > 1.0)
180 return null;
181
182 double t = f * Vector3D.Dot(edge2, q);
183
184 if (t > EPSILON)
185 {
186 return new Point3D(
187 rayOrigin.X + rayDir.X * t,
188 rayOrigin.Y + rayDir.Y * t,
189 rayOrigin.Z + rayDir.Z * t);
190 }
191
192 return null;
193 }

207 public List<Point3D>? AttemptShellReroute(Point3D from, Point3D to, int maxIterations = 5)
208 {
209 if (_shellMesh?.Vertices == null)
210 {
211 Debug.WriteLine(" ⚠️ Cannot reroute: no shell mesh available");
212 return null;
213 }
214
215 Debug.WriteLine($" 🔄 Attempting shell reroute ({maxIterations} max iterations)...");
216
217 // Find nearest shell points to start and end
218 Point3D shellStart = FindNearestShellPoint(from);
219 Point3D shellEnd = FindNearestShellPoint(to);
220
221 // Use geodesic navigator if available, otherwise simple waypoint approach
222 var alternatePath = new List<Point3D> { from };
223
224 // Add intermediate waypoints along shell
225 int numWaypoints = Math.Max(3, (int)(from.DistanceTo(to) / 20.0)); // One waypoint per 20mm
226 for (int i = 1; i <= numWaypoints; i++)
227 {
228 double t = (double)i / (numWaypoints + 1);
229 Point3D interpolated = InterpolatePoint(shellStart, shellEnd, t);
230 Point3D snapped = FindNearestShellPoint(interpolated);
231 alternatePath.Add(snapped);
232 }
233
234 alternatePath.Add(to);
235
236 // Validate the alternate path
237 var checkResult = ValidatePathClearsPartVolume(alternatePath);
238 if (checkResult.IsCollisionFree)
239 {
240 Debug.WriteLine($" ✅ Reroute successful with {alternatePath.Count} waypoints");
241 return alternatePath;
242 }
243
244 Debug.WriteLine($" ⚠️ Reroute still has collisions, trying expansion...");
245
246 // Try expanding the path further from part center
247 for (int iteration = 1; iteration <= maxIterations; iteration++)
248 {
249 double expansionFactor = 1.0 + (iteration * 0.2); // 20%, 40%, 60%, etc.
250 var expandedPath = ExpandPathFromCenter(alternatePath, expansionFactor);
251
252 checkResult = ValidatePathClearsPartVolume(expandedPath);
253 if (checkResult.IsCollisionFree)
254 {
255 Debug.WriteLine($" ✅ Reroute successful after {iteration} expansion(s)");
256 return expandedPath;
257 }
258 }
259
260 Debug.WriteLine($" ❌ Reroute failed after {maxIterations} attempts");
261 return null;
262 }

267 private Point3D FindNearestShellPoint(Point3D point)
268 {
269 if (_shellMesh?.Vertices == null || _shellMesh.Vertices.Length == 0)
270 return point;
271
272 double minDist = double.MaxValue;
273 Point3D nearest = point;
274
275 foreach (var vertex in _shellMesh.Vertices)
276 {
277 double dist = point.DistanceTo(vertex);
278 if (dist < minDist)
279 {
280 minDist = dist;
281 nearest = vertex;
282 }
283 }
284
285 return nearest;
286 }

302 private List<Point3D> ExpandPathFromCenter(List<Point3D> path, double expansionFactor)
303 {
304 if (path.Count < 2)
305 return path;
306
307 // Compute path centroid as proxy for part center
308 double sumX = 0, sumY = 0, sumZ = 0;
309 foreach (var pt in path)
310 {
311 sumX += pt.X;
312 sumY += pt.Y;
313 sumZ += pt.Z;
314 }
315 Point3D center = new Point3D(sumX / path.Count, sumY / path.Count, sumZ / path.Count);
316
317 var expanded = new List<Point3D>(path.Count);
318
319 // Keep first and last points fixed
320 expanded.Add(path[0]);
321
322 for (int i = 1; i < path.Count - 1; i++)
323 {
324 var pt = path[i];
325 Vector3D fromCenter = new Vector3D(pt.X - center.X, pt.Y - center.Y, pt.Z - center.Z);
326
327 Point3D expandedPt = new Point3D(
328 pt.X + fromCenter.X * (expansionFactor - 1),
329 pt.Y + fromCenter.Y * (expansionFactor - 1),
330 pt.Z + fromCenter.Z * (expansionFactor - 1));
331
332 // Snap to shell if available
333 expandedPt = FindNearestShellPoint(expandedPt);
334 expanded.Add(expandedPt);
335 }
336
337 expanded.Add(path[path.Count - 1]);
338 return expanded;
339 }

347 public Point3D PushPointToShellSurface(Point3D interiorPoint, Point3D partCentroid)
348 {
349 if (_shellMesh?.Vertices == null)
350 return interiorPoint;
351
352 // First try: find nearest shell point
353 Point3D nearest = FindNearestShellPoint(interiorPoint);
354
355 // Validate it's further from centroid
356 if (nearest.DistanceTo(partCentroid) > interiorPoint.DistanceTo(partCentroid))
357 {
358 return nearest;
359 }
360
361 // Second try: project radially outward
362 Vector3D outward = new Vector3D(
363 interiorPoint.X - partCentroid.X,
364 interiorPoint.Y - partCentroid.Y,
365 interiorPoint.Z - partCentroid.Z);
366
367 if (outward.Length > 0.0001)
368 {
369 outward.Normalize();
370
371 // Raycast outward to find shell intersection
372 return ShellProjectionUtils.RaycastToShellSurface(interiorPoint, outward, _shellMesh) ?? nearest;
373 }
374
375 return nearest;
376 }
