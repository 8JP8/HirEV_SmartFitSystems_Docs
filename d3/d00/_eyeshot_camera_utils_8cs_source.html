263 private Point3D CalculateMeshCenter(Mesh mesh)
264 {
265 if (mesh.Vertices == null || mesh.Vertices.Length == 0)
266 return new Point3D(0, 0, 0);
267
268 double sumX = 0, sumY = 0, sumZ = 0;

271 sumX += vertex.X;

738 public void OrientCameraToNormal(Point3D targetPoint, Vector3D normal)
739 {
740 try
741 {
742 var viewport = _designControl.ActiveViewport;
743 if (viewport == null) return;
744
745 double camDist = viewport.Camera.Distance;
746 if (camDist < 1.0) camDist = 150.0;
747
748 Vector3D zAxis = (Vector3D)normal.Clone(); // opposite to look direction (target-to-location)
749 if (zAxis.LengthSquared < 1e-6) zAxis = Vector3D.AxisZ;
750 zAxis.Normalize();
751
752 Vector3D xAxis = Vector3D.Cross(Vector3D.AxisZ, zAxis);
753 if (xAxis.LengthSquared < 1e-6) xAxis = Vector3D.Cross(Vector3D.AxisY, zAxis);
754 xAxis.Normalize();
755
756 Vector3D yAxis = Vector3D.Cross(zAxis, xAxis); // camera up direction
757 yAxis.Normalize();
758
759 // Build 3x3 rotation matrix
760 double[,] rotMatrix = new double[3, 3];
761 rotMatrix[0, 0] = xAxis.X; rotMatrix[0, 1] = yAxis.X; rotMatrix[0, 2] = zAxis.X;
762 rotMatrix[1, 0] = xAxis.Y; rotMatrix[1, 1] = yAxis.Y; rotMatrix[1, 2] = zAxis.Y;
763 rotMatrix[2, 0] = xAxis.Z; rotMatrix[2, 1] = yAxis.Z; rotMatrix[2, 2] = zAxis.Z;
764
765 Quaternion rotation = new Quaternion(rotMatrix);
766 double zoomFactor = viewport.Camera.ZoomFactor;
767 if (zoomFactor < 1e-6) zoomFactor = 1.0;
768
769 viewport.SetView(rotation, targetPoint, camDist, zoomFactor);
770 _designControl.Invalidate();
771 }
772 catch (Exception ex)
773 {
774 Debug.WriteLine($"Error orienting camera to normal: {ex.Message}");

860 private bool TryResolveBrepFace(CADFace face, out Brep? brep, out Brep.Face? brepFace)
861 {
862 brep = null;
863 brepFace = null;
864 if (face == null || face.FaceIndex < 0 || _designControl?.Entities == null)
865 return false;
866
867 if (face.EntityIndex >= 0 &&
868 face.EntityIndex < _designControl.Entities.Count &&
869 _designControl.Entities[face.EntityIndex] is Brep directBrep &&
870 face.FaceIndex < directBrep.Faces.Length)
871 {
872 brep = directBrep;
873 brepFace = directBrep.Faces[face.FaceIndex];
874 return brepFace != null;
875 }
876
877 Point3D savedCenter = EyeshotGeometryUtils.GetFaceCenter(face);
878 double bestScore = double.MaxValue;
879
880 foreach (var entity in _designControl.Entities)
881 {
882 if (entity is not Brep candidate)
883 continue;
884
885 for (int i = 0; i < candidate.Faces.Length; i++)
886 {
887 var candidateFace = candidate.Faces[i];
888 var candidateCenter = EyeshotGeometryUtils.GetFaceCentroid(candidateFace);
889 if (candidateCenter == null)
890 continue;
891
892 double score = candidateCenter.DistanceTo(savedCenter);
893 if (!double.IsNaN(face.FaceArea))
894 {
895 double area = EyeshotGeometryUtils.GetFaceArea(candidateFace);
896 if (!double.IsNaN(area))
897 score += Math.Abs(area - face.FaceArea) * 0.01;
898 }
899
900 if (score < bestScore)
901 {
902 bestScore = score;
903 brep = candidate;
904 brepFace = candidateFace;
905 }
906 }
907 }
908
909 return brepFace != null && bestScore < Math.Max(5.0, GetPartRadius(brep) * 0.05);
910 }
