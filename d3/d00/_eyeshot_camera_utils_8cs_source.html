295 private Point3D CalculateMeshCenter(Mesh mesh)
296 {
297 if (mesh.Vertices == null || mesh.Vertices.Length == 0)
298 return new Point3D(0, 0, 0);
299
300 double sumX = 0, sumY = 0, sumZ = 0;
301 foreach (var vertex in mesh.Vertices)
302 {
303 sumX += vertex.X;
304 sumY += vertex.Y;
305 sumZ += vertex.Z;
306 }
307
308 int count = mesh.Vertices.Length;
309 return new Point3D(sumX / count, sumY / count, sumZ / count);

592 private void ApplyProbeOrbitalFollow(ProbeCameraPoseEventArgs pose)
593 {
594 try
595 {
596 if (!_isCameraFollowEnabled || pose == null)
597 return;
598
599 var viewport = _designControl.ActiveViewport;
600 if (viewport == null || viewport.Camera == null)
601 return;
602
603 var camera = viewport.Camera;
604 Point3D orbitCenter = Point3D.Origin;
605 double safeRadius = Math.Max(_followDistance, _boundingBoxDistance);
606 Vector3D desiredDirection = Vector3D.AxisZ;
607 double orbitRadius;
608 bool updateDirection = true;
609 bool isProbeFixation = string.Equals(_cameraFollowMode, "ProbeFixation", StringComparison.OrdinalIgnoreCase);
610
611 if (!TryGetProbeFollowOrbit(pose, out Point3D partCenter, out safeRadius, out CameraFollowPartInfo? partInfo))
612 return;
613
614 if (isProbeFixation)
615 {
616 orbitCenter = pose.ProbePosition;
617 desiredDirection = new Vector3D(partCenter, pose.ProbePosition);
618 if (desiredDirection.LengthSquared < 1e-6)
619 desiredDirection = GetCurrentCameraOrbitDirection(camera, orbitCenter);
620 if (desiredDirection.LengthSquared < 1e-6)
621 desiredDirection = Vector3D.AxisZ;
622 desiredDirection.Normalize();
623
624 // Tilt 30 degrees to avoid looking exactly straight down
625 Vector3D tiltAxis = Vector3D.Cross(desiredDirection, Vector3D.AxisZ);
626 if (tiltAxis.LengthSquared < 1e-6) tiltAxis = Vector3D.AxisX;
627 tiltAxis.Normalize();
628
629 // Create a rotation matrix for 30 degrees around the tilt axis
630 Transformation tiltRotation = Transformation.CreateRotation(Math.PI / 6.0, tiltAxis);
631 desiredDirection = tiltRotation * desiredDirection;
632 desiredDirection.Normalize();
633
634 safeRadius = Math.Max(20.0, _followDistance);
635 orbitRadius = Math.Max(10.0, _followDistance + _cameraOrbitalOffset);
636 updateDirection = false; // Preserve the existing free-rotation behavior for ProbeFixation.
637 }
638 else // SphericalOrbital
639 {
640 orbitCenter = partCenter;
641 orbitRadius = ResolveSphericalOrbitRadius(safeRadius);
642 desiredDirection = GetProbeForegroundOrbitDirection(camera, orbitCenter, pose);
643
644 if (TryGetHoleAxisBiasDirection(pose, desiredDirection, out Vector3D holeDirection))
645 _holeFollowBlend = Math.Min(1.0, _holeFollowBlend + 0.18 * _followOrbitSpeedMultiplier);
646 else
647 _holeFollowBlend = Math.Max(0.0, _holeFollowBlend - 0.06 * _followOrbitSpeedMultiplier);
648
649 if (_holeFollowBlend > 0.001)
650 {
651 double holeBias = SmoothStep(_holeFollowBlend) * 0.65;
652 desiredDirection = SlerpUnitVectors(desiredDirection, holeDirection, holeBias);
653 }
654
655 desiredDirection = ResolveForegroundSafeDirection(orbitCenter, pose, desiredDirection, orbitRadius, partInfo);
656 }
657
658 bool isFirstOrbitFrame = _followOrbitDirection == null;
659 Vector3D currentDirection = _followOrbitDirection ?? GetCurrentCameraOrbitDirection(camera, orbitCenter);
660 if (currentDirection.LengthSquared < 1e-6)
661 currentDirection = desiredDirection;
662 currentDirection.Normalize();
663
664 double dot = Math.Max(-1.0, Math.Min(1.0, Vector3D.Dot(currentDirection, desiredDirection)));
665 double angleError = Math.Acos(dot);
666 double alpha = GetSphericalFollowAlpha(angleError);
667
668 Vector3D nextDirection = isFirstOrbitFrame
669 ? desiredDirection
670 : SlerpUnitVectors(currentDirection, desiredDirection, alpha);
671
672 ApplyOrbitalCameraPose(viewport, orbitCenter, nextDirection, orbitRadius, updateDirection);
673
674 _followOrbitDirection = nextDirection;
675 _lastAppliedOrbitRadius = orbitRadius;
676 _lastAppliedOrbitZoomFactor = 1.0;
677 _lastProbePosition = new Point3D(pose.ProbePosition.X, pose.ProbePosition.Y, pose.ProbePosition.Z);
678 }
679 catch (Exception ex)
680 {
681 Debug.WriteLine($"Error in orbital probe camera follow: {ex.Message}");
682 }
683 }

690 private Vector3D GetProbeForegroundOrbitDirection(Camera camera, Point3D orbitCenter, ProbeCameraPoseEventArgs pose)
691 {
692 Vector3D radialDirection = new Vector3D(orbitCenter, pose.ProbePosition);
693 if (radialDirection.LengthSquared < 1e-6)
694 radialDirection = GetCurrentCameraOrbitDirection(camera, orbitCenter);
695 if (radialDirection.LengthSquared < 1e-6)
696 radialDirection = Vector3D.AxisZ;
697 radialDirection.Normalize();
698
699 Vector3D surfaceNormal = CopyVector(pose.SurfaceNormal);
700 if (surfaceNormal.LengthSquared < 1e-6 || pose.IsBridge)
701 return radialDirection;
702
703 surfaceNormal.Normalize();
704
705 if (Vector3D.Dot(surfaceNormal, radialDirection) < 0.0)
706 surfaceNormal = Negate(surfaceNormal);
707
708 double sameSideAlignment = Vector3D.Dot(surfaceNormal, radialDirection);
709 if (sameSideAlignment < 0.25)
710 return radialDirection;
711
712 Vector3D foregroundDirection = radialDirection + (surfaceNormal * 0.25);
713 if (foregroundDirection.LengthSquared < 1e-6)
714 foregroundDirection = radialDirection;
715 foregroundDirection.Normalize();
716 return foregroundDirection;
717 }

719 private Vector3D ResolveForegroundSafeDirection(
720 Point3D orbitCenter,
721 ProbeCameraPoseEventArgs pose,
722 Vector3D preferredDirection,
723 double orbitRadius,
724 CameraFollowPartInfo? partInfo)
725 {
726 if (preferredDirection.LengthSquared < 1e-6)
727 preferredDirection = Vector3D.AxisZ;
728 preferredDirection.Normalize();
729
730 Vector3D focusDirection = new Vector3D(orbitCenter, pose.ProbePosition);
731 if (focusDirection.LengthSquared < 1e-6)
732 focusDirection = preferredDirection;
733 focusDirection.Normalize();
734
735 // Keep the camera broadly on the centroid-to-probe beam. This prevents
736 // concave-face normals from dragging the camera toward the base side.
737 if (Vector3D.Dot(preferredDirection, focusDirection) < 0.35)
738 preferredDirection = SlerpUnitVectors(focusDirection, preferredDirection, 0.35);
739
740 if (partInfo == null)
741 return preferredDirection;
742
743 Point3D preferredCamera = orbitCenter + (preferredDirection * orbitRadius);
744 if (!IsProbeOccludedByPart(partInfo, preferredCamera, pose.ProbePosition))
745 return preferredDirection;
746
747 foreach (var candidate in BuildForegroundDirectionCandidates(focusDirection, preferredDirection, pose.SurfaceNormal))
748 {
749 Point3D candidateCamera = orbitCenter + (candidate * orbitRadius);
750 if (!IsProbeOccludedByPart(partInfo, candidateCamera, pose.ProbePosition))
751 return candidate;
752 }
753
754 return focusDirection;
755 }

757 private static List<Vector3D> BuildForegroundDirectionCandidates(Vector3D focusDirection, Vector3D preferredDirection, Vector3D surfaceNormal)
758 {
759 var candidates = new List<Vector3D>();
760 AddForegroundCandidate(candidates, preferredDirection, focusDirection);
761 AddForegroundCandidate(candidates, focusDirection, focusDirection);
762
763 Vector3D tangentA = Vector3D.Cross(focusDirection, Vector3D.AxisZ);
764 if (tangentA.LengthSquared < 1e-6)
765 tangentA = Vector3D.Cross(focusDirection, Vector3D.AxisY);
766 if (tangentA.LengthSquared < 1e-6)
767 tangentA = Vector3D.AxisX;
768 tangentA.Normalize();
769
770 Vector3D tangentB = Vector3D.Cross(focusDirection, tangentA);
771 if (tangentB.LengthSquared < 1e-6)
772 tangentB = Vector3D.AxisY;
773 tangentB.Normalize();
774
775 AddForegroundCandidate(candidates, focusDirection + (tangentA * 0.45), focusDirection);
776 AddForegroundCandidate(candidates, focusDirection - (tangentA * 0.45), focusDirection);
777 AddForegroundCandidate(candidates, focusDirection + (tangentB * 0.45), focusDirection);
778 AddForegroundCandidate(candidates, focusDirection - (tangentB * 0.45), focusDirection);
779 AddForegroundCandidate(candidates, preferredDirection + (tangentA * 0.30), focusDirection);
780 AddForegroundCandidate(candidates, preferredDirection - (tangentA * 0.30), focusDirection);
781 AddForegroundCandidate(candidates, preferredDirection + (tangentB * 0.30), focusDirection);
782 AddForegroundCandidate(candidates, preferredDirection - (tangentB * 0.30), focusDirection);
783
784 Vector3D normal = CopyVector(surfaceNormal);
785 if (normal.LengthSquared > 1e-6)
786 {
787 normal.Normalize();
788 if (Vector3D.Dot(normal, focusDirection) > 0.0)
789 AddForegroundCandidate(candidates, focusDirection + (normal * 0.25), focusDirection);
790 }
791
792 return candidates;
793 }

795 private static void AddForegroundCandidate(List<Vector3D> candidates, Vector3D candidate, Vector3D focusDirection)
796 {
797 if (candidate.LengthSquared < 1e-6)
798 return;

813 private bool IsProbeOccludedByPart(CameraFollowPartInfo partInfo, Point3D cameraLocation, Point3D probePosition)
814 {
815 double probeDistance = cameraLocation.DistanceTo(probePosition);
816 if (probeDistance <= 1e-6)
817 return false;
818
819 Vector3D rayDirection = new Vector3D(cameraLocation, probePosition);
820 rayDirection.Normalize();
821
822 double tolerance = Math.Max(1.0, Math.Min(8.0, partInfo.BoundingRadius * 0.02));
823 if (!SegmentIntersectsBoxBeforeProbe(cameraLocation, rayDirection, probeDistance, partInfo.BoxMin, partInfo.BoxMax, tolerance))
824 return false;
825
826 if (partInfo.OcclusionMesh != null &&
827 EyeshotGeometryUtils.IntersectWithMesh(partInfo.OcclusionMesh, cameraLocation, rayDirection, out Point3D hit))
828 {
829 double hitDistance = cameraLocation.DistanceTo(hit);
830 return hitDistance < probeDistance - tolerance && hit.DistanceTo(probePosition) > tolerance;

858 private static bool ClipRayToSlab(double origin, double direction, double min, double max, ref double tMin, ref double tMax)
859 {
860 if (Math.Abs(direction) < 1e-9)
861 return origin >= min && origin <= max;
862
863 double inv = 1.0 / direction;
864 double t1 = (min - origin) * inv;
865 double t2 = (max - origin) * inv;
866 if (t1 > t2)
867 {
868 double temp = t1;
869 t1 = t2;
870 t2 = temp;
871 }
872
873 tMin = Math.Max(tMin, t1);
874 tMax = Math.Min(tMax, t2);
875 return tMin <= tMax;
876 }
