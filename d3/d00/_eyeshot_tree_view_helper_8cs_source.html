308 public void UpdateMeasurementPointsSubNode(TreeViewItem faceNode, CADFace face)
309 {
310 if (faceNode == null || face == null) return;
311
312 TreeViewItem? measurePtsNode = null;
313 string mpLocalizedPrefix = GetLocalizedText("TreeView_MeasurementPoints");
314
315 foreach (var item in faceNode.Items)
316 {
317 if (item is TreeViewItem tvi)
318 {
319 if (Equals(tvi.Tag, "MeasurementPointsGroup"))
320 {
321 measurePtsNode = tvi;
322 break;
323 }
324 string? headerStr = tvi.Header as string ?? (tvi.Header as TextBlock)?.Text;
325 if (!string.IsNullOrEmpty(headerStr) && headerStr.StartsWith(mpLocalizedPrefix, StringComparison.OrdinalIgnoreCase))
326 {
327 measurePtsNode = tvi;
328 break;
329 }
330 }
331 }
332
333 bool hasPoints = face.MeasurementPoints != null && face.MeasurementPoints.Count > 0;
334
335 if (!hasPoints)
336 {
337 if (measurePtsNode != null)
338 {
339 faceNode.Items.Remove(measurePtsNode);
340 }
341 return;
342 }
343
344 bool wasExpanded = measurePtsNode?.IsExpanded ?? true;
345
346 if (measurePtsNode == null)
347 {
348 measurePtsNode = new TreeViewItem
349 {
350 Tag = "MeasurementPointsGroup",
351 HorizontalContentAlignment = HorizontalAlignment.Stretch
352 };
353 faceNode.Items.Add(measurePtsNode);
354 faceNode.IsExpanded = true;
355 }
356 else
357 {
358 measurePtsNode.Tag = "MeasurementPointsGroup";
359 measurePtsNode.Items.Clear();
360 }
361
362 string measurePointsHeader = face.HasManualMeasurementPoints
363 ? $"{mpLocalizedPrefix} ({face.MeasurementPoints!.Count}) [{GetLocalizedText("TreeView_Manual")}]"
364 : $"{mpLocalizedPrefix} ({face.MeasurementPoints!.Count})";
365
366 measurePtsNode.Header = measurePointsHeader;
367 measurePtsNode.IsExpanded = wasExpanded;
368
369 for (int i = 0; i < face.MeasurementPoints.Count; i++)
370 {
371 var mp = face.MeasurementPoints[i];
372 measurePtsNode.Items.Add(new TreeViewItem
373 {
374 Header = $"MP{i}: ({mp.X:F3}, {mp.Y:F3}, {mp.Z:F3})",
375 Tag = new CADMeasurementPoint(face, i, mp),
376 HorizontalContentAlignment = HorizontalAlignment.Stretch

690 private PathTreeViewNode? SearchPathNode(
691 PathTreeViewNode node,
692 devDept.Eyeshot.Entities.LinearPath? segmentEntity,
693 devDept.Geometry.Point3D? pointEntity,
694 bool expandedOnly)
695 {
696 // Check this node
697 if (segmentEntity != null && node.SegmentEntity == segmentEntity)
698 return node;
699
700 if (pointEntity != null && node.PointEntity != null)
701 {
702 if (Math.Abs(node.PointEntity.X - pointEntity.X) < 1e-6 &&
703 Math.Abs(node.PointEntity.Y - pointEntity.Y) < 1e-6 &&
704 Math.Abs(node.PointEntity.Z - pointEntity.Z) < 1e-6)
705 return node;
706 }
707
708 // Only recurse into children if we're not in expanded-only mode, or the node is expanded
709 if (expandedOnly && !node.IsExpanded && node.Children.Count > 0)
710 return null;
711
712 foreach (var child in node.Children)
713 {
714 var result = SearchPathNode(child, segmentEntity, pointEntity, expandedOnly);
715 if (result != null)
716 {
717 // Expand this parent so the found node is visible
718 node.IsExpanded = true;
719 return result;
720 }
721 }
722 return null;
723 }

809 private string LocalizeGeometricInfo(string? geometricInfo)
810 {
811 if (string.IsNullOrWhiteSpace(geometricInfo))
812 return string.Empty;
813
814 var info = geometricInfo.Trim();
815
816 if (string.Equals(info, "NURBS Surface", StringComparison.OrdinalIgnoreCase))
817 return GetLocalizedText("TreeView_GeoInfo_NurbsSurface");
818
819 if (string.Equals(info, "N/A", StringComparison.OrdinalIgnoreCase))
820 return GetLocalizedText("TreeView_GeoInfo_NotAvailable");
821
822 info = Regex.Replace(info, @"\bMajor\s+R:", $"{GetLocalizedText("TreeView_GeoInfo_MajorRadius")}:", RegexOptions.IgnoreCase);
823 info = Regex.Replace(info, @"\bMinor\s+R:", $"{GetLocalizedText("TreeView_GeoInfo_MinorRadius")}:", RegexOptions.IgnoreCase);
824 info = Regex.Replace(info, @"\bRadius:", $"{GetLocalizedText("TreeView_GeoInfo_Radius")}:", RegexOptions.IgnoreCase);
825 info = Regex.Replace(info, @"\bAxis:", $"{GetLocalizedText("TreeView_GeoInfo_Axis")}:", RegexOptions.IgnoreCase);
826 info = Regex.Replace(info, @"\bAngle:", $"{GetLocalizedText("TreeView_GeoInfo_Angle")}:", RegexOptions.IgnoreCase);
827 info = Regex.Replace(info, @"\bDirection:", $"{GetLocalizedText("TreeView_GeoInfo_Direction")}:", RegexOptions.IgnoreCase);
828 info = Regex.Replace(info, @"\bNormal:", $"{GetLocalizedText("TreeView_GeoInfo_Normal")}:", RegexOptions.IgnoreCase);
829
830 var inwardNormalText = GetLocalizedText("TreeView_GeoInfo_CylinderInwardNormalDot");
831 if (string.IsNullOrWhiteSpace(inwardNormalText))
832 inwardNormalText = GetLocalizedText("FaceProperties_CylinderInwardNormalDot");
833 if (!string.IsNullOrWhiteSpace(inwardNormalText))
834 {
835 if (!inwardNormalText.EndsWith(":")) inwardNormalText += ":";
836 info = Regex.Replace(info, @"\bCylinderInwardNormalDot:", inwardNormalText, RegexOptions.IgnoreCase);
837 }
838
839 return info;
840 }

848 public void PopulateTreeView(
849 Dictionary<string, List<CADFace>> facesByColor,
850 Dictionary<string, List<CADHole>> holesByColor,
851 Dictionary<string, List<CADFace>> allFaces,
852 bool showExteriorOnly = false,
853 bool showVisibleOnly = false)
854 {
855 // Ensure we're on the UI thread since we're creating WPF UI elements
856 if (!Application.Current.Dispatcher.CheckAccess())
857 {
858 Application.Current.Dispatcher.Invoke(() =>
859 PopulateTreeView(facesByColor, holesByColor, allFaces, showExteriorOnly, showVisibleOnly));
860 return;
861 }
862
863 _visibleEntityIndices.Clear();
864 if (mvm.MainParent?.designControl != null)
865 {
866 for (int i = 0; i < mvm.MainParent.designControl.Entities.Count; i++)
867 {
868 if (mvm.MainParent.designControl.Entities[i].Visible)
869 {
870 _visibleEntityIndices.Add(i);
871 }
872 }
873 }
874
875 // Phase 1: Build the complete hierarchical UI cache if it doesn't exist or data has changed.
876 if (_cachedHierarchy == null || _allFacesCached != allFaces)
877 {
878 BuildHierarchyCache(facesByColor, holesByColor, allFaces);
879 _allFacesCached = allFaces;
880 }
881
882 // Phase 2: Rapidly assemble the visible TreeView from the cache.
883 if (mvm.TreeViewItems != null)
884 mvm.TreeViewItems = new System.Collections.ObjectModel.ObservableCollection<TreeViewItem>();
885 if (_cachedHierarchy == null) return;
886
887 foreach (var hierarchy in _cachedHierarchy.Values)
888 {
889 // Detach all children from their group parents to make them available for re-adding.
890 hierarchy.FacesGroupNode.Items.Clear();
891 hierarchy.HolesGroupNode.Items.Clear();
892
893 // Filter the face nodes for this color.
894 var visibleFaceNodes = new List<TreeViewItem>();
895 foreach (var cachedNode in hierarchy.FaceNodes)
896 {
897 if (showExteriorOnly && !cachedNode.IsExterior) continue;

961 private void BuildHierarchyCache(
962 Dictionary<string, List<CADFace>> facesByColor,
963 Dictionary<string, List<CADHole>> holesByColor,
964 Dictionary<string, List<CADFace>> allFaces)
965 {
966 _cachedHierarchy = new Dictionary<string, CachedColorHierarchy>(StringComparer.OrdinalIgnoreCase);
967 _totalFaceCount = 0;
968 _totalHoleCount = 0;
969
970 var pathManager = mvm.GetPathManagerForMeasuringPanel();
971 var executionOrderMap = new Dictionary<string, int>(StringComparer.OrdinalIgnoreCase);
972 if (pathManager != null && pathManager.ExecutionOrder != null)
973 {
974 int index = 0;
975 foreach (var color in pathManager.ExecutionOrder)
976 {
977 string colorName = ColorNamer.GetColorName(color);
978 string colorKey = $"{colorName} (R{color.R}_G{color.G}_B{color.B})";
979 executionOrderMap[colorKey] = index++;
980 }
981 }
982
983 var allColorKeys = (facesByColor?.Keys ?? Enumerable.Empty<string>())
984 .Union(holesByColor?.Keys ?? Enumerable.Empty<string>())
985 .Distinct()
986 .OrderBy(k => executionOrderMap.TryGetValue(k, out int order) ? order : int.MaxValue)
987 .ThenBy(k => k);
988
989 var allFacesList = allFaces?.SelectMany(kvp => kvp.Value).ToList() ?? new List<CADFace>();
990
991 foreach (var colorKey in allColorKeys)
992 {
993 // Create the parent/group nodes for this color ONCE.
994 var colorNode = CreateColorNode(colorKey, 0, 0);
995 var facesGroupNode = new TreeViewItem { HorizontalContentAlignment = HorizontalAlignment.Stretch };
996 var holesGroupNode = new TreeViewItem { HorizontalContentAlignment = HorizontalAlignment.Stretch };
997 var hierarchy = new CachedColorHierarchy(colorNode, facesGroupNode, holesGroupNode);
998
999 // Process and cache all faces for this color.
1000 if (facesByColor != null && facesByColor.TryGetValue(colorKey, out var faces))
1001 {
1002 foreach (var face in faces)
1003 {
1004 var faceNode = CreateFaceNode(face);
1005 hierarchy.FaceNodes.Add(new CachedTreeViewNode(faceNode, face.EntityIndex, face.IsExterior));
1006 _totalFaceCount++;
1007 }
1008 }
1009
1010 // Process and cache all holes for this color.
1011 if (holesByColor != null && holesByColor.TryGetValue(colorKey, out var holes))
1012 {
1013 foreach (var hole in holes)
1014 {
1015 if (hole.Depth <= 1e-6) continue;
1016 int entityIdx = GetHoleEntityIndex(hole, allFacesList);
1017 bool isExterior = IsHoleExterior(hole, allFacesList);
1018 var holeNode = CreateHoleNode(hole);
1019 hierarchy.HoleNodes.Add(new CachedTreeViewNode(holeNode, entityIdx, isExterior));
1020 _totalHoleCount++;
1021 }
1022 }
1023
1024 // Add group nodes to the color node if they will ever contain children.
1025 if (hierarchy.FaceNodes.Any()) colorNode.Items.Add(facesGroupNode);
1026 if (hierarchy.HoleNodes.Any()) colorNode.Items.Add(holesGroupNode);
1027
1028 _cachedHierarchy[colorKey] = hierarchy;
1029 }
