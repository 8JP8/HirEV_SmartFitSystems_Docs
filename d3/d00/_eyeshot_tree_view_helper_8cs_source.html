531 private PathTreeViewNode? SearchPathNode(
532 PathTreeViewNode node,
533 devDept.Eyeshot.Entities.LinearPath? segmentEntity,
534 devDept.Geometry.Point3D? pointEntity,
535 bool expandedOnly)
536 {
537 // Check this node
538 if (segmentEntity != null && node.SegmentEntity == segmentEntity)
539 return node;
540
541 if (pointEntity != null && node.PointEntity != null)
542 {
543 if (Math.Abs(node.PointEntity.X - pointEntity.X) < 1e-6 &&
544 Math.Abs(node.PointEntity.Y - pointEntity.Y) < 1e-6 &&
545 Math.Abs(node.PointEntity.Z - pointEntity.Z) < 1e-6)
546 return node;
547 }
548
549 // Only recurse into children if we're not in expanded-only mode, or the node is expanded
550 if (expandedOnly && !node.IsExpanded && node.Children.Count > 0)
551 return null;
552
553 foreach (var child in node.Children)
554 {
555 var result = SearchPathNode(child, segmentEntity, pointEntity, expandedOnly);
556 if (result != null)
557 {
558 // Expand this parent so the found node is visible
559 node.IsExpanded = true;
560 return result;
561 }
562 }
563 return null;
564 }

615 private int GetHoleEntityIndex(CADHole hole, List<CADFace> allFacesList)
616 {
617 if (hole.FaceGlobalIndices == null || !hole.FaceGlobalIndices.Any()) return -1;
618 foreach (var faceGlobalIndex in hole.FaceGlobalIndices)
619 {
620 if (faceGlobalIndex < allFacesList.Count && allFacesList[faceGlobalIndex] != null)
621 return allFacesList[faceGlobalIndex].EntityIndex;
622 }
623 return -1;

632 private bool IsHoleExterior(CADHole hole, List<CADFace> allFacesList)
633 {
634 if (hole.FaceGlobalIndices == null || !hole.FaceGlobalIndices.Any()) return false;
635 foreach (var faceGlobalIndex in hole.FaceGlobalIndices)
636 {
637 if (faceGlobalIndex < allFacesList.Count && allFacesList[faceGlobalIndex] != null)
638 {
639 if (!allFacesList[faceGlobalIndex].IsExterior) return false;
640 }
641 }
642 return true;

650 private string LocalizeGeometricInfo(string? geometricInfo)
651 {
652 if (string.IsNullOrWhiteSpace(geometricInfo))
653 return string.Empty;
654
655 var info = geometricInfo.Trim();
656
657 if (string.Equals(info, "NURBS Surface", StringComparison.OrdinalIgnoreCase))
658 return GetLocalizedText("TreeView_GeoInfo_NurbsSurface");
659
660 if (string.Equals(info, "N/A", StringComparison.OrdinalIgnoreCase))
661 return GetLocalizedText("TreeView_GeoInfo_NotAvailable");
662
663 info = Regex.Replace(info, @"\bMajor\s+R:", $"{GetLocalizedText("TreeView_GeoInfo_MajorRadius")}:", RegexOptions.IgnoreCase);
664 info = Regex.Replace(info, @"\bMinor\s+R:", $"{GetLocalizedText("TreeView_GeoInfo_MinorRadius")}:", RegexOptions.IgnoreCase);
665 info = Regex.Replace(info, @"\bRadius:", $"{GetLocalizedText("TreeView_GeoInfo_Radius")}:", RegexOptions.IgnoreCase);
666 info = Regex.Replace(info, @"\bAxis:", $"{GetLocalizedText("TreeView_GeoInfo_Axis")}:", RegexOptions.IgnoreCase);
667 info = Regex.Replace(info, @"\bAngle:", $"{GetLocalizedText("TreeView_GeoInfo_Angle")}:", RegexOptions.IgnoreCase);
668 info = Regex.Replace(info, @"\bDirection:", $"{GetLocalizedText("TreeView_GeoInfo_Direction")}:", RegexOptions.IgnoreCase);
669 info = Regex.Replace(info, @"\bNormal:", $"{GetLocalizedText("TreeView_GeoInfo_Normal")}:", RegexOptions.IgnoreCase);
670
671 return info;

680 public void PopulateTreeView(
681 Dictionary<string, List<CADFace>> facesByColor,
682 Dictionary<string, List<CADHole>> holesByColor,
683 Dictionary<string, List<CADFace>> allFaces,
684 bool showExteriorOnly = false,
685 bool showVisibleOnly = false)
686 {
687 // Ensure we're on the UI thread since we're creating WPF UI elements
688 if (!Application.Current.Dispatcher.CheckAccess())
689 {
690 Application.Current.Dispatcher.Invoke(() =>
691 PopulateTreeView(facesByColor, holesByColor, allFaces, showExteriorOnly, showVisibleOnly));
692 return;
693 }
694
695 _visibleEntityIndices.Clear();
696 if (mvm.MainParent?.designControl != null)
697 {
698 for (int i = 0; i < mvm.MainParent.designControl.Entities.Count; i++)
699 {
700 if (mvm.MainParent.designControl.Entities[i].Visible)
701 {
702 _visibleEntityIndices.Add(i);
703 }
704 }
705 }
706
707 // Phase 1: Build the complete hierarchical UI cache if it doesn't exist or data has changed.
708 if (_cachedHierarchy == null || _allFacesCached != allFaces)
709 {
710 BuildHierarchyCache(facesByColor, holesByColor, allFaces);
711 _allFacesCached = allFaces;
712 }
713
714 // Phase 2: Rapidly assemble the visible TreeView from the cache.
715 if (mvm.TreeViewItems != null)
716 mvm.TreeViewItems = new System.Collections.ObjectModel.ObservableCollection<TreeViewItem>();
717 if (_cachedHierarchy == null) return;
718
719 foreach (var hierarchy in _cachedHierarchy.Values)
720 {
721 // Detach all children from their group parents to make them available for re-adding.
722 hierarchy.FacesGroupNode.Items.Clear();
723 hierarchy.HolesGroupNode.Items.Clear();
724
725 // Filter the face nodes for this color.
726 var visibleFaceNodes = new List<TreeViewItem>();
727 foreach (var cachedNode in hierarchy.FaceNodes)
728 {
729 if (showExteriorOnly && !cachedNode.IsExterior) continue;
730 if (showVisibleOnly && !_visibleEntityIndices.Contains(cachedNode.EntityIndex)) continue;
731
732 // FIXED: Dynamically apply strikethrough style based on current ignored state.
733 if (cachedNode.Node.Header is TextBlock tb && cachedNode.Node.Tag is CADFace face)
734 {
735 tb.TextDecorations = mvm.IgnoredFaces.Contains(face) ? TextDecorations.Strikethrough : null;
736 }
737
738 visibleFaceNodes.Add(cachedNode.Node);
739 }
740
741 // Filter the hole nodes for this color.
742 var visibleHoleNodes = new List<TreeViewItem>();
743 foreach (var cachedNode in hierarchy.HoleNodes)
744 {
745 if (showExteriorOnly && !cachedNode.IsExterior) continue;
746 if (showVisibleOnly && !_visibleEntityIndices.Contains(cachedNode.EntityIndex)) continue;
747
748 // FIXED: Dynamically apply strikethrough style based on current ignored state.
749 if (cachedNode.Node.Header is TextBlock tb && cachedNode.Node.Tag is CADHole hole)
750 {
751 tb.TextDecorations = mvm.IgnoredHoles.Contains(hole) ? TextDecorations.Strikethrough : null;
752 }
753
754 visibleHoleNodes.Add(cachedNode.Node);
755 }
756
757 // If any nodes for this color are visible, rebuild its part of the tree.
758 if (visibleFaceNodes.Any() || visibleHoleNodes.Any())
759 {
760 if (visibleFaceNodes.Any())
761 {
762 hierarchy.FacesGroupNode.Header = $"{GetLocalizedText("TreeView_FacesCapital")} ({visibleFaceNodes.Count})";
763 visibleFaceNodes.ForEach(node => hierarchy.FacesGroupNode.Items.Add(node));
764 }
765
766 if (visibleHoleNodes.Any())
767 {
768 hierarchy.HolesGroupNode.Header = $"{GetLocalizedText("TreeView_HolesCapital")} ({visibleHoleNodes.Count})";
769 visibleHoleNodes.ForEach(node => hierarchy.HolesGroupNode.Items.Add(node));
770 }
771
772 UpdateColorNodeHeader(hierarchy.ColorNode, visibleFaceNodes.Count, visibleHoleNodes.Count);
773 mvm.TreeViewItems?.Add(hierarchy.ColorNode);
774 }
775 }
776
777 // Update the status text with total counts
778 if (mvm != null)
779 {
780 int colorCount = mvm.TreeViewItems?.Count ?? 0;
781 var statusFormat = MainViewModel.GetXMLElementValue(mvm.DocLanguage, "Eyeshot_FacesTreeViewPopulated")
782 ?? "{0} colors, {1} faces, {2} holes";
783 mvm.EyeshotStatus_Text = string.Format(
784 statusFormat,
785 colorCount, _totalFaceCount, _totalHoleCount);
786 }

793 private void BuildHierarchyCache(
794 Dictionary<string, List<CADFace>> facesByColor,
795 Dictionary<string, List<CADHole>> holesByColor,
796 Dictionary<string, List<CADFace>> allFaces)
797 {
798 _cachedHierarchy = new Dictionary<string, CachedColorHierarchy>(StringComparer.OrdinalIgnoreCase);
799 _totalFaceCount = 0;
800 _totalHoleCount = 0;
801
802 var pathManager = mvm.GetPathManagerForMeasuringPanel();
803 var executionOrderMap = new Dictionary<string, int>(StringComparer.OrdinalIgnoreCase);
804 if (pathManager != null && pathManager.ExecutionOrder != null)
805 {
806 int index = 0;
807 foreach (var color in pathManager.ExecutionOrder)
808 {
809 string colorName = ColorNamer.GetColorName(color);
810 string colorKey = $"{colorName} (R{color.R}_G{color.G}_B{color.B})";
811 executionOrderMap[colorKey] = index++;
812 }
813 }
814
815 var allColorKeys = (facesByColor?.Keys ?? Enumerable.Empty<string>())
816 .Union(holesByColor?.Keys ?? Enumerable.Empty<string>())
817 .Distinct()
818 .OrderBy(k => executionOrderMap.TryGetValue(k, out int order) ? order : int.MaxValue)
819 .ThenBy(k => k);
820
821 var allFacesList = allFaces?.SelectMany(kvp => kvp.Value).ToList() ?? new List<CADFace>();
822
823 foreach (var colorKey in allColorKeys)
824 {
825 // Create the parent/group nodes for this color ONCE.
826 var colorNode = CreateColorNode(colorKey, 0, 0);
827 var facesGroupNode = new TreeViewItem { HorizontalContentAlignment = HorizontalAlignment.Stretch };
828 var holesGroupNode = new TreeViewItem { HorizontalContentAlignment = HorizontalAlignment.Stretch };
829 var hierarchy = new CachedColorHierarchy(colorNode, facesGroupNode, holesGroupNode);
830
831 // Process and cache all faces for this color.
832 if (facesByColor != null && facesByColor.TryGetValue(colorKey, out var faces))
833 {
834 foreach (var face in faces)
835 {
836 var faceNode = CreateFaceNode(face);
837 hierarchy.FaceNodes.Add(new CachedTreeViewNode(faceNode, face.EntityIndex, face.IsExterior));
838 _totalFaceCount++;
839 }
840 }
841
842 // Process and cache all holes for this color.
843 if (holesByColor != null && holesByColor.TryGetValue(colorKey, out var holes))
844 {
845 foreach (var hole in holes)
846 {
847 if (hole.Depth <= 1e-6) continue;
848 int entityIdx = GetHoleEntityIndex(hole, allFacesList);
849 bool isExterior = IsHoleExterior(hole, allFacesList);
850 var holeNode = CreateHoleNode(hole);
851 hierarchy.HoleNodes.Add(new CachedTreeViewNode(holeNode, entityIdx, isExterior));
852 _totalHoleCount++;
853 }
854 }
855
856 // Add group nodes to the color node if they will ever contain children.
857 if (hierarchy.FaceNodes.Any()) colorNode.Items.Add(facesGroupNode);
858 if (hierarchy.HoleNodes.Any()) colorNode.Items.Add(holesGroupNode);
859
860 _cachedHierarchy[colorKey] = hierarchy;

869 private void UpdateColorNodeHeader(TreeViewItem colorNode, int faceCount, int holeCount)
870 {
871 if (!(colorNode.Header is StackPanel headerPanel)) return;
872
873 // Clear existing content
874 headerPanel.Children.Clear();
875
876 var colorKey = colorNode.Tag.ToString();
877
878 // Create color rectangle with border
879 var colorBorder = new System.Windows.Controls.Border
880 {
881 Width = 12,
882 Height = 12,
883 Margin = new Thickness(0, 0, 4, 0),
884 VerticalAlignment = VerticalAlignment.Center,
885 BorderBrush = new SolidColorBrush(Colors.Black),
886 BorderThickness = new Thickness(0.5)
887 };
888
889 // Parse RGB and set color
890 try
891 {
892 var match = Regex.Match(colorKey!, @"R(\d+)_G(\d+)_B(\d+)");
893 if (match.Success)
894 {
895 byte r = byte.Parse(match.Groups[1].Value);
896 byte g = byte.Parse(match.Groups[2].Value);
897 byte b = byte.Parse(match.Groups[3].Value);
898 colorBorder.Background = new SolidColorBrush(Color.FromRgb(r, g, b));
899 }
900 else
901 {
902 colorBorder.Background = new SolidColorBrush(Colors.Gray); // Fallback
903 }
904 }
905 catch (Exception)
906 {
907 colorBorder.Background = new SolidColorBrush(Colors.Gray); // Fallback
908 }
909
910 // Create text block with color name and counts
911 var textBlock = new TextBlock
912 {
913 VerticalAlignment = VerticalAlignment.Center,
914 Text = $"{colorKey} ({faceCount} {GetLocalizedText("TreeView_Faces")}, {holeCount} {GetLocalizedText("TreeView_Holes")})"
915 };
916
917 // Add border and text to panel
918 headerPanel.Children.Add(colorBorder);
919 headerPanel.Children.Add(textBlock);
920 }

948 private TreeViewItem CreateFaceNode(CADFace face)

1020 Tag = mp,
