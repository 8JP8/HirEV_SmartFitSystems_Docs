506 private string LocalizeGeometricInfo(string? geometricInfo)
507 {
508 if (string.IsNullOrWhiteSpace(geometricInfo))
509 return string.Empty;
510
511 var info = geometricInfo.Trim();
512
513 if (string.Equals(info, "NURBS Surface", StringComparison.OrdinalIgnoreCase))
514 return GetLocalizedText("TreeView_GeoInfo_NurbsSurface");
515
516 if (string.Equals(info, "N/A", StringComparison.OrdinalIgnoreCase))
517 return GetLocalizedText("TreeView_GeoInfo_NotAvailable");
518
519 info = Regex.Replace(info, @"\bMajor\s+R:", $"{GetLocalizedText("TreeView_GeoInfo_MajorRadius")}:", RegexOptions.IgnoreCase);
520 info = Regex.Replace(info, @"\bMinor\s+R:", $"{GetLocalizedText("TreeView_GeoInfo_MinorRadius")}:", RegexOptions.IgnoreCase);
521 info = Regex.Replace(info, @"\bRadius:", $"{GetLocalizedText("TreeView_GeoInfo_Radius")}:", RegexOptions.IgnoreCase);
522 info = Regex.Replace(info, @"\bAxis:", $"{GetLocalizedText("TreeView_GeoInfo_Axis")}:", RegexOptions.IgnoreCase);
523 info = Regex.Replace(info, @"\bAngle:", $"{GetLocalizedText("TreeView_GeoInfo_Angle")}:", RegexOptions.IgnoreCase);
524 info = Regex.Replace(info, @"\bDirection:", $"{GetLocalizedText("TreeView_GeoInfo_Direction")}:", RegexOptions.IgnoreCase);
525 info = Regex.Replace(info, @"\bNormal:", $"{GetLocalizedText("TreeView_GeoInfo_Normal")}:", RegexOptions.IgnoreCase);
526
527 return info;
528 }

536 public void PopulateTreeView(
537 Dictionary<string, List<CADFace>> facesByColor,
538 Dictionary<string, List<CADHole>> holesByColor,
539 Dictionary<string, List<CADFace>> allFaces,
540 bool showExteriorOnly = false,
541 bool showVisibleOnly = false)
542 {
543 // Ensure we're on the UI thread since we're creating WPF UI elements
544 if (!Application.Current.Dispatcher.CheckAccess())
545 {
546 Application.Current.Dispatcher.Invoke(() =>
547 PopulateTreeView(facesByColor, holesByColor, allFaces, showExteriorOnly, showVisibleOnly));
548 return;
549 }
550
551 _visibleEntityIndices.Clear();
552 if (mvm.MainParent?.designControl != null)
553 {
554 for (int i = 0; i < mvm.MainParent.designControl.Entities.Count; i++)
555 {
556 if (mvm.MainParent.designControl.Entities[i].Visible)
557 {
558 _visibleEntityIndices.Add(i);
559 }
560 }
561 }
562
563 // Phase 1: Build the complete hierarchical UI cache if it doesn't exist or data has changed.
564 if (_cachedHierarchy == null || _allFacesCached != allFaces)
565 {
566 BuildHierarchyCache(facesByColor, holesByColor, allFaces);
567 _allFacesCached = allFaces;
568 }
569
570 // Phase 2: Rapidly assemble the visible TreeView from the cache.
571 mvm.TreeViewItems?.Clear();
572 if (_cachedHierarchy == null) return;
573
574 foreach (var hierarchy in _cachedHierarchy.Values)
575 {
576 // Detach all children from their group parents to make them available for re-adding.
577 hierarchy.FacesGroupNode.Items.Clear();
578 hierarchy.HolesGroupNode.Items.Clear();
579
580 // Filter the face nodes for this color.
581 var visibleFaceNodes = new List<TreeViewItem>();
582 foreach (var cachedNode in hierarchy.FaceNodes)
583 {
584 if (showExteriorOnly && !cachedNode.IsExterior) continue;
585 if (showVisibleOnly && !_visibleEntityIndices.Contains(cachedNode.EntityIndex)) continue;
586
587 // FIXED: Dynamically apply strikethrough style based on current ignored state.
588 if (cachedNode.Node.Header is TextBlock tb && cachedNode.Node.Tag is CADFace face)
589 {
590 tb.TextDecorations = mvm.IgnoredFaces.Contains(face) ? TextDecorations.Strikethrough : null;
591 }
592
593 visibleFaceNodes.Add(cachedNode.Node);
594 }
595
596 // Filter the hole nodes for this color.
597 var visibleHoleNodes = new List<TreeViewItem>();
598 foreach (var cachedNode in hierarchy.HoleNodes)
599 {
600 if (showExteriorOnly && !cachedNode.IsExterior) continue;
601 if (showVisibleOnly && !_visibleEntityIndices.Contains(cachedNode.EntityIndex)) continue;
602
603 // FIXED: Dynamically apply strikethrough style based on current ignored state.
604 if (cachedNode.Node.Header is TextBlock tb && cachedNode.Node.Tag is CADHole hole)
605 {
606 tb.TextDecorations = mvm.IgnoredHoles.Contains(hole) ? TextDecorations.Strikethrough : null;
607 }
608
609 visibleHoleNodes.Add(cachedNode.Node);
610 }
611
612 // If any nodes for this color are visible, rebuild its part of the tree.
613 if (visibleFaceNodes.Any() || visibleHoleNodes.Any())
614 {
615 if (visibleFaceNodes.Any())
616 {
617 hierarchy.FacesGroupNode.Header = $"{GetLocalizedText("TreeView_FacesCapital")} ({visibleFaceNodes.Count})";
618 visibleFaceNodes.ForEach(node => hierarchy.FacesGroupNode.Items.Add(node));
619 }
620
621 if (visibleHoleNodes.Any())
622 {
623 hierarchy.HolesGroupNode.Header = $"{GetLocalizedText("TreeView_HolesCapital")} ({visibleHoleNodes.Count})";
624 visibleHoleNodes.ForEach(node => hierarchy.HolesGroupNode.Items.Add(node));
625 }
626
627 UpdateColorNodeHeader(hierarchy.ColorNode, visibleFaceNodes.Count, visibleHoleNodes.Count);
628 mvm.TreeViewItems?.Add(hierarchy.ColorNode);
629 }
630 }
631
632 // Update the status text with total counts
633 if (mvm != null)
634 {
635 int colorCount = mvm.TreeViewItems?.Count ?? 0;
636 var statusFormat = MainViewModel.GetXMLElementValue(mvm.DocLanguage, "Eyeshot_FacesTreeViewPopulated")
637 ?? "{0} colors, {1} faces, {2} holes";
638 mvm.EyeshotStatus_Text = string.Format(
639 statusFormat,
640 colorCount, _totalFaceCount, _totalHoleCount);
641 }

648 private void BuildHierarchyCache(
649 Dictionary<string, List<CADFace>> facesByColor,
650 Dictionary<string, List<CADHole>> holesByColor,
651 Dictionary<string, List<CADFace>> allFaces)
652 {
653 _cachedHierarchy = new Dictionary<string, CachedColorHierarchy>(StringComparer.OrdinalIgnoreCase);
654 _totalFaceCount = 0;
655 _totalHoleCount = 0;
656 var allColorKeys = (facesByColor?.Keys ?? Enumerable.Empty<string>())
657 .Union(holesByColor?.Keys ?? Enumerable.Empty<string>()).Distinct().OrderBy(k => k);
658
659 var allFacesList = allFaces?.SelectMany(kvp => kvp.Value).ToList() ?? new List<CADFace>();
660
661 foreach (var colorKey in allColorKeys)
662 {
663 // Create the parent/group nodes for this color ONCE.
664 var colorNode = CreateColorNode(colorKey, 0, 0);
665 var facesGroupNode = new TreeViewItem { HorizontalContentAlignment = HorizontalAlignment.Stretch };
666 var holesGroupNode = new TreeViewItem { HorizontalContentAlignment = HorizontalAlignment.Stretch };
667 var hierarchy = new CachedColorHierarchy(colorNode, facesGroupNode, holesGroupNode);
668
669 // Process and cache all faces for this color.
670 if (facesByColor != null && facesByColor.TryGetValue(colorKey, out var faces))
671 {
672 foreach (var face in faces)
673 {
674 var faceNode = CreateFaceNode(face);
675 hierarchy.FaceNodes.Add(new CachedTreeViewNode(faceNode, face.EntityIndex, face.IsExterior));
676 _totalFaceCount++;
677 }
678 }
679
680 // Process and cache all holes for this color.
681 if (holesByColor != null && holesByColor.TryGetValue(colorKey, out var holes))
682 {
683 foreach (var hole in holes)
684 {
685 if (hole.Depth <= 1e-6) continue;
686 int entityIdx = GetHoleEntityIndex(hole, allFacesList);
687 bool isExterior = IsHoleExterior(hole, allFacesList);
688 var holeNode = CreateHoleNode(hole);
689 hierarchy.HoleNodes.Add(new CachedTreeViewNode(holeNode, entityIdx, isExterior));
690 _totalHoleCount++;
691 }
692 }
693
694 // Add group nodes to the color node if they will ever contain children.
695 if (hierarchy.FaceNodes.Any()) colorNode.Items.Add(facesGroupNode);
696 if (hierarchy.HoleNodes.Any()) colorNode.Items.Add(holesGroupNode);
697
698 _cachedHierarchy[colorKey] = hierarchy;
699 }

707 private void UpdateColorNodeHeader(TreeViewItem colorNode, int faceCount, int holeCount)
708 {
709 if (!(colorNode.Header is StackPanel headerPanel)) return;
710
711 // Clear existing content
712 headerPanel.Children.Clear();
713
714 var colorKey = colorNode.Tag.ToString();
715
716 // Create color rectangle with border
717 var colorBorder = new System.Windows.Controls.Border
718 {
719 Width = 12,
720 Height = 12,
721 Margin = new Thickness(0, 0, 4, 0),
722 VerticalAlignment = VerticalAlignment.Center,
723 BorderBrush = new SolidColorBrush(Colors.Black),
724 BorderThickness = new Thickness(0.5)
725 };
726
727 // Parse RGB and set color
728 try
729 {
730 var match = Regex.Match(colorKey!, @"R(\d+)_G(\d+)_B(\d+)");
731 if (match.Success)
732 {
733 byte r = byte.Parse(match.Groups[1].Value);
734 byte g = byte.Parse(match.Groups[2].Value);
735 byte b = byte.Parse(match.Groups[3].Value);
736 colorBorder.Background = new SolidColorBrush(Color.FromRgb(r, g, b));
737 }
738 else
739 {
740 colorBorder.Background = new SolidColorBrush(Colors.Gray); // Fallback
741 }
742 }
743 catch (Exception)
744 {
745 colorBorder.Background = new SolidColorBrush(Colors.Gray); // Fallback
746 }
747
748 // Create text block with color name and counts
749 var textBlock = new TextBlock
750 {
751 VerticalAlignment = VerticalAlignment.Center,
752 Text = $"{colorKey} ({faceCount} {GetLocalizedText("TreeView_Faces")}, {holeCount} {GetLocalizedText("TreeView_Holes")})"
753 };
754
755 // Add border and text to panel
756 headerPanel.Children.Add(colorBorder);
757 headerPanel.Children.Add(textBlock);
758 }

786 private TreeViewItem CreateFaceNode(CADFace face)

790
791 string faceHeaderStr = $"{GetLocalizedText("TreeView_Face")} {face.FaceGlobalIndex} ({localizedFaceType})";

795 }

801 Tag = face,
802 HorizontalContentAlignment = HorizontalAlignment.Stretch
803 };
804
805 if (!string.IsNullOrEmpty(face.GeometricInfo))

808 }

811 foreach (var p in face.KeyPoints.OrderBy(pt => pt.PointType))
812 {

814 pointsNode.Items.Add(new TreeViewItem

817 Tag = p,
818 HorizontalContentAlignment = HorizontalAlignment.Stretch
819 });
820 }
821 faceNode.Items.Add(pointsNode);
822
823 if (face.ControlPoints != null && face.ControlPoints.Any())
824 {
825 var ctrlPtsNode = new TreeViewItem { Header = $"{GetLocalizedText("TreeView_ControlPoints")} ({face.ControlPoints.Count})", HorizontalContentAlignment = HorizontalAlignment.Stretch };

827 {

838

845

848 Header = measurePointsHeader,
849 HorizontalContentAlignment = HorizontalAlignment.Stretch
850 };
851

856 {
857 Header = $"MP{i}: ({mp.X:F3}, {mp.Y:F3}, {mp.Z:F3})",
858 Tag = mp,
859 HorizontalContentAlignment = HorizontalAlignment.Stretch
860 });
861 }

863 }

871 private TreeViewItem CreateHoleNode(CADHole hole)
872 {
