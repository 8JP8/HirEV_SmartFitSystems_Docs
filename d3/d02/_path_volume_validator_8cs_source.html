49 public bool IsInsidePartVolume(Point3D point)
50 {
51 double sdf = _shellGenerator.GetDistance(point);
52 if (sdf != double.MaxValue)
53 return sdf <= 0.0;
54
55 // The SDF does not cover this point, so fall back to the source BReps.
56 foreach (var brep in _partObstacles)
57 {
58 if (brep == null) continue;
59 try
60 {
61 var status = brep.IsPointInside(point);
62 if (status != pointStatusType.Outside)
63 return true;
64 }
65 catch { /* ignore single-brep failures */ }
66 }
67
68 return false;
69 }

81 public int ValidateAndCorrectApproachPoints(List<ApproachMeasurementPair> pairs, double tolerance = 1.5)
82 {
83 int corrected = 0;
84 double halfOffset = _shellGenerator.OffsetDistance / 2.0;
85
86 foreach (var pair in pairs)
87 {
88 if (pair == null) continue;
89
90 double sdf = _shellGenerator.GetDistance(pair.ApproachPoint);
91
92 // Point is acceptable when SDF >= halfOffset (clearly outside the inner half of the shell wall)
93 bool isViolating = sdf == double.MaxValue // outside SDF grid – treat as unknown, be conservative
94 ? false // unknown → leave as-is (avoid false corrections)
95 : sdf < halfOffset; // too close to / inside the shell
96
97 if (!isViolating) continue;
98
99 // An approach point is part of a measurement pair, not a free shell
100 // waypoint. Projecting it to the nearest shell point can move it
101 // sideways and turns the final approach into a diagonal surface hit.
102 // Keep the point on its measurement-to-approach normal line instead.
103 var outDir = new Vector3D(pair.Normal.X, pair.Normal.Y, pair.Normal.Z);
104 if (outDir.Length <= 1e-6)
105 outDir = new Vector3D(pair.MeasurementPoint, pair.ApproachPoint);
106
107 if (outDir.Length <= 1e-6)
108 {
109 pair.ApproachPoint = new Point3D(
110 pair.MeasurementPoint.X,
111 pair.MeasurementPoint.Y,
112 pair.MeasurementPoint.Z);
113 continue;
114 }
115
116 outDir.Normalize();
117 double offset = Math.Max(
118 pair.MeasurementPoint.DistanceTo(pair.ApproachPoint),
119 Math.Max(0.5, _shellGenerator.OffsetDistance));
120 Point3D correctedPoint = pair.MeasurementPoint + outDir * offset;
121 bool cleared = false;
122
123 for (int attempt = 0; attempt < 6; attempt++)
124 {
125 double candidateSdf = _shellGenerator.GetDistance(correctedPoint);
126 if (candidateSdf == double.MaxValue || candidateSdf >= halfOffset)
127 {
128 cleared = true;
129 break;
130 }
131
132 double deficit = Math.Max(0.5, halfOffset - candidateSdf + 0.5);
133 offset += deficit;
134 correctedPoint = pair.MeasurementPoint + outDir * offset;
135 }
136
137 if (cleared)
138 {
139 Debug.WriteLine(string.Format(
140 GetXMLElement("PVV_CorrectedApproach", " ⚠ PVV: corrected approach {0:F1} -> {1:F1} (sdf={2:F2} < halfOffset={3:F2}) (Face ID: {4})"),
141 pair.ApproachPoint,
142 correctedPoint,
143 sdf,
144 halfOffset,
145 pair.ParentFace.FaceGlobalIndex));
146 pair.ApproachPoint = correctedPoint;
147 corrected++;
148 }
149 else
150 {
151 // Let the exclusive-pair filter discard this target rather than
152 // substituting a diagonal or surface-level navigation point.
153 pair.ApproachPoint = new Point3D(
154 pair.MeasurementPoint.X,
155 pair.MeasurementPoint.Y,
156 pair.MeasurementPoint.Z);
157 }
158 }
159
160 return corrected;
161 }

164 public int ValidateAndCorrectPath(List<Point3D> pathPoints, List<bool> measurementMask, double tolerance = 1.5)
165 {
166 int corrected = 0;
167 for (int i = 0; i < pathPoints.Count; i++)
168 {
169 // Only correct non-measurement (transition / approach) points
170 if (measurementMask != null && i < measurementMask.Count && measurementMask[i])
171 continue;
172
173 if (IsInsidePartVolume(pathPoints[i]))
174 {
175 var pushed = PushToShellSurface(pathPoints[i], tolerance);
176 if (pushed != null)
177 {
178 pathPoints[i] = pushed;
179 corrected++;
180 }
181 }
182 }
183 return corrected;
184 }

203 public bool ValidateSegmentsClearOfVolume(List<Point3D> pathPoints, int samplesPerSegment = 5)
204 {
205 for (int i = 0; i < pathPoints.Count - 1; i++)
206 {
207 var a = pathPoints[i];
208 var b = pathPoints[i + 1];
209 for (int s = 1; s < samplesPerSegment; s++)
210 {
211 double t = (double)s / samplesPerSegment;
212 var sample = new Point3D(
213 a.X + (b.X - a.X) * t,
214 a.Y + (b.Y - a.Y) * t,
215 a.Z + (b.Z - a.Z) * t);
216 if (IsInsidePartVolume(sample))
217 return false;
218 }
219 }
220 return true;
221 }
