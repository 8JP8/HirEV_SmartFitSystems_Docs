33 public bool IsInsidePartVolume(Point3D point)
34 {
35 // Primary: use Brep.IsPointInside (authoritative)
36 foreach (var brep in _partObstacles)
37 {
38 if (brep == null) continue;
39 try
40 {
41 var status = brep.IsPointInside(point);
42 if (status != pointStatusType.Outside)
43 return true;
44 }
45 catch { /* ignore single-brep failures */ }
46 }
47
48 // Secondary: use SDF sign (negative == inside part, before offset)
49 double sdf = _shellGenerator.GetDistance(point);
50 return sdf < 0 && sdf != double.MaxValue;
51 }

63 public int ValidateAndCorrectApproachPoints(List<ApproachMeasurementPair> pairs, double tolerance = 1.5)
64 {
65 int corrected = 0;
66 double halfOffset = _shellGenerator.OffsetDistance / 2.0;
67
68 foreach (var pair in pairs)
69 {
70 if (pair == null) continue;
71
72 double sdf = _shellGenerator.GetDistance(pair.ApproachPoint);
73
74 // Point is acceptable when SDF >= halfOffset (clearly outside the inner half of the shell wall)
75 bool isViolating = sdf == double.MaxValue // outside SDF grid – treat as unknown, be conservative
76 ? false // unknown → leave as-is (avoid false corrections)
77 : sdf < halfOffset; // too close to / inside the shell
78
79 if (!isViolating) continue;
80
81 // Attempt 1: project to nearest shell surface point
82 var nearest = _shellGenerator.FindNearestShellPoint(pair.ApproachPoint);
83 if (nearest != null)
84 {
85 Debug.WriteLine($" 🔧 PVV: corrected approach {pair.ApproachPoint:F1} → {nearest:F1} (sdf={sdf:F2} < halfOffset={halfOffset:F2}) (Face ID: {pair.ParentFace.FaceGlobalIndex})");
86 pair.ApproachPoint = nearest;
87 corrected++;
88 continue;
89 }
90
91 // Attempt 2: push outward along the measurement→approach normal by the deficit
92 double deficit = halfOffset - sdf; // how much further we need to go
93 var outDir = new Vector3D(pair.ApproachPoint.X - pair.MeasurementPoint.X,
94 pair.ApproachPoint.Y - pair.MeasurementPoint.Y,
95 pair.ApproachPoint.Z - pair.MeasurementPoint.Z);
96 if (outDir.Length > 1e-6)
97 {
98 outDir.Normalize();
99 pair.ApproachPoint = pair.ApproachPoint + outDir * (deficit + 0.5);
100 Debug.WriteLine($" 🔧 PVV: pushed approach along normal by {deficit + 0.5:F2}mm (sdf={sdf:F2}) (Face ID: {pair.ParentFace.FaceGlobalIndex})");
101 corrected++;
102 }
103 }
104
105 return corrected;
106 }

109 public int ValidateAndCorrectPath(List<Point3D> pathPoints, List<bool> measurementMask, double tolerance = 1.5)
110 {
111 int corrected = 0;
112 for (int i = 0; i < pathPoints.Count; i++)
113 {
114 // Only correct non-measurement (transition / approach) points
115 if (measurementMask != null && i < measurementMask.Count && measurementMask[i])
116 continue;
117
118 if (IsInsidePartVolume(pathPoints[i]))
119 {
120 var pushed = PushToShellSurface(pathPoints[i], tolerance);
121 if (pushed != null)
122 {
123 pathPoints[i] = pushed;
124 corrected++;
125 }
126 }
127 }
128 return corrected;
129 }

148 public bool ValidateSegmentsClearOfVolume(List<Point3D> pathPoints, int samplesPerSegment = 5)
149 {
150 for (int i = 0; i < pathPoints.Count - 1; i++)
151 {
152 var a = pathPoints[i];
153 var b = pathPoints[i + 1];
154 for (int s = 1; s < samplesPerSegment; s++)
155 {
156 double t = (double)s / samplesPerSegment;
157 var sample = new Point3D(
158 a.X + (b.X - a.X) * t,
159 a.Y + (b.Y - a.Y) * t,
160 a.Z + (b.Z - a.Z) * t);
161 if (IsInsidePartVolume(sample))
162 return false;
163 }
164 }
165 return true;
166 }
