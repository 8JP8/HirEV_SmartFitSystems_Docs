52 public static (List<Point3D> filteredControlPoints, List<Point3D> auxiliaryGridPoints) GenerateGridPointsForFace(
53 Brep brep,
54 Brep.Face face,
55 int faceIndex,
56 List<Point4D> controlPoints,
57 GridConfig config,
58 System.Threading.CancellationToken cancellationToken = default)
59 {
60 var filteredControlPoints = new List<Point3D>();
61 var auxiliaryGridPoints = new List<Point3D>();
62
63 if (!config.EnableGridGeneration)
64 {
65 // If grid generation disabled, just return control points as-is
66 if (config.UseControlPoints && controlPoints != null)
67 {
68 filteredControlPoints = controlPoints.Select(cp => new Point3D(cp.X, cp.Y, cp.Z)).ToList();
69 }
70 return (filteredControlPoints, auxiliaryGridPoints);
71 }
72
73 try
74 {
75 // Step 1: Process control points
76 if (config.UseControlPoints && controlPoints != null && controlPoints.Any())
77 {
78 var cpPoints = controlPoints.Select(cp => new Point3D(cp.X, cp.Y, cp.Z)).ToList();
79
80 // Deduplicate if enabled
81 if (config.DeduplicateControlPoints)
82 {
83 var dedupedPoints = new List<Point3D>();
84 double minDistSq = Math.Pow(config.GridSpacing * 0.3, 2);
85
86 foreach (var cp in cpPoints)
87 {
88 cancellationToken.ThrowIfCancellationRequested();
89 // Validate point is on face
90 if (!EyeshotGeometryUtils.IsPointOnFaceSurface(cp, brep, faceIndex, config.BoundaryTolerance))
91 continue;
92
93 // Check distance to already added points
94 bool tooClose = dedupedPoints.Any(existing =>
95 {
96 double dx = cp.X - existing.X;
97 double dy = cp.Y - existing.Y;
98 double dz = cp.Z - existing.Z;
99 return (dx * dx + dy * dy + dz * dz) < minDistSq;
100 });
101
102 if (!tooClose)
103 {
104 dedupedPoints.Add(cp);
105 }
106 }
107 filteredControlPoints = dedupedPoints;
108 }
109 else
110 {
111 // Just validate points are on face
112 filteredControlPoints = cpPoints
113 .Where(cp => EyeshotGeometryUtils.IsPointOnFaceSurface(cp, brep, faceIndex, config.BoundaryTolerance))
114 .ToList();
115 }
116 }
117
118 // Step 2: Generate auxiliary grid by directly sampling mesh vertices
119 // Then densifying by interpolating between vertices
120 cancellationToken.ThrowIfCancellationRequested();
121 var mesh = face.ConvertToMesh();
122 if (mesh == null || mesh.Vertices == null || mesh.Vertices.Length == 0)
123 {
124 return (filteredControlPoints, auxiliaryGridPoints);
125 }
126
127 // Get mesh centroid and size
128 Point3D meshMin = mesh.BoxMin;
129 Point3D meshMax = mesh.BoxMax;
130 double faceSize = Math.Max(
131 Math.Max(meshMax.X - meshMin.X, meshMax.Y - meshMin.Y),
132 meshMax.Z - meshMin.Z);
133
134 if (faceSize < 1e-6)
135 {
136 return (filteredControlPoints, auxiliaryGridPoints);
137 }
138
139 // Calculate grid density
140 int gridDivisions = Math.Max(5, (int)Math.Ceiling(faceSize / config.GridSpacing));
141 gridDivisions = Math.Min(gridDivisions, 30); // Cap to prevent excessive computation
142
143 // Filter parameters
144 double minCpDist = config.GridSpacing * config.ControlPointClearance;
145 double minGridDistSq = Math.Pow(config.GridSpacing * 0.6, 2); // Slightly looser for denser coverage
146 int pointsAdded = 0;
147
148 // Sample points from mesh triangles
149 // For each triangle in the mesh, generate points on its surface
150 if (mesh.Triangles != null && mesh.Triangles.Length > 0)
151 {
152 int triangleStride = Math.Max(1, mesh.Triangles.Length / (gridDivisions * gridDivisions));
153
154 for (int t = 0; t < mesh.Triangles.Length; t += triangleStride)
155 {
156 cancellationToken.ThrowIfCancellationRequested();
157 var triangle = mesh.Triangles[t];
158
159 // Get triangle vertices
160 Point3D v0 = mesh.Vertices[triangle.V1];
161 Point3D v1 = mesh.Vertices[triangle.V2];
162 Point3D v2 = mesh.Vertices[triangle.V3];
163
164 // Calculate triangle size
165 double triSize = Math.Max(
166 Math.Max(v0.DistanceTo(v1), v1.DistanceTo(v2)),
167 v2.DistanceTo(v0));
168
169 // Number of points to sample on this triangle
170 int samplesPerSide = Math.Max(1, (int)(triSize / config.GridSpacing));
171 samplesPerSide = Math.Min(samplesPerSide, 5); // Cap per triangle
172
173 // Sample points using barycentric coordinates
174 for (int i = 0; i <= samplesPerSide; i++)
175 {
176 for (int j = 0; j <= (samplesPerSide - i); j++)
177 {
178 double u = (double)i / samplesPerSide;
179 double v = (double)j / samplesPerSide;
180 double w = 1.0 - u - v;
181
182 if (w < 0) continue; // Outside triangle
183
184 // Barycentric interpolation
185 Point3D samplePoint = new Point3D(
186 v0.X * w + v1.X * u + v2.X * v,
187 v0.Y * w + v1.Y * u + v2.Y * v,
188 v0.Z * w + v1.Z * u + v2.Z * v
189 );
190
191 // Validate point is on face
192 if (!EyeshotGeometryUtils.IsPointOnFaceSurface(samplePoint, brep, faceIndex, config.BoundaryTolerance))
193 continue;
194
195 // Check clearance from control points
196 if (filteredControlPoints.Any(cp => samplePoint.DistanceTo(cp) < minCpDist))
197 continue;
198
199 // Check spacing from other grid points
200 bool tooClose = auxiliaryGridPoints.Any(gp =>
201 {
202 double dx = samplePoint.X - gp.X;
203 double dy = samplePoint.Y - gp.Y;
204 double dz = samplePoint.Z - gp.Z;
205 return (dx * dx + dy * dy + dz * dz) < minGridDistSq;
206 });
207
208 if (tooClose)
209 continue;
210
211 auxiliaryGridPoints.Add(samplePoint);
212 pointsAdded++;
213
214 if (auxiliaryGridPoints.Count >= config.MaxGridPointsPerFace)
215 {
216 System.Diagnostics.Debug.WriteLine(
217 $" Grid: Triangle sampling → {pointsAdded} points added (hit max)");
218 return (filteredControlPoints, auxiliaryGridPoints);
219 }
220 }
221 }
222 }
223 }
224
225 System.Diagnostics.Debug.WriteLine(
226 $" Grid: Triangle sampling → {pointsAdded} points added " +
227 $"(CP: {filteredControlPoints.Count}, Aux: {auxiliaryGridPoints.Count})");
228 }
229 catch (Exception ex)
230 {
231 System.Diagnostics.Debug.WriteLine($" ⚠️ Grid generation error: {ex.Message}");
232 }
233
234 return (filteredControlPoints, auxiliaryGridPoints);
235 }
236
237 }
