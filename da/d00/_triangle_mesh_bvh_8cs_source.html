113 private static int BuildTree(BvhTriangle[] triangles, int start, int count, List<BvhNode> nodes)
114 {
115 if (count == 0)
116 return -1;
117
118 double minX = double.MaxValue, minY = double.MaxValue, minZ = double.MaxValue;
119 double maxX = double.MinValue, maxY = double.MinValue, maxZ = double.MinValue;
120
121 for (int i = start; i < start + count; i++)
122 {
123 var t = triangles[i];
124 if (t.Min.X < minX) minX = t.Min.X;
125 if (t.Min.Y < minY) minY = t.Min.Y;
126 if (t.Min.Z < minZ) minZ = t.Min.Z;
127 if (t.Max.X > maxX) maxX = t.Max.X;
128 if (t.Max.Y > maxY) maxY = t.Max.Y;
129 if (t.Max.Z > maxZ) maxZ = t.Max.Z;
130 }
131
132 int nodeIndex = nodes.Count;
133 var node = new BvhNode
134 {
135 BoxMin = new Point3D(minX, minY, minZ),
136 BoxMax = new Point3D(maxX, maxY, maxZ),
137 TriangleStart = start,
138 TriangleCount = count,
139 LeftChild = -1,
140 RightChild = -1
141 };
142 nodes.Add(node);
143
144 const int maxLeafTriangles = 8;
145 if (count <= maxLeafTriangles)
146 {
147 return nodeIndex;
148 }
149
150 // Choose split axis along largest extent
151 double extentX = maxX - minX;
152 double extentY = maxY - minY;
153 double extentZ = maxZ - minZ;
154
155 int axis = 0; // 0=X, 1=Y, 2=Z
156 if (extentY > extentX && extentY > extentZ) axis = 1;
157 else if (extentZ > extentX && extentZ > extentY) axis = 2;
158
159 double splitPos = axis switch
160 {
161 0 => (minX + maxX) * 0.5,
162 1 => (minY + maxY) * 0.5,
163 _ => (minZ + maxZ) * 0.5
164 };
165
166 // Partition triangles around splitPos
167 int mid = start;
168 for (int i = start; i < start + count; i++)
169 {
170 var t = triangles[i];
171 double center = axis switch
172 {
173 0 => (t.Min.X + t.Max.X) * 0.5,
174 1 => (t.Min.Y + t.Max.Y) * 0.5,
175 _ => (t.Min.Z + t.Max.Z) * 0.5
176 };
177
178 if (center < splitPos)
179 {
180 if (i != mid)
181 {
182 var tmp = triangles[i];
183 triangles[i] = triangles[mid];
184 triangles[mid] = tmp;
185 }
186 mid++;
187 }
188 }
189
190 // Fallback to equal split if partitioning fails to separate
191 if (mid == start || mid == start + count)
192 {
193 mid = start + count / 2;
194 }
195
196 int leftChild = BuildTree(triangles, start, mid - start, nodes);
197 int rightChild = BuildTree(triangles, mid, start + count - mid, nodes);
198
199 node.LeftChild = leftChild;
200 node.RightChild = rightChild;
201 nodes[nodeIndex] = node;
202
203 return nodeIndex;
204 }

218 public bool IntersectsMesh(Mesh toolMesh, Transformation transform, Point3D toolWorldMin, Point3D toolWorldMax)
219 {
220 if (_rootIndex < 0 || toolMesh?.Vertices == null || toolMesh.Triangles == null || toolMesh.Triangles.Length == 0)
221 return false;
222
223 if (!BoxesOverlap(toolWorldMin, toolWorldMax, BoxMin, BoxMax))
224 return false;
225
226 var vertices = toolMesh.Vertices;
227 var triangles = toolMesh.Triangles;
228
229 // Stack-based traversal to query candidate triangles from BVH against tool mesh
230 var stack = new Stack<int>(32);
231
232 for (int i = 0; i < triangles.Length; i++)
233 {
234 var tri = triangles[i];
235 if (tri.V1 >= vertices.Length || tri.V2 >= vertices.Length || tri.V3 >= vertices.Length)
236 continue;
237
238 Point3D t0 = transform * vertices[tri.V1];
239 Point3D t1 = transform * vertices[tri.V2];
240 Point3D t2 = transform * vertices[tri.V3];
241
242 var tMin = new Point3D(
243 Math.Min(t0.X, Math.Min(t1.X, t2.X)),
244 Math.Min(t0.Y, Math.Min(t1.Y, t2.Y)),
245 Math.Min(t0.Z, Math.Min(t1.Z, t2.Z)));
246 var tMax = new Point3D(
247 Math.Max(t0.X, Math.Max(t1.X, t2.X)),
248 Math.Max(t0.Y, Math.Max(t1.Y, t2.Y)),
249 Math.Max(t0.Z, Math.Max(t1.Z, t2.Z)));
250
251 if (!BoxesOverlap(tMin, tMax, BoxMin, BoxMax))
252 continue;
253
254 stack.Clear();
255 stack.Push(_rootIndex);
256
257 while (stack.Count > 0)
258 {
259 int nIdx = stack.Pop();
260 ref readonly var node = ref _nodes[nIdx];
261
262 if (!BoxesOverlap(tMin, tMax, node.BoxMin, node.BoxMax))
263 continue;
264
265 if (node.IsLeaf)
266 {
267 for (int k = node.TriangleStart; k < node.TriangleStart + node.TriangleCount; k++)
268 {
269 ref readonly var partTri = ref _triangles[k];
270 if (BoxesOverlap(tMin, tMax, partTri.Min, partTri.Max) &&
271 TrianglesIntersect(t0, t1, t2, partTri.V0, partTri.V1, partTri.V2))
272 {
273 return true;
274 }
275 }
276 }
277 else
278 {
279 if (node.RightChild >= 0) stack.Push(node.RightChild);
280 if (node.LeftChild >= 0) stack.Push(node.LeftChild);
281 }
282 }
283 }
284
285 return false;
286 }

292 public bool IsPointInside(Point3D point)
293 {
294 if (_rootIndex < 0) return false;
295
296 if (point.X < BoxMin.X || point.X > BoxMax.X ||
297 point.Y < BoxMin.Y || point.Y > BoxMax.Y ||
298 point.Z < BoxMin.Z || point.Z > BoxMax.Z)
299 {
300 return false;
301 }
302
303 int hitCount = 0;
304 var stack = new Stack<int>(32);
305 stack.Push(_rootIndex);
306
307 while (stack.Count > 0)
308 {
309 int nIdx = stack.Pop();
310 ref readonly var node = ref _nodes[nIdx];
311
312 // Ray is along +Z: check if ray (point.X, point.Y, Z >= point.Z) hits node AABB
313 if (point.X < node.BoxMin.X || point.X > node.BoxMax.X ||
314 point.Y < node.BoxMin.Y || point.Y > node.BoxMax.Y ||
315 point.Z > node.BoxMax.Z)
316 {
317 continue;
318 }
319
320 if (node.IsLeaf)
321 {
322 for (int i = node.TriangleStart; i < node.TriangleStart + node.TriangleCount; i++)
323 {
324 ref readonly var tri = ref _triangles[i];
325 if (point.X >= tri.Min.X && point.X <= tri.Max.X &&
326 point.Y >= tri.Min.Y && point.Y <= tri.Max.Y &&
327 point.Z <= tri.Max.Z)
328 {
329 if (RayIntersectsTriangleZ(point, tri.V0, tri.V1, tri.V2))
330 {
331 hitCount++;
332 }
333 }
334 }
335 }
336 else
337 {
338 if (node.RightChild >= 0) stack.Push(node.RightChild);
339 if (node.LeftChild >= 0) stack.Push(node.LeftChild);
340 }
341 }
342
343 return (hitCount % 2) == 1;
344 }

353 private static bool RayIntersectsTriangleZ(Point3D orig, Point3D v0, Point3D v1, Point3D v2)
354 {
355 const double eps = 1e-8;
356 double e1x = v1.X - v0.X;
357 double e1y = v1.Y - v0.Y;
358 double e1z = v1.Z - v0.Z;
359
360 double e2x = v2.X - v0.X;
361 double e2y = v2.Y - v0.Y;
362 double e2z = v2.Z - v0.Z;
363
364 // rayDir = (0, 0, 1)
365 // pvec = rayDir x e2 = (-e2y, e2x, 0)
366 double pvecX = -e2y;
367 double pvecY = e2x;
368
369 double det = e1x * pvecX + e1y * pvecY;
370 if (Math.Abs(det) < eps) return false;
371
372 double invDet = 1.0 / det;
373 double tx = orig.X - v0.X;
374 double ty = orig.Y - v0.Y;
375 double tz = orig.Z - v0.Z;
376
377 double u = (tx * pvecX + ty * pvecY) * invDet;
378 if (u < 0.0 || u > 1.0) return false;
379
380 // qvec = t x e1
381 double qvecX = ty * e1z - tz * e1y;
382 double qvecY = tz * e1x - tx * e1z;
383 double qvecZ = tx * e1y - ty * e1x;
384
385 // v = dot(rayDir, qvec) * invDet = qvecZ * invDet
386 double v = qvecZ * invDet;
387 if (v < 0.0 || u + v > 1.0) return false;
388
389 // t = dot(e2, qvec) * invDet
390 double t = (e2x * qvecX + e2y * qvecY + e2z * qvecZ) * invDet;
391 return t > 1e-6;
392 }

397 public static bool TrianglesIntersect(Point3D v0, Point3D v1, Point3D v2, Point3D u0, Point3D u1, Point3D u2)
398 {
399 // Plane of triangle 2: N2 . X + d2 = 0
400 var e1 = new Vector3D(u1.X - u0.X, u1.Y - u0.Y, u1.Z - u0.Z);
401 var e2 = new Vector3D(u2.X - u0.X, u2.Y - u0.Y, u2.Z - u0.Z);
402 var n2 = Vector3D.Cross(e1, e2);
403 double d2 = -Vector3D.Dot(n2, new Vector3D(u0.X, u0.Y, u0.Z));
404
405 double dv0 = Vector3D.Dot(n2, new Vector3D(v0.X, v0.Y, v0.Z)) + d2;
406 double dv1 = Vector3D.Dot(n2, new Vector3D(v1.X, v1.Y, v1.Z)) + d2;
407 double dv2 = Vector3D.Dot(n2, new Vector3D(v2.X, v2.Y, v2.Z)) + d2;
408
409 if (Math.Abs(dv0) < 1e-6) dv0 = 0.0;
410 if (Math.Abs(dv1) < 1e-6) dv1 = 0.0;
411 if (Math.Abs(dv2) < 1e-6) dv2 = 0.0;
412
413 double dv0dv1 = dv0 * dv1;
414 double dv0dv2 = dv0 * dv2;
415
416 if (dv0dv1 > 0.0 && dv0dv2 > 0.0) return false; // All on same side
417
418 // Plane of triangle 1: N1 . X + d1 = 0
419 var e1a = new Vector3D(v1.X - v0.X, v1.Y - v0.Y, v1.Z - v0.Z);
420 var e2a = new Vector3D(v2.X - v0.X, v2.Y - v0.Y, v2.Z - v0.Z);
421 var n1 = Vector3D.Cross(e1a, e2a);
422 double d1 = -Vector3D.Dot(n1, new Vector3D(v0.X, v0.Y, v0.Z));
423
424 double du0 = Vector3D.Dot(n1, new Vector3D(u0.X, u0.Y, u0.Z)) + d1;
425 double du1 = Vector3D.Dot(n1, new Vector3D(u1.X, u1.Y, u1.Z)) + d1;
426 double du2 = Vector3D.Dot(n1, new Vector3D(u2.X, u2.Y, u2.Z)) + d1;
427
428 if (Math.Abs(du0) < 1e-6) du0 = 0.0;
429 if (Math.Abs(du1) < 1e-6) du1 = 0.0;
430 if (Math.Abs(du2) < 1e-6) du2 = 0.0;
431
432 double du0du1 = du0 * du1;
433 double du0du2 = du0 * du2;
434
435 if (du0du1 > 0.0 && du0du2 > 0.0) return false;
436
437 // Line of intersection of the two planes
438 var d = Vector3D.Cross(n1, n2);
439 double maxAxis = Math.Max(Math.Abs(d.X), Math.Max(Math.Abs(d.Y), Math.Abs(d.Z)));
440
441 // If coplanar, test 2D overlap
442 if (maxAxis < 1e-7)
443 {
444 return CoplanarTrianglesIntersect(n1, v0, v1, v2, u0, u1, u2);
445 }
446
447 // Project onto intersection axis
448 int index = 0;
449 if (Math.Abs(d.Y) > Math.Abs(d.X) && Math.Abs(d.Y) > Math.Abs(d.Z)) index = 1;
450 else if (Math.Abs(d.Z) > Math.Abs(d.X) && Math.Abs(d.Z) > Math.Abs(d.Y)) index = 2;
451
452 double vp0 = GetCoord(v0, index), vp1 = GetCoord(v1, index), vp2 = GetCoord(v2, index);
453 double up0 = GetCoord(u0, index), up1 = GetCoord(u1, index), up2 = GetCoord(u2, index);
454
455 ComputeIntervals(vp0, vp1, vp2, dv0, dv1, dv2, dv0dv1, dv0dv2, out double isect1A, out double isect1B);
456 ComputeIntervals(up0, up1, up2, du0, du1, du2, du0du1, du0du2, out double isect2A, out double isect2B);
457
458 if (isect1A > isect1B) { double tmp = isect1A; isect1A = isect1B; isect1B = tmp; }
459 if (isect2A > isect2B) { double tmp = isect2A; isect2A = isect2B; isect2B = tmp; }
460
461 return !(isect1B < isect2A || isect2B < isect1A);
462 }

466 private static void ComputeIntervals(double vv0, double vv1, double vv2, double d0, double d1, double d2, double d0d1, double d0d2, out double a, out double b)
467 {
468 if (d0d1 > 0.0)
469 {
470 a = vv2 + (vv0 - vv2) * d2 / (d2 - d0);
471 b = vv2 + (vv1 - vv2) * d2 / (d2 - d1);
472 }
473 else if (d0d2 > 0.0)
474 {
475 a = vv1 + (vv0 - vv1) * d1 / (d1 - d0);
476 b = vv1 + (vv2 - vv1) * d1 / (d1 - d2);
477 }
478 else if (d1 * d2 > 0.0 || d0 != 0.0)
479 {
480 a = vv0 + (vv1 - vv0) * d0 / (d0 - d1);
481 b = vv0 + (vv2 - vv0) * d0 / (d0 - d2);
482 }
483 else if (d1 != 0.0)
484 {
485 a = vv1 + (vv0 - vv1) * d1 / (d1 - d0);
486 b = vv1 + (vv2 - vv1) * d1 / (d1 - d2);
487 }
488 else if (d2 != 0.0)
489 {
490 a = vv2 + (vv0 - vv2) * d2 / (d2 - d0);
491 b = vv2 + (vv1 - vv2) * d2 / (d2 - d1);
492 }
493 else
494 {
495 a = vv0;
496 b = vv1;
497 }
498 }

500 private static bool CoplanarTrianglesIntersect(Vector3D normal, Point3D a0, Point3D a1, Point3D a2, Point3D b0, Point3D b1, Point3D b2)
501 {
502 // Quick 2D projection test along dominant axis
503 int aX = 0, aY = 1;
504 if (Math.Abs(normal.X) > Math.Abs(normal.Y) && Math.Abs(normal.X) > Math.Abs(normal.Z)) { aX = 1; aY = 2; }
505 else if (Math.Abs(normal.Y) > Math.Abs(normal.Z)) { aX = 0; aY = 2; }
506
507 // Check if any edge of B intersects triangle A, or if A is inside B / B is inside A
508 return EdgeIntersectsTriangle2D(a0, a1, a2, b0, b1, aX, aY) ||
509 EdgeIntersectsTriangle2D(a0, a1, a2, b1, b2, aX, aY) ||
510 EdgeIntersectsTriangle2D(a0, a1, a2, b2, b0, aX, aY) ||
511 PointInTriangle2D(b0, a0, a1, a2, aX, aY) ||
512 PointInTriangle2D(a0, b0, b1, b2, aX, aY);
513 }

522 private static bool LineSegmentsIntersect2D(Point3D p1, Point3D p2, Point3D p3, Point3D p4, int ax, int ay)
523 {
524 double x1 = GetCoord(p1, ax), y1 = GetCoord(p1, ay);
525 double x2 = GetCoord(p2, ax), y2 = GetCoord(p2, ay);
526 double x3 = GetCoord(p3, ax), y3 = GetCoord(p3, ay);
527 double x4 = GetCoord(p4, ax), y4 = GetCoord(p4, ay);
528
529 double denom = (y4 - y3) * (x2 - x1) - (x4 - x3) * (y2 - y1);
530 if (Math.Abs(denom) < 1e-9) return false;
531
532 double ua = ((x4 - x3) * (y1 - y3) - (y4 - y3) * (x1 - x3)) / denom;
533 double ub = ((x2 - x1) * (y1 - y3) - (y2 - y1) * (x1 - x3)) / denom;
534
535 return ua >= 0.0 && ua <= 1.0 && ub >= 0.0 && ub <= 1.0;
536 }

538 private static bool PointInTriangle2D(Point3D p, Point3D a, Point3D b, Point3D c, int ax, int ay)
539 {
540 double px = GetCoord(p, ax), py = GetCoord(p, ay);
541 double ax_ = GetCoord(a, ax), ay_ = GetCoord(a, ay);
542 double bx_ = GetCoord(b, ax), by_ = GetCoord(b, ay);
543 double cx_ = GetCoord(c, ax), cy_ = GetCoord(c, ay);
544
545 double v0x = cx_ - ax_, v0y = cy_ - ay_;
546 double v1x = bx_ - ax_, v1y = by_ - ay_;
547 double v2x = px - ax_, v2y = py - ay_;
548
549 double dot00 = v0x * v0x + v0y * v0y;
550 double dot01 = v0x * v1x + v0y * v1y;
551 double dot02 = v0x * v2x + v0y * v2y;
552 double dot11 = v1x * v1x + v1y * v1y;
553 double dot12 = v1x * v2x + v1y * v2y;
554
555 double invDenom = 1.0 / (dot00 * dot11 - dot01 * dot01);
556 double u = (dot11 * dot02 - dot01 * dot12) * invDenom;
557 double v = (dot00 * dot12 - dot01 * dot02) * invDenom;
558
559 return u >= 0.0 && v >= 0.0 && (u + v) <= 1.0;
560 }
