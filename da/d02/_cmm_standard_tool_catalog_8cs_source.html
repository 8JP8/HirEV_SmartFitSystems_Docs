6 public class CmmHeadDefinition
7 {

78 public class CmmToolMontage
79 {
80 public string MontageId { get; set; } = string.Empty;
81 public string DisplayName { get; set; } = string.Empty;
82 public string Description { get; set; } = string.Empty;
83
84 public CmmHeadDefinition Head { get; set; } = new CmmHeadDefinition();
85 public CmmExtensionBarDefinition? Adapter { get; set; }
86 public CmmExtensionBarDefinition? Extension { get; set; }
87 public CmmProbeSensorDefinition Probe { get; set; } = new CmmProbeSensorDefinition();
88 public CmmStylusExtensionDefinition? StylusExtension { get; set; }
89 public CmmStylusDefinition Stylus { get; set; } = new CmmStylusDefinition();
90
94 public double TotalReachLength =>
95 (Adapter?.Length ?? 0.0) +
96 (Extension?.Length ?? 0.0) +
97 Probe.BodyLength +
98 (StylusExtension?.Length ?? 0.0) +
99 Stylus.OverallLength;
100
104 public double LateralClearance => TotalReachLength - (Head.BodyWidth / 2.0);
105
106 public double TotalAssemblyLength =>
107 Head.TotalHeadHeight + TotalReachLength;
108
109 public int DefaultRackSlot { get; set; } = 1;
110 public string TargetDmisSensorLabel => MontageId;
111 }

741 public static CmmToolMontage ResolveToolMontage(string? toolName)
742 {
743 if (string.IsNullOrWhiteSpace(toolName))
744 return StandardMontages[0];
745
746 string clean = toolName.Trim();
747
748 // 1. Direct match on MontageId or DisplayName
749 var directMatch = StandardMontages.FirstOrDefault(m =>
750 string.Equals(m.MontageId, clean, System.StringComparison.OrdinalIgnoreCase) ||
751 string.Equals(m.DisplayName, clean, System.StringComparison.OrdinalIgnoreCase));
752 if (directMatch != null)
753 return directMatch;
754
755 // 2. Legacy / alias mappings (remap old EXT names to clean SE or bare montages)
756 if (clean.StartsWith("PH10M_TP20_EXT50", System.StringComparison.OrdinalIgnoreCase) ||
757 clean.StartsWith("PH10M_TP20_EM1", System.StringComparison.OrdinalIgnoreCase))
758 return StandardMontages.First(m => m.MontageId == "PH10M_TP20_SE30_3x30");
759
760 if (clean.StartsWith("PH10M_TP20_EXT100", System.StringComparison.OrdinalIgnoreCase) ||
761 clean.StartsWith("PH10M_TP20_EM2", System.StringComparison.OrdinalIgnoreCase))
762 return StandardMontages.First(m => m.MontageId == "PH10M_TP20_SE50_4x50");
763
764 if (clean.StartsWith("PH10M_TP20_EXT200", System.StringComparison.OrdinalIgnoreCase))
765 return StandardMontages.First(m => m.MontageId == "PH10M_TP20_SE100_5x50");
766
767 if (string.Equals(clean, "PH10M_TP20", System.StringComparison.OrdinalIgnoreCase))
768 return StandardMontages.First(m => m.MontageId == "PH10M_TP20_2x20");
769
770 if (clean.StartsWith("PH10M_TP200_EXT50", System.StringComparison.OrdinalIgnoreCase))
771 return StandardMontages.First(m => m.MontageId == "PH10M_TP200_SE20_3x30");
772
773 if (clean.StartsWith("PH10M_TP200_EXT100", System.StringComparison.OrdinalIgnoreCase))
774 return StandardMontages.First(m => m.MontageId == "PH10M_TP200_SE50_4x50");
775
776 if (string.Equals(clean, "PH10M_TP200", System.StringComparison.OrdinalIgnoreCase))
777 return StandardMontages.First(m => m.MontageId == "PH10M_TP200_2x20");
778
779 if (clean.StartsWith("PH10M_SP25M_EXT", System.StringComparison.OrdinalIgnoreCase))
780 return StandardMontages.First(m => m.MontageId == "PH10M_SP25M_SE20_3x30");
781
782 if (string.Equals(clean, "PH10M_SP25M", System.StringComparison.OrdinalIgnoreCase))
783 return StandardMontages.First(m => m.MontageId == "PH10M_SP25M_3x30");
784
785 if (clean.StartsWith("PH20_TP20_EXT", System.StringComparison.OrdinalIgnoreCase))
786 return StandardMontages.First(m => m.MontageId == "PH20_TP20_SE20_3x30");
787
788 if (string.Equals(clean, "PH20_TP20", System.StringComparison.OrdinalIgnoreCase))
789 return StandardMontages.First(m => m.MontageId == "PH20_TP20_2x20");
790
791 if (IsStandaloneStylusTool(clean))
792 return GetBestMontageForStylus(clean);
793
794 // Default fallback
795 return StandardMontages[0];
796 }

812 public static CmmToolMontage GetBestMontageForStylus(string? stylusName)
813 {
814 if (string.IsNullOrWhiteSpace(stylusName))
815 return StandardMontages[0];
816
817 string clean = stylusName.Trim();

819 if (clean.IndexOf("D1", System.StringComparison.OrdinalIgnoreCase) >= 0)
820 return StandardMontages.FirstOrDefault(m => m.MontageId == "PH10M_TP20_2x20") ?? StandardMontages[0];
821
822 if (clean.IndexOf("D2", System.StringComparison.OrdinalIgnoreCase) >= 0)
823 return StandardMontages.FirstOrDefault(m => m.MontageId == "PH10M_TP20_2x20") ?? StandardMontages[0];
824
825 if (clean.IndexOf("D3", System.StringComparison.OrdinalIgnoreCase) >= 0)
826 return StandardMontages.FirstOrDefault(m => m.MontageId == "PH10M_TP20_3x30") ?? StandardMontages[0];
827
828 if (clean.IndexOf("D4", System.StringComparison.OrdinalIgnoreCase) >= 0)
829 return StandardMontages.FirstOrDefault(m => m.MontageId == "PH10M_TP20_4x50") ?? StandardMontages[0];
830
831 if (clean.IndexOf("D5", System.StringComparison.OrdinalIgnoreCase) >= 0)
832 return StandardMontages.FirstOrDefault(m => m.MontageId == "PH10M_TP20_5x50") ?? StandardMontages[0];
833
834 return StandardMontages.FirstOrDefault(m => m.MontageId == "PH10M_TP20_3x30") ?? StandardMontages[0];
835 }
